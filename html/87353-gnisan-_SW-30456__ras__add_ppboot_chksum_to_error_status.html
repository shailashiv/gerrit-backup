+u32 btl_crc32(u32 crc, unsigned char const *p, size_t len)
+{
+	u32 data;
+
+	while (len >= sizeof(u32)) {
+		data = __builtin_bswap32(*(u32 const *)p);
+		crc = crc32(crc, (unsigned char const *)&data, sizeof(u32));
+		len -= sizeof(u32);
+		p += sizeof(u32);
+	}
+
+	return crc;
+}

+bool hl_ppboot_is_valid(bool is_prime)
+{
+	u8 *buf = (u8 *)FIT_IMG_BASE; /* Use fit base as playground */
+	u32 offset, memoff, memsz, crc_calc, crc_cmd;
+	u16 blk_num;
+	int i, op;
+
+	if (is_prime)
+		offset = PPBOOT_OFFSET_PRIMARY;
+	else
+		offset = PPBOOT_OFFSET_SECONDARY;
+
+	spi_copy(buf, offset, PPBOOT_MAX_SIZE, NULL);
+
+	i = 0;
+	while (i < PPBOOT_MAX_SIZE) {
+		crc_calc = btl_crc32(0, &buf[i], BTL_CMD_SIZE);
+		crc_cmd = __builtin_bswap32(*(u32 *)(&buf[i] + BTL_CMD_SIZE));
+		if (crc_calc != crc_cmd)
+			return false;
+
+		op = buf[i + BTL_OP_OFFSET] >> BTL_OP_SHIFT;
+		switch (op) {
+		case BTL_CMD_MEM:
+			blk_num = (buf[i + BTL_BLK_OFFSET_HIGH] << 8) |
+				buf[i + BTL_BLK_OFFSET_LOW];
+			memsz = blk_num * BTL_BLK_MUL;
+			memoff = BTL_CMD_SIZE + BTL_CMD_CRC;
+
+			crc_calc = btl_crc32(0, &buf[i] + memoff, memsz);
+			crc_cmd = __builtin_bswap32(*(u32 *)(&buf[i] + memoff +
+							     memsz));
+			if (crc_calc != crc_cmd)
+				return false;
+
+			i += memsz;
+		case BTL_CMD_END:
+			return true;
+		default:
+			break;
+		}
+
+		i += BTL_CMD_SIZE + BTL_CMD_CRC;
+	}
+
+	return false; /* No end */
+}
