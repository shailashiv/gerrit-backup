+static void i2csi_req_handle_read_clear(void)
+{
+	const struct i2csi_read_clear *rc = NULL;
+	int i, sz;
+	int rel_off, rel_size;
+
+	if (!req_rd.data)
+		return;
+
+	switch (req_wd.hdr.opcode) {
+	case HL_I2CI_REQ_OP_ERROR_STATUS:
+		rc = op_err_status_rc;
+		sz = ARRAY_SIZE(op_err_status_rc);
+		break;
+	default:
+		return;
+	}
+
+	for (i = 0; i < sz; i++) {
+		if (req_wd.hdr.offset > rc[i].offset + rc[i].size)
+			continue;
+
+		if (req_wd.hdr.offset + req_wd.hdr.rqdl < rc[i].offset)
+			continue;
+
+		if (req_wd.hdr.offset > rc[i].offset) {
+			/* Reading started after the beginning of the field */
+			rel_off = req_wd.hdr.offset - rc[i].offset;
+			if (rc[i].size - rel_off < req_wd.hdr.rqdl)
+				rel_size = rc[i].size - rel_off;
+			else
+				rel_size = req_wd.hdr.rqdl;
+
+			memset(req_rd.data, 0, rel_size);
+		} else {
+			/* Reading started before the beginning of the field */
+			rel_off = rc[i].offset - req_wd.hdr.offset;
+			if (req_wd.hdr.rqdl - rel_off < rc[i].size)
+				rel_size = req_wd.hdr.rqdl - rel_off;
+			else
+				rel_size = rc[i].size;
+
+			memset(&req_rd.data[rel_off], 0, rel_size);
+		}
+	}
+}
