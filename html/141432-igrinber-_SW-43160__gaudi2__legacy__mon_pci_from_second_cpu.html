Commit Message

[SW-43160] gaudi2: legacy: mon pci from second cpu

PCI link may appear at any stage, not just on boot. It needs to be
monitored and handled swiftly or else the link remains broken.
Preboot contains a good number of routines which may cause a delay to
pci link monitor polling module since they share the same core.

This patch utilizes the secondary ARM core for PCI link monitoring. This
way it can be done in parallel with the rest of preboot logic.

+void preboot_pcie_monitoring(void)
+{
+	/* Wait for PRSTn */
+	while (!regmap_readl(mmPSOC_GLOBAL_CONF_PRSTN))
+		;
+
+	/* Only after PRSTn is de-asserted */
+	preboot_pcie_enable();
+
+	while (1) {
+		unsigned int prstn, ltssm, smlh, rdlh;
+
+		prstn = regmap_readl(mmPSOC_GLOBAL_CONF_PRSTN);
+		ltssm = regmap_readl(mmPCIE_AUX_LTSSM_EN);
+		smlh = regmap_readl(mmPCIE_AUX_SMLH_LINK_UP);
+		rdlh = regmap_readl(mmPCIE_AUX_RDLH_LINK_UP);
+
+		if (prstn && smlh && rdlh)
+			break;
+
+		/*
+		 * Some host assert prstn during link up while moving to GEN3
+		 * This cause BAR size to be reset
+		 * We stop link up process by disabling ltssm
+		 * till BAR is reconfigured
+		 */
+		if (!prstn && ltssm) {
+			regmap_writel(0, mmPCIE_AUX_LTSSM_EN);
+			continue;
+		}
+
+		if (prstn && !ltssm) {
+			regmap_writel(3, mmPCIE_AUX_LTSSM_EN);
+			continue;
+		}
+	}
+
+	/*
+	 * W/A H/W bug (Gaudi1) that may lead to have a wrong mask (0x1)
+	 * in LTSSM. Probably can be removed for Gaudi2, but we keep it as it
+	 * does not hurt.
+	 */
+	regmap_writel(3, mmPCIE_AUX_LTSSM_EN);
+}
