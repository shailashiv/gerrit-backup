Commit Message

[SW-50880] greco: arc: run pcie initialization on arc0

This patch add support to perform PCIe initialization on secondary
CPU ARC0 and let primary CPU ARC1 continue with preboot. Upon successful
initialization of PCIe, secondary CPU ARC0 will go in sleep mode or halt
mode which will be indication in preboot for PCIe is initialized.

+static void preboot_pcie_monitoring(void)
+{
+	/* Wait for PRSTn */
+	while (!regmap_readl(mmPSOC_GLOBAL_CONF_PRSTN))
+		;
+
+	/* Only after PRSTn is de-asserted */
+	preboot_pcie_enable();
+
+	while (1) {
+		unsigned int prstn, ltssm, smlh, rdlh;
+
+		prstn = regmap_readl(mmPSOC_GLOBAL_CONF_PRSTN);
+		ltssm = regmap_readl(mmPCIE_AUX_LTSSM_EN);
+		smlh = regmap_readl(mmPCIE_AUX_SMLH_LINK_UP);
+		rdlh = regmap_readl(mmPCIE_AUX_RDLH_LINK_UP);
+
+		if (prstn && smlh && rdlh)
+			break;
+
+		/*
+		 * Some host assert prstn during link up while moving to GEN3
+		 * This cause BAR size to be reset
+		 * We stop link up process by disabling ltssm
+		 * till BAR is reconfigured
+		 */
+		if (!prstn && ltssm) {
+			regmap_writel(0, mmPCIE_AUX_LTSSM_EN);
+			continue;
+		}
+
+		if (prstn && !ltssm) {
+			regmap_writel(3, mmPCIE_AUX_LTSSM_EN);
+			continue;
+		}
+	}
+
+	/* W/A H/W bug that may lead to have a wrong mask (0x1) in LTSSM */
+	regmap_writel(3, mmPCIE_AUX_LTSSM_EN);
+}
+
+void secondary_main(void)
+{
+	preboot_pcie_monitoring();
+
+	/* Put secondary CPU in sleep state until further notice */
+	while (1)
+		__asm__ __volatile__("sleep\n\t" : : );
+}
+
+void preboot_pcie_wait_for_link(void)
+{
+	u32 *secondary_iccm_addr = (u32 *) ARC0_ICCM_BASE;
+
+	/* init dev id as soon as possible, while waiting for prstn */
+	preboot_device_id_init();
+
+	/*
+	 * update secondary entry address to 0th location of ICCM
+	 * ARC once put in running state will jump to address
+	 * specified in 0th location of ICCM
+	 */
+	*secondary_iccm_addr = (u32) &secondary_entry;
+	run_arc(0);
+}
