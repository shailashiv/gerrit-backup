+void fit_load_print_error(enum fit_load_result errtype)
+{
+	switch (errtype) {
+	case FIT_LOAD_BAD_LOAD_ADDR:
+	case FIT_LOAD_NOT_RUNNABLE:
+	case FIT_LOAD_CORRUPTED:
+		preboot_puts("FIT loading failed\n");
+		break;
+	case FIT_LOAD_SEC_FAILED:
+		preboot_puts("FIT security validation failed\n");
+		break;
+	default:
+		preboot_puts_hex("unrecognized error value for: fit_load_result",
+			(unsigned int)errtype, false);
+		break;
+	}
+}
+
+void fit_format_print_error(enum fit_format_result errtype)
+{
+	switch (errtype) {
+	case FIT_FORMAT_HEADER_CHECK_FAILED:
+		preboot_puts("FIT header check failed\n");
+		break;
+	case FIT_FORMAT_NO_DESCRIPTION:
+		preboot_puts("Wrong FIT format: no description\n");

+		preboot_puts("Unsupported load type\n");

+	default:
