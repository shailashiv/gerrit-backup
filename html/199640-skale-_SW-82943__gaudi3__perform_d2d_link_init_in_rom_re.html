[SW-82943] gaudi3: perform d2d link init in rom recovery

As part of rom recovery, rom is required to perform D2D link
initialization. ROM will not make use of D2D link but needs
to enable it for debug purpose. This patch implements D2D link
initialization sequence as part of rom recovery. D2D link init should
be perfomed by both die's.

+#define D2D_LINK_STATUS_RETRIES		100
+#define D2D_LINK_STATUS_UDELAY		100
+static void rom_d2d_link_init(void)
+{
+	uint32_t retry = D2D_LINK_STATUS_RETRIES;
+	uint32_t val;
+
+	/* D2D link init will be performed by both die's */
+	if (arcboot_rom_global.sku)
+		return;
+
+	if (arcboot_rom_global.config2_fuse & EFUSE_SYSCFG2_D2D_DIS_MASK)
+		return; /* skip d2d init */
+
+	arcboot_rom_status_update(PROGRESS_ARCBOOT_ROM_D2D_LINK_START);
+
+	/*
+	 * TODO: replace with register name and masks
+	 * currently not defined in specs git
+	 */
+	/* Check BUSY bit for 0 value, expected to be 0 */
+	val = regmap_readl(0xC48B0FC);
+	if (val & 0x10) {
+		arcboot_rom_status_update(PROGRESS_ARCBOOT_ROM_D2D_LINK_STS_BUSY);
+		return;
+	}
+
+	/* Set START bit for D2D link initialization */
+	regmap_writel(BIT(0), 0xC48B0F8);
+
+	/*
+	 * Poll 10ms for BUSY bit to be 0 indicating link initialization
+	 * is completed successfully.
+	 */
+	while (1) {
+		val = regmap_readl(0xC48B0FC);
+		if (!(val & 0x10))
+			break;
+
+		retry--;
+		if (!retry) {
+			arcboot_rom_status_update(PROGRESS_ARCBOOT_ROM_D2D_LINK_INIT_TIMEDOUT);
+			return;
+		}
+
+		udelay(D2D_LINK_STATUS_UDELAY);
+	}
+
+	arcboot_rom_status_update(PROGRESS_ARCBOOT_ROM_D2D_LINK_INIT_SUCCESS);
+}
