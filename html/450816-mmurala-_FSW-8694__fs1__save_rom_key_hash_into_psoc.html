+void security_load_rom_key_hash()
+{
+	uint8_t key_idx, dword;
+	uint32_t key_reg;
+	uint32_t *hash = (uint32_t *)rom_key_hashes;
+	uint32_t key_reg_base_array[] = {
+		IO0_H0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_PUBLIC_KEY_0_0),
+		IO0_H0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_PUBLIC_KEY_1_0),
+		IO0_H0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_PUBLIC_KEY_2_0),
+		IO0_H0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_PUBLIC_KEY_3_0),
+		IO0_H0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_PUBLIC_KEY_4_0)
+	};
+
+	/* KEY LOCATION: ROM */
+	boot_rom_status_update(PROGRESS_KEY_LOC_ROM);
+
+	/* TODO: Store keys in boot_info if required */
+
+	for (key_idx = 0; key_idx < NUM_KEYS_IN_ROM; key_idx++) {
+		key_reg = key_reg_base_array[key_idx];
+		for (dword = 0; dword < SECURITY_DWORD_PER_KEY; dword++) {
+			regmap_writel(hash[(key_idx * SECURITY_DWORD_PER_KEY) + dword],
+					key_reg);
+
+			key_reg += sizeof(uint32_t);
+		}
+	}
+}
