[SW-12754] gaudi: security: add lbw rrs protection

Add LBW protection with Range Registers.
The LBW RRs protection in Gaudi is implemented by using "min and max"
concept. Therefore we specify the minimum and maximum addresses to
protect.
The "hit" is implemented by H/W as not inclusive - meaning that the
addresses we supply are not protected by the specific register.
Formula for the "hit": min < hit < max.
So we supply min and max outside of the protected area.

+static void preboot_secure_lbw_rr(int rr, u64 min64, u64 max64)
+{
+	int i;
+	u32 min = min64 & ~HL_CFG_BASE;
+	u32 max = max64 & ~HL_CFG_BASE;
+	u32 rr_base, rr_dest, hit;
+
+	for (i = 0; i < IF_RTRS_NUM; i++) {
+		rr_base = rr * sizeof(u32) + IF_RTRS_DIST * i;
+		rr_dest = rr_base + mmSIF_RTR_0_LBW_RANGE_PRIV_MIN_AW_0;
+		regmap_writel(min, rr_dest);
+		rr_dest = rr_base + mmSIF_RTR_0_LBW_RANGE_PRIV_MAX_AW_0;
+		regmap_writel(max, rr_dest);
+		rr_dest = rr_base + mmSIF_RTR_0_LBW_RANGE_PRIV_MIN_AR_0;
+		regmap_writel(min, rr_dest);
+		rr_dest = rr_base + mmSIF_RTR_0_LBW_RANGE_PRIV_MAX_AR_0;
+		regmap_writel(max, rr_dest);
+
+		/* Enable (hit) RR */
+		rr_base = IF_RTRS_DIST * i;
+		rr_dest = rr_base + mmSIF_RTR_0_LBW_RANGE_PRIV_HIT_AW;
+		hit = regmap_readl(rr_dest);
+		regmap_writel(hit | BIT(rr), rr_dest);
+
+		rr_dest = rr_base + mmSIF_RTR_0_LBW_RANGE_PRIV_HIT_AR;
+		hit = regmap_readl(rr_dest);
+		regmap_writel(hit | BIT(rr), rr_dest);
+	}
+}
