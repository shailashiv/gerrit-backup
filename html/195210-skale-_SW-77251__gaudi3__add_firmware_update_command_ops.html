module operations which will listen on specific registers to receive

+static int fwupd_create_resp(void *res_fit, uint32_t avail_sz, const int load_type,
+			     const struct fwupd_resp_prop *props, int prop_cnt, uint32_t *out_sz)
+{
+	int err, i;
+	unsigned int sz;
+
+	err = fit_init_resp(res_fit, avail_sz);
+	if (err)
+		goto fit_init_err;
+
+	err = fdt_property_u32(res_fit, HL_FIT_LOAD_TYPE_PROP, load_type);
+	if (err)
+		goto fit_init_err;
+
+	for (i = 0; i < prop_cnt; i++) {
+		err = fdt_property(res_fit, props[i].name, props[i].data, props[i].size);
+		if (err)
+			goto fit_init_err;
+	}
+
+	err = fit_fin_resp(res_fit, &sz);
+	if (err)
+		goto fit_init_err;
+
+	*out_sz = sz;
+
+fit_init_err:
+	return err;
+}

+static void module_fwupd_send_status_fast(struct fwupd_state *state,
+					  enum hl_upd_sts sts, void *offset)
+{
+	struct hl_upd_status status;
+	uintptr_t address = (uintptr_t)offset;
+
+	status.val = 0;
+	status.status = sts;
+
+	/* only if we get a address we need to calc offset */
+	if (address >=  SDRAM_BASE)
+		status.offset = (uint32_t)((address - SDRAM_BASE) &
+				~HL_UPD_STS_OFFSET_MASK);
+	else
+		status.offset = (uint32_t)(address & ~HL_UPD_STS_OFFSET_MASK);
+
+	state->sts_writel(status.val);
+}

+static int module_fwupd_proto_status(struct fwupd_state *state,
+				     struct hl_upd_proto_status *ret_status)
+{
+	struct hl_upd_proto_status status;
+	struct fwupd_resp_prop *resp_prop;
+	struct mem_desc mdesc;
+	uint32_t sts_err, sts_serr;
+	int err;
+
+	resp_prop = &resp_props[RESP_PROP_STATUS];
+	resp_prop->name = HL_FIT_RES_PROTO_STS_PROP;
+	resp_prop->data = &status;
+	resp_prop->size = sizeof(status);
+
+	resp_prop = &resp_props[RESP_PROP_SPI_VERSION];
+	resp_prop->name = HL_FIT_RES_SPI_VERSION_PROP;
+	resp_prop->data = "ARCBOOT";
+	resp_prop->size = VERSION_MAX_LEN;
+
+	err = arcboot_get_mem_region(state->mregion, &mdesc);
+	if (err)
+		return err;
+
+	state->map_offset = (void *)mdesc.base;
+
+	/*
+	 * Here we assume that the error is reported only once and
+	 * regardless of the interface
+	 */
+	arcboot_error_read_and_clr(&sts_err, &sts_serr);
+	fwupd_get_proto_status(&status, sts_err, sts_serr);
+
+	err = fwupd_create_resp(state->map_offset, mdesc.size, FIT_LOAD_TYPE_PROTO_STATUS,
+				resp_props, ARRAY_SIZE(resp_props), &state->map_size);
+
+	/* TODO: add error checks */
+
+	if (ret_status)
+		*ret_status = status;
+
+	return err;
+}

+static void module_fwupd_send_status(struct fwupd_state *state,
+				     enum hl_upd_sts sts, void *offset)
+{
+	int err;
+	void *tmp_offset = offset;
+
+	/*
+	 * can send fast if not error
+	 * if it is error - must be slow
+	 */
+	switch (sts) {
+	case HL_UPD_STS_MAP_ERR:
+	case HL_UPD_STS_ERR:
+	case HL_UPD_STS_RST_ERR:
+	case HL_UPD_STS_VALID_ERR:
+	case HL_UPD_STS_TIMEOUT_ERR:
+		/*
+		 * The error inform function, work in such a way
+		 * that the error will be set only once (assuming no
+		 * concurrency) so if the error inform function is called before
+		 * entering the current function, module_fwupd_send_status,
+		 * the second call will not overrun its values.
+		 * it allows support legacy code by setting the sent error
+		 * status as protocol error and set it's minor value.
+		 */
+		arcboot_error_inform(HL_FW_ERR_PROTO, sts);
+		if (state->exec_res != FOUND_SLOW_RETURN) {
+			/*
+			 * error status response requires FIT creation, that
+			 * implies - SLOW MODE
+			 * if an error status request has been done from within
+			 * a slow mode invocation. The request can be served
+			 * and skip this conditional block.
+			 * otherwise, the control should be returned to the main
+			 * event loop,with request for change into SLOW MODE and
+			 * execute a command state to send the error status.
+			 * There is not enough commands bits to add this
+			 * command state, so the alternative for that, is using
+			 * a flag to ask for running the state. That condition
+			 * is not enough; an error should be inform in order to
+			 * trigger the response.
+			 * Both of those condition should be checked in the
+			 * module's check and run functions.
+			 * The error was set previously (2 lines above), and
+			 * the request to change the state is done by this flag.
+			 *
+			 * WARNING:	don't change state->exec_res it is the
+			 *		check function responsibility
+			 */
+			state->sending_error_flag = true;
+			break;
+		}
+
+		/* if an explicit FIT was attached - use it */
+		if (!offset) {
+			tmp_offset = state->map_offset;
+			err = module_fwupd_proto_status(state, NULL);
+
+			/* if prepare fails - just don't include it*/
+			if (err)
+				tmp_offset = 0;
+		}
+		break;
+	default:
+		break;
+	}
+
+	/* on STS_OK / STS_ACK etc,  the original offset is set */
+	module_fwupd_send_status_fast(state, sts, tmp_offset);
+}

+/* NOOP is a sync command, so here we decide what execution type is needed */
+static enum cmd_check_res module_fwupd_check_noop(struct fwupd_state *state)
+{
+	enum cmd_check_res res = NOT_FOUND;
+
+	/* check the command we got before waiting for sync */
+	switch (state->wfc_cmd.cmd) {
+	case HL_UPD_NOOP:
+		break;
+	case HL_UPD_RUN_SRAM:
+	case HL_UPD_GET_PROTO_STATUS:
+		res = FOUND_SLOW_RETURN;
+		break;
+	case HL_UPD_RST_STATE:
+	case HL_UPD_MAP_SRAM:
+	default:
+		res = FOUND_FAST_RETURN;
+		break;
+	}
+
+	return res;
+}
+
+static enum cmd_check_res __module_fwupd_check(struct fwupd_state *state)
+{
+	enum cmd_check_res res = NOT_FOUND;
+
+	if (state->is_disabled)
+		return res;
+
+	/*
+	 * the error status response, involved
+	 * the creation of a FIT file.
+	 * this implies a slow return
+	 * Due to the lack of commands numbers the cmd state (4 bits)
+	 * can't be change to a specific state therefore a flag was added
+	 * other option is to add an internal cmd level.
+	 */
+	if (state->sending_error_flag && arcboot_error_is_pending())
+		return FOUND_SLOW_RETURN;
+
+	if (state->exec_res != NOT_FOUND)
+		return state->exec_res;
+
+	state->cmd.val = state->cmd_readl();
+
+	if (state->cmd.cmd == state->wfc_cmd.cmd)
+		return NOT_FOUND;
+
+	switch (state->cmd.cmd) {
+	case HL_UPD_NOOP:	/* NOOP is a special command */
+		res = module_fwupd_check_noop(state);
+		break;
+	case HL_UPD_CLR_STS:
+	case HL_UPD_RST_STATE:
+	case HL_UPD_MAP_SRAM:
+	case HL_UPD_RUN_SRAM:
+	case HL_UPD_GET_PROTO_STATUS:
+		res = FOUND_FAST_RETURN;
+		break;
+	default:
+		break;
+	}
+
+	return res;
+}
+
+enum cmd_check_res module_fwupd_check(void)
+{
+	return __module_fwupd_check(&state);
+}

+static enum hl_upd_sts module_fwupd_map(struct fwupd_state *state,
+					unsigned int size, __unused enum hl_upd_cmd cmd)
+{
+	int err;
+	struct mem_desc mdesc;
+	enum hl_upd_sts sts = HL_UPD_STS_MAP_OK;
+	uintptr_t base_offset;
+
+	err = arcboot_get_mem_region(state->mregion, &mdesc);
+	if (err)
+		return HL_UPD_STS_MAP_ERR;
+
+	base_offset = mdesc.base;
+
+	err = !arcboot_is_valid_address_range(base_offset, size);
+	if (err || size > mdesc.size)
+		return HL_UPD_STS_VALID_ERR;
+
+	state->map_offset = (void *)base_offset;
+	state->map_size = size;
+
+	return sts;
+}

+static void module_fwupd_run_fast_noop(struct fwupd_state *state)
+{
+	enum hl_upd_sts sts = HL_UPD_STS_NOOP;
+
+	/* depending on the current state machine, perform action */
+	switch (state->wfc_cmd.cmd) {
+	case HL_UPD_NOOP:
+	case HL_UPD_CLR_STS:
+		break;
+	case HL_UPD_RST_STATE:
+		module_fwupd_reset_state_secure(state);
+		module_fwupd_send_status(state, HL_UPD_STS_RST_OK, NULL);
+		break;
+	case HL_UPD_MAP_SRAM:
+		sts = module_fwupd_map(state, state->wfc_cmd.size, state->wfc_cmd.cmd);
+		module_fwupd_send_status(state, sts, state->map_offset);
+		break;
+	case HL_UPD_RUN_SRAM:
+	case HL_UPD_GET_PROTO_STATUS:
+	default:	/* should never get here */
+		break;
+	}
+}
+
+static void __module_fwupd_run_fast(struct fwupd_state *state)
+{
+	enum hl_upd_sts sts = HL_UPD_STS_NOOP;
+	int err = 0;
+
+	switch (state->cmd.cmd) {
+	case HL_UPD_NOOP:	/* NOOP is a special command */
+		module_fwupd_run_fast_noop(state);
+		break;
+	case HL_UPD_CLR_STS:
+		/* Clear the status */
+		module_fwupd_send_status(state, HL_UPD_STS_NOOP, NULL);
+		break;
+	case HL_UPD_RST_STATE:
+		module_fwupd_send_status(state, HL_UPD_STS_RST_ACK, NULL);
+		break;
+	case HL_UPD_MAP_SRAM:
+		/* Protect from multi-MAP */
+		if (state->map_offset || state->map_size) {
+			sts = HL_UPD_STS_VALID_ERR;
+			err = 1;
+		} else {
+			sts = HL_UPD_STS_MAP_ACK;
+		}
+
+		module_fwupd_send_status(state, sts, NULL);
+		break;
+	case HL_UPD_RUN_SRAM:
+		/* Protect from RUN before MAP */
+		if (state->map_offset && state->map_size) {
+			sts = HL_UPD_STS_ACK;
+		} else {
+			sts = HL_UPD_STS_VALID_ERR;
+			err = 1;
+		}
+
+		module_fwupd_send_status(state, sts, NULL);
+		break;
+	case HL_UPD_GET_PROTO_STATUS:
+		module_fwupd_send_status(state, HL_UPD_STS_ACK, NULL);
+		/* We ACK the command, but handle only on CLR, so save cmd */
+		break;
+	default:
+		module_fwupd_send_status(state, HL_UPD_STS_VALID_ERR, NULL);
+		err = 1;
+		break;
+	}
+
+	if (!err)
+		module_fwupd_state_shift(state);
+}
+
+void module_fwupd_run_fast(void)
+{
+	__module_fwupd_run_fast(&state);
+}

+static enum hl_upd_sts module_fwupd_run_cmd(struct fwupd_state *state,

+static void module_fwupd_push_error_status(struct fwupd_state *state)
+{
+	int err;
+	void *offset = state->map_offset;
+	struct hl_upd_proto_status status;
+
+	err = module_fwupd_proto_status(state, &status);
+	if (err)
+		offset = 0;
+
+	module_fwupd_send_status(state, status.err, offset);
+	module_fwupd_send_status(state, HL_UPD_STS_NOOP, offset);
+}

+static void __module_fwupd_run(struct fwupd_state *state)
+{
+	enum hl_upd_sts sts = HL_UPD_STS_NOOP;
+	int err = 0;
+
+	if (state->sending_error_flag && arcboot_error_is_pending()) {
+		module_fwupd_push_error_status(state);
+		state->sending_error_flag = false;
+	}
+
+	switch (state->wfc_cmd.cmd) {
+	case HL_UPD_RUN_SRAM:
+		err = module_fwupd_run_cmd(state, state->wfc_cmd.cmd, &sts);
+		module_fwupd_send_status(state, sts, NULL);
+		break;
+	case HL_UPD_GET_PROTO_STATUS:
+		err = module_fwupd_proto_status(state, NULL);
+		if (err)
+			module_fwupd_send_status(state, HL_UPD_STS_ERR, NULL);
+		else
+			module_fwupd_send_status(state, HL_UPD_STS_OK,
+						 state->map_offset);
+		break;
+	default:	/* should never get here */
+		break;
+	}
+
+	/* NOTE: at this point we have code that reset the protocol FSM state,
+	 *       on error, and only registers on NOT_FOUND result.
+	 *       Do not run the secured version here. The response memory must
+	 *       be kept open until an explicit request for RST_STATE invoked
+	 *       by the host arrives.
+	 */
+	if (err) {
+		module_fwupd_reset_state(state);
