+#define GPIO_CARD_TYPE		23
+#define GPIO_CARD_TYPE_BIT	BIT(GPIO_CARD_TYPE % 16)
+
+static bool is_card_type_pcie(void)
+{
+	uint32_t val;
+	uint32_t is_pcie;
+
+	val = regmap_readl(D0_PSOC_GPIO1_REG(mmGPIO_DATA));
+
+	is_pcie = val & GPIO_CARD_TYPE_BIT;
+
+	return !!is_pcie;
+}
+
+void preboot_device_id_init(void)
+{
+	if (is_card_type_pcie())
+		regmap_rmwritel(mmD0_PIF_ARC_CONTROL, 1, PIF_CTRL_PCI_CARD);
+}
+
+u32 preboot_device_id_get(void)
+{
+	u32 arc_ctrl = regmap_readl(mmD0_PIF_ARC_CONTROL);
+
+	return (arc_ctrl & PIF_CTRL_PCI_CARD) ? PCI_IDS_GAUDI3_HL_338 : PCI_IDS_GAUDI3;
