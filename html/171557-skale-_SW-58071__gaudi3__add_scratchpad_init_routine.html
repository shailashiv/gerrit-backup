+static void arcboot_rom_init_sp_sram(void)
+{
+	struct gaudi3_cold_rst_data cold_rst_data;
+	uint32_t init_sts;
+	uint8_t retry = HL_INIT_SP_SRAM_RETRIES;
+
+	/* Perform scratchpad init only on cold reset */
+	cold_rst_data.data = regmap_readl(mmCOLD_RST_FLOPS_2);
+	if (cold_rst_data.spsram_init_done)
+		return;
+
+	regmap_writel(D0_PSOC_GLOBAL_CONF_SCRATCHPAD_INIT_CTRL_START_M,
+			mmD0_PSOC_GLOBAL_CONF_SCRATCHPAD_INIT_CTRL);
+	while (1) {
+		init_sts = regmap_readl(mmD0_PSOC_GLOBAL_CONF_SCRATCHPAD_INIT_CTRL);
+		if (init_sts & D0_PSOC_GLOBAL_CONF_SCRATCHPAD_INIT_CTRL_DONE_M) {
+			cold_rst_data.spsram_init_done = 1;
+			regmap_writel(cold_rst_data.data, mmCOLD_RST_FLOPS_2);
+			break;
+		}
+
+		/*
+		 * Do not wait indefinitely on copy status
+		 * Wait for 1 milliseconds (100 retries with 10 micro-
+		 * seconds sleep), else bailout.
+		 */
+		retry--;
+		if (!retry) {
+			arcboot_rom_status_update(
+					PROGRESS_INIT_SP_SRAM_NOT_COMPLETED);
+			break;
+		}
+
+		udelay(HL_INIT_SP_SRAM_UDELAY);
+	}
+}
