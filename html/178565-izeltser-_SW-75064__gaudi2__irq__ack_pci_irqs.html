followed by pci drain irq.
As agreed before, IRQs that are consequence of the reboot sequence

+#define GAUDI2_IRQ_SIZE	512
+static void gaudi2_ack_gic_irq(void)
+{
+	u32 i;
+
+	for (i = 0; i < GAUDI2_IRQ_SIZE; i++)
+		regmap_writel(i, mmGIC_DISTRIBUTOR__6_GICD_CLRSPI_NSR);
+}
+
+static void gaudi2_ack_pci_axi_irq(void)
+{
+	/* Ack PCI addr decoder error */
+	regmap_writel(1, mmPCIE_WRAP_AXI_INTR);
+	/* Ack PCI drain indication */
+	regmap_writel(0, mmPCIE_WRAP_AXI_DRAIN_IND);
+}
+
+void preboot_ack_irq(void)
+{
+	gaudi2_ack_pci_axi_irq();
+	gaudi2_ack_gic_irq();
