+define AGENT_FWL_DEFINE_MODULE_ENVVARS
+	$(eval MODULE_NAME=$(strip $1))

+	$(eval MODULE_NAME_UPPER_CASE=$(strip $(MODULE_NAME_UPPER_CASE)))

+$(foreach module,$(AGENT_FWL_APP_MODULES),$(call AGENT_FWL_DEFINE_MODULE_ENVVARS,$(module),$(AGENT_FWL_APP_PROD_NAME)))

+# Arguments:

+define AGENT_FWL_MODULE_VERSION_HOOK
+	$(eval MODULE_NAME_UPPER_CASE=$(shell echo $1 | tr a-z A-Z))
+	$(eval MODULE_NAME_UPPER_CASE=$(strip $(MODULE_NAME_UPPER_CASE)))

+
+define AGENT_FWL_APP_VERSION_GEN

+	@$(foreach module,$(AGENT_FWL_APP_MODULES),$(call AGENT_FWL_MODULE_VERSION_HOOK,$(module)))
+endef
+
+define AGENT_FWL_CREATE_HL_VERSION
+	@mkdir -p $(AGENT_FWL_BUILD_DIR)/include

+	@echo "#define _HL_VERSION_" >>  $(AGENT_FWL_BUILD_DIR)/include/_hl_version.h
+	@echo "#define HL_VERSION_MAJOR $$(echo $(AGENT_FWL_LOCAL_VERSION) | cut -d. -f1)" >> $(AGENT_FWL_BUILD_DIR)/include/_hl_version.h
+	@echo "#define HL_VERSION_MINOR $$(echo $(AGENT_FWL_LOCAL_VERSION) | cut -d. -f2)" >> $(AGENT_FWL_BUILD_DIR)/include/_hl_version.h
+	@echo "#define HL_VERSION_PATCH $$(echo $(AGENT_FWL_LOCAL_VERSION) | cut -d. -f3)" >> $(AGENT_FWL_BUILD_DIR)/include/_hl_version.h
+	@echo "#endif" >>  $(AGENT_FWL_BUILD_DIR)/include/_hl_version.h
+endef
