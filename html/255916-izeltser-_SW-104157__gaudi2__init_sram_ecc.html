Commit Message

[SW-104157] gaudi2: init sram ecc

SRAM self repair procedure is overwriting the ecc data of the SRAM,
so after running the self repair the ECC must be initialized.
So far it was performed by resetting the whole SRAM subsystem, but
it's also resetting some parts of the NOC which can cause troubles
if other activities are running in the background (like pci init).
So we switch to plan B, and initialize the ECC data by writing the
whole SRAM address space using the mem_cpy engine.

+#define HL_MEMCPY_TIMEOUT_US	100000
+#define HL_MEMCPY_OUTSTANDING	62
+#define HL_MEMCPY_SIZE_SHIFT	4
+
+static int gaudi2_memcpy_init_mem(u64 addr, u32 size)
+{
+	u64 start;
+	u32 mem_cfg, cpy_sts;
+
+	regmap_writel((u32)addr, mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_L);
+	regmap_writel((u32)(addr >> 32), mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_H);
+
+	mem_cfg = regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+	mem_cfg &= ~(PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_MEM_SIZE_MASK);
+	mem_cfg &= ~(PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_WR_OS_MASK);
+
+	/* set size of in granularity of 16 bytes */
+	size >>= HL_MEMCPY_SIZE_SHIFT;
+	mem_cfg |= size & PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_MEM_SIZE_MASK;
+	mem_cfg |= PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_USE_CONST_MASK;
+
+	/* set number of outstanding write transactions */
+	mem_cfg |= HL_MEMCPY_OUTSTANDING << PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_WR_OS_SHIFT;
+
+	regmap_writel(mem_cfg, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+	regmap_writel(0x0, mmPSOC_GLOBAL_CONF_MEM_CPY_CONST);
+
+	/* Enable copy and wait for completion */
+	regmap_writel(PSOC_GLOBAL_CONF_MEM_CPY_CTRL_EN_MASK, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL);
+	start = timestamp_get_us();
+	while (1) {
+		cpy_sts = regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_STATUS);
+		if (cpy_sts & PSOC_GLOBAL_CONF_MEM_CPY_STATUS_DONE_MASK)
+			break;
+
+		if (timeout_expired(start, HL_MEMCPY_TIMEOUT_US)) {
+			preboot_puts_hex64_raw("memcpy timed out, addr ", addr, false);
+			preboot_puts_hex(" size", size, false);
+			return -1;
+		}
+	}
+
+	return 0;
+}

+#define HL_MEMCPY_MAX_SIZE	SZ_512K

+static void gaudi2_mem_init(u64 base, u32 size)
+{
+	u64 mem_base = base;
+	u32 cpy_size, remain_size = size;
+
+	while (remain_size) {
+		if (remain_size > HL_MEMCPY_MAX_SIZE)
+			cpy_size = HL_MEMCPY_MAX_SIZE;
+		else
+			cpy_size = remain_size;
+
+		gaudi2_memcpy_init_mem(mem_base, cpy_size);
+
+		remain_size -= cpy_size;
+		mem_base += cpy_size;
+	}
+}
