+; --------------------------------------------------------------------------------------------------
+; @Title: Startup script for the DALI SoC
+; @Description: This script configures the base addresses for debug and attaches to the core
+; @Author: Oded Gabbay
+; @SoC: DALI
+; @Chip: Cortex-A53
+; @Copyright: (C) 2016-2017 HabanaLabs, Ltd.
+; --------------------------------------------------------------------------------------------------
+; NOTE: Interrupts are disabled with SYStem.Option IMASKASM and IMASKHLL per default
+;
+; -------------------------------------------------------------------------------------------------
+; Calling convention:
+;
+;   CD.DO onix.cmm <cpu_num> <cfg_mode>
+;
+;     <cpu_num>    The number of CPUs to be debugged. If this parameter is omitted, the CPU number
+;                  will default to 2. (2x Cortex-A53).
+;
+;     <cfg_mode>   Determines whether the sript attaches to the CPU or not. If PREPAREONLY is
+;                  passed, then only the Coresight configuration will be done.
+;                  if no parameter or ATTACH is passed, then the script will attach to the cores.
+; -------------------------------------------------------------------------------------------------
+; Examples:
+;
+;   1) Call without parameters. In this case four cores will be debugged.
+;      The script will wait for system power and automatically attach to the CPU.
+;        CD.DO onix.cmm
+;
+;   2) Call with a defined number of CPUs, example 2 CPUs. The script will wait for system power
+;      and automatically attach to the CPU.
+;        CD.DO onix.cmm 2.
+;
+;   3) Call with a defined number of CPUs, example 2 CPUs. Just configure, do not attach.
+;        CD.DO onix.cmm 2. PREPAREONLY
+; -------------------------------------------------------------------------------------------------
+
+LOCAL &cpuNumber &prepare
+ENTRY &cpuNumber &prepare
+
+;--------------------------------------------------------------------------------------------------
+; Parameters
+
+; Possible values: 1. - 2. . 2. means 2x Cortex-A53
+; 1. is default
+IF "&cpuNumber"==""
+(
+  &cpuNumber=1.
+)
+
+; Possible values: ATTACH, PREPAREONLY
+; PREPAREONLY is default
+IF "&prepare"==""
+   &prepare="PREPAREONLY"
+
+; Make sure that case of parameter is ignored
+&prepare=STRing.UPpeR("&prepare")
+
+;--------------------------------------------------------------------------------------------------
+; Configuration
+
+SYStem.RESet
+SYStem.CPU CORTEXA53
+SYStem.JtagClock 20MHz
+SYStem.option dismode aarch64
+
+SYStem.CONFIG CORENUMBER 2.
+
+IF (&cpuNumber<1.)||(&cpuNumber>2.)
+(
+  DIALOG.OK "Invalid number of CPUs: &cpuNumber - End of script"
+  ENDDO
+)
+
+; Assignment:
+IF &cpuNumber==1.
+  CORE.ASSIGN 1.
+IF &cpuNumber==2.
+  CORE.ASSIGN 1. 2.
+
+; Cortex-A53 ROM-Table: 0x81000000
+;
+;
+; Cortex-A53 | debug      | CTI        | BMC        | ETM (Trace)
+;------------------------------------------------------------------
+;   core 0   | 0x81010000 | 0x81020000 | 0x81030000 | 0x81040000
+;   core 1   | 0x81110000 | 0x81120000 | 0x81130000 | 0x81140000
+;
+; Base ROM-Table: 0x80000000
+;
+
+; Access port numbers are
+SYStem.CONFIG APBACCESSPORT 1.
+SYStem.CONFIG DEBUGACCESSPORT 1.
+SYStem.CONFIG AXIACCESSPORT 0.
+SYStem.CONFIG MEMORYACCESSPORT 0.
+
+SYStem.CONFIG COREDEBUG BASE DAP:0xfe810000 DAP:0xfe910000
+SYStem.CONFIG CTI BASE DAP:0xfe820000 DAP:0xfe920000
+SYStem.CONFIG BMC BASE DAP:0xfe830000 DAP:0xfe930000
+SYStem.CONFIG ETM BASE DAP:0xfe840000 DAP:0xfe940000
+
+SYStem.CONFIG.STM.Base DAP:0xfe486000
+SYStem.CONFIG.ETF.Base DAP:0xfe481000 DAP:0xfe482000
+SYStem.CONFIG.ETF1.ATBSource STM1
+
+; Disable interrupts in debug mode.
+; Set to off to debug interrupt routines
+SYStem.Option IMASKASM ON
+SYStem.Option IMASKHLL ON
+SYStem.Option EnReset OFF               ; Avoid that reset line is pulled low during Prepare mode
+
+; <Target register address> < ~mask> < reset value > < value to be written on out of reset >
+system.option.RESetREGister axi:0x7FFC441124 0x210000 0x011000 0x011000
+
+LOCAL &edscr &connected &edlar &wasrunning
+
+&connected=TRUE()
+&wasrunning=TRUE()
+
+IF !STATE.POWER()
+(
+  DIALOG.OK "Board is not powered."
+
+  PRINT "Waiting for SYStem.Power"
+  ON TIME 10.0s GOTO powerdown                  ; Wait max 10 sec for VTREF to go to 1.8V
+
+WHILE !STATE.POWER()
+(
+  WAIT 0.5s
+)
+
+  ON TIME                                       ; Disable time out again
+)
+
+PRINT "System power detected, waiting for debug registers"
+
+ON ERROR GOSUB ConnectHandler                   ; On bus error set connection attempt to false
+SYStem.Mode Prepare                             ; Go to prepare mode for debug register polling
+CORE 0.                                         ; Select core 0 to wait for debug registers
+
+; <Target register address> < ~mask> < reset value > < value to be written on out of reset >
+system.option.RESetREGister axi:0x7FFC441124 0x210000 0x011000 0x011111
+
+; Poll EDLAR register until no further bus error occurs.
+waitconnect:
+&connected=TRUE()                               ; Assume that next read will succeed
+&edlar=Data.Long(eapb:(corebase()+0xFB0))       ; Use EDLAR to check connection. EDLAR should
+IF !&connected                                  ; also be available even when a lock is active
+(
+  WAIT 0.5s
+  GOTO waitconnect
+)
+
+; Make sure reset vector for all cores is 0x7ff0010000
+D.S axi:0x7FFC441104 %LE %Long 0xf0010000
+D.S axi:0x7FFC441108 %LE %Long 0xf0010000
+D.S axi:0x7FFC441114 %LE %Long 0x7f
+D.S axi:0x7FFC441118 %LE %Long 0x7f
+
+;--------------------------------------------------------------------------------------------------
+; Attach
+
+IF "&prepare"=="ATTACH"
+(
+  ; Finally attach to the target
+  SYStem.Mode Attach
+
+  IF RUN()
+    Break.direct
+  ELSE
+    &wasrunning=FALSE()
+
+  ; Toggle ETM to make trace work
+  ETM.OFF
+;  ETM.ON
+
+  IF &wasrunning
+    Go.direct
+
+  PRINT "Target is up and running"
+)
+
+ENDDO
+
+; -------------------------------------------------------------------------------------------------
+; Support handlers
+ConnectHandler:
+
+  &connected=FALSE()
+
+RETURN
+
+; -------------------------------------------------------------------------------------------------
+; Error handlers
+
+powerdown:
+  DIALOG.OK "Board power did not come up. Please check the power supply and if the debug connector is correctly plugged"
+ENDDO
