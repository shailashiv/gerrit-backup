+#define SYS_CLOCK_FREQUENCY	250000000ULL /* PCI PLL DIV3 Emulation clk */
+#define PLDM_UART_REF_CLK	250000000ULL
+#define PLDM_UART_BAUDRATE	300
+
+uint32_t boot_rom_pll_get_freq(enum boot_rom_pll_div div)
+{
+	uint32_t offset = div * sizeof(uint32_t);
+
+	/* 2'd0 - ref_clk, 2'd1 - pll_clk, 2'd2 - divided ref_clk, 2'd3 - divided pll_clk */
+	/* H15_PROJECT ➔ IO_0_CFG ➔ HALF_0 ➔ SOC_0 ➔ PLL_0 ➔ PSOC_PLL ➔ CTRL ➔ DIV_SEL */
+	/* Divider clock select */
+	regmap_writel(PLL_DIV_PLL, (IO0_H0_SOC_0_PLL_0_PSOC_PLL(mmPLL_CONTROL_DIV_SEL_0)) + offset);
+
+	/* Divison factor value */
+	/* H15_PROJECT ➔ IO_0_CFG ➔ HALF_0 ➔ SOC_0 ➔ PLL_0 ➔ PSOC_PLL ➔ CTRL ➔ DIV_FACTOR */
+	regmap_writel(PLL_DIV_PLL, (IO0_H0_SOC_0_PLL_0_PSOC_PLL(mmPLL_CONTROL_DIV_FACTOR_0)) + offset);
+
+	/* H15_PROJECT ➔ IO_0_CFG ➔ HALF_0 ➔ SOC_0 ➔ PLL_0 ➔ PSOC_PLL ➔ CTRL ➔ DIV_EN  */
+	regmap_writel(0x1, (IO0_H0_SOC_0_PLL_0_PSOC_PLL(mmPLL_CONTROL_DIV_EN_0)) + offset); /* Enable dividers */
+
+	/* H15_PROJECT ➔ IO_0_CFG ➔ HALF_0 ➔ SOC_0 ➔ PLL_0 ➔ PSOC_PLL ➔ CTRL ➔ CFG */
+	regmap_writel(0x101, (IO0_H0_SOC_0_PLL_0_PSOC_PLL(mmPLL_CONTROL_CFG))); /* Enables PSOC_PLL and FOUTPOSTDIVEN */
+
+	return PLDM_UART_REF_CLK;
+}
+
+uint32_t boot_rom_get_baudrate()
+{
+	return PLDM_UART_BAUDRATE;
+}
+
+bool fuse_valid_volt_prot_pattern(uint32_t config_fuse)
+{
+	if ((config_fuse & EFUSE_SYSCFG_VOLT_PROT_BITS) == FUSE_VOLT_PROT_PATTERN)
+		return true;
+
+	return false;
+}
+
+uint32_t boot_rom_tpm_spi_baudr_get(void)
+{
+	enum boot_rom_tpm_spi_freq_config freq_config;
+	uint32_t baudr;
+
+	freq_config = boot_rom_get_tpm_spi_freq_config();
+
+	if (freq_config == FREQ_CFG_12_5MHZ)
+		baudr = (SYS_CLOCK_FREQUENCY / SPI_TPM_FREQ_12_5M);
+	else
+		baudr = (SYS_CLOCK_FREQUENCY / SPI_TPM_FREQ_6_25M);
+
+	/* baud rate doesn't support odd number and must not be 0 */
+	baudr = (baudr < 2) ? 2 : baudr;
+	baudr = (baudr + 1) & 0xFFFE;
+
+	return baudr;
+}
