+{
+	return 1;
+}
+
+u64 hl_pll_get_cpu_cycles_per_usec(void)
+{
+	return SYS_CLOCK_FREQUENCY / USEC_PER_SEC;
+}
+
+void hl_uart_init(void)
+{
+	preboot_uart_init(UART_CLOCK_FREQUENCY / PLDM_UART_BAUDRATE);
+}
+
+u32 hl_spi_baudr_get(void)
+{
+	u32 factor = 2; /* Min Divider */
+
+	/* baud rate doesn't support odd number */
+	return (factor + 1) & 0xFFFE;
