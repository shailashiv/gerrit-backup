+ * arcboot - Signature Verification
+ *
+ * Both the protected data and the signature of the protected data can be
+ * changed, but the public key should be locked into the device.
+ *
+ * @arguments:
+ * image     - pointer to blob of protected data
+ * image_len - size of the blob (Image + public key (inc key header))
+ * sig       - pointer to signature of the blob
+ * sig_len   - signature size
+ * key       - pointer to public key
+ * key_len   - public key size + key header (784 + 8 = 792 bytes)
+ *
+ * @return:
+ * SECUREBOOT_VERIFY_SUCCESS    : Signature validation successful
+ * SECUREBOOT_VERIFY_KEY_REVOKED: Selected key index is revoked

+ * SECUREBOOT_VERIFY_FAILED     : Signature validation failed
+ */
+int secureboot_verify_image_v2(const uint8_t *image, uint32_t image_len, uint8_t *key,
+				uint32_t key_len, uint8_t *sig, uint32_t sig_len,
+				uint8_t sec_ver)
+{
+	int ret, key_index;
+
+	if (!security_efuse_image_validation_en())
+		return SECUREBOOT_VERIFY_SUCCESS;

+	if ((key_len > TOTAL_PUB_KEY_PROPS_SZ) || (sig_len > SIG_SIZE))
+		return SECUREBOOT_VERIFY_FAILED;
+
+	if (security_check_valid_fw_version(sec_ver)
+			!= SECURITY_FW_VERSION_SECURE) {
+		return SECUREBOOT_VERIFY_VER_FAILED;
+	}
+
+	key_index = *(uint32_t*) (key + key_len - KEY_PROP_HDR_SZ);
+
+	/* TODO: Confirm if key revocation failure needs a dedicated status bit */
+	ret = security_check_key_revoke_status(key_index);
+	if (ret != 1)
+		return SECUREBOOT_VERIFY_KEY_REVOKED;
+
+	if (secureboot_verify_key_hash(key_index, key, key_len))
+		return SECUREBOOT_VERIFY_KEY_FAILED;
+

+
+	/* 0 - SUCCESS, but arcboot main code, expects 1 for SUCCESS */

+int security_get_key_hash(uint32_t *hash, uint8_t key_index)
+{
+	uint32_t valid_key_base, max_key_index;
+	int i;
+	enum security_key_location key_location;
+	uint32_t *ptr;
+
+	key_location = ((arcboot_rom_global.config_fuse >>
+			EFUSE_SYSCFG_KEY_LOCATION) &
+			EFUSE_SYSCFG_KEY_LOCATION_MASK);
+
+	/* check efuse for key location */
+	if (key_location == KEY_LOCATION_EFUSE) {
+		max_key_index = NUM_KEYS_IN_EFUSE;
+
+		if (key_index >= max_key_index)
+			return 1;
+
+		/* read from efuse/SP SRAM */
+		valid_key_base = (key_index * SECURITY_KEY_HASH_LEN) + SECURITY_EFUSE_KEYS_OFFSET;
+
+		for (i = 0; i < SECURITY_DWORD_PER_KEY; i++)
+			*(hash + i) = sp_sram_readl(valid_key_base + (i *
+					SECURITY_BYTE_PER_DWORD));
+	} else {
+		/*
+		 * All other cases are routed into ROM, since it is
+		 * the more secure way
+		 */
+		max_key_index = NUM_KEYS_IN_ROM;
+
+		if (key_index >= max_key_index)
+			return 1;
+
+		/* read from ROM itself */
+		ptr = (uint32_t *)rom_key_hashes[key_index];
+		memcpy(hash, ptr, SECURITY_KEY_HASH_LEN);
+	}
+
+	return 0;
+}
