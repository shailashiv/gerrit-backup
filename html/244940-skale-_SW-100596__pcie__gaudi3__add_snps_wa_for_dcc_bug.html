+#define PHY_CR_PARA_CMD_DATA		0x2
+#define PHY_CR_PARA_CMD_ADDR		0x5149
+
+void pcie_wa_snps_dcc(void)
+{
+	uint32_t val;
+
+	regmap_writel(D0_PCIE_PHY_CR_PARA_CFG_SEL_M, mmD0_PCIE_PHY_CR_PARA_CFG);
+
+	regmap_writel(D0_PCIE_PHY_CR_PARA_CTRL_WR_EN_M, mmD0_PCIE_PHY_CR_PARA_CTRL);
+	val = ((PHY_CR_PARA_CMD_DATA << D0_PCIE_PHY_CR_PARA_CMD_DATA_S) | PHY_CR_PARA_CMD_ADDR);
+	regmap_writel(val, mmD0_PCIE_PHY_CR_PARA_CMD);
+
+	/* Wait until status is cleared */
+	while (regmap_readl(mmD0_PCIE_PHY_CR_PARA_STS))
+		;
+}
