+#include <boot_rom.h>
+#include <status.h>
+
+void boot_rom_status_update(enum boot_rom_progress progress)
+{
+	uint32_t status_base, reg, offset;
+
+	status_base = boot_rom_get_status_base();
+	offset = progress / BOOT_ROM_STATUS_REG_SZ_BITS;
+	progress = progress % BOOT_ROM_STATUS_REG_SZ_BITS;
+
+	reg = regmap_readl(status_base + offset * 4);
+	reg |= BIT(progress);
+	regmap_writel(reg, (status_base + offset * 4));
+}
+
+void boot_rom_status_update_bringup(enum boot_rom_progress progress, uint32_t position)
+{
+	boot_rom_status_update(progress + position);
+}
+
+void boot_rom_status_update_val(enum boot_rom_progress progress, uint32_t val, uint8_t bits)
+{
+	uint32_t status_base, reg, offset;
+
+	status_base = boot_rom_get_status_base();
+	offset = progress / BOOT_ROM_STATUS_REG_SZ_BITS;
+	progress = progress % BOOT_ROM_STATUS_REG_SZ_BITS;
+
+	reg = regmap_readl(status_base + offset * 4);
+	reg |= val << progress;
+	regmap_writel(reg, status_base + offset * 4);
+
+	if ((progress + bits) > BOOT_ROM_STATUS_REG_SZ_BITS) {
+		offset++;
+		val = val >> (BOOT_ROM_STATUS_REG_SZ_BITS - progress);
+		reg = regmap_readl(status_base + offset * 4);
+		reg |= val;
+		regmap_writel(reg, status_base + offset * 4);
+	}
+}
