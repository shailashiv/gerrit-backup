+// SPDX-License-Identifier: Apache-2.0+
+/*
+ * Copyright 2024 HabanaLabs, Ltd.
+ * All Rights Reserved.
+ */
+
+/*
+ * This is adoptation of zephyr watchdog driver
+ *  wdt_cmsdk_apb.c for baremetal environment
+ */
+
+#include <stdint.h>
+#include <gaudi3_blocks.h>
+
+struct wdog {
+	/* offset: 0x000 (r/w) watchdog load register */
+	volatile uint32_t  load;
+	/* offset: 0x004 (r/ ) watchdog value register */
+	volatile uint32_t  value;
+	/* offset: 0x008 (r/w) watchdog control register */
+	volatile uint32_t  ctrl;
+	/* offset: 0x00c ( /w) watchdog clear interrupt register */
+	volatile uint32_t  intclr;
+	/* offset: 0x010 (r/ ) watchdog raw interrupt status register */
+	volatile uint32_t  rawintstat;
+	/* offset: 0x014 (r/ ) watchdog interrupt status register */
+	volatile uint32_t  maskintstat;
+	volatile uint32_t  reserved0[762];
+	/* offset: 0xc00 (r/w) watchdog lock register */
+	volatile uint32_t  lock;
+	volatile uint32_t  reserved1[191];
+	/* offset: 0xf00 (r/w) watchdog integration test control register */
+	volatile uint32_t  itcr;
+	/* offset: 0xf04 ( /w) watchdog integration test output set register */
+	volatile uint32_t  itop;
+};
+
+#define WDOG_LOAD		(0xFFFFFFFF << 0)
+#define WDOG_RELOAD		(0xE4E1C00 << 0)
+#define WDOG_VALUE		(0xFFFFFFFF << 0)
+#define WDOG_CTRL_RESEN		(0x1 << 1)
+#define WDOG_CTRL_INTEN		(0x1 << 0)
+#define WDOG_INTCLR		(0x1 << 0)
+#define WDOG_RAWINTSTAT		(0x1 << 0)
+#define WDOG_MASKINTSTAT	(0x1 << 0)
+#define WDOG_LOCK		(0x1 << 0)
+#define WDOG_INTEGTESTEN	(0x1 << 0)
+#define WDOG_INTEGTESTOUTSET	(0x1 << 1)
+
+#define REF_CLK_FREQUENCY	50000000
+
+/*
+ * Value written to the LOCK register to lock or unlock the write access
+ * to all of the registers of the watchdog (except the LOCK register)
+ */
+#define WDOG_UNLOCK_VALUE	(0x1ACCE551)
+#define WDOG_LOCK_VALUE		(0x2BDDF662)
+
+#define WDOG_STRUCT ((volatile struct wdog *)(mmD0_PSOC_WDOG_BASE))
+
+/* watchdog reload value in sec */
+static unsigned int reload_s = WDOG_RELOAD;
+
+static void wdog_unlock(void)
+{
+	volatile struct wdog *wdog = WDOG_STRUCT;
+
+	wdog->lock = WDOG_UNLOCK_VALUE;
+}
+
+int wdog_setup(void)
+{
+	volatile struct wdog *wdog = WDOG_STRUCT;
+
+	/* Start the watchdog counter with INTEN bit */
+	wdog->ctrl = (WDOG_CTRL_RESEN | WDOG_CTRL_INTEN);
+	return 0;
+}
+
+int wdog_disable(void)
+{
+	volatile struct wdog *wdog = WDOG_STRUCT;
+
+	/* Stop the watchdog counter with INTEN bit */
+	wdog->ctrl = ~(WDOG_CTRL_RESEN | WDOG_CTRL_INTEN);
+	return 0;
+}
+
+int wdog_install_timeout(uint32_t max_ms)
+{
+	volatile struct wdog *wdog = WDOG_STRUCT;
+
+	/* Reload value:
+	 * The watchdog timer is driven by the clock, which can have
+	 * higher values e.g running at 35000000/50000000 Hz.
+	 * Also timeout given in max is in millisecs this can lead to
+	 * overflow on multiplication as max reload value can be
+	 * 4294967295 (0xffffffff), this can lead to wrong configuration of
+	 * reload value
+	 * Also below formula is wrong as reload value needs to be converted
+	 * into ticks, as at each tick loaded value decrement.
+	 * Since clk freq is ticks per second, and user provided input
+	 * is in millisecs, divide by 1000 to be in same units
+	 */
+	reload_s = max_ms * (REF_CLK_FREQUENCY / 1000);
+	wdog->load = reload_s;
+	return 0;
+}
+
+int wdog_feed(void)
+{
+	volatile struct wdog *wdog = WDOG_STRUCT;
+
+	wdog->intclr = WDOG_INTCLR;
+	wdog->load = reload_s;
+
+	return 0;
+}
+
+int wdog_init(void)
+{
+	volatile struct wdog *wdog = WDOG_STRUCT;
+
+	wdog_unlock();
+	wdog->load = reload_s;
+	return 0;
+}
