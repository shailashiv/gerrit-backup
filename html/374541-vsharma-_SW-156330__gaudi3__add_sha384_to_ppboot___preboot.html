+function append_sha384 {
+	IMG=${1}
+	SHA384_BINFILE=img-hash-${IMG%.*}.sha384
+
+	rm -f ${SHA384_BINFILE}.sha384
+
+	$OPENSSL dgst -sha384 -binary ${IMG} > ${SHA384_BINFILE}
+
+	# Pad magic to be 16 byte - aligned
+	dd status=none conv=sync bs=4 if=/dev/zero count=3 >> ${IMG}
+
+	echo ${SHA384_MAGIC} | xxd -r -p - >> ${IMG}
+	dd status=none conv=sync bs=1 if=${SHA384_BINFILE} >> ${IMG}
