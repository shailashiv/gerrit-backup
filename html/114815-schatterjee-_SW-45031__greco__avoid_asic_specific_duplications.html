+u32 hl_skip_pcie_init(void)
+{
+	/* Use DumpSel[2] to skip PCIe initialization */
+	return regmap_readl(mmPSOC_GLOBAL_CONF_BOOT_STRAP_PINS_L) & BIT(10);

+u64 hl_pll_timestamp_freq_get(void)
+{
+	/* As per Goya2 PLL cfg.xslx, PCI PLL Div[3] drives CPU timestamp */
+	return hl_pll_get_freq(mmPSOC_PCI_PLL_CTRL_BASE, HL_PLL_DIV3);
+}
+
+void hl_uart_init(void)
+{
+	/* As per Goya2 PLL cfg.xlsx, PCI PLL Div[3] drives uart */
+	u32 uart_clk_freq = hl_pll_get_freq(mmPSOC_PCI_PLL_CTRL_BASE,
+					    HL_PLL_DIV3);
+
+	regmap_writel(1, mmPSOC_GLOBAL_CONF_PAD_SEL_8);	/* UART0 RXD */
+	regmap_writel(1, mmPSOC_GLOBAL_CONF_PAD_SEL_9);	/* UART0 TXD */
+	preboot_uart_init(uart_clk_freq / HL_UART_BAUDRATE);
+}
+
+u32 hl_spi_baudr_get(void)
+{
+	u32 factor;
+	/* As per Goya2 PLL cfg.xlsx, PCI PLL Div[3] drives SPI */
+	u32 spi_clk_freq = hl_pll_get_freq(mmPSOC_PCI_PLL_CTRL_BASE,
+					   HL_PLL_DIV3);
+
+	factor = spi_clk_freq / HL_SPI_FLASH_MAX_FREQ;
+
+	return factor > 2 ? factor : 2;
+}
+
+void hl_pci_iatu_init(void)
+{
+	struct hl_pci_inbound_region inbound_region;
+	struct hl_pci_outbound_region outbound_region;
+
+	hl_pcie_disable_access();
+
+	/* Inbound Region 0 - Bar - - Point to SRAM and CFG */
+	inbound_region.mode = PCI_BAR_MATCH_MODE;
+	inbound_region.bar = PCI_BAR_ID_0;
+	inbound_region.addr = SDRAM_FULL_BASE;
+	hl_pci_set_inbound_region(0, &inbound_region);
+
+	/* Inbound Region 1 - Bar 4 - Point to DRAM */
+	inbound_region.mode = PCI_BAR_MATCH_MODE;
+	inbound_region.bar = PCI_BAR_ID_4;
+	inbound_region.addr = DRAM_BASE;
+	hl_pci_set_inbound_region(1, &inbound_region);
+
+	/* Outbound Region 0 - Point to Host */
+	outbound_region.addr = HOST_PHYS_BASE;
+	outbound_region.size = HOST_PHYS_SIZE;
+	hl_pci_set_outbound_region(&outbound_region);
+
+	hl_pcie_enable_access();
+}
