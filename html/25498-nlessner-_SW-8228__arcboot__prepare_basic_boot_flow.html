+;; *******************************************************************************************************************
+;; -----------------
+;; ARC start up file
+;; -----------------
+;; The startup code for the ARC family of processors does the following
+;; before transferring control to user defined main label :
+;;	1. Set sp to __stack_top (link time variable)
+;;	2. Set fp to zero
+;;	3. Zero out the bss section (for uninitialized globals)
+;; After returning from main, the processor is halted and the pipeline is flushed out
+;;
+;; *******************************************************************************************************************
+.file "crt0_ram.S"
+
+.extern main
+
+.section .text.__startup, "ax", @progbits
+
+.global __start
+.type __start, @function
+.align 4
+
+; Startup code for the ARC600, ARC700 and ARCv2 processors
+; NOTE:  The following restrictions apply on zero overhead loops (other
+; restrictions are not pertinent to this code)
+; - loop end should be 4 instruction words away from the lp_count setting
+;   instruction
+; - loop body should have at least two instruction words
+__start:
+	mov gp, @__SDATA_BEGIN__
+
+	mov_s r2, @__sbss_start	; r2 = start of the bss section
+	sub r3, @_end, r2	; r3 = size of the bss section in bytes
+	; set up the loop counter register to the size (in words) of the bss section
+	asr_s r13, r3
+	asr.f lp_count, r13
+
+	mov sp, @__stack_top	; initialize stack pointer
+	mov_s r3, 0
+	; loop to zero out the bss.  Enter loop only if lp_count != 0
+	lpnz @.Lend_zbss
+	st.ab r3, [r2, 4]	; zero out the word
+	nop
+.Lend_zbss:
+
+	mov_s r1, @_etext	; r1 = end of the text section
+	mov_s r2, @_data	; r1 = end of the data section
+
+	sub r3, @_edata, r2	; r3 = size of the data section in bytes
+
+	; set up the loop counter register to the size (in words) of the data section
+	asr_s r13, r3
+	asr.f lp_count, r13
+
+	; loop to copy data from ROM to DCCM.  Enter loop only if lp_count != 0
+	lpnz @.Lend_cdata
+	ld.ab	r5,[r1,4]
+	st.ab	r5,[r2,4]
+.Lend_cdata:
+
+	;; branch to main
+	bl.d @arcboot_ram_main
+	mov fp, 0		; initialize frame pointer
+				; r0 contains exit code
+	j @_exit_halt
+
+.global _exit_halt
+.type _exit_halt, @function
+.align 4
+_exit_halt:
+				; r0 contains exit code
+	flag 0x01
+	nop
+	nop
+	nop
+	j @_exit_halt
+	nop

+.macro CPU_EARLY_SETUP
+	; enable I-cache/D-cache
+	lr	r5, [ic_ctrl]
+	bclr	r5, r5, 0	; 0 - Enable, 1 is Disable
+	sr	r5, [ic_ctrl]
+
+	lr	r5, [dc_ctrl]
+	bclr	r5, r5, 6	; Invalidate (discard w/o wback)
+	bclr	r5, r5, 0	; Enable (+Inv)
+	sr	r5, [dc_ctrl]
+
+	; Disable alignment checking
+	lr	r5, [status32]
+	bset	r5, r5, 19
+	kflag	r5
+.endm

+.section .ivt, "a", @progbits

+; Helper macro to define weak symbols to include into interrupt vector table.
+; User code may define those functions in them, so user function will be
+; referenced in the IVT. By default all handlers point to _exit_halt - so they
+; always cause application halt, because if application causes an exception or
+; interrupt, but doesn't set a handler for it - something is wrong in
+; application. Exception is "start" entry of IVT, which points to __start
+; function.

+    .weak name `\

+/*
+ * arcboot - Signature Verification
+ *
+ * A simple application which call phase0b to validate the signature
+ * of a blob of data.
+ *
+ * Both the protected data and the signature of the protected data can be
+ * changed, but the public key should be locked into the device.
+ *
+ * test_phase0b_data     - blob of protected data
+ * test_phase0b_data_len - size of the blob
+ * test_phase0b_sig      - signature of the blob
+ * test_phase0b_sig_len  - signature size
+ * test_public_key       - public key
+ * test_public_key_len   - public key size
+ */
+
+int arcboot_verify_image(uint32_t image_size)
+{
+	int st, err;
+	unsigned char block_buf[512];
+	uint16_t test_phase0b_data_len = image_size - SIG_SIZE - BLOB_START;
+
+	unsigned char *test_phase0b_data = &fw_sram[BLOB_START];
+	unsigned char *test_phase0b_sig = &fw_sram[test_phase0b_data_len + BLOB_START];
+	unsigned char *test_public_key = test_phase0b_sig - KEY_SIZE;
+	uint32_t test_phase0b_sig_len = SIG_SIZE;
+	uint32_t test_public_key_len = KEY_SIZE;
+
+	XSTACK_INIT();
+
+	/* Phase0b uses the public key (test_public_key) to validate signature (test_phase0b_sig)
+	 * of the data blob (test_phase0b_data) */
+	err = elp_secureboot_phase0b(sb_proxy_read, test_phase0b_data, test_phase0b_data_len, test_public_key, test_public_key_len,
+			test_phase0b_sig, test_phase0b_sig_len, &st, block_buf, sizeof block_buf);
+
+	if (err || !st) {
+		regmap_writel(0x800, mmPSOC_SECURITY_STATUS);
+		regmap_writel(err, mmPSOC_SECURITY_SW_STATUS_0);
+		regmap_writel(st, mmPSOC_SECURITY_SW_STATUS_1);
+	}
+
+	regmap_writel(1, mmPSOC_SECURITY_STATUS);
+	return 0;
+}
