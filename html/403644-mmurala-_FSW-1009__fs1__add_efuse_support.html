[FSW-1009] fs1: add efuse support

This patch added efuse support, bootrom will
trigger fsm to start efuse data of size 256dwords
(bank0 and bank1) copy from efuse device to psoc
config space efuse data registers. if any failure
in efuse read, system will goto halt.

+// SPDX-License-Identifier:     GPL-2.0+
+/*
+ * Copyright (C) 2024 HabanaLabs Ltd.
+ * All Rights Reserved.
+ */
+
+#include "boot_rom.h"
+#include "efuse.h"
+#include "delay.h"
+
+uint32_t boot_rom_fuse_read(uint32_t word, uint32_t *val)
+{
+	uint32_t offset;
+
+	if (word > EFUSE_WORDS_PER_BANK)
+		return BOOT_ROM_CFG_ERROR;
+
+	if (boot_rom_global.efuse_read == 0)
+		return BOOT_ROM_CFG_ERROR;
+
+	if (boot_rom_global.fake_fuse_en)
+		return spi_flash_fuse_read(word, val);
+
+	/* read value from bank0 */
+	offset = (word * sizeof(uint32_t));
+	*val = fuse_read_data(offset);
+
+	/* read value from bank1 and OR with value from bank0 */
+	offset = ((EFUSE_WORDS_PER_BANK + word) * sizeof(uint32_t));
+	*val |= fuse_read_data(offset);
+
+	return BOOT_ROM_SUCCESS;
+}
+
+uint32_t boot_rom_fuse_init(void)
+{
+	uint32_t i;
+
+	fuse_prog_read_address();
+	fuse_prog_range_config();
+
+	if (boot_rom_global.efuse_read == 1)
+		return BOOT_ROM_SUCCESS;
+
+	fuse_set_read_ctrl_reg_val(FUSE_READ_FSM_TRIGGER);
+
+	for (i = 0; i < FUSE_RETRIES; i++)
+		if (fuse_get_read_ctrl_reg_val())
+			udelay(FUSE_DELAY_US);
+		else
+			break;
+
+	if (i == FUSE_RETRIES)
+		return BOOT_ROM_TIMEOUT;
+
+	boot_rom_global.efuse_read = 1;
+
+	return BOOT_ROM_SUCCESS;
+}
