+

+SERDES_OFFSET="SPI_SERDES_CONFIG_OFFSET"
+FLASH_OFFSET=0x"$(grep -E "^\#define[[:space:]]+$SERDES_OFFSET[[:space:]]+" "$SPECS_FILE" | sed "s/^\#define[[:space:]]\+$SERDES_OFFSET[[:space:]]\+//")"
+
+VERSION=`cat ${BR2_EXTERNAL}/output/${DEV_PREFIX}_arc/build/agent-fwl-custom/build/buildversion-habanalabs`
+

+SERDES_ITS_TEMPLATE=${BR2_EXTERNAL}/board/habanalabs/gaudi3/serdes-config.its
+SERDES_ITS_FILE=${STAGE_DIR}/${DEV_PREFIX}-serdes-config.its
+SERDES_ITB_FILE=${ITBS_DIR}/${DEV_PREFIX}-serdes-config.itb
+
+ITS_SED_PARAMS+="s/@HL_VERSION@/${VERSION}/g;"
+ITS_SED_PARAMS+="s/@DEV_PREFIX@/${DEV_PREFIX}/g;"
+
+ITS_SED_PARAMS+="s#@SERDES_CONFIG_IMAGE_BIN@#${IMAGE_BIN}#g;"
+ITS_SED_PARAMS+="s/@SERDES_CONFIG_FLASH_OFFSET@/${FLASH_OFFSET}/g;"
+
+sed ${ITS_SED_PARAMS} ${SERDES_ITS_TEMPLATE} > ${SERDES_ITS_FILE}
+$MKIMAGE_FILE -D "-W no-unit_address_vs_reg" -f ${SERDES_ITS_FILE} ${SERDES_ITB_FILE}
