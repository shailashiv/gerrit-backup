[SW-34719] gaudi: ras: ERR_STS for FIT failure

Support added for direct command ERR_STS (242) via
i2c slave. In this current patch ERR_STS reg values
are updated in case of FIT and security failures.
On receiving this ERR_STS status BMC can further read
indirect "Error Status" opcode for more details.

+#define ERR_STS_REG_MASK_LOW	0xFF
+#define ERR_STS_REG_MASK_HI	0xFF00
+u8 i2cslave_error_status_cmd(u8 cmd)
+{
+	switch (cmd) {
+	case HL_ERROR_STATUS_ALERT_CAUSE_1:
+		/* clear the low byte as reg is RC(read and clear) */
+		err_sts_reg_val &= ERR_STS_REG_MASK_HI;
+		/* return the stored low byte */
+		return (err_sts_reg_val & ERR_STS_REG_MASK_LOW);
+	case HL_ERROR_STATUS_ALERT_CAUSE_2:
+		/* clear the high byte as reg is RC(read and clear) */
+		 err_sts_reg_val &= ERR_STS_REG_MASK_LOW;
+		/* return the stored high byte */
+		return ((err_sts_reg_val & ERR_STS_REG_MASK_HI) >> 8);
+	default:
+		break;
+	}
+
+	return 0;
+}
