Commit Message

[SW-94962] erom: tpm: allocate sha384 pcr bank

by default SHA384 PCR bank is not allocated in the TPM version we are
using.
In order to make sure this bank is usable before extending it, the
PCR capabilities are being read on startup, and the PCR bank is
allocated if needed.

+#define TPM2_GET_CAPAB_CMD_SIZE	22

+int tpm2_get_capability(u32 capability, u32 property, void *data, size_t *data_len)
+{
+	const u8 command_v2[TPM2_GET_CAPAB_CMD_SIZE] = {
+		tpm_u16(TPM2_ST_NO_SESSIONS),		/* TAG */
+		tpm_u32(TPM2_GET_CAPAB_CMD_SIZE),	/* Length */
+		tpm_u32(TPM2_CC_GET_CAPABILITY),	/* Command code */
+
+		tpm_u32(capability),			/* Capability */
+		tpm_u32(property),			/* Property */
+		tpm_u32(1),				/* Property count */
+	};
+	size_t response_len = COMMAND_BUFFER_SIZE;
+	int ret;
+	/*
+	 * In the response buffer, the properties are located after the:
+	 * tag (u16), response size (u32), response code (u32),
+	 * YES/NO flag (u8), TPM_CAP (u32).
+	 */
+	const size_t properties_off = sizeof(uint16_t) + sizeof(uint32_t) + sizeof(uint32_t) +
+				      sizeof(uint8_t) + sizeof(uint32_t);
+
+	ret = tpm_spi_sendrecv_command(command_v2, response_buffer, &response_len);
+	if (ret)
+		return TPM_LIB_ERROR;
+
+	response_len -= properties_off;
+	memcpy(data, &response_buffer[properties_off], response_len);
+
+	if (data_len)
+		*data_len = response_len;
+
+	return ret;
+}

+int tpm2_pcr384_allocate(void)
+{
+	uint8_t command_v2[COMMAND_BUFFER_SIZE] = {
+		tpm_u16(TPM2_ST_SESSIONS),	/* TAG */
+		tpm_u32(10 + 4 + 13 + 16),	/* Length */
+		tpm_u32(TPM2_CC_PCR_ALLOCATE),	/* Command code */
+
+		/* handles 4 bytes */
+		tpm_u32(TPM2_RH_PLATFORM),	/* Primary platform seed */
+
+		/* session header 13 bytes */
+		tpm_u32(9),			/* Header size */
+		tpm_u32(TPM2_RS_PW),		/* Password authorisation */
+		tpm_u16(0),			/* nonce_size */
+		0,				/* session_attrs */
+		tpm_u16(0),			/* auth_size */
+
+		/* TPML_PCR_SELECTION  16 bytes*/
+		tpm_u32(2),			/* num of PCR banks */
+		/* deallocate SHA1 bank*/
+		tpm_u16(TPM2_ALG_SHA1),		/* alg */
+		3,				/* Size of PCR mask */
+		tpm_u16(0x0),			/* 24 PCRs mask */
+		0x0,
+		/* allocate SHA384 bank */
+		tpm_u16(TPM2_ALG_SHA384),	/* alg */
+		3,				/* Size of PCR mask */
+		tpm_u16(0xFFFF),		/* 24 PCRs mask */
+		0xFF,
+	};
+
+	return tpm_spi_sendrecv_command(command_v2, NULL, NULL);
+}
