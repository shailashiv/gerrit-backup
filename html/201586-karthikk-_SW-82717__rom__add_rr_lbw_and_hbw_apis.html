[SW-82717] rom: add rr lbw and hbw apis

In order to configure range registers in recovery, apis to support
multiple rr index is required. This patch implements the lbw
and hbw apis for the same.

pcie, jtag and twi (i2c + smi) range registers are to be configured
in recovery flow.

+static uint32_t security_rr_calc_rr_dist(struct rom_rr_ctx *rr_ctx)
+{
+	uint32_t rr_dist;
+
+	if (rr_ctx->rr_idx > RR_LBW_MAX_SHORT_IDX)
+		rr_dist = (rr_ctx->rr_idx - RR_LBW_MAX_SHORT_RR) * sizeof(uint32_t);
+	else
+		rr_dist = rr_ctx->rr_idx * sizeof(uint32_t);
+
+	return rr_dist;
+}

+static void security_rr_update_hit_reg(uint32_t reg, uint8_t hit_idx)
+{
+	uint32_t val = regmap_readl(reg);
+
+	val |= BIT(hit_idx);
+	regmap_writel(val, reg);
+
+	val = regmap_readl(reg + RR_HIT_WR_TO_RD_DIST);
+	val |= BIT(hit_idx);
+	regmap_writel(val, reg + RR_HIT_WR_TO_RD_DIST);
+}
