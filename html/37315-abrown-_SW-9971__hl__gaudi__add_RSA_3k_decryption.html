+struct rsa_public_key {
+	u32 len;	/* len of modulus[] in number of u32 */
+	u32 n0inv;	/* -1 / modulus[0] mod 2^32 */
+	u32 *modulus;	/* modulus as little endian array */
+	u32 *rr;	/* R^2 as little endian array */
+	u64 exponent;	/* public exponent */
+};

+#define CHECKSUM_LENGTH		(SHA256_SUM_LEN)
+#define CHECKSUM_DER_LENGTH	(SHA256_DER_LEN)
+#define CHECKSUM_DER_PREFIX	(sha256_der_prefix)

+#define RSA4096_BITS		(4096)
+#define RSA4096_BYTES		(RSA4096_BITS / 8)
+#define RSA3072_BITS		(3072)
+#define RSA3072_BYTES		(RSA3072_BITS / 8)
+#define CRYPTO_RSA_LENGTH	(RSA3072_BYTES)
+#define RSA_DEFAULT_PUBEXP	(65537)

+#if (CHECKSUM_LENGTH > CRYPTO_RSA_LENGTH)
+#error "CHECKSUM_LENGTH > CRYPTO_RSA_LENGTH"
+#endif

+#define get_unaligned_be32(a) fdt32_to_cpu(*(u32 *)a)
+#define put_unaligned_be32(a, b) (*(u32 *)(b) = cpu_to_fdt32(a))

+/**
+ * is_public_exponent_bit_set() - Check if a bit in the public exponent is set
+ *
+ * @key:	RSA key
+ * @pos:	The bit position to check
+ */
+static int is_public_exponent_bit_set(const struct rsa_public_key *key,
+				      int pos)
+{
+	return key->exponent & (1ULL << pos);
+}
+
+/**
+ * subtract_modulus() - subtract modulus from the given value
+ *
+ * @key:	Key containing modulus to subtract
+ * @num:	Number to subtract modulus from, as little endian word array
+ */
+static void subtract_modulus(const struct rsa_public_key *key, u32 num[])
+{
+	long long acc = 0;
+	u32 i;
+
+	for (i = 0; i < key->len; i++) {
+		acc += (u64)num[i] - key->modulus[i];
+		num[i] = (u32)acc;
+		acc >>= 32;
+	}
+}
+
+
+/**
+ * greater_equal_modulus() - check if a value is >= modulus
+ *
+ * @key:	Key containing modulus to check
+ * @num:	Number to check against modulus, as little endian word array
+ * @return 0 if num < modulus, 1 if num >= modulus
+ */
+static int greater_equal_modulus(const struct rsa_public_key *key,
+				 u32 num[])
+{
+	int i;
+
+	for (i = (int)key->len - 1; i >= 0; i--) {
+		if (num[i] < key->modulus[i])
+			return 0;
+		if (num[i] > key->modulus[i])
+			return 1;
+	}
+
+	return 1;	/* equal */
+}
+
+/**
+ * num_pub_exponent_bits() - Number of bits in the public exponent
+ *
+ * @key:	RSA key
+ * @num_bits:	Storage for the number of public exponent bits
+ */
+static int num_public_exponent_bits(const struct rsa_public_key *key,
+				    int *num_bits)
+{
+	u64 exponent;
+	int exponent_bits;
+	const u32 max_bits = (sizeof(exponent) * 8);
+
+	exponent = key->exponent;
+	exponent_bits = 0;
+
+	if (!exponent) {
+		*num_bits = exponent_bits;
+		return 0;
+	}
+
+	for (exponent_bits = 1; exponent_bits < max_bits + 1; ++exponent_bits)
+		if (!(exponent >>= 1)) {
+			*num_bits = exponent_bits;
+			return 0;
+		}
+
+	return 1;
+}

+/**
+ * montgomery_mul_add_step() - Perform montgomery multiply-add step
+ *
+ * Operation: montgomery result[] += a * b[] / n0inv % modulus
+ *
+ * @key:	RSA key
+ * @result:	Place to put result, as little endian word array
+ * @a:		Multiplier
+ * @b:		Multiplicand, as little endian word array
+ */
+static void montgomery_mul_add_step(const struct rsa_public_key *key,
+				    u32 result[], const u32 a, const u32 b[])
+{
+	u64 acc_a, acc_b;
+	u32 d0;
+	u32 i;
+
+	acc_a = (u64)a * b[0] + result[0];
+	d0 = (u32)acc_a * key->n0inv;
+	acc_b = (u64)d0 * key->modulus[0] + (u32)acc_a;
+	for (i = 1; i < key->len; i++) {
+		acc_a = (acc_a >> 32) + (u64)a * b[i] + result[i];
+		acc_b = (acc_b >> 32) + (u64)d0 * key->modulus[i] +
+				(u32)acc_a;
+		result[i - 1] = (u32)acc_b;
+	}
+
+	acc_a = (acc_a >> 32) + (acc_b >> 32);
+
+	result[i - 1] = (u32)acc_a;
+
+	if (acc_a >> 32)
+		subtract_modulus(key, result);
+}
+
+/**
+ * montgomery_mul() - Perform montgomery mutitply
+ *
+ * Operation: montgomery result[] = a[] * b[] / n0inv % modulus
+ *
+ * @key:	RSA key
+ * @result:	Place to put result, as little endian word array
+ * @a:		Multiplier, as little endian word array
+ * @b:		Multiplicand, as little endian word array
+ */
+static void montgomery_mul(const struct rsa_public_key *key,
+			   u32 result[], u32 a[], const u32 b[])
+{
+	u32 i;
+
+	for (i = 0; i < key->len; ++i)
+		result[i] = 0;
+	for (i = 0; i < key->len; ++i)
+		montgomery_mul_add_step(key, result, a[i], b);
+}

+static int pow_mod(const struct rsa_public_key *key, u32 *out, u32 *in)
+{
+	u32 *result, *ptr;
+	u32 i;
+	int j, k;
+
+	u32 val[key->len], acc[key->len], tmp[key->len];
+	u32 a_scaled[key->len];
+
+	result = tmp;	/* Re-use location. */
+
+	/* Convert from big endian byte array to little endian word array. */
+	for (i = 0, ptr = in + key->len - 1; i < key->len; i++, ptr--)
+		val[i] = get_unaligned_be32(ptr);
+
+	if (num_public_exponent_bits(key, &k) != 0) {
+		preboot_puts("num_public_exponent_bits failed!\n");
+		return 1;
+	}
+
+	if (k < 2) {
+		preboot_puts("Public exponent is too short\n");
+		return 1;
+	}
+
+	if (!is_public_exponent_bit_set(key, 0)) {
+		preboot_puts("LSB of RSA public exponent must be set.\n");
+		return 1;
+	}
+
+	/* the bit at e[k-1] is 1 by definition, so start with: C := M */
+	montgomery_mul(key, acc, val, key->rr); /* acc = a * RR / R mod n */
+	/* retain scaled version for intermediate use */
+	memcpy(a_scaled, acc, key->len * sizeof(a_scaled[0]));
+
+	for (j = k - 2; j > 0; --j) {
+		montgomery_mul(key, tmp, acc, acc); /* tmp = acc^2 / R mod n */
+
+		if (is_public_exponent_bit_set(key, j)) {
+			/* acc = tmp * val / R mod n */
+			montgomery_mul(key, acc, tmp, a_scaled);
+		} else {
+			/* e[j] == 0, copy tmp back to acc for next operation */
+			memcpy(acc, tmp, key->len * sizeof(acc[0]));
+		}
+	}
+
+	/* the bit at e[0] is always 1 */
+	montgomery_mul(key, tmp, acc, acc); /* tmp = acc^2 / R mod n */
+	montgomery_mul(key, acc, tmp, val); /* acc = tmp * a / R mod M */
+	memcpy(result, acc, key->len * sizeof(result[0]));
+
+	/* Make sure result < mod; result is at most 1x mod too large. */
+	if (greater_equal_modulus(key, result))
+		subtract_modulus(key, result);
+
+	/* Convert to bigendian byte array */
+	for (i = key->len - 1, ptr = out; (int)i >= 0; i--, ptr++)
+		put_unaligned_be32(result[i], ptr);
+	return 0;
+}

+static void rsa_convert_big_endian(u32 *dst, const u32 *src, int len)
+{
+	int i;
+
+	for (i = 0; i < len; i++)
+		dst[i] = fdt32_to_cpu(src[len - 1 - i]);
+}

+int rsa_mod_exp_sw(const u8 *sig, struct rsa_key_prop *prop, u32 *out)
+{
+	struct rsa_public_key key;
+	int ret;
+
+	key.n0inv = prop->n0inv;
+	key.len = prop->num_bits / (sizeof(u32) * 8);
+	if (prop->public_exponent)
+		key.exponent = fdt64_to_cpu(*(prop->public_exponent));
+	else
+		key.exponent = RSA_DEFAULT_PUBEXP;
+
+	u32 key1[key.len], key2[key.len];
+
+	key.modulus = key1;
+	key.rr = key2;
+
+	rsa_convert_big_endian(key.modulus, (u32 *)prop->modulus, key.len);
+	rsa_convert_big_endian(key.rr, (u32 *)prop->rr, key.len);
+
+	ret = pow_mod(&key, out, (u32 *)sig);
+	if (ret)
+		return ret;
+
+	return 0;
+}

+static int rsa_verify_padding(const u8 *msg, const int pad_len)
+{
+	int ff_len;
+	int ret;
+
+	/* first byte must be 0x00 */
+	ret = *msg++;
+	/* second byte must be 0x01 */
+	ret |= *msg++ ^ 0x01;
+	/* next ff_len bytes must be 0xff */
+	ff_len = pad_len - CHECKSUM_DER_LENGTH - 3;
+	ret |= *msg ^ 0xff;
+	ret |= memcmp(msg, msg+1, ff_len-1);
+	msg += ff_len;
+	/* next byte must be 0x00 */
+	ret |= *msg++;
+	/* next der_len bytes must match der_prefix */
+	ret |= memcmp(msg, CHECKSUM_DER_PREFIX, CHECKSUM_DER_LENGTH);
+
+	return ret;
+}
+
+int padding_pkcs_15_verify(u8 *msg, int msg_len, const u8 *hash)
+{
+	int pad_len = msg_len - CHECKSUM_LENGTH;
+
+	if (rsa_verify_padding(msg, pad_len)) {
+		preboot_puts("RSAVerify(): rsa_verify_padding failed!\n");
+		return 1;
+	}
+
+	/* Check hash. */
+	if (memcmp((u8 *)msg + pad_len, hash, msg_len - pad_len)) {
+		preboot_puts("RSAVerify(): Hash check failed!\n");
+		return 1;
+	}
+
+	return 0;
+}

+static int rsa_verify_key(struct rsa_key_prop *prop, const u8 *sig,
+			  const u32 sig_len, const u8 *hash, const u32 key_len)
+{
+	u32 buf[sig_len>>2];
+
+	if (rsa_mod_exp_sw(sig, prop, buf)) {
+		preboot_puts("rsa_mod_exp_sw failed\n");
+		return 1;
+	}
+
+	if (padding_pkcs_15_verify((u8 *)buf, key_len, hash))
+		return 1;
+
+	return 0;
+}
+
+/* based on int rsa_verify(struct image_sign_info *info,
+ *		const struct image_region region[], int region_count,
+ *		uint8_t *sig, uint sig_len)
+ * from: u-boot/lib/rsa/rsa-verify.c
+ */
+int rsa_verify(const u8 *data, int data_len, const u8 *sig, u32 sig_len,
+	       struct rsa_key_prop *prop)
+{
+	u8 hash[CRYPTO_RSA_LENGTH];
+	int ret;
+
+	if (sig_len != (prop->num_bits >> 3)) {
+		preboot_puts("sig_len != (prop->num_bits / 8)\n");
+		return 1;
+	}
+
+	sha256_csum(data, data_len, hash);
+
+	ret = rsa_verify_key(prop, sig, sig_len, hash, CRYPTO_RSA_LENGTH);
+	return ret;
+}
