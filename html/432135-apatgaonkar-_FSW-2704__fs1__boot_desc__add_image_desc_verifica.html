+#define PPBOOT_MAX_IMAGE_SIZE	(2 * MAX_DMA_SIZE_NO_CRC)
+uint32_t boot_rom_verify_img_desc_address_range(struct boot_desc *desc,
+						enum boot_rom_progress img_sts_base,
+						enum hl_fw_img_type img)
+{
+	if (!boot_rom_is_valid_address_range(desc->loadaddr, desc->img_size)) {
+		boot_rom_status_update_bringup(img_sts_base, PROGRESS_BRINGUP_LOADADDR_INVALID);
+		return BOOT_ROM_BRINGUP_ERROR;
+	}
+
+	/* TODO: Veirfy DMA destination address */
+
+	if (!boot_rom_is_valid_start_addr(desc)) {
+		boot_rom_status_update_bringup(img_sts_base, PROGRESS_BRINGUP_STARTADDR_INVALID);
+		return BOOT_ROM_BRINGUP_ERROR;
+	}
+
+	if ((img == HL_FW_IMG_PPBOOT) && (desc->img_size > PPBOOT_MAX_IMAGE_SIZE)) {
+		boot_rom_status_update_bringup(img_sts_base, PROGRESS_BRINGUP_TOO_BIG);
+		return BOOT_ROM_BRINGUP_ERROR;
+	}
+
+	return BOOT_ROM_SUCCESS;
+}
