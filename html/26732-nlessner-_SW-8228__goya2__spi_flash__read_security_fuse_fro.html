+#include "arcboot.h"
+#include "spi_flash.h"
+
+static void spi_flash_set_byte_swap(uint8_t enable)
+{
+	uint32_t reg_spi_direct_wr_rd_ctrl = 0;
+
+	/* clear BYTE SWAP in SPI DMA so we dont switch endianness */
+	reg_spi_direct_wr_rd_ctrl =
+		regmap_readl(mmPSOC_GLOBAL_CONF_SPI_DIRECT_WR_RD_CTRL);
+	if (enable)
+		reg_spi_direct_wr_rd_ctrl |=
+			PSOC_GLOBAL_CONF_SPI_DIRECT_WR_RD_CTRL_BYTE_SWAP_MASK;
+	else
+		reg_spi_direct_wr_rd_ctrl &=
+			~PSOC_GLOBAL_CONF_SPI_DIRECT_WR_RD_CTRL_BYTE_SWAP_MASK;
+
+	regmap_writel(reg_spi_direct_wr_rd_ctrl,
+		      mmPSOC_GLOBAL_CONF_SPI_DIRECT_WR_RD_CTRL);
+}
+
+u32 spi_flash_readl(u32 offset)
+{
+	spi_flash_set_byte_swap(ENABLE_BYTE_SWAP);
+	u32 reg = base_readl(HL_FLASH_BASE + offset);
+
+	return reg;
+}
+
+void spi_flash_writel(u32 val, u32 offset)
+{
+	spi_flash_set_byte_swap(ENABLE_BYTE_SWAP);
+	base_writel(val, HL_FLASH_BASE + offset);
+}
+
+void spi_flash_get_revoke(uint32_t *key_revoke)
+{
+	uint32_t flash_efuse_offset, revoke_status;
+	uint8_t i;
+	*key_revoke = 0;
+	flash_efuse_offset = spi_flash_readl(FLASH_EFUSE_OFFSET);
+	revoke_status =  spi_flash_readl(flash_efuse_offset +
+					 FLASH_EFUSE_KEYS_REVOKE_OFFSET);
+	/* check for revoke habana key bit and customer bit 0 - 3 */
+	for (i = 0; i < 4; i++) {
+		if ((revoke_status >> i) & 1)
+			*key_revoke |= BIT(i);
+		revoke_status = revoke_status >> 4;
+	}
+
+	flash_efuse_offset += 4;
+	revoke_status =  spi_flash_readl(flash_efuse_offset +
+					 FLASH_EFUSE_KEYS_REVOKE_OFFSET);
+	/* check for revoke bit of customer key 4 */
+	if ((revoke_status & 1) == 1)
+		*key_revoke |= BIT(4);
+}
+
+void spi_flash_get_hash(uint32_t *key_hash)
+{
+	uint8_t i;
+	uint32_t flash_efuse_offset = spi_flash_readl(FLASH_EFUSE_OFFSET) +
+						FLASH_EFUSE_KEYS_OFFSET;
+	for (i = 0; i < 40; i++)
+		*(key_hash + i) = spi_flash_readl(flash_efuse_offset + i);
+}
+
+/* minimal allowed security version is defined by a 128bit map, to find this
+ * we scan from end to beginning of map looking for the first set bit. once
+ * it's found this indicated the minimal version allowed to run on the system
+ */
+uint8_t spi_flash_get_fw_version_security(void)
+{
+	uint8_t i, row, colume;
+	uint32_t fw_version[4];
+	uint8_t fw_version_sec;
+	uint32_t flash_efuse_offset = spi_flash_readl(FLASH_EFUSE_OFFSET) +
+						FLASH_EFUSE_FW_VERSION_OFFSET;
+
+	/* read min security version bit map from flash */
+	for (i = 0; i < 4; i++)
+		fw_version[i] = spi_flash_readl(flash_efuse_offset + i);
+
+	/* init max version */
+	fw_version_sec = 128;
+
+	/* iterate back from 127 -> 0 till a set bit found */
+	for (row = 3; row > 0; row--) {
+		for (colume = 31; colume > 0; colume--) {
+			/* find highest bit set in valid FW */
+			fw_version_sec--;
+			if (fw_version[row] & BIT(colume))
+				return fw_version_sec;
+		}
+	}
+	return fw_version_sec;
+}
