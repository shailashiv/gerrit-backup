+void hl_secure_update_hit_reg(int rr_idx, int inst_dist,
+				struct hit_regs *rr_reg, bool write_only, int die)
+{
+	uint32_t rr_dest, hit;
+
+	/* Enable (hit) RR */
+	rr_dest = rr_reg->hit_wr + inst_dist;
+	if (die) {
+		hit = mirror_readl(rr_dest);
+		mirror_writel(hit | BIT(rr_idx), rr_dest);
+	} else {
+		hit = regmap_readl(rr_dest);
+		regmap_writel(hit | BIT(rr_idx), rr_dest);
+	}
+
+	if (write_only)
+		return;
+
+	rr_dest = rr_reg->hit_rd + inst_dist;
+	if (die) {
+		hit = mirror_readl(rr_dest);
+		mirror_writel(hit | BIT(rr_idx), rr_dest);
+	} else {
+		hit = regmap_readl(rr_dest);
+		regmap_writel(hit | BIT(rr_idx), rr_dest);
+	}
+}

+void hl_secure_range_enable(uint32_t rr_dest, bool write_only, int die)
+{
+	if (die)
+		mirror_rmwritel(rr_dest, 1, MSTR_IF_RR_LBW_PRIV_RANGE_EN_WR_M);
+	else
+		regmap_rmwritel(rr_dest, 1, MSTR_IF_RR_LBW_PRIV_RANGE_EN_WR_M);

+	if (write_only)
+		return;
+
+	if (die)
+		mirror_rmwritel(rr_dest, 1, MSTR_IF_RR_LBW_PRIV_RANGE_EN_RD_M);
+	else
+		regmap_rmwritel(rr_dest, 1, MSTR_IF_RR_LBW_PRIV_RANGE_EN_RD_M);

+void hl_secure_update_rr_reg(int rr_idx, uint32_t inst_offset, struct lbw_rr_reg *rr_reg,
+									bool enable, int die)
+{
+	uint32_t rr_base, rr_dest;
+
+	rr_base = rr_reg->rr_idx_calc * sizeof(uint32_t) + inst_offset;
+	rr_dest = rr_base + rr_reg->reg_min;
+	if (die)
+		mirror_writel(rr_reg->data_min, rr_dest);
+	else
+		regmap_writel(rr_reg->data_min, rr_dest);
+
+	rr_dest = rr_base + rr_reg->reg_max;
+	if (die)
+		mirror_writel(rr_reg->data_max, rr_dest);
+	else
+		regmap_writel(rr_reg->data_max, rr_dest);
+
+	rr_dest = rr_base + rr_reg->range_en;
+	if (enable)
+		hl_secure_range_enable(rr_dest, false, die);
+	else
+		hl_secure_range_disable(rr_dest, false, die);
+
+	hl_secure_update_hit_reg(rr_idx, inst_offset, &rr_reg->hit, false, die);
+}

+/* shift bits to get partial address regions for 4KB granularity */
+#define HL_SEC_SHIFT_4K_POS	12
+
+static void hl_secure_lbw_rr_gen(int rr_idx, uint64_t min, uint64_t max, int dist, int inst_num,
+				enum hl_sec_engine_type type, bool cfg_all_inst, bool enable, int die)
+{
+	int i;
+	const struct hl_sec_rrs_cfg *cfg = &hl_sec_lbw_rrs_tbl[type];
+	struct lbw_rr_reg rr_reg;
+
+	rr_reg.data_min = (min & ~LBW_BASE);
+	rr_reg.data_max = (max & ~LBW_BASE);
+
+	if (rr_idx < HL_SEC_LBW_RR_FULL_0) {
+		rr_reg.rr_idx_calc = rr_idx;
+		rr_reg.data_min &= ~GEN_BIT_MASK(HL_SEC_SHIFT_4K_POS);
+		rr_reg.data_max &= ~GEN_BIT_MASK(HL_SEC_SHIFT_4K_POS);
+		rr_reg.reg_min = cfg->min_short_breg;
+		rr_reg.reg_max = cfg->max_short_breg;
+		rr_reg.range_en = cfg->range_en_short_breg;
+	} else {
+		rr_reg.rr_idx_calc = rr_idx - HL_SEC_LBW_RR_FULL_0;
+		rr_reg.reg_min = cfg->min_full_breg;
+		rr_reg.reg_max = cfg->max_full_breg;
+		rr_reg.range_en = cfg->range_en_full_breg;
+	}
+
+	rr_reg.reg_min += dist;
+	rr_reg.reg_max += dist;
+	rr_reg.hit.hit_wr = cfg->hit_wr_breg + dist;
+	rr_reg.hit.hit_rd = cfg->hit_rd_breg + dist;
+	rr_reg.range_en += dist;
+
+	/* Configure specific instance num if cfg_all_inst is false */
+	if (cfg_all_inst == false) {
+		hl_secure_update_rr_reg(rr_idx, cfg->instances_dist * inst_num, &rr_reg, enable, die);
+		return;
+	}
+
+	for (i = 0; i < inst_num; i++)
+		hl_secure_update_rr_reg(rr_idx, cfg->instances_dist * i, &rr_reg, enable, die);
+}

+#define HL_SEC_BITS_IN_REG			32
+/* shift bits to get address regions for 16KB granularity */
+#define HL_SEC_SHIFT_16K_POS			14
+
+static void hl_secure_update_rr_reg_hbw(int rr_idx, uint32_t inst_offset, struct hbw_rr_reg *rr_reg, int die)
+{
+	uint64_t rr_base, rr_dest;
+	uint32_t data_max_hi = ((uint32_t)(rr_reg->data_max >> HL_SEC_BITS_IN_REG));
+	uint32_t data_min_hi = ((uint32_t)(rr_reg->data_min >> HL_SEC_BITS_IN_REG));
+
+	rr_base = rr_reg->rr_idx_calc * sizeof(uint32_t) + inst_offset;
+	rr_dest = rr_base + rr_reg->reg_min.lo;
+	if (die)
+		mirror_writel(rr_reg->data_min, rr_dest);
+	else
+		regmap_writel(rr_reg->data_min, rr_dest);
+
+	rr_dest = rr_base + rr_reg->reg_min.hi;
+	if (die)
+		mirror_writel(data_min_hi, rr_dest);
+	else
+		regmap_writel(data_min_hi, rr_dest);
+
+	rr_dest = rr_base + rr_reg->reg_max.lo;
+	if (die)
+		mirror_writel(rr_reg->data_max, rr_dest);
+	else
+		regmap_writel(rr_reg->data_max, rr_dest);
+
+	rr_dest = rr_base + rr_reg->reg_max.hi;
+	if (die)
+		mirror_writel(data_max_hi, rr_dest);
+	else
+		regmap_writel(data_max_hi, rr_dest);
+
+	rr_dest = rr_base + rr_reg->range_en;
+	hl_secure_range_enable(rr_dest, false, die);
+
+	hl_secure_update_hit_reg(rr_idx, inst_offset, &rr_reg->hit, false, die);
+}

+static void hl_secure_hbw_rr_gen(int rr_idx, uint64_t min, uint64_t max, int dist,
+						int inst_num, enum hl_sec_engine_type type, int die)
+{
+	int i;
+	struct hbw_rr_reg rr_reg;
+	const struct hl_sec_rrs_cfg_hbw *cfg = &hl_sec_hbw_rrs_tbl[type];
+
+	/* Following masks is used for 16K granularity
+	 * As lower 32 bits takes address bits from 31:14
+	 */
+	rr_reg.data_min = (min & ~GEN_BIT_MASK(HL_SEC_SHIFT_16K_POS));
+	rr_reg.data_max = (max & ~GEN_BIT_MASK(HL_SEC_SHIFT_16K_POS));
+
+	rr_reg.rr_idx_calc = rr_idx;
+	rr_reg.reg_min = cfg->min_full;
+	rr_reg.reg_max = cfg->max_full;
+
+	rr_reg.reg_min.lo += dist;
+	rr_reg.reg_min.hi += dist;
+	rr_reg.reg_max.lo += dist;
+	rr_reg.reg_max.hi += dist;
+	rr_reg.hit.hit_wr = cfg->hit_wr_breg + dist;
+	rr_reg.hit.hit_rd = cfg->hit_rd_breg + dist;
+	rr_reg.range_en = cfg->range_en_full_breg + dist;
+
+	for (i = 0; i < inst_num; i++)
+		hl_secure_update_rr_reg_hbw(rr_idx, cfg->instances_dist * i, &rr_reg, die);
+}

+#define HL_SEC_LBW_RR_MAX_NUMBER	18
+#define HL_SEC_HBW_RR_MAX_NUMBER	8
+
+#define HL_SEC_BW_RR_REPEAT_NUM		4
+
+/* index to track LBW RR debug array */
+static unsigned int lbw_idx;
+
+/* index to track HBW RR debug array */
+static unsigned int hbw_idx;
+
+/* Array to count each RR programmed number of times */
+static uint8_t lbw_rr_dbg_arr[HL_SEC_LBW_RR_MAX_NUMBER];
+
+/* Array to count each RR programmed number of times */
+static uint8_t hbw_rr_dbg_arr[HL_SEC_HBW_RR_MAX_NUMBER];
+
+static bool hl_secure_xbw_rr_debug_update(void *rr_reg_dbg_arr,
+			int rr_idx, uint64_t min, uint64_t max, bool enable, bool lbw_n_rbw)
+{
+	uint32_t xbw_idx;
+	uint8_t *rr_dbg_arr;
+	uint32_t hl_sec_xbw_rr_limit;
+	struct xbw_rr_reg_dbg rr_reg_dbg;
+
+	if (lbw_n_rbw) {
+		rr_reg_dbg.lbw_rr_reg_dbg_arr = (struct lbw_rr_reg_dbg *)rr_reg_dbg_arr;
+		xbw_idx = lbw_idx;
+		rr_dbg_arr = lbw_rr_dbg_arr;
+		hl_sec_xbw_rr_limit = HL_SEC_LBW_RR_MAX_NUMBER * HL_SEC_BW_RR_REPEAT_NUM;
+	} else {
+		rr_reg_dbg.hbw_rr_reg_dbg_arr = (struct hbw_rr_reg_dbg *)rr_reg_dbg_arr;
+		xbw_idx = hbw_idx;
+		rr_dbg_arr = hbw_rr_dbg_arr;
+		hl_sec_xbw_rr_limit = HL_SEC_HBW_RR_MAX_NUMBER * HL_SEC_BW_RR_REPEAT_NUM;
+	}
+
+	/* if g_lbw_rr_reg_dbg_arr is full, skip logging to avoid overflow */
+	if (xbw_idx < hl_sec_xbw_rr_limit) {
+		/* An RR is configured 4 times already, set the flag */
+		if (rr_dbg_arr[rr_idx] >= HL_SEC_BW_RR_REPEAT_NUM) {
+			rr_reg_dbg.lbw_rr_reg_dbg_arr[xbw_idx].log_overflow = 1;
+		}  else  { /* "else":- if still logging required, */
+			/* removing "else"  skip latest RR data update */
+			rr_reg_dbg.lbw_rr_reg_dbg_arr[xbw_idx].status = enable;
+			rr_reg_dbg.lbw_rr_reg_dbg_arr[xbw_idx].sym = 0;
+			rr_reg_dbg.lbw_rr_reg_dbg_arr[xbw_idx].rr_index = rr_idx;
+			if (lbw_n_rbw) {
+				rr_reg_dbg.lbw_rr_reg_dbg_arr[xbw_idx].data_min = (min & ~LBW_BASE);
+				rr_reg_dbg.lbw_rr_reg_dbg_arr[xbw_idx].data_max = (max & ~LBW_BASE);
+			} else {
+				rr_reg_dbg.hbw_rr_reg_dbg_arr[xbw_idx].data_min = min;
+				rr_reg_dbg.hbw_rr_reg_dbg_arr[xbw_idx].data_max = max;
+			}
+		}
+
+		if (rr_dbg_arr[rr_idx] >= HL_SEC_BW_RR_REPEAT_NUM)
+			rr_reg_dbg.lbw_rr_reg_dbg_arr[xbw_idx].log_overflow = 1;
+		return true;
+
+	} else {
+		return false;
+	}
+}

+#define HL_SEC_PSOC_INST_NUM    3 /* JTAG, SMI, I2C */
+#define DIE1_HDCORE_OFFSET      (HDCORE_OFFSET * GAUDI3_NUM_HDCORES_PER_DIE)
+
+static inline void hl_secure_lbw_rr_type_debug_update(struct lbw_rr_reg_dbg *lbw_rr_reg_dbg_arr,
+							bool log_require, uint8_t rr_type_val)
+{
+	if (true == log_require)
+		lbw_rr_reg_dbg_arr[lbw_idx].rr_type |= rr_type_val;
+}

+void hl_secure_lbw_rr_common(struct hl_security_drv_data *data, int rr_idx, uint64_t min,
+			uint64_t max, struct hl_security_rr_engine_select rr_sel, bool enable)
+{
+	uint32_t die1_hdcore_offset, num_of_hdcores, hdcore, i, die;
+	/*
+	 * ROT and SEDMA is available in HD1, HD3, HD4 and HD6
+	 * Hence below array.
+	 */
+	uint32_t hdcore_rot_sedma[GAUDI3_NUM_HDCORES] = {0, 1, 0, 1, 1, 0, 1, 0};
+	bool log_require = false;
+	struct lbw_rr_reg_dbg *p_g_lbw_rr_reg_dbg_arr = (struct lbw_rr_reg_dbg *)
+							data->lbw_rr_dbg_offset;
+
+	/* Capture RR info in debug array */
+	if (!data->sec_dbg)
+		log_require = hl_secure_xbw_rr_debug_update(p_g_lbw_rr_reg_dbg_arr,
+							rr_idx, min, max, enable, true);
+	if (rr_sel.noc) {
+		/* update RR debug info for rr_type NOC */
+		hl_secure_lbw_rr_type_debug_update(p_g_lbw_rr_reg_dbg_arr,
+				log_require, HL_SEC_RR_SEL_NOC);
+
+		for (die = 0; die <= data->sku; die++) {
+			die1_hdcore_offset = (DIE1_HDCORE_OFFSET * die);
+			for (hdcore = 0; hdcore < GAUDI3_NUM_HDCORES_PER_DIE; hdcore++)
+				hl_secure_lbw_rr_gen(rr_idx, min, max, ((HDCORE_OFFSET * hdcore) +
+						die1_hdcore_offset), GAUDI3_NUM_RRTR_PER_HDCORE,
+						HL_SEC_ENGINE_IF_RTR, true, enable, die);
+
+			hl_secure_lbw_rr_gen(rr_idx, min, max, (DIE_OFFSET * die), 0,
+					HL_SEC_ENGINE_NRTR, false, enable, die);
+		}
+
+		num_of_hdcores = (data->sku) ? GAUDI3_NUM_HDCORES : GAUDI3_NUM_HDCORES_PER_DIE;
+		for (hdcore = 0; hdcore < num_of_hdcores; hdcore++) {
+
+			/*
+			 * As we have only 2 ROT and 2 SEDMA per DCORE.
+			 * In case of dual die we need to configure only in HD1, HD3, HD4 and HD6.
+			 * and in case of single die we need to configure only in HD1, HD3.
+			 */
+			if (!hdcore_rot_sedma[hdcore])
+				continue;
+
+			die = (hdcore < GAUDI3_NUM_HDCORES_PER_DIE) ? 0 : 1;
+
+			hl_secure_lbw_rr_gen(rr_idx, min, max, (HDCORE_OFFSET * hdcore), 0,
+							HL_SEC_ENGINE_RIF_RTR, false, enable, die);
+		}
+	}
+	/*
+	 * JTAG, SMI, I2C MSTR_IFs used for blocking JTAG/I2C/SMI
+	 * PSOC traffics.
+	 */
+	if (rr_sel.psoc) {
+		/* update rr_type for PSOC */
+		hl_secure_lbw_rr_type_debug_update(p_g_lbw_rr_reg_dbg_arr,
+				log_require, HL_SEC_RR_SEL_PSOC);
+
+		for (die = 0; die <= data->sku; die++)
+			hl_secure_lbw_rr_gen(rr_idx, min, max, (DIE_OFFSET * die),
+					HL_SEC_PSOC_INST_NUM, HL_SEC_ENGINE_PSOC, true, enable, die);
+	}
+
+	if (rr_sel.pcie) {
+		/* update rr_type for PCIE elbi, lbw, mstr */
+		hl_secure_lbw_rr_type_debug_update(p_g_lbw_rr_reg_dbg_arr, log_require,
+				(HL_SEC_RR_SEL_PCIE | (0xff & (rr_sel.pcie << 3))));
+		/* pcie on die1 is not used, So configure for only die0 pcie */
+		for (i = 0; i < HL_SEC_PCIE_RR_MAX; i++) {
+			if (rr_sel.pcie & BIT(i))
+				hl_secure_lbw_rr_gen(rr_idx, min, max, 0, i,
+						HL_SEC_ENGINE_PCIE, false, enable, 0);
+		}
+		hl_secure_lbw_rr_gen(rr_idx, min, max, 0, 0, HL_SEC_ENGINE_NW12_DEC, false, enable, 0);
+	}
+
+	if (rr_sel.sync_mngr) {
+		for (die = 0; die <= data->sku; die++) {
+			die1_hdcore_offset = (DIE1_HDCORE_OFFSET * die);
+			for (hdcore = 0; hdcore < GAUDI3_NUM_HDCORES_PER_DIE; hdcore++)
+				hl_secure_lbw_rr_gen(rr_idx, min, max, ((HDCORE_OFFSET * hdcore) +
+						die1_hdcore_offset), 0, HL_SEC_ENGINE_SYNC_MNGR,
+						false, enable, die);
+		}
+	}
+
+	if (true == log_require) {
+		lbw_idx++; /*Increment LBW RR configure logging debug buffer index */
+		lbw_rr_dbg_arr[rr_idx] += 1; /* counting each RR configured how many times */
+	}
+}

+static inline void hl_secure_hbw_rr_type_debug_update(struct hbw_rr_reg_dbg
+					*hbw_rr_reg_dbg_arr, bool log_require, uint8_t rr_type_val)
+{
+	if (true == log_require)
+		hbw_rr_reg_dbg_arr[hbw_idx].rr_type |= rr_type_val;
+}

+	struct hbw_rr_reg_dbg *p_g_hbw_rr_reg_dbg_arr = (struct hbw_rr_reg_dbg *)
+								data->hbw_rr_dbg_offset;
+
+
+	/* Capture RR info in debug array */

+		log_require = hl_secure_xbw_rr_debug_update(p_g_hbw_rr_reg_dbg_arr,
