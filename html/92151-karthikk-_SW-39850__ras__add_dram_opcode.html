+{
+	u32 size = MAX_INFO_DESC_DATA;
+	int ret, table_info_size;

+	u16 offset;
+
+	ret = rma_info_get(RMA_HBM_ROW_REPLACEMENT_TABLE, err_sts_rma_buf, &size);
+	if (ret)
+		return;
+
+	table_info_size = size - RMA_TABLE_HEADER_SIZE;
+	op->number_of_replaced_rows =  table_info_size / RMA_HBM_ROW_REP_ENTRY_SIZE;
+
+	/*
+	 * Opcode's query of row index must
+	 * be an index within the replaced rows
+	 */
+	if (op->index_replaced_row_selection < op->number_of_replaced_rows) {
+		offset = RMA_TABLE_HEADER_SIZE +
+				(op->index_replaced_row_selection * RMA_HBM_ROW_REP_ENTRY_SIZE);
+		hbm_row_replacement =
+				(struct rma_hbm_row_replacement_entry *)&err_sts_rma_buf[offset];
+		op->replaced_row_hbm_index = hbm_row_replacement->device;
+		op->replaced_row_pc_index = hbm_row_replacement->pc;
+		op->replaced_row_bank_index = hbm_row_replacement->bank;
+		op->replaced_row_cause.data = hbm_row_replacement->header.data_cause;
+		op->replaced_row_address = hbm_row_replacement->row;
+
+	}
+	/* Clear out the data for buffer re-use */
+	memset(err_sts_rma_buf, 0, size);
