[SW-20154] rsa: support for pkcs#1 v 1.5 decryption

This patch adds support for pkcs#1 v1.5 decryption
from Uboot master version 2020.10.
Instead of computing the public key properties (or)
extracting them from a fit file, this decryption
expects the properties of public key to be appended
to the end of public key.

+#include <rsa.h>
+#include <rsa-mod-exp.h>
+#include <string.h>
+#include <hl_rsa_arc.h>
+
+/* Default public exponent for backward compatibility */
+#define RSA_DEFAULT_PUBEXP	65537
+
+/**
+ * subtract_modulus() - subtract modulus from the given value
+ *
+ * @key:	Key containing modulus to subtract
+ * @num:	Number to subtract modulus from, as little endian word array
+ */
+static void subtract_modulus(const struct rsa_public_key *key, uint32_t num[])
+{
+	int64_t acc = 0;
+	uint32_t i;
+
+	for (i = 0; i < key->len; i++) {
+		acc += (uint64_t)num[i] - key->modulus[i];
+		num[i] = (uint32_t)acc;
+		acc >>= 32;
+	}
+}
+
+/**
+ * greater_equal_modulus() - check if a value is >= modulus
+ *
+ * @key:	Key containing modulus to check
+ * @num:	Number to check against modulus, as little endian word array
+ * @return 0 if num < modulus, 1 if num >= modulus
+ */
+static int greater_equal_modulus(const struct rsa_public_key *key,
+				 uint32_t num[])
+{
+	int i;
+
+	for (i = (int)key->len - 1; i >= 0; i--) {
+		if (num[i] < key->modulus[i])
+			return 0;
+		if (num[i] > key->modulus[i])
+			return 1;
+	}
+
+	return 1;  /* equal */
+}
+
+/**
+ * montgomery_mul_add_step() - Perform montgomery multiply-add step
+ *
+ * Operation: montgomery result[] += a * b[] / n0inv % modulus
+ *
+ * @key:	RSA key
+ * @result:	Place to put result, as little endian word array
+ * @a:		Multiplier
+ * @b:		Multiplicand, as little endian word array
+ */
+static void montgomery_mul_add_step(const struct rsa_public_key *key,
+		uint32_t result[], const uint32_t a, const uint32_t b[])
+{
+	uint64_t acc_a, acc_b;
+	uint32_t d0;
+	uint32_t i;
+
+	acc_a = (uint64_t)a * b[0] + result[0];
+	d0 = (uint32_t)acc_a * key->n0inv;
+	acc_b = (uint64_t)d0 * key->modulus[0] + (uint32_t)acc_a;
+	for (i = 1; i < key->len; i++) {
+		acc_a = (acc_a >> 32) + (uint64_t)a * b[i] + result[i];
+		acc_b = (acc_b >> 32) + (uint64_t)d0 * key->modulus[i] +
+				(uint32_t)acc_a;
+		result[i - 1] = (uint32_t)acc_b;
+	}
+
+	acc_a = (acc_a >> 32) + (acc_b >> 32);
+
+	result[i - 1] = (uint32_t)acc_a;
+
+	if (acc_a >> 32)
+		subtract_modulus(key, result);
+}
+
+/**
+ * montgomery_mul() - Perform montgomery mutitply
+ *
+ * Operation: montgomery result[] = a[] * b[] / n0inv % modulus
+ *
+ * @key:	RSA key
+ * @result:	Place to put result, as little endian word array
+ * @a:		Multiplier, as little endian word array
+ * @b:		Multiplicand, as little endian word array
+ */
+static void montgomery_mul(const struct rsa_public_key *key,
+		uint32_t result[], uint32_t a[], const uint32_t b[])
+{
+	uint32_t i;
+
+	for (i = 0; i < key->len; ++i)
+		result[i] = 0;
+	for (i = 0; i < key->len; ++i)
+		montgomery_mul_add_step(key, result, a[i], b);
+}
+
+/**
+ * num_pub_exponent_bits() - Number of bits in the public exponent
+ *
+ * @key:	RSA key
+ * @num_bits:	Storage for the number of public exponent bits
+ */
+static int num_public_exponent_bits(const struct rsa_public_key *key,
+		int *num_bits)
+{
+	uint64_t exponent;
+	uint32_t exponent_bits;
+	const uint32_t max_bits = (sizeof(exponent) * 8);
+
+	exponent = key->exponent;
+	exponent_bits = 0;
+
+	if (!exponent) {
+		*num_bits = exponent_bits;
+		return 0;
+	}
+
+	for (exponent_bits = 1; exponent_bits < max_bits + 1; ++exponent_bits)
+		if (!(exponent >>= 1)) {
+			*num_bits = exponent_bits;
+			return 0;
+		}
+
+	return -EINVAL;
+}
+
+/**
+ * is_public_exponent_bit_set() - Check if a bit in the public exponent is set
+ *
+ * @key:	RSA key
+ * @pos:	The bit position to check
+ */
+static int is_public_exponent_bit_set(const struct rsa_public_key *key,
+		int pos)
+{
+	return key->exponent & (1ULL << pos);
+}
+
+/**
+ * pow_mod() - in-place public exponentiation
+ *
+ * @key:	RSA key
+ * @inout:	Big-endian word array containing value and result
+ */
+
+/*
+ * These arrays have been declared here, to be in the same
+ * context as Uboot RSA code, because the original code
+ * declares them on the stack.
+ */
+static uint32_t val[RSA3072_DWORDS], acc[RSA3072_DWORDS], tmp[RSA3072_DWORDS];
+static uint32_t a_scaled[RSA3072_DWORDS];
+
+static int pow_mod(const struct rsa_public_key *key, uint32_t *inout)
+{
+	uint32_t *result, *ptr;
+	uint32_t i;
+	int j, k;
+
+	/* Sanity check for stack size - key->len is in 32-bit words */
+	if (key->len > RSA_MAX_KEY_BITS / 32) {
+		return -EINVAL;
+	}
+
+	/* Uboot RSA declares these arrays on stack.
+	 * These are moved to global arrays to minimize stack usage
+	 * uint32_t val[key->len], acc[key->len], tmp[key->len];
+	 * uint32_t a_scaled[key->len];
+	 */
+	result = tmp;  /* Re-use location. */
+
+	/* Convert from big endian byte array to little endian word array. */
+	for (i = 0, ptr = inout + key->len - 1; i < key->len; i++, ptr--)
+		val[i] = get_unaligned_be32(ptr);
+
+	if (0 != num_public_exponent_bits(key, &k))
+		return -EINVAL;
+
+	if (k < 2) {
+		return -EINVAL;
+	}
+
+	if (!is_public_exponent_bit_set(key, 0)) {
+		return -EINVAL;
+	}
+
+	/* the bit at e[k-1] is 1 by definition, so start with: C := M */
+	montgomery_mul(key, acc, val, key->rr); /* acc = a * RR / R mod n */
+	/* retain scaled version for intermediate use */
+	memcpy(a_scaled, acc, key->len * sizeof(a_scaled[0]));
+
+	for (j = k - 2; j > 0; --j) {
+		montgomery_mul(key, tmp, acc, acc); /* tmp = acc^2 / R mod n */
+
+		if (is_public_exponent_bit_set(key, j)) {
+			/* acc = tmp * val / R mod n */
+			montgomery_mul(key, acc, tmp, a_scaled);
+		} else {
+			/* e[j] == 0, copy tmp back to acc for next operation */
+			memcpy(acc, tmp, key->len * sizeof(acc[0]));
+		}
+	}
+
+	/* the bit at e[0] is always 1 */
+	montgomery_mul(key, tmp, acc, acc); /* tmp = acc^2 / R mod n */
+	montgomery_mul(key, acc, tmp, val); /* acc = tmp * a / R mod M */
+	memcpy(result, acc, key->len * sizeof(result[0]));
+
+	/* Make sure result < mod; result is at most 1x mod too large. */
+	if (greater_equal_modulus(key, result))
+		subtract_modulus(key, result);
+
+	/* Convert to bigendian byte array */
+	for (i = key->len - 1, ptr = inout; (int)i >= 0; i--, ptr++)
+		put_unaligned_be32(result[i], ptr);
+
+	return 0;
+}
+
+/*
+ * These arrays have been declared here, to be in the same
+ * context as Uboot RSA code, because the original code
+ * declares them on the stack.
+ */
+static uint32_t key1[RSA3072_DWORDS], key2[RSA3072_DWORDS];
+static uint32_t buf_mod_exp[RSA3072_DWORDS];
+
+int rsa_mod_exp_sw(const uint8_t *sig, uint32_t sig_len,
+		struct key_prop *prop, uint8_t *out)
+{
+	struct rsa_public_key key;
+	int ret;
+
+	if (!prop) {
+		return -EBADF;
+	}
+	key.n0inv = prop->n0inv;
+	key.len = prop->num_bits;
+
+	if (!prop->public_exponent)
+		key.exponent = RSA_DEFAULT_PUBEXP;
+	else {
+		rsa_convert_big_endian((uint32_t *)&key.exponent,
+				       prop->public_exponent, 2);
+	}
+
+	if (!key.len || !prop->modulus || !prop->rr) {
+		return -EFAULT;
+	}
+
+	/* Sanity check for stack size */
+	if (key.len > RSA_MAX_KEY_BITS || key.len < RSA_MIN_KEY_BITS) {
+		return -EFAULT;
+	}
+	key.len /= sizeof(uint32_t) * 8;
+
+	/*
+	 * Uboot RSA declares the below arryas on stack.
+	 * These are made global to minimise stack usage.
+	 *
+	 * uint32_t key1[key.len], key2[key.len];
+	 */
+
+	key.modulus = key1;
+	key.rr = key2;
+
+	rsa_convert_big_endian(key.modulus, (uint32_t *)prop->modulus, key.len);
+	rsa_convert_big_endian(key.rr, (uint32_t *)prop->rr, key.len);
+
+	if (!key.modulus || !key.rr) {
+		return -ENOMEM;
+	}
+
+	/*
+	 * Uboot RSA delares this array on stack.
+	 * Make this global, to reduce stack usage.
+	 *
+	 * uint32_t buf[sig_len / sizeof(uint32_t)];
+	 */
+
+	memcpy(buf_mod_exp, sig, sig_len);
+
+	ret = pow_mod(&key, buf_mod_exp);
+	if (ret)
+		return ret;
+
+	memcpy(out, buf_mod_exp, sig_len);
+
+	return 0;
+}

+/*

+#include <rsa-mod-exp.h>
+#include <rsa.h>
+#include <sha512.h>
+#include <string.h>
+
+/*
+ * This array is declared as global, to minimise
+ * the usage of stack
+ */
+static uint8_t buf_verify[RSA3072_BYTES];
+
+/**
+ * Based on Original function - rsa_verify_padding()
+ * from Uboot in the same file
+ *
+ * Verify a RSA message's padding is consistent with PKCS1.5
+ * padding as described in the RSA PKCS#1 v2.1 standard.
+ *
+ * @msg:	Padded message
+ * @pad_len:	Number of expected padding bytes
+ * @return 0 on success, != 0 on failure
+ */
+static int rsa_verify_padding(const uint8_t *msg, const int pad_len)
+{
+	int ff_len;
+	int ret;
+
+	/* first byte must be 0x00 */
+	ret = *msg++;
+	/* second byte must be 0x01 */
+	ret |= *msg++ ^ 0x01;
+	/* next ff_len bytes must be 0xff */
+	ff_len = pad_len - SHA384_DER_LEN - 3;
+	ret |= *msg ^ 0xff;
+	ret |= memcmp(msg, msg+1, ff_len-1);
+	msg += ff_len;
+	/* next byte must be 0x00 */
+	ret |= *msg++;
+	/* next der_len bytes must match der_prefix */
+	ret |= memcmp(msg, sha384_der_prefix, SHA384_DER_LEN);
+
+	return ret;
+}
+
+int padding_pkcs_15_verify(uint8_t *msg, int msg_len,
+			   const uint8_t *hash,
+		__attribute__ ((unused))int hash_len)
+{
+	int ret, pad_len = msg_len - SHA384_SUM_LEN;
+
+	/* Check pkcs1.5 padding bytes. */
+	ret = rsa_verify_padding(msg, pad_len);
+	if (ret) {
+		return -EINVAL;
+	}
+
+	/* Check hash. */
+	if (memcmp((uint8_t *)msg + pad_len, hash, msg_len - pad_len)) {
+		return -EACCES;
+	}
+
+	return 0;
+}
+
+/**
+ * Based on Original function from U-boot -
+ * rsa_verify_key() - Verify a signature against some data using RSA Key
+ *
+ * Verify a RSA PKCS1.5 signature against an expected hash using
+ * the RSA Key properties in prop structure.
+ *
+ * @prop:	Specifies key
+ * @sig:	Signature
+ * @sig_len:	Number of bytes in signature
+ * @hash:	Pointer to the expected hash
+ * @key_len:	Number of bytes in rsa key
+ * @return 0 if verified, -ve on error
+ */
+static int rsa_verify_key(struct key_prop *prop, const uint8_t *sig,
+			  const uint32_t sig_len, const uint8_t *hash,
+			  const uint32_t key_len)
+{
+	int ret;
+	int hash_len;
+
+	if (!prop || !sig || !hash)
+		return -EIO;
+
+	if ((int)sig_len != (prop->num_bits / 8)) {
+		return -EINVAL;
+	}
+
+
+	/* Sanity check for stack size */
+	if (sig_len > RSA_MAX_SIG_BITS / 8) {
+		return -EINVAL;
+	}
+
+	/*
+	 * Uboot RSA code declares this static array on the stack.
+	 * Move this to global, to minimise usage of stack
+	 *
+	 * uint8_t buf[sig_len];
+	 */
+	hash_len = SHA384_SUM_LEN;
+
+	ret = rsa_mod_exp_sw(sig, sig_len, prop, buf_verify);
+	if (ret) {
+		return ret;
+	}
+
+	ret = padding_pkcs_15_verify(buf_verify, key_len, hash, hash_len);
+	if (ret) {
+		return ret;
+	}
+
+	return 0;
+}
+
+/**
+ * Based on Original function from U-boot -
+ * rsa_verify_with_pkey() - Verify a signature against some data using
+ * only modulus and exponent as RSA key properties.
+ * @info:	Specifies key information
+ * @key:	Pointer to the key
+ * @key_len:	Length of the key
+ * @hash:	Pointer to the expected hash
+ * @sig:	Signature
+ * @sig_len:	Number of bytes in signature
+ *
+ * Return	0 if verified, -ve on error
+ */
+
+/*
+ * Packing of public key and key properties is done as below -
+ *
+ * All the sizes mentioned are fixed for RSA3072 encryption.
+ *  ------------------------
+ * |  Public key (384 bytes) |
+ *  ------------------------
+ * |    r^2 (384 bytes)      |
+ *  ------------------------
+ * |    modulus (384 bytes)  |
+ *  ------------------------
+ * |   exponent (8 bytes)    |
+ *  ------------------------
+ * |   n0 Inverse (4 bytes)  |
+ *  ------------------------
+ * |   num bits (4 bytes)    |
+ *  ------------------------
+ * |   key header (8 bytes)  |
+ *  ------------------------
+ */
+int rsa_verify_with_pkey(uint8_t *key,
+		 	__attribute__ ((unused))uint32_t key_len,
+			 const void *hash, uint8_t *sig, uint32_t sig_len)
+{
+	struct key_prop prop;
+	int ret;
+	uint8_t *hl_props = (uint8_t *)(key + RSA3072_BYTES);
+
+	/*
+	 * Uboot RSA code calls the rsa_gen_key_prop function here,
+	 * to extract the public key properties. Since we already precomputed
+	 * them, and appened them to the public key in the above format,
+	 * we skip calling this function here.
+	 *
+	 * Public key is self-described to fill key_prop
+	 * ret = rsa_gen_key_prop(key, key_len, &prop);
+	 * if (ret) {
+	 * 	return ret;
+	 * }
+	 */
+
+	prop.rr = hl_props;
+	hl_props += RSA3072_BYTES;
+
+	prop.modulus = hl_props;
+	hl_props += RSA3072_BYTES;
+
+	prop.public_exponent = hl_props;
+	prop.exp_len = 8;
+	hl_props += 8;
+
+	prop.n0inv = *(uint32_t *)(hl_props);
+	hl_props += 4;
+
+	prop.num_bits = *(uint32_t *)(hl_props);
+
+	ret = rsa_verify_key(&prop, sig, sig_len, hash, RSA3072_BYTES);
+
+	return ret;
+}
+
+int rsa_verify_hash(uint8_t *key, uint32_t key_len,
+		    const uint8_t *hash, uint8_t *sig, uint32_t sig_len)
+{
+	int ret;
+
+	ret = rsa_verify_with_pkey(key, key_len, hash, sig, sig_len);
+
+	return ret;
+}
+
+/**
+ * Based on Original function from Uboot -
+ * int rsa_verify(struct image_sign_info *info,
+ *        const struct image_region region[], int region_count,
+ *        uint8_t *sig, uint32_t sig_len)
+ */
+int rsa_verify(const uint8_t *data, uint32_t data_len,
+		uint8_t *key, uint32_t key_len,
+		uint8_t *sig, uint32_t sig_len)
+{
+	/* Reserve memory for maximum checksum-length */
+	uint8_t hash[SHA384_SUM_LEN];
+
+	/*
+	 * Verify that the checksum-length does not exceed the
+	 * rsa-signature-length
+	 */
+	sha384_csum(data, data_len, hash, 0);
+
+	return rsa_verify_hash(key, key_len, hash, sig, sig_len);
+}
