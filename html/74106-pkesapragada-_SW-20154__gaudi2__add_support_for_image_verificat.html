[SW-20154] gaudi2: add support for image verification

This patch enables RSA3072 verification of images using
Uboot APIs and SHA384 checksum. In Gaudi2, there are only
two possible locations for Keys - eFUSE and ROM. Hence
the key hash checking is modified accordingly.

+static int security_check_key_revoke_status(uint32_t key_index)
+{
+	uint8_t max_key_index = NUM_KEYS_IN_EFUSE, key_revoke;
+
+	/*
+	 * Key index ranges from:
+	 * 0 - 4: if key location == ROM (1)
+	 * 0 - 2: if key location == eFUSE (0)
+	 */
+	/* check efuse for key location */
+	if ((arcboot_rom_global.config_fuse >> EFUSE_SYSCFG_KEY_LOCATION) &
+			EFUSE_SYSCFG_KEY_LOCATION_MASK)
+		max_key_index = NUM_KEYS_IN_ROM;
+
+	if (key_index >= max_key_index)
+		return -1;
+
+	key_revoke = sp_sram_readl(SECURITY_EFUSE_KEYS_REVOKE_OFFSET);
+
+	/* if the key is not revoked, return 1 */
+	if ((key_revoke & BIT(key_index)) == 0)
+		return 1;
+
+	/* no valid key found */
+	return -1;
+}
