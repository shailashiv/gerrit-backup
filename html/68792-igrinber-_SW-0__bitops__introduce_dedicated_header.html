[SW-0] bitops: introduce dedicated header

As a preparation for the Gaudi HBM code addition, we need to introduce
the ffs() function for finding set bits.
This already gives a rationale for a dedicated bitops header.
Introduce such a header and move all the relevant bit macros to the new
header.
This also remove the duplication of the EXTRACT_BITS() macro.

+#ifndef _BITOPS_H_
+#define _BITOPS_H_
+
+#define BIT(nr)		(1 << (nr))
+
+/* extracts "n" number of bits from position "p" in the given number "v" */
+#define EXTRACT_BITS(v, p, n) (((v) & (((1 << (n)) - 1) << (p))) >> (p))
+
+
+#ifndef __ASSEMBLY__
+
+/*
+ * __ffs32: find first bit set. This is defined the same way as
+ * the libc and compiler builtin ffs routines, therefore
+ * differs in spirit from the above ffz (man ffs).
+ */
+
+static inline int __ffs32(unsigned int x)
+{
+	int r = 1;
+
+	if (!x)
+		return 0;
+
+	if (!(x & 0xffff)) {
+		x >>= 16;
+		r += 16;
+	}
+	if (!(x & 0xff)) {
+		x >>= 8;
+		r += 8;
+	}
+	if (!(x & 0xf)) {
+		x >>= 4;
+		r += 4;
+	}
+	if (!(x & 3)) {
+		x >>= 2;
+		r += 2;
+	}
+	if (!(x & 1)) {
+		x >>= 1;
+		r += 1;
+	}
+	return r;
+}
+
+#endif /* __ASSEMBLY__ */
+
+#endif /* _BITOPS_H_ */
