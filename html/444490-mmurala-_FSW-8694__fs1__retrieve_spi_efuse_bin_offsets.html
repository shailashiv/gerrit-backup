Commit Message

[FSW-8694] fs1: retrieve spi efuse bin offsets

SPI Efuse has separate manifest, but instead of
locating it from constant SPI offset a separate
img descriptor is used inside main manifest.
Once fake fuse image descriptor is located from
main manifest, desc offset used for spi dma copy
spi efuse manifest to iccm.
Each img desc inside spi efuse manifest used to
locate spi efuse binaries.

+	spi_manifest_consts.dest_addr_arc = SPI_MANIFEST_IMG_2_DMA_DEST_ARC;
+
+	fake_fuse_base = boot_flow_get_spi_efuse_manifest_offset(manifest_idx);
+	if (fake_fuse_base == 0) {
+		boot_rom_status_update(PROGRESS_INIT_EFUSE_SPI_IMG_DESC_FAIL);
+		return BOOT_ROM_FAIL;
+	}
+
+	fake_fuse_boot_ptr_size = spi_flash_readl(fake_fuse_base);
+	if (fake_fuse_boot_ptr_size > HL_IMG_PTRS_MAX_SIZE) {
+		boot_rom_status_update(PROGRESS_INIT_EFUSE_SPI_EXCEEDS_MAX_SIZE);
+		return BOOT_ROM_FAIL;
+	}
+
+	/* Fetch boot desc using DMA into dest offset */
+	ret = boot_rom_spi_dma(fake_fuse_boot_ptr_size, fake_fuse_base,
+		spi_manifest_consts.dest_addr_dma, false, false, 0);
+
+	if (ret != DMA_SUCCESS) {
+		boot_rom_status_update(PROGRESS_INIT_EFUSE_SPI_DMA_COPY_FAIL);
+		return BOOT_ROM_FAIL;
+	}
+
+	memcpy(&fake_fuse_boot_ptr, (uint32_t *)(spi_manifest_consts.dest_addr_arc),
+		sizeof(struct hl_fw_boot_ptrs));
+
+	if (fake_fuse_boot_ptr.magic.val != (FAKE_FUSE_BOOT_PTR_MAGIC_NAME | FW_SPI_BOOT_PTR_SUB)) {
+		boot_rom_status_update(PROGRESS_INIT_EFUSE_SPI_MAGIC_FAIL);
+		return BOOT_ROM_FAIL;
+	}
+
+	offset_array_base = (uint32_t *)(spi_manifest_consts.dest_addr_arc +
+		fake_fuse_boot_ptr.offset2array);
+
+	if ((offset_array_base == NULL) || (fake_fuse_boot_ptr.count == 0)) {
+		boot_rom_status_update(PROGRESS_INIT_EFUSE_SPI_DESC_COUNT_FAIL);
+		return BOOT_ROM_FAIL;
+	}
+
+	if (die_id_index == MAX_DIE_COUNT) {
+		for(index = 0; index < fake_fuse_boot_ptr.count; index++) {
+			fake_fuse_img_desc = (struct hl_fw_img_desc *)(spi_manifest_consts.dest_addr_arc +
+				offset_array_base[index]);
+
+			/* Fake fuse bin order controlled by enum fake_fuse_bin_index */
+			if (((fake_fuse_img_desc->magic.type - 1) != (uint8_t)index) ||
+				(fake_fuse_img_desc->magic.type > MAX_DIE_COUNT))
+				return BOOT_ROM_FAIL;
+
+			fake_fuse_offset_array[index] = fake_fuse_base + fake_fuse_img_desc->offset;
+		}

+			offset_array_base[die_id_index]);
+		fake_fuse_offset_array[die_id_index] = fake_fuse_base + fake_fuse_img_desc->offset;
+	}
+
+	return BOOT_ROM_SUCCESS;
