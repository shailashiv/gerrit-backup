+#define BITS_IN_REG			32
+#define GAUDI2_HBW_SHORT_MASK		0xF
+#define GAUDI2_MAX_HBW_SHORT_RR		6
+#define GAUDI2_HBW_WRITE_ONLY_RR_IDX	8
+
+static void hl_secure_update_rr_reg_hbw(int rr_idx, u32 inst_offset,
+		struct hbw_rr_reg *rr_reg)
+{
+	u32 rr_base, rr_dest;
+	u32 data_max_hi = ((u32)(rr_reg->data_max >> BITS_IN_REG));
+	u32 data_min_hi = ((u32)(rr_reg->data_min >> BITS_IN_REG));
+	bool write_only = rr_idx >= GAUDI2_HBW_WRITE_ONLY_RR_IDX;
+
+	if (rr_idx < GAUDI2_MAX_HBW_SHORT_RR) {
+		data_max_hi &= GAUDI2_HBW_SHORT_MASK;
+		data_min_hi &= GAUDI2_HBW_SHORT_MASK;
+	}
+
+	rr_base = rr_reg->rr_idx_calc * sizeof(u32) + inst_offset;
+	rr_dest = rr_base + rr_reg->reg_min.lo;
+	regmap_writel((u32)rr_reg->data_min, rr_dest);
+	rr_dest = rr_base + rr_reg->reg_min.hi;
+	regmap_writel(data_min_hi, rr_dest);
+	rr_dest = rr_base + rr_reg->reg_max.lo;
+	regmap_writel((u32)rr_reg->data_max, rr_dest);
+	rr_dest = rr_base + rr_reg->reg_max.hi;
+	regmap_writel(data_max_hi, rr_dest);
+
+	hl_secure_update_hit_reg(rr_idx, inst_offset, &rr_reg->hit, write_only);
+}
+
+static void hl_secure_hbw_rr_gen(int rr_idx, u64 min, u64 max,
+		int dist, enum hl_sec_engine_type type)
+{
+	int i;
+	struct hbw_rr_reg rr_reg;
+	struct hl_sec_rrs_cfg_hbw *cfg = &hl_sec_hbw_rrs_tbl[type];
+
+	rr_reg.data_min = min >> SHIFT_4K_POS;
+	rr_reg.data_max = max >> SHIFT_4K_POS;
+
+	if (rr_idx < GAUDI2_MAX_HBW_SHORT_RR) {
+		rr_reg.rr_idx_calc = rr_idx;
+		rr_reg.reg_min = cfg->min_short;
+		rr_reg.reg_max = cfg->max_short;
+	} else {
+		rr_reg.rr_idx_calc = rr_idx - GAUDI2_MAX_HBW_SHORT_RR;
+		rr_reg.reg_min = cfg->min_full;
+		rr_reg.reg_max = cfg->max_full;
+	}
+
+	rr_reg.reg_min.lo += dist;
+	rr_reg.reg_min.hi += dist;
+	rr_reg.reg_max.lo += dist;
+	rr_reg.reg_max.hi += dist;
+	rr_reg.hit.hit_wr = cfg->hit_wr_breg + dist;
+	rr_reg.hit.hit_rd = cfg->hit_rd_breg + dist;
+
+	for (i = 0; i < cfg->instances_num; i++)
+		hl_secure_update_rr_reg_hbw(rr_idx, cfg->instances_dist * i, &rr_reg);
+}
