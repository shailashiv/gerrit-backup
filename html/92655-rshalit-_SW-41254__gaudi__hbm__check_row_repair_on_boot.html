+static void hl_hbm_row_repair_flash_list(void)
+{
+	int err;
+	struct rma_table *table = NULL;
+	size_t size = 0;
+	u8 *p;
+
+	err = rma_table_read(RMA_HBM_ROW_REPLACEMENT_TABLE, &table, &size);
+	if (err || (size == 0))
+		return; /* no row replacement info, nothing to be done */
+
+	for (p = table->data; p < &table->data[size]; p += table->header.entry_size) {
+		struct rma_hbm_row_replacement_entry *entry =
+			(struct rma_hbm_row_replacement_entry *)p;
+
+		gaudi_hbm_row_repair(entry->device, entry->pc, entry->sid,
+				     entry->bank, entry->row);
+	}
+
+	rma_table_free(table, size);
+}
