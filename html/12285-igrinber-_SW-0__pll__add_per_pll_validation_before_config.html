Commit Message

[SW-0] pll: add per pll validation before config

Add a device hook for controlling the PLL configuration enablement.
This enables a device to make the configuration decision prior to
configurring any PLL from its list.
Implement the hook for each device.
For Goya and Gaudi, do not permit the CPU PLL initialization.
For Gaudi, do not permit if already done.
For Onix and PLDM, there are no PLLs, so always restrict.

+bool hl_pll_is_configurable(struct hl_pll_cfg *cfg)
+{
+	enum hl_pll_bypass_val bval;
+
+	switch (cfg->base) {
+	case mmMESH_W_PLL_BASE:
+	case mmMESH_E_PLL_BASE:
+	case mmIF_W_PLL_BASE:
+	case mmIF_E_PLL_BASE:
+	case mmPSOC_PCI_PLL_BASE:
+		/* DIV0 is the undivided PLL output */
+		bval = hl_pll_bypass_get(cfg->base, HL_PLL_DIV0);
+		if (bval == HL_PLL_PLL_CLK || bval == HL_PLL_DIV_PLL)
+			return false;
+		break;
+	case mmPSOC_CPU_PLL_BASE:
+		/* No CPU PLL configuration while running on it */
+		return false;
+	default:
+		break;
+	}
+
+	return true;
+}
