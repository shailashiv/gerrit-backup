Commit Message

board: habana: add remote_cleanup script

The script should be installed on the lab host at:
/var/lib/tftpboot/tests
It is used by automatic test frameworks (e.g. tbot, jenkins)
for connecting to the PCIe host, uninstalling the habanalabs driver
and cleaning up all the leftovers.
This is needed for clean habana packages install and for the tbot
testing framework to not have false positives from starting the Linux
kernel on target board (Dali) multiple times.

+#!/bin/bash -l
+
+host=$1
+habana_pkg_lst="habanalabs habanatools libhabana"
+result=~/remote_cleanup.result
+
+# TODO: this script can share code with the other scripts of this family.
+
+function local_msg() {
+	local msg=$1
+	echo -e   "********************************************************"
+	echo -e   "$msg"
+	echo -e   "********************************************************"
+}
+
+function remote_purge_leftovers() {
+	local_msg "****************** Purging leftovers *******************"
+	DKMS_HABANA_DIR=/var/lib/dkms/habanalabs
+	ssh -t $host "HABANA_KO=\"/lib/modules/\`uname -r\`/updates/dkms/habanalabs.ko\"
+			if [ \`lsmod | grep -ci habana\` -ne 0 ]; then
+				echo habanalabs is still inserted...;
+				sudo modprobe -r habanalabs;
+			fi;
+			if [ -e $DKMS_HABANA_DIR ]; then
+				echo $DKMS_HABANA_DIR is still present...;
+				sudo rm -fr $DKMS_HABANA_DIR;
+			fi
+			if [ -e $HABANA_KO ]; then
+				sudo rm -f $HABANA_KO;
+			fi;" | tee -a $result
+
+	local pipestat=${PIPESTATUS[0]}
+	echo ssh purge leftovers status: $pipestat
+	return $pipestat;
+}
+
+function remote_remove_packages() {
+	local_msg "************* Removing old habana packages *************"
+	ssh -t $host "sudo apt-get purge --auto-remove $habana_pkg_lst -y" | tee -a $result
+	local pipestat=${PIPESTATUS[0]}
+	echo ssh apt-get purge status: $pipestat
+	if [ $pipestat -ne 0 ]; then
+		return $pipestat;
+	fi
+
+	ssh -t $host "sudo apt-get clean" | tee -a $result
+	pipestat=${PIPESTATUS[0]}
+	echo ssh apt-get clean status: $pipestat
+	return $pipestat;
+}
+
+for ((i=100; i>0; i--)); do
+	echo $i
+	sleep 1;
+	ping -c 1 -W 1 $host
+	if [ $? -eq 0 ]; then
+		rm -f $result
+		ssh -t $host "echo booted"	|| continue;
+		remote_remove_packages		|| break;
+		remote_purge_leftovers		|| break;
+		echo done;
+		exit 0;
+	fi
+done
+
+echo error
+exit 1;
