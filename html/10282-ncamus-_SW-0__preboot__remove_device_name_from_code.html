+/*
+ * SSI device Chip Select (CS0) is muxed with GPIO53
+ * Using this we can realy do SSI bus claim/release
+ * GPIO53 is set through mmPSOC_GPIO3 where each
+ * mmPSOC_GPIO serve 16 GPIOs
+ * Note: CS0 is active low
+ */
+static void spi_ssi_cs0_init(void)
+{
+	unsigned int reg, mask = BIT_MASK(53 % 16);
+
+	reg  = readl(mmPSOC_GPIO3_DATAOUT);
+	reg  &= ~mask;
+	writel(reg,  mmPSOC_GPIO3_DATAOUT);
+
+	reg = readl(mmPSOC_GPIO3_OUTENSET);
+	reg |=  mask;
+	writel(reg, mmPSOC_GPIO3_OUTENSET);
+}
+
+static void spi_claim_bus(void)
+{
+	writel(1, mmPSOC_GLOBAL_CONF_PAD_SEL_53);
+}
+
+static void spi_release_bus(void)
+{
+	writel(0, mmPSOC_GLOBAL_CONF_PAD_SEL_53);
+}
+
+static inline void spi_dev_update_frame_size(unsigned int len)
+{
+	writel(0, mmPSOC_SPI_SSIENR);
+	writel(frame_size_bits((len * 8) - 1), mmPSOC_SPI_CTRLR0);
+	writel(1, mmPSOC_SPI_SSIENR);
+}
+
+static inline int spi_fifo_is_empty(unsigned long level_reg)
+{
+	return !readl(level_reg);
+}
+
+static void spi_dev_dword_cmd(unsigned int cmd, unsigned int *val)
+{
+	unsigned int tmp;
+
+	while (!spi_fifo_is_empty(mmPSOC_SPI_TXFLR))
+		;
+	writel(cmd, mmPSOC_SPI_DR0);
+
+	while (spi_fifo_is_empty(mmPSOC_SPI_RXFLR))
+		;
+	tmp = readl(mmPSOC_SPI_DR0);
+
+	if (val)
+		*val = tmp;
+}
+
+static void spi_special_cmd(unsigned int cmd, unsigned char *val)
+{
+	unsigned char tmp;
+
+	spi_dev_update_frame_size(sizeof(tmp));
+
+	while (!spi_fifo_is_empty(mmPSOC_SPI_TXFLR))
+		;
+	writeb(cmd, mmPSOC_SPI_DR0);
+
+	while (spi_fifo_is_empty(mmPSOC_SPI_RXFLR))
+		;
+	tmp = readb(mmPSOC_SPI_DR0);
+
+	if (val)
+		*val = tmp;
+}

+static void spi_special_cmd_csel(unsigned int cmd, unsigned char *val,
+								size_t size)
+{
+	int i;
+
+	spi_claim_bus();
+
+	spi_dev_update_frame_size(sizeof(*val));
+
+	while (!spi_fifo_is_empty(mmPSOC_SPI_TXFLR))
+		;
+	writeb(cmd, mmPSOC_SPI_DR0);
+	for (i = 0; i < size; i++) {
+		while (!spi_fifo_is_empty(mmPSOC_SPI_TXFLR))
+			;
+		writeb(val[i], mmPSOC_SPI_DR0);
+	}
+
+	while (spi_fifo_is_empty(mmPSOC_SPI_RXFLR))
+		;
+	readb(mmPSOC_SPI_DR0);
+	for (i = 0; i < size; i++) {
+		while (spi_fifo_is_empty(mmPSOC_SPI_RXFLR))
+			;
+		readb(mmPSOC_SPI_DR0);
+	}
+
+	spi_release_bus();
+}

+static void spi_common_write(unsigned int cmd, unsigned char *buf, size_t size)
+{
+	unsigned char sr;
+
+	spi_char_cmd_csel(CMD_WRITE_ENABLE, NULL);
+	spi_special_cmd_csel(cmd, buf, size);
+	do
+		spi_char_cmd_csel(CMD_READ_STATUS, &sr);
+	while (sr & READ_STATUS_WIP);
+	spi_char_cmd_csel(CMD_WRITE_DISABLE, NULL);

+static unsigned char spi_prot(void)
+{
+	unsigned char sr, cr, data[2];
+
+	spi_char_cmd_csel(CMD_READ_CONF, &cr);
+	spi_char_cmd_csel(CMD_READ_STATUS, &sr);
+	sr |= PROT_MASK;
+	data[0] = sr;
+	data[1] = cr;
+	spi_common_write(CMD_WRITE_STATUS, data, sizeof(data));
+	return sr;
+}

+static void spi_read_spl(unsigned int offset)
+{
+	void *dst = (void *)CONFIG_SPL_TEXT_BASE;
+	unsigned int cmd = (CMD_READ << 24) | offset;
+	int i;
+
+	spi_claim_bus();
+	spi_dev_update_frame_size(sizeof(cmd));
+	spi_dev_dword_cmd(cmd, NULL);
+	for (i = 0; i < SPL_SIZE; i += sizeof(unsigned int))
+		spi_dev_dword_cmd(0, dst + i);
+	spi_dev_dword_cmd(0, NULL);
+	spi_release_bus();
+}

+	writel(0x00000000, mmPSOC_SPI_RXFTLR);

+void preboot_puts_raw(const char *str)

+/*
+ * preboot_byte2char	- convert half byte to char.
+ *
+ * This function takes a byte number and converts its value to
+ * a character representing the value (e.g. 0x8c can be converted to
+ * '8' or 'c').
+ */
+static inline char preboot_byte2char(u8 byte, bool upper)
+{
+	/* upper or lower 4 bits of the byte */
+	byte = upper ? byte >> 4 : byte & 0xf;
+
+	/*
+	 * convert: if half byte is a number [0..9] => use '0' as an offset
+	 *	    if half byte is a letter [0xa..0xf] => use 'a' as a base,
+	 *	    but since 'a' is not 0, we need to subtract
+	 *	    its base offset - 0xa.
+	 */
+	return byte + ((byte >= 0xa) ? 'a' - 0xa : '0');
+}
+
+/*
+ * preboot_uint2hexstring	- convert uint to a hex values string.
+ *
+ * Take a number and provide its hexadecimal string representation.
+ */
+int preboot_uint2hexstring(char *str, unsigned int num, bool pad)
+{
+	int j = 0;
+	char tmp;
+
+	/*
+	 * Here we walk on each byte of uint num in the reverse order, so
+	 * we build the string in the correct order.
+	 */
+	for (int i = 3; i >= 0; i--) {
+		u8 byte = (u8)((num >> (i * 8)) & 0xff);
+		/* here we run the loop on 2 parts of byte: upper and lower */
+		for (int k = 0; k < 2; k++) {
+			tmp = preboot_byte2char(byte, !k);
+			if (tmp == '0')	/* care of the leading zeroes (pad) */
+				pad = true;
+			if (pad)
+				str[j++] = tmp;
+		}
+	}
+
+	/* for !pad, return how many characters were actually written */
+	return j;
+}
+
+void preboot_puth_raw(unsigned int val, bool pad)
+{
+	char str[STR_UINT_SZ + 1];
+	int val_sz;
+
+	val_sz = preboot_uint2hexstring(str, val, pad);
+	val_sz = val_sz > STR_UINT_SZ ? STR_UINT_SZ : val_sz;
+	str[val_sz] = '\0';
+	preboot_puts_raw(str);
+}
+
+void preboot_puth(unsigned int val, bool pad)
+{
+	preboot_puth_raw(val, pad);
+	preboot_puts("\n");
+}

+}

+{

+}

+	/* Wait for PRSTn */
+	while (!readl(mmPSOC_GLOBAL_CONF_PRSTN))

+	/* Enable FW FSM after PRSTn */
+	writel(1, mmPSOC_GLOBAL_CONF_PCI_FW_FSM);

+		unsigned int prstn, ltssm, smlh, rdlh, hot_reset;

+		prstn = readl(mmPSOC_GLOBAL_CONF_PRSTN);
+		ltssm = readl(mmPCIE_AUX_LTSSM_EN);
+		smlh = readl(mmPCIE_AUX_SMLH_LINK_UP);
+		rdlh = readl(mmPCIE_AUX_RDLH_LINK_UP);
+		hot_reset = readl(mmPCIE_CORE_HOT_RST);
+
+		if ((hot_reset & BIT(0)) == BIT(0)) {
+			writel(0x10, mmPCIE_CORE_HOT_RST);
+			writel(1, mmPCIE_AUX_LTSSM_EN);
+		}
+
+		if (prstn && smlh && rdlh)
+			break;
+
+		/*
+		 * Some host assert prstn during link up while moving to GEN3
+		 * This cause BAR size to be reset
+		 * We stop link up process by disabling ltssm
+		 * till BAR is reconfigured
+		 */
+		if (!prstn && ltssm) {
+			writel(0, mmPCIE_AUX_LTSSM_EN);
+			continue;
+		}
+
+		if (prstn && !ltssm) {
+			writel(1, mmPCIE_AUX_LTSSM_EN);
+			continue;
+		}
+	}

+	}
