Commit Message

[SW-39414] fit: validate image hash

Add validation of image hash to detect corrupted loadable images, but
more importantly, since signature validation is only done for hash
field and not its data, skipping this step caused a security hole where
an attacker may replace the data binary which does not go through the
signature validation process.

+static const u8 *fit_node_get_hash(const void *fit, int node)
+{
+	int hash;
+	const char *prop;
+
+	hash = fdt_subnode_offset(fit, node, FIT_HASH_PROP);
+	prop = (const char *)fdt_getprop(fit, hash, FIT_ALGO_PROP, NULL);
+	if (strcmp(prop, FIT_SHA256))
+		return NULL;
+
+	return (const u8 *)fdt_getprop(fit, hash, FIT_VALUE_PROP, NULL);
+}
+
+static int fit_node_check_hash(const void *fit, int node, const u8 *buf,
+			       int len)
+{
+	const u8 *node_hash = fit_node_get_hash(fit, node);
+	u8 calc_hash[SHA256_SUM_LEN];
+
+	if (!node_hash) {
+		preboot_puts("Missing hash for loadable\n");
+		return 1;
+	}
+
+	sha256_csum(buf, len, calc_hash);
+	if (memcmp(node_hash, calc_hash, SHA256_SUM_LEN))
+		return 1;
+
+	return 0;
+}
