Commit Message

[SW-50527] gaudi: security: add dbg protection

Add protection for the DBG areas (TPC related) that can be protected
with privilege protection bits.
For TPC_CFG and TPC QMAN, protect only the privilege protection bits
registers and the internal configuration registers areas.
The regular registers in these pages cannot do any harm.

+#define DBG_TPC_DIST	0x200000
+static void preboot_secure_dbg(void)
+{
+	int i;
+	u32 prot_base, base;
+
+	for (i = 0; i < TPCS_NUM; i++) {
+		base = DBG_TPC_DIST * i;
+
+		/* Protect DBG TPC_EML_CFG */
+		prot_base = (mmTPC0_EML_CFG_BASE & ~HL_CFG_BASE) + base;
+		set_prot_bits_range_all(prot_base, SNGL_REG_PROT_SIZE * 29);
+		regmap_writel(0, prot_base + 0xEF8);
+		/*
+		 * The Protection Bits register protecting the privilege bits
+		 * must be accessed the last due to apparent bug in h/w, which
+		 * according to observation changes (strips?) the privilege
+		 * level from or access transaction, so further accesses result
+		 * in exception.
+		 */
+		regmap_writel(0, prot_base + 0xEF4);
+
+		/* Protect DBG TPC_CFG - only Priv protection and int bits */
+		prot_base = (mmTPC0_EML_TPC_CFG_BASE & ~HL_CFG_BASE) + base;
+		regmap_writel(0, prot_base + 0xEF8);
+		/* same reason as above */
+		regmap_writel(0, prot_base + 0xEF4);
+
+		/* Protect DBG TPC QMAN - only Priv protection and int bits */
+		prot_base = (mmTPC0_EML_TPC_QM_BASE & ~HL_CFG_BASE) + base;
+		regmap_writel(0, prot_base + 0xEF8);
+		/* same reason as above */
+		regmap_writel(0, prot_base + 0xEF4);
+	}
+}
