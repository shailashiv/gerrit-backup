+void hl_ts_data_init(void)
+{
+	u32 pll_freq;
+	int i;
+
+	pll_freq = hl_pll_get_freq(mmDCORE0_XBAR_IF_PLL_CTRL_BASE & ~HL_CFG_BASE, HL_PLL_DIV3);
+
+	/* All TS instances use XBAR IF PLL, DIV3 clock */
+	for (i = 0; i < HL_ASIC_TS_MAX_COUNT; i++)
+		gaudi2_hl_ts_data[i].freq = pll_freq;
+
+	memcpy(hl_ts_data, gaudi2_hl_ts_data,
+			sizeof(struct hl_ts_cfg) * HL_ASIC_TS_MAX_COUNT);
+}
+
+void hl_ts_reset(void)
+{
+	preboot_puts("Performing RST of TS UNIT\n");
+	/* Perform unit reset for TS and TS_IF */
+	regmap_writel(PSOC_RESET_CONF_TS_UNIT_RST_EN_MASK,
+			mmPSOC_RESET_CONF_TS_UNIT_RST);
+	regmap_writel(PSOC_RESET_CONF_TS_IF_UNIT_RST_EN_MASK,
+			mmPSOC_RESET_CONF_TS_IF_UNIT_RST);
+	regmap_writel(1, mmPSOC_RESET_CONF_UNIT_RST_N);
+	udelay(1000);
+	regmap_writel(0, mmPSOC_RESET_CONF_TS_UNIT_RST);
+	regmap_writel(0, mmPSOC_RESET_CONF_TS_IF_UNIT_RST);
+	regmap_writel(1, mmPSOC_RESET_CONF_UNIT_RST_N);
+}
