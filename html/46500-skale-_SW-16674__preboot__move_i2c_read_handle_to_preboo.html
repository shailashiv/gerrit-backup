Commit Message

[SW-16674] preboot: move i2c read handle to preboot

Since i2c module is moved to common folder which will be used
by both preboot and recovery image, we need to make sure that
command request from I2C handled by preboot and recovery should
be only those which they are expected to handle. In order to achieve
this, i2cslave_handle_read_req function is moved to preboot only file.

At later part, i2cslave_handle_read_req will be added in recovery file
which handles its own commands.

+#include <preboot.h>
+#include <hl_i2c_slave_events.h>
+
+#include "thermal.h"
+#include "module_i2c_slave.h"
+
+int i2cslave_handle_read_req(struct module_i2c_data *data)
+{
+	int max_tmp, ret;
+
+	switch (data->cmd) {
+	case HL_ASIC_TEMP_CMD_CODE:
+	case HL_ASIC_TEMP_MAX_VAL:
+		max_tmp = hl_ts_get_max_temp();
+		/* Convert from mC to C */
+		ret = max_tmp / 1000;
+		break;
+	case HL_DEVICE_STATUS_CMD_CODE:
+		ret = HL_DEVICE_STATUS_CMD_CODE_LEN;
+		break;
+	case HL_DEVICE_STATUS:
+		ret = HL_DEVICE_STATUS_UBOOT;
+		break;
+	case HL_PCIE_VENDOR_ID_VAL_LOW:
+	case HL_PCIE_VENDOR_ID_VAL_HIGH:
+		ret = i2cslave_pcie_vendor_id_cmd(data->cmd +
+						data->read_bytes_count);
+		break;
+	default:
+		/* NA */
+		ret = 0;
+		break;
+	}
+
+	return ret;
+}
