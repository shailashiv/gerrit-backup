[SW-28857] i2c indirect: add control port

Add I2C indirect control port with the following functionality:
- Abort request
- Get API vesion
- Get OS state
- Reset device

The control port is accessed directly using I2C command 231 (control)
and 237 (status).

+u32 i2csi_ctrl_handle_command(u8 **data, u8 cnt)
+{
+	struct hl_i2ci_ctrl_cpsr *cpsr;
+
+	/* Command is expected */
+	if (!cnt) {
+		*data = ctrl_cmd.data;
+		return HL_I2CI_CTRL_CMD_LEN;
+	}
+
+	ctrl_sts.tag = ctrl_cmd.tag;
+	ctrl_sts.status = HL_I2CI_CTRL_STS_BUSY;
+	ctrl_sts.error = HL_I2CI_CTRL_ERR_NONE;
+	cpsr = &ctrl_sts.cpsr;
+
+	switch (ctrl_cmd.opcode) {
+	case HL_I2CI_CTRL_OP_ABORT:
+		i2csi_req_set_idle();
+		cpsr->abort.status = HL_I2CI_CTRL_ABORT_NONE;
+		break;
+	case HL_I2CI_CTRL_OP_GET_API_VER:
+		cpsr->version.major = HL_I2CI_API_VERSION_MAJOR;
+		cpsr->version.minor = HL_I2CI_API_VERSION_MINOR;
+		cpsr->version.patch = HL_I2CI_API_VERSION_PATCH;
+		break;
+	case HL_I2CI_CTRL_OP_GET_OS_STATE:
+		cpsr->os_state.pcie_access = HL_I2CI_CTRL_ACCESS_FULL;
+		cpsr->os_state.mgmt_access = HL_I2CI_CTRL_ACCESS_FULL;
+		cpsr->os_state.os_state = HL_I2CI_CTRL_OS_PREBOOT;
+		break;
+	case HL_I2CI_CTRL_OP_RESET_DEVICE:
+		/* Remain busy - action will take place from I2C SM */
+		return 0;
+	default:
+		ctrl_sts.error = HL_I2CI_CTRL_ERR_INVALID_OPCODE;
+	}
+
+	ctrl_sts.status = HL_I2CI_CTRL_STS_RESP_RDY;
+
+	return 0;
+}
+
+u8 i2csi_ctrl_handle_status(u8 cmd, u32 read_seq)
+{
+	register int idx = cmd + read_seq - HL_INDIRECT_CTRL_STS_DATA;
+	u8 ret;
+
+	if (idx >= HL_I2CI_CTRL_STS_LEN)
+		return 0;
+
+	ret = ctrl_sts.data[idx];
+
+	/* Read till the end - return to idle */
+	if (idx == HL_I2CI_CTRL_STS_LEN - 1)
+		memset(ctrl_sts.data, 0, HL_I2CI_CTRL_STS_LEN);
+
+	return ret;
+}
+
+bool i2csi_ctrl_is_rst_dev(void)
+{
+	return ctrl_sts.status == HL_I2CI_CTRL_STS_BUSY &&
+		ctrl_cmd.opcode == HL_I2CI_CTRL_OP_RESET_DEVICE;
+}
