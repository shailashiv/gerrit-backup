+static uint32_t arcboot_flash_image_offset(uint8_t image_ptr)

+	uint32_t flash_image_offset;

+	spi_flash_copy(&flash_image_switch, FLASH_IMAGE_SWITCH_OFFSET, 1, "EROM_TPM");

+		if (image_ptr != ARCBOOT_SECONDARY)

+	if (image_ptr == ARCBOOT_SECONDARY)

+static int erom_get_ppboot_data(uint32_t *offset, uint16_t *size)
+{
+	uint8_t image_ptr = ARCBOOT_SECONDARY + 1;
+	uint32_t flash_image_offset;
+	uint32_t flash_ppboot_pcie_offset;
+	uint32_t ppboot_pcie_offset;
+	uint32_t ppboot_pcie_size;
+
+	if (arcboot_rom_status_read(PROGRESS_BRINGUP_RAM_PRI))
+		image_ptr = ARCBOOT_PRIMARY;
+	else if (arcboot_rom_status_read(PROGRESS_BRINGUP_RAM_SEC))
+		image_ptr = ARCBOOT_SECONDARY;
+
+	if (image_ptr > ARCBOOT_SECONDARY) {
+		cpu_boot_err_set(CPU_BOOT_ERR_TPM_FAIL);
+		return -1;
+	}
+
+	flash_image_offset = arcboot_flash_image_offset(image_ptr);
+	/* check if offset is within safe limits */
+	if ((flash_image_offset + FLASH_64KB_PADD) >= FLASH_VALID_MASK) {
+		cpu_boot_err_set(CPU_BOOT_ERR_TPM_FAIL);
+		return -1;
+	}
+
+	flash_ppboot_pcie_offset = (uint32_t)(FLASH_64KB_PADD +
+					      flash_image_offset);
+	spi_flash_copy(&ppboot_pcie_size, flash_ppboot_pcie_offset, 1, "EROM_TPM");
+	if (ppboot_pcie_size > SECUREBOOT_MAX_PPBOOT_IMG_SIZE ||
+	    ppboot_pcie_size <= SECUREBOOT_IMG_HEAD_FOOTER) {
+		cpu_boot_err_set(CPU_BOOT_ERR_TPM_FAIL);
+		return -1;
+	}
+
+	/* check for safe limits */
+	if ((flash_ppboot_pcie_offset + ppboot_pcie_size) >= FLASH_VALID_MASK) {
+		cpu_boot_err_set(CPU_BOOT_ERR_TPM_FAIL);
+		return -1;
+	}
+
+	/* remove headers and footers from image, this is the image to verify */
+	ppboot_pcie_size -= SECUREBOOT_IMG_HEAD_FOOTER;
+	/* offset of ppboot+pcie_fw image to verify, start from fw_version */
+	ppboot_pcie_offset = (HL_FWSRAM_BASE + IMG_LEN_SIZE);
+
+	*offset = ppboot_pcie_size;
+	*size = ppboot_pcie_offset;
+
+	return 0;
+}
