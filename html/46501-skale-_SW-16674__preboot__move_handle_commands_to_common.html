+#include "cmd_module.h"

+void module_handle_cmds(struct cmd_module_ops cmd_ops[], int size)
+{
+	int i = 0, j = 0;
+	enum cmd_check_res res = NOT_FOUND;
+
+	for (j = 0; j < size; j++)
+		cmd_ops[j].setup();
+
+	preboot_puts("Waiting for commands...\n");
+	i = 0;
+	while (1) {
+		res = cmd_ops[i].check();
+		switch (res) {
+		case NOT_FOUND:
+			break;
+		case FOUND_FAST_RETURN:
+			cmd_ops[i].run_fast();
+			break;
+		case FOUND_SLOW_RETURN:
+			for (j = 0; j < size; j++)
+				if (i != j)
+					cmd_ops[j].disable();
+
+			cmd_ops[i].run();
+			for (j = 0; j < size; j++)
+				if (i != j)
+					cmd_ops[j].enable();
+
+			break;
+		case FOUND_NO_RETURN:
+			for (j = 0; j < size; j++)
+				if (i != j)
+					cmd_ops[j].teardown();
+
+			cmd_ops[i].run_noreturn();
+			break;
+		default:
+			break;
+		}
+
+		if (++i == size)
+			i = 0;
+	}
+}
