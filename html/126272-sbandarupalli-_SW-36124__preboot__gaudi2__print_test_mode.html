+/*
+ * TEST_GPIO(89) is set through mmPSOC_GPIO5 BIT(9)
+ * Test Mode Indication
+ * 1: Normal operation
+ * 0: ASIC/GPU enter into electrical compliance mode
+ */
+#define TEST_MODE_GPIO		9
+void preboot_read_test_mode(void)
+{
+	u32 val, mask = BIT(TEST_MODE_GPIO);
+
+	/* configure as input */
+	regmap_writel(mask, mmPSOC_GPIO5_OUTENCLR);
+
+	val  = regmap_readl(mmPSOC_GPIO5_DATA);
+
+	if (val & mask)
+		preboot_puts_raw("Test mode indication: Normal operation");
+	else
+		preboot_puts_raw("Test mode indication: Compliance mode");
+
+	preboot_puts("\n");
+}
