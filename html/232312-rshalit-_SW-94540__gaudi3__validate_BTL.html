+function handle_btl {
+	local INF=$1

+function create_spi_image {
+	local BTL_BIN=$1
+	local PREBOOT_BIN=$2
+	local IMG_PTR_BIN=$3
+	local OUTF=$4
+
+	# [0-5] BTL (pri,sec,prstn1,prstn2,pcie1,pcie2)
+	dd status=none conv=sync bs=64K		if=${BTL_BIN}		 > ${OUTF}
+	# [6] Image pointers ARM
+	dd status=none conv=sync bs=64K		if=${IMG_PTR_BIN}	>> ${OUTF}
+	# [7] U-Boot ENV
+	dd status=none conv=sync bs=64K count=1 if=64K-0xff.bin		>> ${OUTF}
+	# [8-11] Preboot1
+	dd status=none conv=sync bs=256K	if=${PREBOOT_BINFILE}	>> ${OUTF}
+	# [12-15] Preboot2
+	dd status=none conv=sync bs=256K	if=${PREBOOT_BINFILE}	>> ${OUTF}
+}
+
+function hl_chksum_add {
+	# As a part of image descriptor version 3, an additional checksum field
+	# was added to support sha384, just after the sha256 field. So this
+	# function populates a valid checksum for the relavent algorithm and
+	# pads the irrelavant one. For projects using sha256  the first field
+	# is populated with a valid checksum while the second field is padded.
+	# For projects using sha384 it is vice versa.
+	local pad_dwords=8
+	local checksum=$1
+
+	# Pad the sha256 checksum field
+	hl_chksum_padding ${pad_dwords}
+	hl_chksum_padding ${pad_dwords}
+
+	# Update the sha384 checksum field
+	echo $checksum
+	echo $checksum
+}
