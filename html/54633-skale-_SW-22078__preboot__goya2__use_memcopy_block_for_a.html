[SW-22078] preboot: goya2: use memcopy block for arc fit images

Currently on goya2 ASIC, inconsistent behaviour is observed while copying
ARC fit images to ICCM/DCCM regions from ARM and content is not seen as
expected on ARC ICCM/DCCM regions. Till HW team provide resolution to
this issue, this patch adds workaround and make use of memcopy block to
copy ARC ICCM/DCCM images from ARM as per below sequence.

1. Copy FIT image to scratchpad SRAM base region
2. Copy from scratchpad SRAM to ICCM/DCCM region using memcopy block.

The reason to copy to scratchpad SRAM (step 1) is the fit source address
for image may not be 16 bytes aligned and memcopy block has requirement
of source and destination address to be 16 bytes aligned.

+static void fit_copy_arc_image(u64 dst, const void *buf, int len)
+{
+	u32 mem_cfg;
+
+	/*
+	 * For ARC images:
+	 * 1. Copy FIT image to scratchpad SRAM base. This is required
+	 *    for using memcopy block as memcopy requires source and
+	 *    destination addressess to be 16 bytes aligned.
+	 * 2. Copy from scratchpad SRAM to ICCM/DCCM using memcopy block.
+	 */
+	memcpy((void *)HL_SP_SRAM_BASE, buf, len);
+
+	/* Set source and destination address in memcopy block */
+	regmap_writel((u32)dst, mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_L);
+	regmap_writel((u32)(dst >> 32), mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_H);
+	regmap_writel((u32)HL_SP_SRAM_BASE,
+			mmPSOC_GLOBAL_CONF_MEM_CPY_START_ADDR_L);
+	regmap_writel((u32)(HL_SP_SRAM_BASE >> 32),
+			mmPSOC_GLOBAL_CONF_MEM_CPY_START_ADDR_H);
+
+
+	/* Align length to 16 bytes */
+	if (len % 16)
+		len += (16 - (len % 16));
+
+	/* Set length in memcopy register in granuality of 16 bytes */
+	len = len >> 4;
+	mem_cfg = regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+	mem_cfg &= ~(PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_FLD_SIZE_MASK);
+	mem_cfg |= (len & PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_FLD_SIZE_MASK);
+	regmap_writel(mem_cfg, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+
+	/* Enable copy and wait for completion */
+	regmap_writel(0x1, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL);
+	while (!regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_STATUS))
+		;
