+	[MEM_FWUPD_PCI] = {

+	},
+	[MEM_I2C] = {
+		.base = MEM_I2C_BASE,
+		.size = MEM_I2C_SZ,
+	},
+};
+
+__weak int arcboot_get_dev_spec_mem_region(__unused enum mem_region region,
+					   __unused struct mem_desc *desc)
+{
+	return 1; /* not dev spec */
+}
+
+int arcboot_get_mem_region(enum mem_region region, struct mem_desc *desc)
+{
+	int err;
+
+	switch (region) {
+	case MEM_GP:
+	case MEM_LKD_PCI:
+	case MEM_FWUPD_PCI:
+	case MEM_I2C:
+		break;
+	default:
+		return 1;
+	}
+
+	if (!desc)
+		return 1;
+
+	/* device defined memory regions */
+	err = arcboot_get_dev_spec_mem_region(region, desc);
+	if (!err)
+		return err;
+
+	desc->base = mem_desc[region].base;
+	desc->size = mem_desc[region].size;
+
+	return 0;
+}
