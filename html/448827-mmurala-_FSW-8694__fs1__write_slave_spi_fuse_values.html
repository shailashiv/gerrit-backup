+void boot_rom_init_slave_spi_efuse()
+{
+	/* Align with enum fake_fuse_bin_index */
+	uint8_t die_id_arary[MAX_DIE_COUNT] = {FS1_IO_DIE_0, FS1_IO_DIE_1,
+		FS1_COMPUTE_DIE_0, FS1_COMPUTE_DIE_1, FS1_COMPUTE_DIE_2, FS1_COMPUTE_DIE_3};
+	uint32_t bank_offset, spi_offset, syscfg1_data, syscfg2_data;
+	uint8_t die_id_index;
+
+	/* Note: Index 0 master, no need to export. */
+	for(die_id_index = 1 ; die_id_index < MAX_DIE_COUNT; die_id_index++) {
+
+		spi_offset = fake_fuse_offset_array[die_id_index];
+
+		if (spi_offset == 0)
+			continue;
+
+		/* read value from bank0 */
+		bank_offset = (IO_EFUSE_SYSCFG_OFFSET * sizeof(uint32_t));
+		syscfg1_data = spi_flash_readl(spi_offset + bank_offset);
+
+		/* read value from bank1 and OR with value from bank0 */
+		bank_offset = ((EFUSE_WORDS_PER_BANK + IO_EFUSE_SYSCFG_OFFSET) * sizeof(uint32_t));
+		syscfg1_data |= spi_flash_readl(spi_offset + bank_offset);
+
+		/* read value from bank0 */
+		bank_offset = (IO_EFUSE_SYSCFG2_OFFSET * sizeof(uint32_t));
+		syscfg2_data = spi_flash_readl(spi_offset + bank_offset);
+
+		/* read value from bank1 and OR with value from bank0 */
+		bank_offset = ((EFUSE_WORDS_PER_BANK + IO_EFUSE_SYSCFG2_OFFSET) * sizeof(uint32_t));
+		syscfg2_data |= spi_flash_readl(spi_offset + bank_offset);
+
+		/* Notify slave config fuse value in slave scratchpad 0/1 register */
+		regmap_writel_inter_die(
+			IO0_H0_PSOC_ARC_MNG_LOCAL(mmARC_LOCAL_CONF_CFG_ADDR_MNPL_CTRL),
+			syscfg1_data,
+			DIE_BASED_ADDRESS((IO0_H0_PSOC_GLOBAL_CONF(mmGLOBAL_CONF_SCRATCHPAD_0)),
+				die_id_arary[die_id_index]));
+
+		regmap_writel_inter_die(
+			IO0_H0_PSOC_ARC_MNG_LOCAL(mmARC_LOCAL_CONF_CFG_ADDR_MNPL_CTRL),
+			syscfg2_data,
+			DIE_BASED_ADDRESS((IO0_H0_PSOC_GLOBAL_CONF(mmGLOBAL_CONF_SCRATCHPAD_1)),
+				die_id_arary[die_id_index]));
+
+		/* Indicate slave die about fuse value is available */
+		regmap_writel_inter_die(
+			IO0_H0_PSOC_ARC_MNG_LOCAL(mmARC_LOCAL_CONF_CFG_ADDR_MNPL_CTRL),
+			FFUSE_WRITE_IND_MAGIC,
+			DIE_BASED_ADDRESS((IO0_H0_PSOC_GLOBAL_CONF(mmGLOBAL_CONF_SCRATCHPAD_2)),
+				die_id_arary[die_id_index]));
+	}
+}
