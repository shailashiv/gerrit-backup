+bool boot_rom_pcie_gen_limit_enabled(void)
+{
+	if ((boot_rom_global.config_fuse & EFUSE_SYSCFG_PCIE_GEN_LIMIT) == 0)
+		return true;
+
+	return false;
+}
+
+void boot_rom_dbi_ro_write_en(bool enable)
+{
+	uint32_t val;
+
+	val = regmap_readl(IO0_H0_PCIE_DBI(mmPCIE_DBI_MISC_CONTROL_1_OFF));
+
+	if (enable)
+		val |= PCIE_DBI_MISC_CONTROL_1_OFF_DBI_RO_WR_EN_M;
+	else
+		val &= ~PCIE_DBI_MISC_CONTROL_1_OFF_DBI_RO_WR_EN_M;
+
+	regmap_writel(val, IO0_H0_PCIE_DBI(mmPCIE_DBI_MISC_CONTROL_1_OFF));
+}
+
+void boot_rom_shadow_reg_access_en(bool enable)
+{
+	uint32_t val = 0;
+
+	if (enable)
+		val = PCIE_AUX_DBI_CS2_M;
+
+	regmap_writel(val, IO0_H0_PCIE_AUX(mmPCIE_AUX_DBI));
+}
+
+#define BAR_SIZE_1GB			(SZ_1G - 1)
+void boot_rom_set_pf_cfg_bar(void)
+{
+	regmap_writel(BAR_SIZE_1GB, IO0_H0_PCIE_DBI(mmPCIE_DBI_BAR0_REG));
+	regmap_writel(0, IO0_H0_PCIE_DBI(mmPCIE_DBI_BAR1_REG));
+}
+
+void boot_rom_set_vf_cfg_bar(void)
+{
+	regmap_writel(BAR_SIZE_1GB, IO0_H0_PCIE_DBI(mmPCIE_DBI_SRIOV_BAR0_REG));
+	regmap_writel(0, IO0_H0_PCIE_DBI(mmPCIE_DBI_SRIOV_BAR1_REG));
+}
+
+void boot_rom_disable_alt_protocol(void)
+{
+	uint32_t val;
+
+	val = regmap_readl(IO0_H0_PCIE_DBI(mmPCIE_DBI_PL32G_CAPABILITY_REG));
+	val &= ~PCIE_DBI_PL32G_CAPABILITY_REG_MOD_TS_ALT_PROTOCOL_SUPPORT_M;
+	regmap_writel(val, IO0_H0_PCIE_DBI(mmPCIE_DBI_PL32G_CAPABILITY_REG));
+}
+
+#define SRIOV_MAX_NUM_VFS		0x4
+#define SRIOV_DEFAULT_NUM_VFS		0x1
+void boot_rom_set_number_of_vfs(uint32_t num_vfs)
+{
+	uint32_t val;
+
+	if (num_vfs > SRIOV_MAX_NUM_VFS)
+		num_vfs = SRIOV_DEFAULT_NUM_VFS;
+
+	val = num_vfs << PCIE_DBI_TOTAL_VFS_INITIAL_VFS_REG_SRIOV_INITIAL_VFS_S |
+		num_vfs << PCIE_DBI_TOTAL_VFS_INITIAL_VFS_REG_SRIOV_TOTAL_VFS_S;
+	regmap_writel(val, IO0_H0_PCIE_DBI(mmPCIE_DBI_TOTAL_VFS_INITIAL_VFS_REG));
+}
+
+void boot_rom_skip_ats_capability(void)
+{
+	uint32_t val;
+
+	val = regmap_readl(IO0_H0_PCIE_DBI(mmPCIE_DBI_SRIOV_BASE_REG));
+	val &= ~PCIE_DBI_SRIOV_BASE_REG_SRIOV_NEXT_OFFSET_M;
+	val |= mmPCIE_DBI_PRS_EXT_CAP_HDR_REG <<
+		PCIE_DBI_SRIOV_BASE_REG_SRIOV_NEXT_OFFSET_S;
+	regmap_writel(val, IO0_H0_PCIE_DBI(mmPCIE_DBI_SRIOV_BASE_REG));
+}
+
+#define LTTSM_EN_MASK		(PCIE_AUX_LTSSM_EN_IND_M | PCIE_AUX_LTSSM_EN_MASK_M)
+void boot_rom_pcie_set_ltssm_en(void)
+{
+	regmap_writel(LTTSM_EN_MASK, IO0_H0_PCIE_AUX(mmPCIE_AUX_LTSSM_EN));
+}

+// SPDX-License-Identifier:     GPL-2.0+
+/*
+ * Copyright (C) 2024 HabanaLabs Ltd.
+ * All Rights Reserved.

+
+#include <boot_rom.h>
+#include <efuse.h>
+#include <delay.h>
+#include <pcie.h>
+#include <status.h>
+
+#define NUM_VFS			1
+
+static void boot_rom_pcie_bar_config(void)
+{
+	boot_rom_shadow_reg_access_en(true);
+
+	boot_rom_set_pf_cfg_bar();
+
+	boot_rom_set_vf_cfg_bar();
+
+	boot_rom_shadow_reg_access_en(false);
+}
+
+static void boot_rom_pcie_mode_config(void)
+{
+	boot_rom_disable_alt_protocol();
+
+	/* TODO: Confirm how many VFs should be enabled in recovery */
+	boot_rom_set_number_of_vfs(NUM_VFS);
+
+	boot_rom_skip_ats_capability();
+
+	/* TODO: Set Gen 1 limit config */
+	if (boot_rom_pcie_gen_limit_enabled())
+		boot_rom_status_update(PROGRESS_BOOT_ROM_PCIE_LINK_LIMIT_GEN1);
+}
+
+void boot_rom_pcie_link_init(void)
+{
+	boot_rom_status_update(PROGRESS_BOOT_ROM_PCIE_LINK_START);
+
+	/* TODO: Check if PRSTN de-assert loop is required */
+
+	boot_rom_status_update(PROGRESS_BOOT_ROM_PCIE_LINK_INIT);
+
+	/* TODO: Add PCIe Phy fw init code*/
+
+	boot_rom_dbi_ro_write_en(true);
+
+	boot_rom_status_update(PROGRESS_BOOT_ROM_PCIE_LINK_IATU_CONFIG);
+
+	boot_rom_pcie_bar_config();
+
+	boot_rom_pcie_mode_config();
+
+	boot_rom_dbi_ro_write_en(false);
+
+	boot_rom_pcie_set_ltssm_en();
+
+	boot_rom_status_update(PROGRESS_BOOT_ROM_PCIE_LINK_FINISH);
+}
