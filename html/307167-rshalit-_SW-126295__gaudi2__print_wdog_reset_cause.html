[SW-126295] gaudi2: print wdog reset cause

Wdog reset can happen for several reasons:
arm hearbeat failure
arcpid heartbeat failure,
or if arcmgmt did not feed watchdog.

Arcmgmt save the information, and on watchdog
reset the information is passed using cold_rst
flop register.

+void wd_cause_dump(void)
+{
+	struct gaudi2_cold_rst_data cold_rst_data;
+
+	cold_rst_data.data = regmap_readl(mmCOLD_RST_DATA);
+
+	if (cold_rst_data.wd_rst_cause_arm)
+		preboot_puts_raw("wd cause: ARM heartbeat\n");
+
+	if (cold_rst_data.wd_rst_cause_arcpid)
+		preboot_puts_raw("wd cause: ARCPID heartbeat\n");
+
+	if ((!cold_rst_data.wd_rst_cause_arm) &&
+		(!cold_rst_data.wd_rst_cause_arcpid))
+		preboot_puts_raw("wd cause: ARCMGMT heartbeat\n");
+
+
+	cold_rst_data.wd_rst_cause_arm = 0;
+	cold_rst_data.wd_rst_cause_arcpid = 0;
+
+	regmap_writel(cold_rst_data.data, mmCOLD_RST_DATA);
