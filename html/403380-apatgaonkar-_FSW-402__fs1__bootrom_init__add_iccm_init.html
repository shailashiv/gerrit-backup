+void boot_rom_iccm_init(void)
+{
+	struct fs1_cold_rst_data cold_rst_data;
+	int ret;
+
+	if (boot_rom_global.die_type != IO_DIE_MASTER)
+		return;
+
+	/* Perform arc1 iccm init only on cold reset */
+	cold_rst_data.data = regmap_readl(IO0_HALF0_PSOC_BOOT(mmPSOC_BOOT_CONF_COLD_RST_REG_2));
+	if (cold_rst_data.arc1_iccm_init_done)
+		return;
+
+	ret = boot_rom_mem_init(HL_ARC1_ICCM_FULL_BASE, ARC1_ICCM_SIZE);
+	if (ret) {
+		boot_rom_status_update(PROGRESS_INIT_ARC1_ICCM_INIT_FAILED);
+		return;
+	}
+
+	cold_rst_data.arc1_iccm_init_done = 1;
+	regmap_writel(cold_rst_data.data, IO0_HALF0_PSOC_BOOT(mmPSOC_BOOT_CONF_COLD_RST_REG_2));
