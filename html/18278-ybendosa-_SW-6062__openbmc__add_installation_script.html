+function usage
+{
+	echo "Usage: $0 [<options>] <BMC IP ADDRESS>"
+	echo "	-f	<image file>	Path to BMC image file"
+	echo "	-p	<password>	Password to root user in BMC"
+	echo "	-h			Print this help message"
+}
+
+while getopts ":f:p:" o; do
+	case "${o}" in
+		f)
+			f=${OPTARG}
+			;;
+		p)
+			p=${OPTARG}
+			;;
+		*)
+			usage
+			exit 1
+			;;
+	esac
+done
+shift $((OPTIND-1))
+
+if [ -z "${f}" ]; then
+	f=${PWD}/flash-hls.tgz
+fi
+
+if [ -z "${p}" ]; then
+	p=root
+fi
+
+if [ ! -f "${f}" ]; then
+	echo missing image file
+	exit 2
+fi
+
+if [ -z "${1}" ]; then
+	echo missing BMC address
+	exit 2
+fi
+
+BMC_IMAGE=${f}
+BMC_ROOT_PASSWORD=${p}
+BMC_ADDRESS=${1}
+
+sshpass -p root scp -o StrictHostKeyChecking="no" $BMC_IMAGE root@$BMC_ADDRESS:
+sshpass -p $BMC_ROOT_PASSWORD ssh root@$BMC_ADDRESS "update_flash.sh && reboot"
