Commit Message

[SW-102640] gaudi3: set ppboot ver in bootinfo

Gaudi3 ppboot uses arcboot boot info API which
doesn't support data merge, meaning, that as long
as ppboot doesn't add data to boot info there are
no issues, however, as part of support for ppboot version
in generic packet, the ppboot version has to be added
to boot info.

This commit add implementation for bootinfo 'set' which
fetches boot info data from spsram and thus when
boot_info_finalize_no_merge() is called, the existing data
will be merged.

+/*
+ * Gaudi3 ppboot uses bootinfo API from arcboot which doesn't support merge of existing boot info
+ * data, so prior to using boot_info_finalize_no_merge(), the bootinfo data on spsram memory needs
+ * to be fetched to the 'boot_info' structure by calling boot_info_set_no_merge() for all existing
+ * descs.
+ */
+static int ppboot_boot_info_set(u32 type, void *data, u32 size)
+{
+	struct mem_desc mem_region_desc;
+	struct hl_boot_info_desc *desc;
+	u32 desc_count = hl_boot_info_get_num_descs();
+	uintptr_t mem_base, mem_base_end;
+	u32 i, actual_size, ret = 0;
+
+	if (preboot_get_mem_region(MEM_GP, &mem_region_desc)) {
+		preboot_puts("boot_info_set: failed to get mem region MEM_GP\n");
+		return 1;
+	}
+
+	mem_base = mem_region_desc.base;
+	mem_base_end = mem_base + mem_region_desc.size;
+	for (i = 0; i < desc_count; i++) {
+		desc = hl_boot_info_get_desc(i);
+		if (!desc) {
+			preboot_puts_hex("boot_info_set: failed to get pointer to desc index ", i,
+					 false);
+			continue;
+		}
+
+		/* copy boot info memory into GP mem region */
+		if (mem_base + desc->size > mem_base_end) {
+			preboot_puts("boot_info_set: not enough space for bootinfo\n");
+			ret = 1;
+			goto merge_and_exit;
+		}
+
+		if (boot_info_get_no_merge(desc->type, (void *)mem_base, desc->size,
+					   &actual_size)) {
+			preboot_puts_hex("boot_info_set: failed to get type ", desc->type,
+					 false);
+			continue;
+		}
+
+		if (boot_info_set_no_merge(desc->type, (void *)mem_base, desc->size)) {
+			preboot_puts_hex("boot_info_set: bootinfo set failure for type ",
+					 desc->type, false);
+			continue;
+		}
+
+		mem_base += desc->size;
+	}
+
+	if (boot_info_set_no_merge(HL_BOOT_INFO_PPBOOT_VERSION, data, size)) {
+		preboot_puts("boot_info_set: failed to set HL_BOOT_INFO_PPBOOT_VERSION\n");
+		return 1;
+	}
+
+merge_and_exit:
+	boot_info_finalize_no_merge();
+	return ret;
+}
