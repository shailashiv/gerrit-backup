Commit Message

[SW-78885] gaudi3: initialize arc1 iccm memory

ARC1 ICCM is used by ROM to download and validate boot
descriptor image and is also used to copy boot descriptor data
while using boot info. In order to prevent any uninitialized read access
to iccm in ROM code, this patch performs init of ARC1 iccm memory only in
cold reset using memcopy block.

+int arcboot_rom_mem_init(uint64_t addr, uint32_t size)
+{
+	uint32_t retry = HL_INIT_MEM_RETRIES;
+	uint32_t mem_cfg, cpy_sts;
+
+	regmap_writel((uint32_t)addr, mmD0_PSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_L);
+	regmap_writel((uint32_t)(addr >> 32), mmD0_PSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_H);
+
+	mem_cfg = regmap_readl(mmD0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+	mem_cfg &= ~(D0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_MEM_SIZE_M);
+
+	/* set size of in granuality of 16 bytes */
+	size = size >> 4;
+	mem_cfg |= (size & D0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_MEM_SIZE_M);
+	mem_cfg |= D0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_USE_CONST_M;
+
+	regmap_writel(mem_cfg, mmD0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+
+	/* Enable copy and wait for completion */
+	regmap_writel(D0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL_EN_M, mmD0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL);
+	while (1) {
+		cpy_sts = regmap_readl(mmD0_PSOC_GLOBAL_CONF_MEM_CPY_STATUS);
+		if (cpy_sts & D0_PSOC_GLOBAL_CONF_MEM_CPY_STATUS_DONE_M)
+			break;
+
+		retry--;
+		if (!retry)
+			return -1;
+
+		udelay(HL_INIT_MEM_UDELAY);
+	}
+
+	return 0;
+}
+
+static void arcboot_rom_iccm_init(void)
+{
+	struct gaudi3_cold_rst_data cold_rst_data;
+	int ret;
+
+	if (arcboot_rom_global.die_id)
+		return;
+
+	/* Perform arc1 iccm init only on cold reset */
+	cold_rst_data.data = regmap_readl(mmCOLD_RST_FLOPS_2);
+	if (cold_rst_data.arc1_iccm_init_done)
+		return;
+
+	ret = arcboot_rom_mem_init((HL_GEN_MEM_BASE_FULL + HL_ARC1_ICCM_OFFSET), HL_ICCM_SIZE);
+	if (ret) {
+		arcboot_rom_status_update(PROGRESS_INIT_ARC1_ICCM_INIT_FAILED);
+		return;
+	}
+
+	cold_rst_data.arc1_iccm_init_done = 1;
+	regmap_writel(cold_rst_data.data, mmCOLD_RST_FLOPS_2);
+}
