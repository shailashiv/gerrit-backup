+#define MMES_NUM	4

+static void preboot_secure_mme(void)
+{
+	u32 reg;
+	int i;
+
+	/* Set MME engines level to user */
+	for (i = 0; i < MMES_NUM; i++) {
+		/* Set ACC to user */
+		reg = mmMME0_ACC_PROT + MMES_DIST * i;
+		regmap_writel(PROT_OVR_USER, reg);
+		/* Protect the ACC configuration */
+		regmap_writel(PROT_BIT(mmMME0_ACC_PROT), PRIV_PROT_REG(reg));
+		set_prot_bits_priv_int_conf(mmMME0_ACC_BASE + MMES_DIST * i);
+
+		/* Set SBAB to user */
+		reg = mmMME0_SBAB_PROT + MMES_DIST * i;
+		regmap_writel(PROT_OVR_USER, reg);
+		/* Protect the SBAB configuration */
+		regmap_writel(PROT_BIT(mmMME0_SBAB_PROT), PRIV_PROT_REG(reg));
+		set_prot_bits_priv_int_conf(mmMME0_SBAB_BASE + MMES_DIST * i);
+
+		/* Set CTRL to user */
+		reg = mmMME0_CTRL_PROT + MMES_DIST * i;
+		regmap_writel(PROT_OVR_USER, reg);
+		/* Protect the CTRL configuration */
+		regmap_writel(PROT_BIT(mmMME0_CTRL_PROT), PRIV_PROT_REG(reg));
+		set_prot_bits_priv_int_conf(mmMME0_CTRL_BASE + MMES_DIST * i);
+	}
+}
