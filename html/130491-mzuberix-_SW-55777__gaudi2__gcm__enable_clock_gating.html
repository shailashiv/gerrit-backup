+static void gaudi2_enable_all_tpcs_clk_gating(void)
+{
+	int dcore, inst, instances, tpc_seq = -1;
+	u32 offset;
+
+	for (dcore = 0; dcore < NUM_OF_DCORES; dcore++) {
+		instances = (dcore == 0) ?
+				NUM_DCORE0_TPC : NUM_OF_TPC_PER_DCORE;
+
+		for (inst = 0; inst < instances; inst++) {
+			tpc_seq++;
+
+			if (!(TPC_ENABLED_MASK & BIT(tpc_seq)))
+				continue;
+
+			offset = (DCORE_OFFSET * dcore) +
+				 (DCORE_TPC_OFFSET * inst);
+			gaudi2_enable_tpc_clk_gating(dcore, inst, offset);
+		}
+	}
+}

+void gaudi2_init_mme_clk_gating(void)
+{
+	int i, j, dcore_offs = 0;
+
+	for (i = 0; i < NUM_OF_DCORES; i++) {
+		if (!(MME_MASK & (1 << i)))
+			continue;
+
+		dcore_offs = i * DCORE_OFFSET;
+
+		regmap_writel(GAUDI2_QM_CGM_CFG_MASK, mmDCORE0_MME_QM_CGM_CFG +
+			      dcore_offs);
+		regmap_writel(0x210006, mmDCORE0_MME_CTRL_LO_EU + dcore_offs);
+		regmap_writel(0, mmDCORE0_MME_ACC_CLK_GATE_EN + dcore_offs);
+
+		for (j = 0; j < NUM_OF_MME_SBTE_PORTS; j++) {
+			regmap_writel(0x11, mmDCORE0_MME_SBTE0_ENABLE_CGATE +
+				dcore_offs + (j * DCORE_MME_SBTE_OFFSET));
+		}
+	}
