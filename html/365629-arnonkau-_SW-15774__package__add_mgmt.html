+		$($(call UPPERCASE,$(DEV_PREFIX))_CONF_DEV_EXTRA_CFLAGS) \
+		$(ZEPHYR_CONF_DEV_EXTRA_CFLAGS)

+ifeq ($(HL_TARGET),pldm)
+MGMT_CONF_EXTRA_CFLAGS += -DTARGET_IS_PLDM
+ifeq ($(PLDM_SKIP_UBOOT),1)
+MGMT_CONF_EXTRA_CFLAGS += -DTARGET_IS_PLDM_REGRESSION

+MGMT_CONF_EXTRA_CFLAGS += -DTARGET_IS_ONIX

+MGMT_CONF_OPTS += -DCMAKE_EXE_LINKER_FLAGS_INIT=-Wl,-e__start

+MGMT_WEST_DEST_DIR :=

+define MGMT_COPY_ZEPHYR_SRC
+	rsync -au --chmod=u=rwX,go=rX  --exclude .svn --exclude .git --exclude .hg --exclude .bzr --exclude CVS $(MGMT_ZEPHYR_SRCDIR) $(MGMT_BUILD_DIR)
+endef

+# Arguments:
+# - module name
+# - component dir
+# - component name
+#
+define MGMT_DEFINE_MODULE_ENVVARS
+	$(eval MODULE_NAME=$(strip $1))
+	$(eval MODULE_NAME_UPPER_CASE=$(shell echo $(MODULE_NAME) | tr a-z A-Z))
+	$(eval MODULE_NAME_UPPER_CASE=$(strip $(MODULE_NAME_UPPER_CASE)))
+	$(eval $(MODULE_NAME_UPPER_CASE)_SRCDIR=$(EMBEDDED_DIR)/zephyr-modules/$(MODULE_NAME))
+	$(eval $(MODULE_NAME_UPPER_CASE)_DIR=$(@D)/build/modules/$(MODULE_NAME))
+	$(eval $(MODULE_NAME_UPPER_CASE)_NAME=$2_$(MODULE_NAME))
+endef
+
+$(foreach module,$(MGMT_APP_MODULES),$(call MGMT_DEFINE_MODULE_ENVVARS,$(module),$(MGMT_APP_PROD_NAME)))
+
+# Arguments:
+# - module (mbedtls, eeprom, etc.)
+define MGMT_MODULE_VERSION_HOOK
+	$(eval MODULE_NAME_UPPER_CASE=$(shell echo $1 | tr a-z A-Z))
+	$(eval MODULE_NAME_UPPER_CASE=$(strip $(MODULE_NAME_UPPER_CASE)))
+	$(call HL_VERSION_GEN, ${${MODULE_NAME_UPPER_CASE}_SRCDIR}, ${${MODULE_NAME_UPPER_CASE}_DIR}, ${${MODULE_NAME_UPPER_CASE}_NAME})
+endef
+
+define ZEPHR_MGMT_APP_VERSION_GEN
+	@$(call HL_VERSION_GEN, ${MGMT_SRCDIR},.,${MGMT_APP_NAME})
+	@$(call HL_VERSION_GEN, ${MGMT_SRCDIR},build,${MGMT_APP_NAME})
+	@$(foreach module,$(MGMT_APP_MODULES),$(call MGMT_MODULE_VERSION_HOOK,$(module)))
+endef
+
+define MGMT_CREATE_HL_VERSION
+	@mkdir -p $(MGMT_BUILD_DIR)/include
+	@echo "#ifndef _HL_VERSION_" >  $(MGMT_BUILD_DIR)/include/hl_version.h
+	@echo "#define _HL_VERSION_" >>  $(MGMT_BUILD_DIR)/include/hl_version.h
+	@echo "#define HL_VERSION_MAJOR $$(echo $(MGMT_LOCAL_VERSION) | cut -d. -f1)" >> $(MGMT_BUILD_DIR)/include/hl_version.h
+	@echo "#define HL_VERSION_MINOR $$(echo $(MGMT_LOCAL_VERSION) | cut -d. -f2)" >> $(MGMT_BUILD_DIR)/include/hl_version.h
+	@echo "#define HL_VERSION_PATCH $$(echo $(MGMT_LOCAL_VERSION) | cut -d. -f3)" >> $(MGMT_BUILD_DIR)/include/hl_version.h
+	@echo "#endif" >>  $(MGMT_BUILD_DIR)/include/hl_version.h
+endef
+
+define MGMT_CONFIGURE_CMDS
+	@$(call MGMT_CREATE_HL_VERSION)
+	@$(MGMT_CONF_ENV) $(HOST_DIR)/bin/cmake $(MGMT_CONF_OPTS) -S $(MGMT_BUILD_DIR) -B $(MGMT_BUILD_DIR)/build
+endef
+
+MGMT_POST_RSYNC_HOOKS += HL_VERSION_GEN
+MGMT_POST_RSYNC_HOOKS += MGMT_COPY_ZEPHYR_SRC
+MGMT_POST_RSYNC_HOOKS += ZEPHR_MGMT_APP_VERSION_GEN
+
+define MGMT_BUILD_CMDS
+	@$(TARGET_MAKE_ENV) $(MAKE) -C $(MGMT_BUILD_DIR)/build all
+endef
+
+define MGMT_INSTALL_TARGET_CMDS
+	@$(INSTALL) -D -m 0644 $(@D)/build/zephyr/zephyr.elf $(BINARIES_DIR)/zephyr_$(MGMT_APP_NAME).elf
+endef
+
+mgmt-configure: $(MGMT_DEPENDENCIES)
+mgmt-menuconfig mgmt-guiconfig: | mgmt-configure
+	$(TARGET_MAKE_ENV) $(MAKE) -C $(MGMT_BUILD_DIR)/build all
+	rm -f $(MGMT_BUILD_DIR)/.stamp_{configured,built}

+$(eval $(cmake-package))
