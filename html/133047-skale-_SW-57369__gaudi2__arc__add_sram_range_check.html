+bool preboot_sram_is_valid_address_range(u64 addr, int len)
+{
+	/* Prepare the ARC SRAM address */
+	u32 arc_addr = ((u32)addr & ARM_ADDR_MASK) | BIT(28);
+
+	if (arc_addr >= SDRAM_BASE && arc_addr + len <= SDRAM_BASE + SRAM_SIZE)
+		return !preboot_is_overlap_address_range(arc_addr, len);
+
+	if (addr >= FIT_IMG_SPSRAM_BASE && addr + len <= FIT_IMG_SPSRAM_END)
+		return true;
+
 	return false;
 }
