gaudi3: perform sram_mbist in ppboot

This patch incorporates sram_mbist data copying
from spi flash to fwsram and execute sram mbist
at ppboot stage after pcie link is up and before
preboot is loaded.

die_sram_mbist_data.txt file is shared by dft team
and covers sram mbist test of all caches (cse_up/dn)
for all dcores at ppboot stage for die 0.

Entire die 1 mbist for caches will be performed
at preboot stage. Same mbist tables used for die0,
will be used for die1 as well.

At preboot stage, another mbist table, shared by
dft team, is used. Entire die 0 mbist for other
engine srams will be performed, when we exclude
psoc, cpu, pcie and caches. That mbist table
will be part of preboot image.

PPboot SRAM MBIST  is kept under GAUDI3_SRAM_MBIST_ENABLE
flag and this flag is disabled now.

+int hl_ppboot_sram_self_repair_mismatch_logic(u32 ist_addr, u32 *data,
+		u32 *fail_buffer, u32 fail_limit, u32 *fail_index)
+{
+	int status = 0;
+	u32 rdval, expval, failmask;
+	u32 i = 1, readID;
+
+	if (data[0] != 1)
+		return -1;  /* version mismatch */
+
+	readID = data[i];
+	++i;
+
+	while (data[i] != 0) {
+		u32 repeat;
+		u32 opw = data[i];
+		u32 op = (opw & 0x3);
+
+		++i;
+		for (repeat = (opw >> 2); repeat != 0; --repeat) {
+			if (op == IST_OPC_WAIT) {
+				gaudi3_mbist_wait(data[i]);
+				++i;
+				continue;
+			}
+
+			regmap_writel(data[i], ist_addr);
+			++i;
+
+			if (op == IST_OPC_WRITE_X_COMPARE) {
+				++readID;
+				rdval = regmap_readl(ist_addr) & data[i];
+				++i;
+				expval = data[i];
+				++i;
+
+				if (rdval != expval) {
+					status = 1;
+					if (*fail_index < (fail_limit * 2)) {
+						failmask = rdval ^ expval;
+						fail_buffer[*fail_index] = readID;
+						(*fail_index)++;
+						fail_buffer[*fail_index] = failmask;
+						(*fail_index)++;
+					}
+				}
+
+			} else if (op == IST_OPC_WRITE_BIN_COMPARE) {
+				++readID;
+				rdval = regmap_readl(ist_addr);
+				expval = data[i];
+				++i;
+
+				if (rdval != expval) {
+					status = 1;
+					if (*fail_index < (fail_limit * 2)) {
+						failmask = rdval ^ expval;
+						fail_buffer[*fail_index] = readID;
+						(*fail_index)++;
+						fail_buffer[*fail_index] = failmask;
+						(*fail_index)++;
+					}
+				}
+
+			} else {
+				/* no compare */
+			}
+		}
+	}
+
+	return status;
