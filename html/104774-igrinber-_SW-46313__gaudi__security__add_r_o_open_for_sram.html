[SW-46313] gaudi: security: add r/o open for sram

Add support for enabling the read-only access to the whole SRAM region.
To make this happen, preboot_security_sram_pci_set_ro() function walks
over all the SRAM protecting dynamic RRs and opens the read-only access.

+/*
+ * preboot_security_sram_pci_set_ro() function assumes the TPC3 has already been
+ * moved to be secured through the secure level RRs and thus only opens the PCIe
+ * RRs for reading.
+ */
+void preboot_security_sram_pci_set_ro(bool en)
+{
+	int i;
+	u64 min, max;
+	struct preboot_sec_dyn_pcie_rr_cfg *cfg;
+
+
+	for (i = SRAM_BGN_TO_VERS; i <= IMG_END_TO_SRAM_END; i++) {
+		cfg = &preboot_sec_dyn_cfg[i];
+		min = en ? 0 : cfg->min;
+		max = en ? 0 : cfg->max;
+
+		if (cfg->rr >= 0 && cfg->min && cfg->max) {
+			preboot_secure_pcie_rr_wr(cfg->rr, cfg->min, cfg->max);
+			preboot_secure_pcie_rr_rd(cfg->rr, min, max);
+		}
+	}
+
+}
