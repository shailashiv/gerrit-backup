+static int read_flash_chunk(uint32_t addr, uint32_t size, uint8_t *buf)
+{
+	const struct device *storage_dev = FLASH_DEV_INST;
+	const struct flash_driver_api *api = storage_dev->api;
+	int loop_counter = 0, rc = -EPERM;
+	int spi_disabled_timeout_ms = 1000;
+
+	while (rc == -EPERM) {
+		rc = api->read(FLASH_DEV_INST, addr, buf, size);
+		if (rc == -EPERM) {
+			k_msleep(1);
+			if (++loop_counter == spi_disabled_timeout_ms)
+				break;
+		} else if (rc)
+			LOG_ERR("Failed to read SPI NOR flash. Address=0x%X Size=%d", addr, size);
+	}
+
+	return rc;

+static int validate_lfs_serdes(void)
+{
+	int rc;
+	uint32_t start_addr;
+	uint32_t size;
+	uint8_t serdes_lfs_partition_hash[SHA384_SUM_LEN];
+	uint8_t digest[SHA384_SUM_LEN];
+	uint8_t *buf = NULL;
+
+	rc = hl_fw_info_rd_field(&start_addr, serdes_lfs_partition_offset);
+	if (!rc)
+		rc = hl_fw_info_rd_field(&size, serdes_lfs_partition_size);
+
+	if (!rc)
+		rc = hl_fw_info_rd_field(serdes_lfs_partition_hash, serdes_lfs_partition_checksum);
+
+	if (rc) {
+		LOG_ERR("Serdes: Failed to read LFS partition data, rc=%d", rc);
+		goto val_error;
+	}
+
+	rc = -ENOMEM;
+	if (!sys_heap_hbm) {
+		LOG_INF("Serdes: HBM heap not available");
+		return 0;
+	}
+
+	buf = sys_heap_alloc(sys_heap_hbm, size);
+	if (!buf) {
+		LOG_ERR("Serdes: failed to allocate validation buffer");
+		goto val_error;
+	}
+
+	rc = read_flash_chunk(start_addr, size, buf);
+	if (rc)
+		goto val_error;
+
+	sha384_csum(buf, size, digest, 0);
+	if (memcmp(serdes_lfs_partition_hash, digest, SHA384_SUM_LEN)) {
+		LOG_ERR("Serdes: Validation of LFS Serdes partition failed");
+		rc = -1;
+		goto val_error;
+	}
+
+	LOG_INF("Serdes: LFS Serdes partition validated successfully");
+
+val_error:
+	if (buf)
+		sys_heap_free(sys_heap_hbm, buf);
+
+	return rc;
+}
