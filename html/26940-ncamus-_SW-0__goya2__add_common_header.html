+#define FW_WORD_NUM		0x2000	/* FW_SIZE / 2 */
+static void preboot_init_pcie_phy_fw_update(void)
+{
+	/* Configure FW update size */
+	regmap_writel(FW_WORD_NUM, mmPCIE_WRAP_PHY_FW_FSM_SIZE);
+
+	/*
+	 * PHY FW FSM EN and Direct FW write mechanism enable
+	 * FW FSM assumes that the data is present in the SRAM
+	 * The FSM will wait for SRAM init done before updating
+	 * the SRAM contents
+	 */
+	regmap_writel(9, mmPCIE_WRAP_PHY_FW_FSM);
+
+	/* Take the PHY out of reset */
+	regmap_writel(1, mmPSOC_GLOBAL_CONF_PCIE_PHY_RST_N);
+}
+
+__attribute__((weak)) void preboot_init_pcie_phy(void)
+{
+	/* Reset PCIe PHY */
+	regmap_writel(0, mmPSOC_GLOBAL_CONF_PCIE_PHY_RST_N);
+
+	/*
+	 * Set PCIe PHY CR Para I/F to CFG.
+	 * Select CR PARA I/F instead of JTAG
+	 */
+	regmap_writel(1, mmPCIE_PHY_CR_PARA_SEL_0);
+	regmap_writel(1, mmPCIE_PHY_CR_PARA_SEL_1);
+	regmap_writel(1, mmPCIE_PHY_CR_PARA_SEL_2);
+	regmap_writel(1, mmPCIE_PHY_CR_PARA_SEL_3);
+
+	/* Select indirect access to CR PARA I/F instead of PRBI */
+	regmap_writel(1, mmPCIE_PHY_CR_PARA_CFG);
+
+	/*
+	 * Use PHY configuration from SoC online
+	 * instead of default values
+	 */
+	regmap_writel(1, mmPCIE_PHY_PHY_EXT_CTRL_SEL);
+	preboot_init_pcie_phy_fw_update();
+}
+
+static void preboot_init_pcie(void)
+{
+	preboot_init_pcie_phy();
+
+	/* Update Vendor ID */
+	regmap_writel(0x10001DA3, mmPCIE_DBI_DEVICE_ID_VENDOR_ID_REG);
+
+	/* Update subsystem Vendor ID */
+	regmap_writel(0x10001DA3,
+		      mmPCIE_DBI_SUBSYSTEM_ID_SUBSYSTEM_VENDOR_ID_REG);
+
+	/* Disable the EQ REDO */
+	regmap_writel(0x00402801, mmPCIE_DBI_GEN3_RELATED_OFF);
+
+	/* delay before executing Synopsys W/A > 300us */
+	hl_pll_udelay(300);
+
+	/* Execute the Synopsys W/A */
+	regmap_writel(0xF, mmPCIE_PHY_CR_PARA_CTRL);
+	regmap_writel(0x0004A07D, mmPCIE_PHY_CR_PARA_CMD);
+	hl_pll_udelay(20);
+
+	/* Enable and Mask LTSSM (erratum 102) */
+	regmap_writel(3, mmPCIE_AUX_LTSSM_EN);
+}
+
+/*
+ * This logic copied from BTL PCIe section (#4)
+ * In BTL it is called when DIP switch for PCIe enable is set
+ * Now we wish to do this logic also here as a measure of security
+ * It means that in new PCB's version by default PCIe won't be enabled
+ * So without proper FW device won't be accessible through PCIe
+ * Hence device is more secured
+ */
+static void preboot_pcie_enable(void)
+{
+	u32 boot_strap = regmap_readl(mmPSOC_GLOBAL_CONF_BOOT_STRAP_PINS_L);
+	u32 reset_cause;
+
+	/* Check if already done by BTL */
+	if (boot_strap & 0x2)
+		return;
+
+	/* During hard reset triggered by LKD keep PCIe intact */
+	reset_cause = regmap_readl(mmPREBOOT_PCIE_EN);
+	regmap_writel(0, mmPREBOOT_PCIE_EN);
+	if (reset_cause == LKD_HARD_RESET_MAGIC)
+		return;
+
+	preboot_init_pcie();
+}
+
+void preboot_pcie_wait_for_link(void)
+{
+	/* Wait for PRSTn */
+	while (!regmap_readl(mmPSOC_GLOBAL_CONF_PRSTN))
+		;
+
+	/* Only after PRSTn is de-asserted */
+	preboot_pcie_enable();
+
+	while (1) {
+		unsigned int prstn, ltssm, smlh, rdlh;
+
+		prstn = regmap_readl(mmPSOC_GLOBAL_CONF_PRSTN);
+		ltssm = regmap_readl(mmPCIE_AUX_LTSSM_EN);
+		smlh = regmap_readl(mmPCIE_AUX_SMLH_LINK_UP);
+		rdlh = regmap_readl(mmPCIE_AUX_RDLH_LINK_UP);
+
+		if (prstn && smlh && rdlh)
+			break;
+
+		/*
+		 * Some host assert prstn during link up while moving to GEN3
+		 * This cause BAR size to be reset
+		 * We stop link up process by disabling ltssm
+		 * till BAR is reconfigured
+		 */
+		if (!prstn && ltssm) {
+			regmap_writel(0, mmPCIE_AUX_LTSSM_EN);
+			continue;
+		}
+
+		if (prstn && !ltssm) {
+			regmap_writel(3, mmPCIE_AUX_LTSSM_EN);
+			continue;
+		}
+	}
+
+	/* W/A H/W bug that may lead to have a wrong mask (0x1) in LTSSM */
+	regmap_writel(3, mmPCIE_AUX_LTSSM_EN);
+}
+
+void preboot_delay(void) {}

+#define mmPCIE_WRAP_VUART_TX_DATA	0xC01F60

+	 * update vUART to HOST

+	regmap_writel(c, mmPCIE_WRAP_VUART_TX_DATA);
+	regmap_writel(1, mmPCIE_WRAP_VUART_TX_STATUS);
+}
