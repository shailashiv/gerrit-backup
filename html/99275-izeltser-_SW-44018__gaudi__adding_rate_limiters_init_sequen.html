+/*
+ * rate limiter parameters for 800MHz clock. that's the only clock currently
+ * supported for gaudi hbm in preboot
+ */
+#define GAUDI_HBM_RL_SAT		4
+#define GAUDI_HBM_RL_RST		11
+#define GAUDI_HBM_RL_TIMEOUT		15
+
+#define GAUDI_HBM_MID_WGHT		0x111
+#define GAUDI_HBM_CH_DIST		0x20000
+#define GAUDI_HBM_NUM_DEVICES		4
+#define GAUDI_FUSE_DEV_ID_WORD_OFFSET	2
+
+void preboot_rate_limiter_init(void)
+{
+	int i, ret;
+	u32 base, dev_type;
+	char dev_maj, dev_min;
+
+	ret = preboot_fuse_read(GAUDI_FUSE_DEV_ID_WORD_OFFSET, &dev_type);
+	if (ret)
+		return;
+
+	dev_type = __builtin_bswap32(dev_type);
+	dev_maj = (u8)((dev_type >> 8) & 0xff);
+	dev_min = (u8)(dev_type & 0xff);
+	if (!dev_min)
+		dev_min = '0';
+
+	/* rate limiters are initialized for A0 and B0 devices only */
+	if ((dev_min != '0') || ((dev_maj != 'A') && (dev_maj != 'B')))
+		return;
+
+	for (i = 0; i < GAUDI_HBM_NUM_DEVICES; i++) {
+		base = i * GAUDI_HBM_CH_DIST;
+		regmap_writel(GAUDI_HBM_MID_WGHT,
+			      base + mmDMA_IF_W_S_DOWN_RSP_MID_WGHT_0);
+		regmap_writel(GAUDI_HBM_MID_WGHT,
+			      base + mmDMA_IF_W_S_DOWN_RSP_MID_WGHT_1);
+		regmap_writel(GAUDI_HBM_RL_SAT,
+			      base + mmDMA_IF_W_S_DOWN_CH0_RL_HBM_SAT);
+		regmap_writel(GAUDI_HBM_RL_SAT,
+			      base + mmDMA_IF_W_S_DOWN_CH1_RL_HBM_SAT);
+		regmap_writel(GAUDI_HBM_RL_RST,
+			      base + mmDMA_IF_W_S_DOWN_CH0_RL_HBM_RST);
+		regmap_writel(GAUDI_HBM_RL_RST,
+			      base + mmDMA_IF_W_S_DOWN_CH1_RL_HBM_RST);
+		regmap_writel(GAUDI_HBM_RL_TIMEOUT,
+			      base + mmDMA_IF_W_S_DOWN_CH0_RL_HBM_TIMEOUT);
+		regmap_writel(GAUDI_HBM_RL_TIMEOUT,
+			      base + mmDMA_IF_W_S_DOWN_CH1_RL_HBM_TIMEOUT);
+		regmap_writel(1, base + mmDMA_IF_W_S_DOWN_CH0_RL_HBM_EN);
