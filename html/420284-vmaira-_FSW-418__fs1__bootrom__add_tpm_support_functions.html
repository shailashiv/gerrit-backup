+/**
+ * __ffs - find first bit in word.
+ * @word: The word to search
+ *
+ * Undefined if no bit exists, so code should check against 0 first.
+ */
+static inline int __ffs32(unsigned int word)
+{
+	int num = 0;
+
+	if ((word & 0xffff) == 0) {
+		num += 16;
+		word >>= 16;
+	}
+	if ((word & 0xff) == 0) {
+		num += 8;
+		word >>= 8;
+	}
+	if ((word & 0xf) == 0) {
+		num += 4;
+		word >>= 4;
+	}
+	if ((word & 0x3) == 0) {
+		num += 2;
+		word >>= 2;
+	}
+	if ((word & 0x1) == 0)
+		num += 1;
+	return num;
+}

+/**
+ * FIELD_PREP() - prepare a bitfield element
+ * @_mask: shifted mask defining the field's length and position
+ * @_val:  value to put in the field
+ *
+ * FIELD_PREP() masks and shifts up the value.  The result should
+ * be combined with other fields of the bitfield using logical OR.
+ */
+#define FIELD_PREP(_mask, _val)						  \
+	(								  \
+	    {								  \
+		    ((typeof(_mask))(_val) << __ffs32(_mask)) & (_mask);  \
+	    }								  \
+	)
+
+#define BITS_PER_LONG		32

+struct xfer_priv {
+	struct spi_ctx *ctx;
+	const void *tx;
+	const void *tx_end;
+	void *rx;
+	void *rx_end;
+	uint32_t fifo_len;			/* depth of the FIFO buffer */
+	uint32_t max_xfer;			/* Maximum transfer size (in bits) */
+	int len;
+	int bits_per_word;
+};

+static uint32_t dw_spi_dw32_update_cr0(int bits_per_word, int tmode)
+{
+	return FIELD_PREP(SPI_CTRLR0_DFS_32_M, bits_per_word - 1)
+	     | FIELD_PREP(SPI_CTRLR0_FRF_M, CTRLR0_FRF_SPI)
+	     | FIELD_PREP(CTRLR0_MODE_MASK, 0x0)
+	     | FIELD_PREP(SPI_CTRLR0_TMOD_M, tmode);
+}
+
+/* Return the max entries we can fill into tx fifo */
+static inline uint32_t tx_max(struct xfer_priv *priv)
+{
+	uint32_t tx_left, tx_room, rxtx_gap;
+
+	tx_left = (priv->tx_end - priv->tx) / (priv->bits_per_word >> 3);
+	tx_room = priv->fifo_len - spi_dw_readl(priv->ctx, mmSPI_TXFLR);
+
+	/*
+	 * Another concern is about the tx/rx mismatch, we
+	 * thought about using (priv->fifo_len - rxflr - txflr) as
+	 * one maximum value for tx, but it doesn't cover the
+	 * data which is out of tx/rx fifo and inside the
+	 * shift registers. So a control from sw point of
+	 * view is taken.
+	 */
+	rxtx_gap = ((priv->rx_end - priv->rx) - (priv->tx_end - priv->tx)) /
+		(priv->bits_per_word >> 3);
+
+	return MIN3(tx_left, tx_room, (uint32_t)(priv->fifo_len - rxtx_gap));
+}
+
+/* Return the max entries we should read out of rx fifo */
+static inline uint32_t rx_max(struct xfer_priv *priv)
+{
+	uint32_t rx_left = (priv->rx_end - priv->rx) / (priv->bits_per_word >> 3);
+
+	return min_t(uint32_t, rx_left, spi_dw_readl(priv->ctx, mmSPI_RXFLR));
+}
+
+static void dw_writer(struct xfer_priv *priv)
+{
+	uint32_t max = tx_max(priv);
+	uint32_t txw = 0x0;
+
+	while (max--) {
+		/* Set the tx word if the transfer's	*/
+		/* original "tx" is not null		*/
+		if (priv->tx_end - priv->len) {
+			if (priv->bits_per_word == 8)
+				txw = *(uint8_t *)(priv->tx);
+			else
+				txw = *(uint32_t *)(priv->tx);
+		}
+		spi_dw_writel(priv->ctx, txw, mmSPI_DR0);
+		priv->tx += priv->bits_per_word >> 3;
+	}
+}
+
+static void dw_reader(struct xfer_priv *priv)
+{
+	uint32_t max = rx_max(priv);
+	uint32_t rxw;
+
+	while (max--) {
+		rxw = spi_dw_readl(priv->ctx, mmSPI_DR0);
+
+		/* Care about rx if the transfer's original "rx" is not null */
+		if (priv->rx_end - priv->len) {
+			if (priv->bits_per_word == 8)
+				*(uint8_t *)(priv->rx) = rxw;
+			else
+				*(uint32_t *)(priv->rx) = rxw;
+		}
+		priv->rx += priv->bits_per_word >> 3;
+	}
+}
+
+static int poll_transfer(struct xfer_priv *priv)
+{
+	uint32_t count = RX_TIMEOUT;
+
+	do {
+		dw_writer(priv);
+		dw_reader(priv);
+		count--;
+	} while ((priv->rx_end > priv->rx) && (count > 0));
+
+	return 0;
+}
+
+int spi_poll_timeout(struct spi_ctx *ctx)
+{
+	uint32_t val = spi_dw_readl(ctx, mmSPI_SR), count = RX_TIMEOUT;
+
+	while (count > 0) {
+		if ((val & SPI_SR_TFE_M) && !(val & SPI_SR_BUSY_M))
+			return 0;
+		count--;
+	}
+
+	return -1;
+}
+
+/* based on u-boot designware SPI driver */
+int spi_xfer(struct spi_ctx *ctx, unsigned int bitlen,
+		const void *dout, void *din, unsigned long flags)
+{
+	const uint8_t *tx = dout;
+	uint8_t *rx = din;
+	int ret = 0;
+	uint32_t cr0 = 0;
+	uint8_t tmode;			/* TR/TO/RO/EEPROM */
+	struct xfer_priv priv = {0};
+
+	priv.ctx = ctx;
+
+	if ((flags & SPI_XFER_BEGIN) && (priv.ctx->claim_bus))
+		priv.ctx->claim_bus();
+
+	/* spi core configured to do 8 bit transfers */
+	if (bitlen % 8)
+		return -1;
+
+	if (priv.ctx->spi_dword_access_support)
+		priv.bits_per_word = ((bitlen % 32) ||
+				(flags & SPI_XFER_BEGIN)) ? 8 : 32;
+	else
+		priv.bits_per_word = 8;
+
+	if (rx && tx)
+		tmode = CTRLR0_TMOD_TR;
+	else if (rx)
+		tmode = CTRLR0_TMOD_RO;
+	else
+		/*
+		 * In transmit only mode (CTRL0_TMOD_TO) input FIFO never gets
+		 * any data which breaks our logic in poll_transfer() above.
+		 */
+		tmode = CTRLR0_TMOD_TR;
+
+	cr0 = dw_spi_dw32_update_cr0(priv.bits_per_word, tmode);
+
+	priv.len = bitlen >> 3;
+
+	priv.tx = (void *)tx;
+	priv.tx_end = priv.tx + priv.len;
+	priv.rx = rx;
+	priv.rx_end = priv.rx + priv.len;
+
+	/* Disable controller before writing control registers */
+	spi_dw_writel(priv.ctx, 0, mmSPI_SSIENR);
+
+	/* Reprogram cr0 only if changed */
+	if (spi_dw_readl(priv.ctx, mmSPI_CTRLR0) != cr0)
+		spi_dw_writel(priv.ctx, cr0, mmSPI_CTRLR0);
+
+	/* Enable controller after writing control registers */
+	spi_dw_writel(priv.ctx, 1, mmSPI_SSIENR);
+
+	priv.fifo_len = ctx->fifo_len;
+
+	/* Start transfer in a polling loop */
+	ret = poll_transfer(&priv);
+
+	/*
+	 * Wait for current transmit operation to complete.
+	 * Otherwise if some data still exists in Tx FIFO it can be
+	 * silently flushed, i.e. dropped on disabling of the controller,
+	 * which happens when writing 0 to DW_SPI_SSIENR which happens
+	 * in the beginning of new transfer.
+	 */
+	ret = spi_poll_timeout(ctx);
+
+	/* Stop the transaction */
+	if ((flags & SPI_XFER_END) && (priv.ctx->release_bus))
+		priv.ctx->release_bus();
+
+	return ret;
+}
+
+static void get_fifo_len(struct spi_ctx *ctx)
+{
+	uint32_t fifo;
+
+	for (fifo = 1; fifo < 256; fifo++) {
+		spi_dw_writel(ctx, fifo, mmSPI_TXFTLR);
+		if (fifo != spi_dw_readl(ctx, mmSPI_TXFTLR))
+			break;
+	}
+
+	ctx->fifo_len = (fifo == 1) ? 0 : fifo;
+	spi_dw_writel(ctx, 0, mmSPI_TXFTLR);
+}

+void spi_init(struct spi_ctx *ctx)
+{
+	uint32_t baudr;
+
+	if (ctx->spi_initialized)
+		return;
+
+	baudr = hl_spi_baudr_get();
+	/* Validate that baudr is even */
+	baudr = (baudr + 1) & ~1;
+
+	ctx->cs_init();
+	/* Disable ssi module */
+	spi_dw_writel(ctx, 0x00000000, mmSPI_SSIENR);
+	/* Set Baud rate */
+	spi_dw_writel(ctx, baudr, mmSPI_BAUDR);
+	/* Set Rx FIFO threshold value to default value */
+	spi_dw_writel(ctx, 0x00000000, mmSPI_RXFTLR);
+	/* Set Tx FIFO threshold value to default value */
+	spi_dw_writel(ctx, 0x00000000, mmSPI_TXFTLR);
+	/* Disable Interrupts */
+	spi_dw_writel(ctx, 0x00000000, mmSPI_IMR);
+	/* Set CTRL0 to default value */
+	spi_dw_writel(ctx, 0x00070000, mmSPI_CTRLR0);
+	/* Enable Slave */
+	spi_dw_writel(ctx, 0x00000001, mmSPI_SER);
+	/* Enable SSI module */
+	spi_dw_writel(ctx, 0x00000001, mmSPI_SSIENR);
+
+	get_fifo_len(ctx);
+
+	ctx->spi_initialized = true;
+}

+/*
+ * tpm_tis_spi_xfer() - TPM <-> SPI abstraction api.
+ *
+ * @description : This api is used to communicate with the spi
+ * driver in order to send commands as per TPM 2.0 specification
+ * and to send/receive data from the TPM slave.
+ *
+ * @addr: register address to read from
+ * @out: buffer in case caller sends data to tpm slave
+ * @in: buffer in case caller requests data from tpm slave
+ *
+ * @return -1 on error, 0 on success.
+ */
+
+#define TPM_ADDR_COMMAND_OPCODE		0xD4
+static int tpm_tis_spi_xfer(uint32_t addr, const uint8_t *out, uint8_t *in, uint16_t len)
+{
+	int transfer_len, ret, i;
+	uint8_t tx_buf[MAX_SPI_FRAMESIZE];
+	uint8_t rx_buf[MAX_SPI_FRAMESIZE];
+
+	if (in && out)
+		return -1;
+
+	while (len) {
+		/* Request */
+		transfer_len = min_t(uint16_t, len, MAX_SPI_FRAMESIZE);
+		tx_buf[0] = (in ? BIT(7) : 0) | (transfer_len - 1);
+		tx_buf[1] = TPM_ADDR_COMMAND_OPCODE;
+		tx_buf[2] = addr >> 8;
+		tx_buf[3] = addr;
+
+		ret = spi_xfer(&tpm_spi, 4 * 8, tx_buf, rx_buf, SPI_XFER_BEGIN);
+		if (ret)
+			goto release_bus;
+
+		/* Wait state */
+		if (!(rx_buf[3] & 0x1)) {
+			for (i = 0; i < TPM_WAIT_STATES; i++) {
+				ret = spi_xfer(&tpm_spi, 1 * 8, NULL, rx_buf,
+						0);
+				if (ret)
+					goto release_bus;
+
+				if (rx_buf[0] & 0x1)
+					break;
+			}
+
+			if (i == TPM_WAIT_STATES) {
+				ret = -1;
+				goto release_bus;
+			}
+		}
+
+		/* Read/Write */
+		if (out) {
+			memcpy(tx_buf, out, transfer_len);
+			out += transfer_len;
+		} else {
+			memset(tx_buf, 0, transfer_len);
+		}
+
+		ret = spi_xfer(&tpm_spi, transfer_len * 8,
+			       tx_buf,
+			       in ? rx_buf : NULL,
+			       SPI_XFER_END);
+		if (ret)
+			goto release_bus;
+

+static int tpm_spi_recv_data(uint8_t *buf, size_t count)
+{
+	size_t size = 0;
+	int burstcnt, len, ret;
+	uint8_t status;
+
+	while (size < count &&
+	       tpm_spi_wait_for_stat(TPM_STS_DATA_AVAIL | TPM_STS_VALID,
+					 TIS_SHORT_TIMEOUT_MS, &status) == 0) {
+		burstcnt = tpm_spi_get_burstcount();
+		if (burstcnt < 0)
+			return burstcnt;
+
+		len = min_t(int, burstcnt, count - size);
+		ret = tpm_tis_spi_read(TPM_DATA_FIFO(locality),
+				       buf + size, len);
+		if (ret)
+			return ret;
+
+		size += len;
+	}

+	return size;

+static int tpm_spi_recv(uint8_t *buf, size_t count)
+{
+	size_t size, expected;
+
+	if (count < TPM_HEADER_SIZE) {
+		size = -1;
+		goto out;
+	}
+
+	size = tpm_spi_recv_data(buf, TPM_HEADER_SIZE);
+	if (size < TPM_HEADER_SIZE) {
+		/* TODO: Check if size invalid requires to be returned */
+		goto out;
+	}
+
+	expected = get_unaligned_be32(buf + 2);
+	if (expected > count) {
+		size = -1;
+		goto out;
+	}
+
+	size += tpm_spi_recv_data(&buf[TPM_HEADER_SIZE],
+				   expected - TPM_HEADER_SIZE);
+	if (size < expected) {
+		size = -1;
+		goto out;
+	}
+
+out:
+	tpm_tis_spi_cancel();
+	tpm_tis_spi_release_locality(locality, false);
+
+	return size;
+}

+static int tpm_spi_send(const uint8_t *buf, size_t len)
+{
+	uint32_t i, size;
+	uint8_t status;
+	int burstcnt, ret;
+	uint8_t data;
+
+	if (len > TPM_DEV_BUFSIZE)
+		return -E2BIG;  /* Command is too long for our tpm, sorry */
+
+	ret = tpm_tis_spi_request_locality(locality);
+	if (ret < 0)
+		return -EBUSY;
+
+	/*
+	 * Check if the TPM is ready. If not, if not, cancel the pending command
+	 * and poll on the status to be finally ready.
+	 */
+	ret = tpm_spi_status(&status);
+	if (ret)
+		return ret;
+
+	if (!(status & TPM_STS_COMMAND_READY)) {
+		/* Force the transition, usually this will be done at startup */
+		ret = tpm_tis_spi_cancel();
+		if (ret)
+			goto out_err;
+
+		ret = tpm_spi_wait_for_stat(TPM_STS_COMMAND_READY,
+						TIS_LONG_TIMEOUT_MS, &status);
+		if (ret < 0 || !(status & TPM_STS_COMMAND_READY))
+			goto out_err;
+	}
+
+	for (i = 0; i < len - 1;) {
+		burstcnt = tpm_spi_get_burstcount();
+		if (burstcnt < 0)
+			return burstcnt;
+
+		size = min_t(int, len - i - 1, burstcnt);
+		ret = tpm_tis_spi_write(TPM_DATA_FIFO(locality),
+					buf + i, size);
+		if (ret < 0)
+			goto out_err;
+
+		i += size;
+	}
+
+	ret = tpm_tis_spi_valid_status(&status);
+	if (ret)
+		goto out_err;
+
+	if ((status & TPM_STS_DATA_EXPECT) == 0) {
+		ret = -EIO;
+		goto out_err;
+	}
+
+	ret = tpm_tis_spi_write(TPM_DATA_FIFO(locality),
+				buf + len - 1, 1);
+	if (ret)
+		goto out_err;
+
+	ret = tpm_tis_spi_valid_status(&status);
+	if (ret)
+		goto out_err;
+
+	if ((status & TPM_STS_DATA_EXPECT) != 0) {
+		ret = -EIO;
+		goto out_err;
+	}
+
+	data = TPM_STS_GO;
+	ret = tpm_tis_spi_write(TPM_STS(locality), &data, 1);
+	if (ret)
+		goto out_err;
+
+	return len;
+
+out_err:
+	tpm_tis_spi_cancel();
+	tpm_tis_spi_release_locality(locality, false);
+
+	return ret;
+}

+static int tpm_xfer(const uint8_t *sendbuf, size_t send_size,
+	uint8_t *recvbuf, size_t *recv_size)
+{
+	unsigned int count;
+	int ret = 0;
+
+	/* switch endianness: big->little */
+	count = get_unaligned_be32(sendbuf + TPM_CMD_COUNT_BYTE);
+
+	if (count == 0)
+		return -1;
+
+	if (count > send_size)
+		return -1;
+
+	ret = tpm_spi_send(sendbuf, send_size);
+	if (ret < 0)
+		return ret;
+
+	ret = tpm_spi_recv(tpm_buf, sizeof(tpm_buf));
+	if (ret >= 0) {
+		if (ret > (int)*recv_size)
+			return -1;
+		memcpy(recvbuf, tpm_buf, ret);
+		*recv_size = ret;
+		ret = 0;
+	}
+
+	return ret;
+}

+static int tpm_spi_sendrecv_command(const void *command,
+			 void *response, size_t *size_ptr)
+{
+	int err;
+	size_t response_length;
+
+	if (response) {
+		response_length = *size_ptr;
+	} else {
+		response = cmd_response_buffer;
+		response_length = sizeof(cmd_response_buffer);
+	}
+
+	err = tpm_xfer(command, tpm_command_size(command),
+		       response, &response_length);
+
+	if (err < 0)
+		return err;
+
+	if (size_ptr)
+		*size_ptr = response_length;
+
+	return tpm_return_code(response);

+static int tpm2_pack_byte_string(uint8_t *str, size_t size, const char *format, ...)
+{
+	va_list args;
+	size_t offset = 0, length = 0;
+	uint8_t *data = NULL;
+	uint32_t value = 0;
+
+	va_start(args, format);
+	for (; *format; format++) {
+		switch (*format) {
+		case 'b':
+			offset = va_arg(args, size_t);
+			value = va_arg(args, int);
+			length = 1;
+			break;
+		case 'w':
+			offset = va_arg(args, size_t);
+			value = va_arg(args, int);
+			length = 2;
+			break;
+		case 'd':
+			offset = va_arg(args, size_t);
+			value = va_arg(args, uint32_t);
+			length = 4;
+			break;
+		case 's':
+			offset = va_arg(args, size_t);
+			data = va_arg(args, uint8_t *);
+			length = va_arg(args, uint32_t);
+			break;
+		default:
+			va_end(args);
+			return -1;
+		}
+
+		if (offset + length > size) {
+			va_end(args);
+			return -1;
+		}
+
+		switch (*format) {
+		case 'b':
+			str[offset] = value;
+			break;
+		case 'w':
+			put_unaligned_be16(value, str + offset);
+			break;
+		case 'd':
+			put_unaligned_be32(value, str + offset);
+			break;
+		case 's':
+			memcpy(str + offset, data, length);
+			break;
+		}
+	}
+	va_end(args);
+
+	return 0;
+}
