+static void greco_ack_gic_irq(void)
+{
+	u32 i;
+
+	for (i = 0; i < GRECO_IRQ_SIZE; i++)
+		regmap_writel(i, mmGIC_DISTRIBUTOR__6_GICD_CLRSPI_NSR);
+}
+
+#define PCIE_INTR_ERR_INTR_BIT		(BIT(0) | BIT(1))
+static void greco_ack_pci_axi_irq(void)
+{
+	u32 reg = regmap_readl(mmPCIE_WRAP_PCIE_IC_SEI_INTR_IND);
+	u32 bitmask = reg & PCIE_INTR_ERR_INTR_BIT;
+
+	/* Reset the indication */
+	regmap_writel(reg & ~bitmask, mmPCIE_WRAP_PCIE_IC_SEI_INTR_IND);
+	/* Clear the interrupt */
+	regmap_writel(BIT(0), mmPCIE_WRAP_AXI_INTR);
+	/* Ack PCI drain indication */
+	regmap_writel(0, mmPCIE_WRAP_AXI_DRAIN_IND);
+}
+
+void preboot_ack_irq(void)
+{
+	greco_ack_pci_axi_irq();
+	greco_ack_gic_irq();
+}
