+static void fs1_rst_work(struct k_work *work)
+{
+	uint32_t  retry = FSM_STATE_RETRIES;
+
+	/* Early reset notification to CXL */
+	fs1_rst_notify_cxl_arc();
+
+	/* Prepare xHA for reset */
+	fs1_rst_prepare_xha_reset();
+
+	/* Prepare CXL arc for reset */
+	fs1_rst_send_cxl_reset();
+
+	/* Set ISO for the blocks which undergo reset */
+	fs1_rst_isolation_cfg(true);
+
+	/* Issue unit reset to block under reset */
+	fs1_rst_issue_unit_rst(true);
+
+	/* wait for 0.5 usec */
+	k_busy_wait(500);
+
+	/* End unit reset */
+	fs1_rst_issue_unit_rst(false);
+
+	/* Clear ISO for the blocks which underwent reset */
+	fs1_rst_isolation_cfg(true);
+
+	/* set the BTM FSM state to FSM_GRAD_RST_DONE */
+	CFG_WREG32(IO0_H0_PSOC_BOOT(mmPSOC_BOOT_CONF_BTM_FSM), FSM_GRAD_RST_DONE);
+
+	/* wait for BTM FSM state done 0x0 */
+	while (CFG_RREG32(IO0_H0_PSOC_BOOT(mmPSOC_BOOT_CONF_BTM_FSM))) {
+		retry--;
+		k_busy_wait(1);
+		if (!retry) {
+			LOG_ERR("Timedout waiting for FSM state to be idle");
+			return;
+		}
+	}
+
+}
+
+__unused static void fs1_rst_callback(const struct device *dev, void *data)
+{
+	k_work_submit(&reset_work);
+}
