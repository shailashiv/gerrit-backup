[SW-36216] greco: security: lbw rr preparation

Add api for lbw range register configuration.

There are 18(0..17 rr idx) RRs for LBW
1) 14(0..13 rr idx) RRs with 4KB granularity
API: preboot_secure_lbw_rr_short(rr idx, min addr, max addr)

2) 4(14..17 rr idx) RRs with full 39 bit address range.
API: preboot_secure_lbw_rr_full(rr idx, min addr, max addr)
rr idx type : greco_sec_if_lbw_rrs

+static void preboot_secure_update_rr_reg(int rr_idx, u32 inst_offset, u32 min_reg, u32 max_reg,
+		u32 min, u32 max)
+{
+	u32 rr_base, rr_dest;
+
+	rr_base = rr_idx * sizeof(u32) + inst_offset;
+	rr_dest = rr_base + min_reg;
+	regmap_writel(min, rr_dest);
+	rr_dest = rr_base + max_reg;
+	regmap_writel(max, rr_dest);
+}
+
+static void preboot_secure_update_hit_reg(int rr_idx, struct preboot_sec_rrs_cfg *cfg, int inst_num,
+		bool write_only)
+{
+	u32 rr_base, rr_dest, hit;
+	/* Enable (hit) RR */
+	rr_base = cfg->instances_dist * inst_num;
+	rr_dest = rr_base + cfg->hit_wr_breg;
+	hit = regmap_readl(rr_dest);
+	regmap_writel(hit | BIT(rr_idx), rr_dest);
+
+	if (write_only)
+		return;
+
+	rr_dest = rr_base + cfg->hit_rd_breg;
+	hit = regmap_readl(rr_dest);
+	regmap_writel(hit | BIT(rr_idx), rr_dest);
+}
+
+#define MAX_LBW_SHORT_RR 14
+
+static void preboot_secure_lbw_rr_gen(int rr_idx, u32 min, u32 max,
+		enum preboot_sec_engine_type type, bool is_short)
+{
+	int i;
+	u32 min_breg, max_breg, rr_calc;
+	struct preboot_sec_rrs_cfg *cfg = &preboot_sec_lbw_rrs_tbl[type];
+
+	if (is_short) {
+		rr_calc = rr_idx;
+		min_breg = cfg->min_short_breg;
+		max_breg = cfg->max_short_breg;
+	} else {
+		rr_calc = rr_idx - MAX_LBW_SHORT_RR;
+		min_breg = cfg->min_full_breg;
+		max_breg = cfg->max_full_breg;
+	}
+
+	for (i = 0; i < cfg->instances_num; i++) {
+		preboot_secure_update_rr_reg(rr_calc, cfg->instances_dist * i,
+				min_breg, max_breg, min, max);
+
+		preboot_secure_update_hit_reg(rr_idx, cfg, i, false);
+	}
+}
+
+static void preboot_secure_lbw_rr_wrap(int rr_idx, u32 min, u32 max, bool is_short)
+{
+	int i;
+
+	for (i = TPC_MME_IF; i <= ELBI_OR_COMMON_IF2; i++)
+		preboot_secure_lbw_rr_gen(rr_idx, min, max, (enum preboot_sec_engine_type)i,
+				is_short);
+}
+
+__unused static void preboot_secure_lbw_rr_full(int rr_idx, u64 min64, u64 max64)
+{
+	u32 min = min64 & ~HL_CFG_BASE;
+	u32 max = max64 & ~HL_CFG_BASE;
+
+	preboot_secure_lbw_rr_wrap(rr_idx, min, max, false);
+}
+
+__unused static void preboot_secure_lbw_rr_short(int rr_idx, u64 min64, u64 max64)
+{
+	u32 min = (min64 & ~HL_CFG_BASE) >> SHIFT_4K_POS;
+	u32 max = (max64 & ~HL_CFG_BASE) >> SHIFT_4K_POS;
+
+	preboot_secure_lbw_rr_wrap(rr_idx, min, max, true);
+}
