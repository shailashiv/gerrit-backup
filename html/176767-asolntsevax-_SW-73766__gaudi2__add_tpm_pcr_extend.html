+{

+{

+
+	if (arcboot_rom_status_read(PROGRESS_BRINGUP_PCI_SEC))

+		image_ptr = ARCBOOT_PRIMARY;
+
+	if (image_ptr > ARCBOOT_SECONDARY) {

+	flash_image_offset = arcboot_flash_image_offset(image_ptr);
+	/* check if offset is within safe limits */
+	if ((flash_image_offset + FLASH_64KB_PADD) >= FLASH_VALID_MASK) {
+		cpu_boot_err_set(CPU_BOOT_ERR_TPM_FAIL);
+		return -1;

+	}
