[SW-20719] gaudi: introduce hbm init code

Add HBM initialization code as a part of requirement for cleaning the
HBM memory after FLR/reset/on power up.
The code is mainly copied from U-Boot, but multiple adjustments were
made to lower the compiled binary size.
The HBM frequency is set to a safe default of 800MHz.
All the debug abilities were stripped.
Error messages shortened or removed - depending on case.

+static int hbm_poll_timeout(u32 cfg_addr, u32 ref_val, u64 tout_us, u32 *val)
+{
+	u64 current, start = timestamp_get_us();
+	u64 timeout = start + tout_us;
+	bool overlap = false;
+	u32 reg;
+
+	if (timeout < start)
+		overlap = true;
+
+	while (1) {
+		reg = regmap_readl(cfg_addr);
+		if (reg == ref_val)
+			return 0;
+
+		current = timestamp_get_us();
+		if (!overlap) {
+			if (current > timeout || current <= start)
+				break;
+		} else {
+			if (current > timeout && current < start)
+				break;
+		}
+	}
+
+	*val = reg;
+
+	return -1;
+}

+static void ieee1500_inst(int device, u32 ch,
+				enum ieee1500_instruction instruction,
+				bool write)
+{
+	/*
+	 * Write instructions - WDR registers (0x9100 - 0x9188) should be
+	 * written prior to calling this function.
+	 * Read instructions - WDR registers (0x9800 - 0x9F88) should be read
+	 * after calling this function.
+	 */
+	u32 base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+	u32 wir, op, status, err;
+
+	/* Configure WDR length */
+	WREG32(base + 0x9020, ieee_wdr_len_t[instruction] - 1);
+
+	/*
+	 * Configure WIR
+	 * {ch, instruction} - ch may be 0xF to indicate ALL channels
+	 */
+	wir = (instruction & 0xFF) | ((ch & 0xF) << 8);
+	WREG32(base + 0x9024, wir);
+
+	/* Send WIR and write/read WDR */
+	op = write ? 0x6 : 0x5;
+	WREG32(base + 0x9028, op);
+
+	/* Poll for MLB interrupt */
+	err = hbm_poll_timeout(base + 0x9034, op,
+			       GAUDI_HBM_CFG_TIMEOUT_US, &status);
+	if (err) {
+		preboot_puts_hex_raw("MLB interrupt poll failed, dev: ",
+				     device, false);
+		preboot_puts_hex_raw(", ch: ", ch, false);
+		preboot_puts_hex(", status: ", status, false);
+	}
+
+	/* Clear MLB interrupts */
+	WREG32(base + 0x9034, 0);
+}

+				}

+				if (((received >> 0) & 1) !=
+						((expected  >>  22) & 1)) {
+					err++;
+				}
+			}
+
+			if (ch == 1) {
+				received = RREG32(base + 0x8000 + 0x04c);
+				if (((received >> 3) & 1) !=
+						((expected >> 20) & 1)) {
+					err++;
+				}
+			}
+		}
+	}
+
+	/*
+	 * Patterns 8 - 15
+	 */
+
+	/* Patterns loop */
+	for (pat = 8; pat < 15; pat++) {
+		for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+			/* Configure WDR write values */
+			for (i = 0; i < EXTEST_WDR_REG_NUM; i++) {
+				addr = base +
+					(GAUDI_HBM_CENTER_CHANNEL * 0x1000) +
+					0x100 + (i*4);
+				if (ch == pat - 8)
+					idx = 9 * EXTEST_WDR_REG_NUM + i;
+				else
+					idx = 8 * EXTEST_WDR_REG_NUM + i;
+
+				WREG32(addr, extest_wdr_pattern[idx]);
+			}
+
+			/* Perfrom EXTEST_TX command to the selected channel */
+			ieee1500_inst(device, hbm_ch_mapping[ch],
+				      ieee1500_extest_tx, EXTEST_WRITE_INST);
+
+			/* Compare results */
+			for (i = 0; i < EXTEST_WDR_REG_NUM; i++) {
+				if (ch == pat - 8)
+					idx = 9 * EXTEST_WDR_REG_NUM + i;
+				else
+					idx = 8 * EXTEST_WDR_REG_NUM + i;
+
+				expected = extest_wdr_pattern[idx];
+
+				received = RREG32(base + (ch * 0x1000) + 0x960
+								+ (i * 4));
+
+				mask = extest_wdr_tx_mask[i];
+
+				if ((received & mask) != (expected & mask))
+					err++;
+			}
+
+			/* Special case - pins TEMP[2:0] + CATTRIP */
+			if (ch == 0) {
+				received = RREG32(base + 0x8000 + 0x04c);
+				if (((received >> 2) & 1) !=
+						((expected  >>  20) & 1)) {
+					err++;
+				}
+
+				if (((received >> 1) & 1) !=
+						((expected >> 21) & 1)) {
+					err++;
+				}
+
+				if (((received >> 0) & 1) !=
+						((expected  >>  22) & 1)) {
+					err++;
+				}
+			}
+
+			if (ch == 1) {
+				received = RREG32(base + 0x8000 + 0x04c);
+				if (((received >> 3) & 1) !=
+						((expected >> 20) & 1)) {
+					err++;
+				}
+			}
+		}
+	}
+
+	return err;
+}

+	int i;

+static void config_mlb_state(u32 base, int ch, enum phy_state next_state)
+{
+	enum phy_state curr_state =
+				RREG32_MASK(base + ch * 0x1000 + 0x200, 0xFF00);
+	int offset = next_state - curr_state;
+	int i;
+
+	if (!offset)
+		return;
+
+	/* mdelay(20); */
+
+	for (i = 0; i < abs(offset); i++)
+		RMWREG32(base + ch * 0x1000 + 0x200,
+				(offset > 0) ? 1 : 2, 0xFF);
+
+	/* mdelay(20); */
+}

+static void set_mode_reg_by_mlb(u32 base, int ch, u32 num, u32 op)
+{
+	u32 count = 0, i, one = 1, parity;
+	u32 value = (num << 8) | op;
+
+	for (i = 0; i < 12; i++)
+		if (value & (one << i))
+			count++;
+
+	parity = count & 1;
+
+	WREG32(base + ch * 0x1000 + 0x210, (parity << 16) | value);
+	WREG32(base + ch * 0x1000 + 0x214, 4);
+
+	mdelay(1);
+}

+static void config_dram_state(struct hbm *hbm, u32 base,
+					int ch, enum dram_state state)
+{
+	config_mlb_state(base, ch, phy_state_initial);
+
+	if (state == dram_state_training) {
+		/* CA Par=en, Wr Par=en, Rd Par=en, WDBI=en, RDBI=en */
+		set_mode_reg_by_mlb(base, ch, 0, 0x73);
+		/* drive strength=15mA, WR=16 */
+		set_mode_reg_by_mlb(base, ch, 1, 0x70);
+		/* RL=22, WL=7 */
+		set_mode_reg_by_mlb(base, ch, 2, 0xA6);
+		/* BL=4, BG=enable, tRAS=33 */
+		set_mode_reg_by_mlb(base, ch, 3, 0xE1);
+		/* PL=0, DM=disable, ECC=enable */
+		set_mode_reg_by_mlb(base, ch, 4, 0x03);
+		/* Vref=50% */
+		set_mode_reg_by_mlb(base, ch, 15, 0x00);
+
+		/* Adjust MLB for PL=0 */
+		RMWREG32(base + ch * 0x1000 + 0x204, 0x0, 0x300);
+	} else if (state == dram_state_bypass) {
+		/* CA Par=en, Wr Par=en, Rd Par=en, WDBI=en, RDBI=en */
+		set_mode_reg_by_mlb(base, ch, 0, 0x70 |
+					(hbm->dbi_enable << 1) |
+					(hbm->dbi_enable << 0));
+		/* drive strength=15mA, WR=16 */
+		set_mode_reg_by_mlb(base, ch, 1, 0x70);
+		/* RL=22, WL=7 */
+		set_mode_reg_by_mlb(base, ch, 2, 0xA6);
+		/* BL=4, BG=enable, tRAS=33 */
+		set_mode_reg_by_mlb(base, ch, 3, 0xE1);
+		/* PL=2, DM=disable, ECC=enable */
+		set_mode_reg_by_mlb(base, ch, 4, 0x8 |
+					(hbm->ecc_enable << 1) |
+					(hbm->ecc_enable << 0));
+		/* Vref=50% */
+		set_mode_reg_by_mlb(base, ch, 15, 0x00);
+
+		/* Adjust MLB for PL=2 */
+		RMWREG32(base + ch * 0x1000 + 0x204, 0x2, 0x300);
+	}
+}

+static int mcbist_poll_completion(int sid, int dev)
+{
+	int err, ch, rc = 0;
+	u32 base, status;
+
+	base = GAUDI_HBM_CFG_BASE + dev * GAUDI_HBM_CFG_OFFSET;
+
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+		err = hbm_poll_timeout(base + ch * 0x1000 + 0x11C, 0x03000000,
+				       GAUDI_HBM_ECC_CFG_TIMEOUT_US, &status);
+		if (!err)
+			continue;
+
+		rc = -1;
+	}
+
+	return rc;

+static int mcbist_write_read_full(struct hbm *hbm, int unused)
+{
+	int device, ch, sid, acc_err = 0;
+	u32 base, pseudo_rand = 0x1, interleave = 0x0;
+	enum mc_bist_mode bist_mode = mc_bist_mode_once;
+
+	/* Clean transition from Functional mode to MCBIST mode */
+	for (device = 0; device < GAUDI_HBM_DEVICES; device++) {
+		base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+		for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+			config_mlb_state(base, ch, phy_state_initial);
+			config_mc_state(hbm, device, ch, mc_state_bist);
+			config_mlb_state(base, ch, phy_state_bypass);
+		}
+	}
+
+	for (sid = 0; sid < 2; sid++) {
+		/* Configure MCBIST for all channels */
+		for (device = 0; device < GAUDI_HBM_DEVICES; device++) {
+			base = GAUDI_HBM_CFG_BASE +
+					device * GAUDI_HBM_CFG_OFFSET;
+
+			for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+				WREG32(base + ch * 0x1000 + 0x100, pseudo_rand |
+						(interleave << 1) |
+						(bist_mode << 2) |
+						(hbm->ecc_enable  << 3));
+				WREG32(base + ch * 0x1000 + 0x104,
+						(0x0 | sid << 28 | sid << 29));
+				WREG32(base + ch * 0x1000 + 0x108, 0x800000);
+				WREG32(base + ch * 0x1000 + 0x10c, 0x12345678);
+				WREG32(base + ch * 0x1000 + 0x110, 0x9ABCDEF0);
+				WREG32(base + ch * 0x1000 + 0x114, 0x5A5A5A5A);
+				WREG32(base + ch * 0x1000 + 0x118, 0xA5A5A5A5);
+			}
+		}
+
+		/* Start MCBIST */
+		for (device = 0; device < GAUDI_HBM_DEVICES; device++) {
+			base = GAUDI_HBM_CFG_BASE +
+					device * GAUDI_HBM_CFG_OFFSET;
+
+			for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++)
+				WREG32(base + ch * 0x1000 + 0x11C, 0x3);
+		}
+
+		/* Poll for MCBIST completion */
+		for (device = 0; device < GAUDI_HBM_DEVICES; device++)
+			acc_err |= mcbist_poll_completion(sid, device);
+
+		if (acc_err)
+			break;
+	}
+
+	/* Clean transition from MCBIST mode to Functional mode */
+	for (device = 0; device < GAUDI_HBM_DEVICES; device++) {
+		base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+		for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+			config_mlb_state(base, ch, phy_state_initial);
+			config_mc_state(hbm, device, ch, mc_state_normal);
+			config_mlb_state(base, ch, phy_state_bypass);
+		}
+	}
+
+	return acc_err;
+}

+static int do_loopback(struct hbm *hbm, u32 base, int ch,
+			enum loopback_state state)
+{
+	int i, j = ch * 16/*, cnt = 0*/;
+	u32 val;
+
+	if (state == loopback_state_initial) {
+		/* Internal Loopback mode */
+		RMWREG32(base + ch * 0x1000 + 0x370, 3, 0xFF);
+		for (i = 0; i < 16; i++)
+			WREG32(base + ch * 0x1000 + 0x380 + i * 4,
+					hbm_golden_registers[i + j]);
+		return 0;
+	}
+
+	if (state != loopback_state_check)
+		return -1;
+
+	WREG32(base + ch * 0x1000 + 0x294, 1);
+	WREG32(base + ch * 0x1000 + 0x294, 0);
+
+	RMWREG32(base + ch * 0x1000 + 0x378, 0xF, 0xFF);
+	RMWREG32(base + ch * 0x1000 + 0x374, 0xF, 0xFF);
+
+	val = RREG32(base + ch * 0x1000 + 0x37C);
+	if (val)
+		return -1;
+
+	return 0;
+}

+static int internal_loopback_test(struct hbm *hbm, int device)
+{
+	u32 base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+	int ch, dw;
+
+	/* Add delay from previous SRX */
+	udelay(1);
+
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+		/* SRE */
+		WREG32(base + ch * 0x1000 + 0x214, 2);
+		/* ILB ctrl_rddv_sel */
+		WREG32(base + ch * 0x1000 + 0x298, 5);
+		/* Reset read FIFO */
+		WREG32(base + ch * 0x1000 + 0x294, 1);
+		WREG32(base + ch * 0x1000 + 0x294, 0);
+
+		do_loopback(hbm, base, ch, loopback_state_initial);
+
+		for (dw = 0; dw < 4; dw++)
+			set_dword_read_dll(base, ch, dw, 0);
+
+		for (dw = 0; dw < 4; dw++)
+			set_dword_write_dll(base, ch, dw, 0x50);
+
+		if (do_loopback(hbm, base, ch, loopback_state_check))
+			return 1;
+
+		WREG32(base + ch * 0x1000 + 0x378, 0);
+		WREG32(base + ch * 0x1000 + 0x370, 0);
+		/* functional ctrl_rddv_sel */
+		WREG32(base + ch * 0x1000 + 0x298, 4);
+		/* SRX */
+		WREG32(base + ch * 0x1000 + 0x214, 1);
+	}
+
+	return 0;
+}

+static void acc_training(struct hbm *hbm, int device)
+{
+	u32 base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+	u32 dll_inc, status;
+	int ch, dw, err;
+
+	if (hbm->inc == 16)
+		dll_inc = 3;
+	else if (hbm->inc == 4)
+		dll_inc = 2;
+	else
+		dll_inc = 1;
+
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+		for (dw = 0; dw < 5; dw++)
+			set_acc_dll(base, ch, dw, hbm->acc_initial);
+
+		WREG32(base + ch * 0x1000 + 0x444,
+				(hbm->training_delay << 8) | dll_inc);
+	}
+
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch += 2) {
+		WREG32(base + ch * 0x1000 + 0x440, 0x1F);
+		WREG32(base + (ch + 1) * 0x1000 + 0x440, 0xF);
+	}
+
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+		err = hbm_poll_timeout(base + ch * 0x1000 + 0x440, 0,
+				       GAUDI_HBM_CFG_TIMEOUT_US, &status);
+		if (err) {
+			preboot_puts("ACC done polling failed\n");
+			return;
+		}
+	}
+
+	copy_acc(base);
+
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+		if (!aword_acc_window(base, ch))
+			preboot_puts("aword acc_window can't be 0\n");
+
+		for (dw = 0; dw < 4; dw++)
+			if (!dword_acc_window(base, ch, dw))
+				preboot_puts("dword acc_window can't be 0\n");
+	}
+}

+static void check_training_result(struct hbm *hbm, int dev)
+{
+	int ch, dw, min_window_size = 50; /* percentage of UI */
+
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+		for (dw = 0; dw < 5; dw++)
+			if (acc[ch][dw] <= hbm->acc_initial)
+				preboot_puts("HBM err: ACC UI size not obtained\n");
+
+		if (aword[ch][1] == 0 || aword[ch][2] == 0 ||
+			aword[ch][3] < ((acc[ch][0] * min_window_size) / 100))
+			preboot_puts("HBM err: valid AWORD window not found\n");
+
+		for (dw = 0; dw < 4; dw++) {
+			if (dword_wr[ch][dw][2] == 0 ||
+			    dword_wr[ch][dw][3] < ((acc[ch][dw + 1] * min_window_size) / 100))
+				preboot_puts("HBM err: valid DWORD Write window not found\n");
+
+			if (dword_rd[ch][dw][2] == 0 ||
+			    dword_rd[ch][dw][3] < ((acc[ch][dw + 1] * min_window_size) / 100))
+				preboot_puts("HBM err: valid DWORD Read window not found\n");
+		}
+	}
+}

+static void phy_init(int device)
+{
+	u32 base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+	u32 ctrl_rddv_sel = 4;
+	u32 rdatvld_dly;
+	u32 dev_id;
+	int ch;
+
+	/* bits [31:16] are the device id */
+	dev_id = preboot_get_vendor_id_device_id() >> 16;
+
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+		switch (dev_id) {
+		case PCI_IDS_GAUDI_HL2000M:
+			if (ch == 0 || ch == 4) {
+				/* Release PHY RSTN (set bit 1) */
+				RMWREG32(base + ch * 0x1000 + 0x284, 1, 0x2);
+				mdelay(1);
+
+				/* non-deskew PLL settings 2Gbps */
+				WREG32(base + ch * 0x1000 + 0x2B8, 0x19180F3F);
+				WREG32(base + ch * 0x1000 + 0x2BC, 0x0000001B);
+
+				/* Release PLL power-down (clear bit 0) */
+				RMWREG32(base + ch * 0x1000 + 0x284, 0, 0x1);
+				mdelay(1);
+
+				/* Release PLL clock divider rstn */
+				WREG32(base + ch * 0x1000 + 0x288, 0x0000001);
+				mdelay(1);
+
+				/* Release PLL stable_n (clear bit 7) */
+				RMWREG32(base + ch * 0x1000 + 0x284, 0, 0x80);
+				mdelay(1);
+			}
+
+			/* Release MLB & PHASE FIFO RSTN (set bit 7) */
+			RMWREG32(base + ch * 0x1000 + 0x174, 1, 0x80);
+			mdelay(1);
+			/* PHASE FIFO start (set bit 15) */
+			RMWREG32(base + ch * 0x1000 + 0x174, 1, 0x8000);
+
+			if (ch == 0 || ch == 4) {
+				/* Release DLL power-down */
+				WREG32(base + ch * 0x1000 + 0x28C, 0x0000000);
+				mdelay(1);
+
+				/* Drive strength */
+				WREG32(base + ch * 0x1000 + 0x2A0, 0x681F0);
+				WREG32(base + ch * 0x1000 + 0x2A4, 0x6A1F0);
+			}
+			break;
+		default:
+			if (ch == 0 || ch == 4) {
+				/* PLL settings */
+				WREG32(base + ch * 0x1000 + 0x2B8, 0x0CD80303);
+				WREG32(base + ch * 0x1000 + 0x2BC, 0x0000001E);
+
+				/* PLL and DLL power up */
+				RMWREG32(base + ch * 0x1000 + 0x288, 0x1, 0x1);
+				mdelay(1);
+				RMWREG32(base + ch * 0x1000 + 0x284, 0x0, 0x1);
+				mdelay(1);
+				RMWREG32(base + ch * 0x1000 + 0x28C, 0x0, 0x1);
+				mdelay(1);
+
+				/* Drive strength */
+				WREG32(base + ch * 0x1000 + 0x2A0, 0x681F0);
+				WREG32(base + ch * 0x1000 + 0x2A4, 0x6A1F0);
+			}
+		}
+
+		/* Reset read FIFO */
+		RMWREG32(base + ch * 0x1000 + 0x294, 0x1, 0x1);
+		RMWREG32(base + ch * 0x1000 + 0x294, 0x0, 0x1);
+
+		/* DFI setting - PHY */
+		WREG32(base + ch * 0x1000 + 0x208, 0x140106);
+
+		/* DFI setting - MC:
+		 * r_t_rddata_en=0x15, r_t_phy_wrlat=0x6, r_t_phy_wrdata=0x1,
+		 * r_rrq_switch_lvl_low=0x0. TODO: remove later
+		 */
+		WREG32(base + (ch * 0x1000) + 0x054, 0x10615);
+
+		/* trfc_sb_fix=152, r_r2w = 18 to support PL=2
+		 * TODO: remove later
+		 */
+		WREG32(base + (ch * 0x1000) + 0x04C, 0xA0030812);
+
+		/* ctrl_rddv_sel */
+		WREG32(base + ch * 0x1000 + 0x298, ctrl_rddv_sel);
+
+		/* Enable HBM CK, set PL=0, set rdatvld_dly */
+		if (dev_id != PCI_IDS_GAUDI_HL2000M)
+			rdatvld_dly = ctrl_rddv_sel + 8;
+		else
+			rdatvld_dly = ctrl_rddv_sel + 10;
+
+		if (ch == 0 || ch == 1 || ch == 4 || ch == 5)
+			WREG32(base + ch * 0x1000 + 0x204,
+					0x1 | (rdatvld_dly << 16));
+		else if (ch == 2 || ch == 3 || ch == 6 || ch == 7)
+			WREG32(base + ch * 0x1000 + 0x204,
+					0x1 | ((rdatvld_dly + 1) << 16));
+
+		/* Enable ECC. TODO: remove later */
+		WREG32(base + (ch * 0x1000) + 0x000, 0x413000);
+	}
+}

+static void dram_rstn(int device, enum cke_state cke)
+{
+	u32 base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+	int ch;
+
+	/* SRE (CKE low) */
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++)
+		WREG32(base + ch * 0x1000 + 0x214, 0x2);
+
+	/* WRCK disable */
+	WREG32(base + GAUDI_HBM_CENTER_CHANNEL * 0x1000 + 0x18, 0x0);
+
+	/* WRSTn low */
+	WREG32(base + GAUDI_HBM_CENTER_CHANNEL * 0x1000 + 0x14, 0x0);
+
+	/* hbm_RESETn low */
+	WREG32(base + GAUDI_HBM_CENTER_CHANNEL * 0x1000 + 0x10, 0x0);
+
+	/* tINIT1=200us for initial power-up or tPW_RESET=1us with stable
+	 * power
+	 */
+	mdelay(1);
+
+	/* hbm_RESETn high */
+	WREG32(base + GAUDI_HBM_CENTER_CHANNEL * 0x1000 + 0x10, 0x1);
+
+	/* tINIT3: 500us from RESETn deassertion to WRSTn/CKE high */
+	mdelay(1);
+
+	/* WRCK divider */
+	WREG32(base + GAUDI_HBM_CENTER_CHANNEL * 0x1000 + 0x00, 0x1);
+
+	/* WRSTn high */
+	WREG32(base + GAUDI_HBM_CENTER_CHANNEL * 0x1000 + 0x14, 0x1);
+
+	/* WRCK enable */
+	WREG32(base + GAUDI_HBM_CENTER_CHANNEL * 0x1000 + 0x18, 0x1);
+
+	/* IEEE1500 IO strength
+	 * RMWREG32(base + GAUDI_HBM_CENTER_CHANNEL * 0x1000 + 0x40, 0xF, 0xF);
+	 */
+
+	/* tINIT4 / tWINIT1 */
+	udelay(1);
+
+	if (cke == cke_state_high) {
+		/* SRX (CKE high) */
+		for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++)
+			WREG32(base + ch * 0x1000 + 0x214, 0x1);
+
+		/* tINIT5 */
+		udelay(1);
+	}
+}

+static void init_mc(struct hbm *hbm, int device)
+{
+	u32 base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+	int ch;
+
+	/* Cleanup: Reset HBM + MRS setup */
+	dram_rstn(device, cke_state_high);
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++)
+		config_dram_state(hbm, base, ch, dram_state_bypass);
+
+	/* Set dfi_init_complete */
+	WREG32(base + 0x9004, 1);
+
+	/* Enable DBI (MCCTL) */
+	if (hbm->dbi_enable) {
+		WREG32(base + 0x8004, 0xFFFFFFFF);
+		WREG32(base + 0x8008, 0xFFFFFFFF);
+		WREG32(base + 0x800C, 0xFFFFFFFF);
+		WREG32(base + 0x8010, 0xFFFFFFFF);
+		WREG32(base + 0x8014, 0xFFFFFFFF);
+		WREG32(base + 0x8018, 0xFFFFFFFF);
+		WREG32(base + 0x801C, 0xFFFFFFFF);
+		WREG32(base + 0x8020, 0xFFFFFFFF);
+	}
+
+	/* Enable CATTRIP interrupt */
+	WREG32(base + 0x8000 + 0x044, 0x8);
+
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+		/* Reset PHY read FIFO */
+		WREG32(base + (ch * 0x1000) + 0x294, 1);
+		WREG32(base + (ch * 0x1000) + 0x294, 0);
+
+		/* Pass control to MC */
+		config_mlb_state(base, ch, phy_state_bypass);
+		config_mc_state(hbm, device, ch, mc_state_normal);
+
+		/* Basic configuration */
+		WREG32(base + (ch * 0x1000) + 0x000,
+				0x21012010 | (hbm->ecc_enable << 22));
+
+		/* Enable interrupts. Clear them from leftovers from init. */
+		RMWREG32(base + (ch * 0x1000) + 0x060, 0x8, 0xF);
+		RMWREG32(base + (ch * 0x1000) + 0x070, 0x8, 0xF);
+		WREG32(base + (ch * 0x1000) + 0x06C, 0x1F1F);
+		WREG32(base + (ch * 0x1000) + 0x07C, 0x1F1F);
+		WREG32(base + (ch * 0x1000) + 0x068, 0x817);
+		WREG32(base + (ch * 0x1000) + 0x078, 0x817);
+		RMWREG32(base + (ch * 0x1000) + 0x060, 0x0, 0xF);
+		RMWREG32(base + (ch * 0x1000) + 0x070, 0x0, 0xF);
+	}
+}

+static void hardware_training(struct hbm *hbm, int device)
+{
+	u32 base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+	u32 dll_inc, status;
+	int ch, err;
+
+	reduce_pll();
+
+	if (hbm->inc == 16)
+		dll_inc = 3;
+	else if (hbm->inc == 4)
+		dll_inc = 2;
+	else
+		dll_inc = 1;
+
+	/* Reset read FIFO */
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+		WREG32(base + ch * 0x1000 + 0x294, 1);
+		WREG32(base + ch * 0x1000 + 0x294, 0);
+	}
+
+	acc_training(hbm, device);
+
+	/* AWORD training */
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++)
+		WREG32(base + ch * 0x1000 + 0x454,
+			((hbm->hardware_training_repeat - 1) << 16) |
+				(hbm->training_delay << 8) | dll_inc);
+
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+		WREG32(base + ch * 0x1000 + 0x450, 1);
+		err = hbm_poll_timeout(base + ch * 0x1000 + 0x450, 0,
+				       GAUDI_HBM_CFG_TIMEOUT_US, &status);
+		if (err) {
+			preboot_puts("AWORD write done test failed\n");
+			goto hw_train_out;
+		}
+	}
+
+	/* Cleanup: Reset HBM + MRS setup */
+	dram_rstn(device, cke_state_high);
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++)
+		config_dram_state(hbm, base, ch, dram_state_training);
+
+	/* DWORD read training */
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++)
+		WREG32(base + ch * 0x1000 + 0x484,
+			((hbm->hardware_training_repeat - 1) << 16) |
+			(hbm->training_delay << 8) | dll_inc);
+
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+		WREG32(base + ch * 0x1000 + 0x480, 1);
+		err = hbm_poll_timeout(base + ch * 0x1000 + 0x480, 0,
+				       GAUDI_HBM_CFG_TIMEOUT_US, &status);
+		if (err) {
+			preboot_puts("DWORD read done test failed\n");
+			goto hw_train_out;
+		}
+	}
+
+	/* Cleanup: Reset HBM + MRS setup */
+	dram_rstn(device, cke_state_high);
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++)
+		config_dram_state(hbm, base, ch, dram_state_training);
+
+	/* DWORD write training */
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++)
+		WREG32(base + ch * 0x1000 + 0x464,
+			((hbm->hardware_training_repeat - 1) << 16) |
+			(hbm->training_delay << 8) | dll_inc);
+
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+		WREG32(base + ch * 0x1000 + 0x460, 1);
+		err = hbm_poll_timeout(base + ch * 0x1000 + 0x460, 0,
+				       GAUDI_HBM_CFG_TIMEOUT_US, &status);
+		if (err) {
+			preboot_puts("DWORD write done test failed\n");
+			goto hw_train_out;
+		}
+	}
+
+hw_train_out:
+	read_hardware_training_result(device);
+
+	restore_pll();
+}

+static int init_ecc(struct hbm *hbm, int unused)
+{
+	u32 base;
+	u32 status;
+	int device, err, ch, sid;
+
+	/* Clean transition from Functional mode to MCBIST mode */
+	for (device = 0; device < GAUDI_HBM_DEVICES; device++) {
+		base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+		for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+			config_mlb_state(base, ch, phy_state_initial);
+			config_mc_state(hbm, device, ch, mc_state_bist);
+			config_mlb_state(base, ch, phy_state_bypass);
+		}
+	}
+
+	/* Configure MCBIST engine for ECC init in all channels */
+	for (device = 0; device < GAUDI_HBM_DEVICES; device++) {
+		base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+		for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+			WREG32(base + ch * 0x1000 + 0x100, 0x100);
+			WREG32(base + ch * 0x1000 + 0x108, 0x800000);
+			WREG32(base + ch * 0x1000 + 0x10c, 0x0);
+			WREG32(base + ch * 0x1000 + 0x110, 0x0);
+			WREG32(base + ch * 0x1000 + 0x114, 0xFFFFFFFF);
+			WREG32(base + ch * 0x1000 + 0x118, 0xFFFFFFFF);
+		}
+	}
+
+	for (sid = 0; sid < 2; sid++) {
+		/* Start MCBIST engine */
+		for (device = 0; device < GAUDI_HBM_DEVICES; device++) {
+			base = GAUDI_HBM_CFG_BASE +
+						device * GAUDI_HBM_CFG_OFFSET;
+			for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+				WREG32(base + ch * 0x1000 + 0x104,
+						(0x0 | sid << 28 | sid << 29));
+				WREG32(base + ch * 0x1000 + 0x11C, 0x3);
+			}
+		}
+
+		/* Poll for MCBIST completion */
+		for (device = 0; device < GAUDI_HBM_DEVICES; device++) {
+			base = GAUDI_HBM_CFG_BASE +
+						device * GAUDI_HBM_CFG_OFFSET;
+			for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+				err = hbm_poll_timeout(
+						base + ch * 0x1000 + 0x11C,
+						0x03000000,
+						GAUDI_HBM_ECC_CFG_TIMEOUT_US,
+						&status);
+				if (err) {
+					preboot_puts("ECC init failed\n");
+					config_mc_state(hbm, device, ch,
+							mc_state_normal);
+					return -1;
+				}
+			}
+		}
+	}
+
+	/* Clean transition from MCBIST mode to Functional mode */
+	for (device = 0; device < GAUDI_HBM_DEVICES; device++) {
+		base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+		for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+			config_mlb_state(base, ch, phy_state_initial);
+			config_mc_state(hbm, device, ch, mc_state_normal);
+			config_mlb_state(base, ch, phy_state_bypass);
+		}
+	}
+
+	return 0;
+}

+static int lane_detection(int device)
+{
+	u32 base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+	int ch, rc;
+
+	/* Enter IODC mode */
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++)
+		WREG32(base + ch * 0x1000 + 0x900, 1);
+
+	rc = extest_rx(device);
+	if (rc)
+		return rc;
+
+	rc = extest_tx(device);
+	if (rc)
+		return rc;
+
+	/* Exit IODC mode */
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++)
+		WREG32(base + ch * 0x1000 + 0x900, 0);
+
+	return 0;
+}

+static void lane_remap(int device)
+{
+	u32 base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+	u32 reg_800, reg_804, reg_808;
+	u32 remap_info_dw01, remap_info_dw23, remap_info_aw;
+	int ch;
+
+	for (ch = 0; ch < GAUDI_HBM_CHANNELS; ch++) {
+		/* Read HBM lane repair eFuse information */
+		ieee1500_inst(device, ch, ieee1500_hard_lane_repair,
+			      EXTEST_READ_INST);
+		reg_800 = RREG32(base + (GAUDI_HBM_CENTER_CHANNEL * 0x1000) +
+							0x800 + (ch * 0x100));
+		reg_804 = RREG32(base + (GAUDI_HBM_CENTER_CHANNEL * 0x1000) +
+							0x804 + (ch * 0x100));
+		reg_808 = RREG32(base + (GAUDI_HBM_CENTER_CHANNEL * 0x1000) +
+							0x808 + (ch * 0x100));

+		/* Configure MLB according to remapped lanes */
+		remap_info_dw01 = reg_800;
+		remap_info_aw   = reg_804 & 0xFF;
+		remap_info_dw23 = ((reg_804 & 0xFFFFFF00) >> 8) |
+						((reg_808 & 0xFF) << 24);
+		WREG32(base + (ch*0x1000) + 0x2F0, remap_info_dw01);
+		WREG32(base + (ch*0x1000) + 0x2F4, remap_info_dw23);
+		WREG32(base + (ch*0x1000) + 0x2F8, remap_info_aw);
+	}

+static int temperature_read(int device)

+	base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;

+static void hbm_reset(void)
+{
+	/* assert reset: set bits 33-36 (HBM) */
+	regmap_writel(0, mmPSOC_GLOBAL_CONF_UNIT_RST_N_L);
+	regmap_writel(0x1E, mmPSOC_GLOBAL_CONF_UNIT_RST_N_H);
+	regmap_writel(1, mmPSOC_GLOBAL_CONF_UNIT_RST_N);
+	udelay(250);
+	/* deassert reset */
+	regmap_writel(0, mmPSOC_GLOBAL_CONF_UNIT_RST_N_L);
+	regmap_writel(0, mmPSOC_GLOBAL_CONF_UNIT_RST_N_H);
+	regmap_writel(1, mmPSOC_GLOBAL_CONF_UNIT_RST_N);
+}

+/**
+ * gaudi_hbm_init - Initialize HBM controller of the chip and HBM memories
+ *		    Scrub the memories.
+ *
+ * Return: 0 for success, 1 for failure
+ *
+ */
+int gaudi_hbm_init(void)
+{
+	struct hbm *hbm = &hbm_data;
+	int dev, err;
+
+	hbm_reset();
+
+	for (dev = 0; dev < GAUDI_HBM_DEVICES; dev++) {
+		/* Initialization */
+
+		/* PHY requires 100ns delay from mc_rstn */
+		udelay(1);
+
+		phy_init(dev);
+		dram_rstn(dev, cke_state_low);
+
+		/* Internal Loopback (bring-up only) */
+		if (internal_loopback_test(hbm, dev)) {
+			preboot_puts("HBM intern loopback failed\n");
+			return 1;
+		}
+
+		/* IO connectivity test (bring-up only) */
+		if (lane_detection(dev)) {
+			preboot_puts("HBM lane det failed\n");
+			return 1;
+		}
+		dram_rstn(dev, cke_state_low);
+
+		/* Lane remapping */
+		lane_remap(dev);
+
+		/* Training */
+		hardware_training(hbm, dev);
+		check_training_result(hbm, dev);
+
+		/* Initialize MC for functional operation */
+		init_mc(hbm, dev);
+	}
+
+	/* MC BIST (validate Training result) */
+	err = mcbist_write_read_full(hbm, 0);
+	if (err)
+		return 1;
+
+	/* If ECC enabled, perform ECC scrubbing */
+	if (hbm->ecc_enable)
+		if (init_ecc(hbm, 0))
+			return 1;
+
+	/* Temperature readout and Enable traffic */
+	for (dev = 0; dev < GAUDI_HBM_DEVICES; dev++) {
+		temperature_check(dev);
+		enable_traffic(dev);
+	}
+
+	return 0;
