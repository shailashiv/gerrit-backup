Commit Message

[SW-184188] gaudi3: clear parc intr at ppboot

At bootup, PARC SEI interrupt is set due to invalid register access
by BOOT FSM [H9-5613]. Currently this was being cleared
at preboot but is moved to ppboot to avoid intr detection at early
preboot.

+static void clear_parc_interrupt(void)
+{
+	uint32_t cause, cause_offset, clr_offset, rz_clr_offset, mstrif_clr_offset;
+	uint32_t die_offset = mmD1_CPU_INT_AGG_SHARED_SEI_INT_MSG_BASE -
+				mmD0_CPU_INT_AGG_SHARED_SEI_INT_MSG_BASE;
+
+	/*
+	 * PARC interrupt is generated due to invalid register
+	 * access in boot FSM [H9-5613].
+	 * Clear it in early preboot stage
+	 */
+	cause_offset =
+		D0_PARC_GLOBAL_CONF_REG(mmPARC_GLOBAL_CONF_SEI_INTR_CTRL_CAUSE);
+	clr_offset =
+		D0_PARC_GLOBAL_CONF_REG(mmPARC_GLOBAL_CONF_SEI_INTR_CTRL_CLEAR);
+	rz_clr_offset =
+		D0_NRTR0_2CH_CTRL0_LBW_CH_RAZWI_LBW_CH1_REG(
+			mmRTR_CTRL_LBW_2_CH_RAZWI_LBW_CH1_ADEC_RAZWI_AW_CLR);
+	mstrif_clr_offset =
+		D0_PARC_MSTR_IF_XRESP_LBW_REG(mmMSTR_IF_V1_XRESP_LBW_INTR_CTRL_CLEAR);
+
+	/* Die 0 */
+	cause = regmap_readl(cause_offset);
+	/* Clear interrupt (W1C) */
+	regmap_writel(BIT(0), rz_clr_offset);
+	regmap_writel(MSTR_IF_V1_XRESP_LBW_INTR_CTRL_CLEAR_BRESP_ERR_M,
+		      mstrif_clr_offset);
+	regmap_writel(cause, clr_offset);
+	/* Clear interrupt at agg */
+	regmap_writel(AGG_SEI_PARC_INT_IDX_STS,
+		      mmD0_CPU_INT_AGG_SHARED_SEI_INT_MSG_BASE +
+		      mmINT_AGG_SHARED_SEI_INT_MSG_STS_0);
+
+	if (hl_is_dual_die_sku()) {
+		/* Die 1 */
+		cause = mirror_readl(cause_offset + die_offset);
+		/* Clear interrupt (W1C) */
+		mirror_writel(BIT(0), rz_clr_offset + die_offset);
+		mirror_writel(MSTR_IF_V1_XRESP_LBW_INTR_CTRL_CLEAR_BRESP_ERR_M,
+			      mstrif_clr_offset + die_offset);
+		mirror_writel(cause, clr_offset + die_offset);
+		/* Clear interrupt at agg */
+		mirror_writel(AGG_SEI_PARC_INT_IDX_STS,
+			      mmD0_CPU_INT_AGG_SHARED_SEI_INT_MSG_BASE +
+			      mmINT_AGG_SHARED_SEI_INT_MSG_STS_0 + die_offset);
+	}
+}
