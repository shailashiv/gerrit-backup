+static void pcie_config(void)
+{
+	uint8_t pcie_prstn, sram_init_done;
+	uint32_t boot_strap;
+
+	if (pcie_ltssm_en()) {
+		do {
+			pcie_prstn = regmap_readl(mmPSOC_GLOBAL_CONF_PRSTN);
+		} while (pcie_prstn != 1);
+
+		regmap_writel(0, mmPSOC_GLOBAL_CONF_PCIE_PHY_RST_N);
+		regmap_writel(1, mmPCIE_PHY_CR_PARA_SEL_0);
+		regmap_writel(1, mmPCIE_PHY_CR_PARA_SEL_1);
+
+		boot_strap = regmap_readl(mmPSOC_GLOBAL_CONF_BOOT_STRAP_PINS_L);
+
+		if (boot_strap &
+		    PSOC_GLOBAL_CONF_BOOT_STRAP_PINS_L_SRIS_MODE_MASK)
+			regmap_writel(1, mmPCIE_PHY_PIPE_RX_SRIS_MODE_EN);
+
+		regmap_writel(1, mmPSOC_GLOBAL_CONF_PCIE_PHY_RST_N);
+		do {
+			sram_init_done =
+				regmap_readl(mmPCIE_PHY_PHYX_SRAM_INIT_DONE_0);
+		} while (sram_init_done != 1);
+
+		do {
+			sram_init_done =
+				regmap_readl(mmPCIE_PHY_PHYX_SRAM_INIT_DONE_1);
+		} while (sram_init_done != 1);
+
+		regmap_writel(1, mmPCIE_PHY_PHYX_SRAM_EXT_LD_DONE_0);
+		regmap_writel(1, mmPCIE_PHY_PHYX_SRAM_EXT_LD_DONE_1);
+
+		/* Configure IATU */
+		regmap_writel(0x300000, mmPCIE_AUX_DBI);
+		regmap_writel(0xF0000000, mmPCIE_DBI_CORR_ERR_MASK_OFF);
+		regmap_writel(0x7F, mmPCIE_DBI_ADV_ERR_CAP_CTRL_OFF);
+		regmap_writel(0, mmPCIE_DBI_AER_EXT_CAP_HDR_OFF);
+		regmap_writel(0xC0080000, mmPCIE_DBI_UNCORR_ERR_STATUS_OFF);
+		regmap_writel(0x3, mmPCIE_AUX_LTSSM_EN);
+	}
+
+}
