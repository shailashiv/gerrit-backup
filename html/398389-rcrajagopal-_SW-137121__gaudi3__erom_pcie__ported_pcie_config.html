+static void pcie_phy_wait_for_sram_init_done(void)

+void hl_erom_bringup_pcie_link(void)
+{
+	u8 ltssm_en;
+
+	/* Disable PIF isolation */
+	regmap_writel(0x00, D0_PSOC_BOOT_CONF_REG(mmPSOC_BOOT_CONF_D0_PIF_ISO));
+
+	/* Disable PCIe error responses (both HBW and LBW) */
+	regmap_writel(0x1, D0_PCIE_WRAP_REG(mmPCIE_WRAP_PCIE_RSP_OK_0));
+	regmap_writel(0x1, D0_PCIE_WRAP_REG(mmPCIE_WRAP_PCIE_RSP_OK_1));
+
+	/* Wait for PRSTn */
+	while (!regmap_readl(D0_PSOC_GLOBAL_CONF_REG(mmGLOBAL_CONF_PRSTN)))
+		;
+
+	ltssm_en = regmap_readl(D0_PCIE_AUX_REG(mmPCIE_AUX_LTSSM_EN)) &
+					PCIE_AUX_LTSSM_EN_IND_M;
+
+	if (!ltssm_en) {
+		/* Reset PCIe PHY */
+		regmap_writel(0, D0_PSOC_GLOBAL_CONF_REG(mmGLOBAL_CONF_PCIE_PHY_RST_N));
+
+		pcie_phy_para_sel_set();
+
+		if (get_sris_mode_from_strap()) {
+			regmap_writel(1, D0_PCIE_PHY_REG(mmPCIE_PHY_PIPE_RX_SRIS_MODE_EN));
+			pcie_phy_set_lane_refclk_mode();
+			pcie_phy_set_pll_short_lock();
+			regmap_writel(1, D0_PCIE_AUX_REG(mmPCIE_AUX_APP_SRIS_MODE));
+		}
+
+		if (get_srns_mode_from_strap()) {
+			pcie_phy_set_lane_refclk_mode();
+			pcie_phy_set_pll_short_lock();
+		}
+
+		pcie_phy_ssc_en_set();
+
+		/* Take the PHY out of reset */
+		regmap_writel(1, D0_PSOC_GLOBAL_CONF_REG(mmGLOBAL_CONF_PCIE_PHY_RST_N));
+
+		pcie_phy_wait_for_sram_init_done();
+
+		pcie_phy_wa_snps_dcc();
+
+		pcie_phy_set_sram_ext_ld_done();
+
+		pcie_set_pcie_device_vendor_id();
+
+		hl_pcie_clear_boot_termination();
+
+		regmap_writel(0x3, D0_PCIE_AUX_REG(mmPCIE_AUX_LTSSM_EN));
+	} else {
+		hl_pcie_clear_boot_termination();
