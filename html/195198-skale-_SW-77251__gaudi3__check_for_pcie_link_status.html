Commit Message

[SW-77251] gaudi3: check for pcie link status

Currenlty as part of ROM recovery, PCIe link configuration
is performed and ROM doesn't check whether PCIe link is up or not.

In order to support firmware update via inband, it is required
for ROM to check for pcie link status and update progress accordingly.
This patch add supports to check for link as part of command modules
of PCIe and update progress bit once link is up.

+bool is_pcie_initialized(void)

+	uint32_t prstn, ltssm, smlh, rdlh;
+
+	if (arcboot_rom_global.pci_link_up)
+		return true;
+
+	prstn = regmap_readl(mmD0_PSOC_GLOBAL_CONF_PRSTN);
+	ltssm = regmap_readl(mmD0_PCIE_AUX_LTSSM_EN);
+	smlh = regmap_readl(mmD0_PCIE_AUX_SMLH_LINK_UP);
+	rdlh = regmap_readl(mmD0_PCIE_AUX_RDLH_LINK_UP);
+
+	if (prstn && smlh && rdlh) {
+		arcboot_rom_global.pci_link_up = true;
+		regmap_writel(0x3, mmD0_PCIE_AUX_LTSSM_EN);
+		arcboot_rom_status_update(PROGRESS_ARCBOOT_ROM_PCIE_LINK_UP);
+		return true;
+	}
+
+	if (!prstn && ltssm) {
+		regmap_writel(0, mmD0_PCIE_AUX_LTSSM_EN);
+		return false;
+	}
+
+	if (prstn && !ltssm) {
+		regmap_writel(0x3, mmD0_PCIE_AUX_LTSSM_EN);
+		return false;
+	}
+
+	return false;

+#include <arcboot.h>
+#include <cmd_module.h>
+
+
+void module_handle_cmds(struct cmd_module_ops cmd_ops[], int size)
+{
+	int i = 0, j;
+	enum cmd_check_res res = NOT_FOUND;
+
+	for (j = 0; j < size; j++)
+		cmd_ops[j].setup();
+
+	while (1) {
+		res = cmd_ops[i].check();
+		switch (res) {
+		case NOT_FOUND:
+			break;
+		case FOUND_FAST_RETURN:
+			cmd_ops[i].run_fast();
+			break;
+		case FOUND_SLOW_RETURN:
+			for (j = 0; j < size; j++)
+				if (i != j)
+					cmd_ops[j].disable();
+
+			cmd_ops[i].run();
+			for (j = 0; j < size; j++)
+				if (i != j)
+					cmd_ops[j].enable();
+
+			break;
+		case FOUND_NO_RETURN:
+			for (j = 0; j < size; j++)
+				if (i != j)
+					cmd_ops[j].teardown();
+
+			cmd_ops[i].run_noreturn();
+			break;
+		default:
+			break;
+		}
+
+		if (++i == size)
+			i = 0;
+	}
+}

+#ifndef _HL_CMD_MODULE_H_
+#define _HL_CMD_MODULE_H_
+
+#include <stdbool.h>
+#include <hl_boot_if.h>
+/*
+ * enum cmd_check_res	- used to specify various results that check() routine
+ *			  can return following its check.
+ *
+ * NOT_FOUND		- check() routine has not discovered any request which
+ *			  is related to its module.
+ * FOUND_FAST_RETURN	- check() routine has discovered a request that is very
+ *			  fast to handle.
+ * FOUND_SLOW_RETURN	- check() routine has discovered a request that will
+ *			  take relatively large amount time to handle and after
+ *			  it has been handled, the execution will return back
+ * FOUND_NO_RETURN	- check() routine has discovered a request and its
+ *			  handling will never return back.
+ */
+enum cmd_check_res {
+	NOT_FOUND,
+	FOUND_FAST_RETURN,
+	FOUND_SLOW_RETURN,
+	FOUND_NO_RETURN,
+};
+
+/*
+ * struct cmd_module_ops	- operations defined for all registered modules
+ *
+ * setup()		- setup the module and prepare checking for
+ *			  commands/requests and handling them
+ * check()		- check if there is a command/request pending and return
+ *			  the appropriate result
+ * disable()		- temporarily disable the module to enable another
+ *			  module execution of run() routine
+ * run()		- run the handling of the command/request discovered by
+ *			  the check() routine - can be slow since other modules
+ *			  have been disabled by disable() routine
+ * enable()		- enable the module after it was disabled by disable()
+ *			  callback
+ * run_fast()		- run the handling of the command/request discovered by
+ *			  the check() routine - must take low amount of time
+ *			  since other modules have not been disabled
+ * teardown()		- the opposite from setup(), the module will no longer
+ *			  be checked or handled - stop the module
+ * run_noreturn()	- all modules have been teared down, so run the handling
+ *			  w/o returning
+ */
+struct cmd_module_ops {
+	void (*setup)(void);
+	enum cmd_check_res (*check)(void);
+	void (*disable)(void);
+	void (*run)(void);
+	void (*enable)(void);
+	void (*run_fast)(void);
+	void (*teardown)(void);
+	void (*run_noreturn)(void);
+};
+
+/* a dummy to fill in irrelevant entries */
+static inline void noop(void) {}
+
+/* PCI */

+void module_pci_disable(void);
+void module_pci_run(void);
+void module_pci_run_fast(void);
+void module_pci_enable(void);
+void module_pci_teardown(void);
+
+void module_handle_cmds(struct cmd_module_ops cmd_ops[], int size);
+#endif /* _HL_CMD_MODULE_H_ */
