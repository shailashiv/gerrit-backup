+bool boot_rom_skip_boot_info(void)
+{
+	if (boot_rom_global.die_type != IO_DIE_MASTER)
+		return true;
+
+	/* TODO: efuse check can be added here */
+
+	return false;
+}
+
+void boot_rom_init_boot_info(void)
+{
+	struct fs1_cold_rst_data cold_rst_data;
+
+	if(boot_rom_skip_boot_info())
+		return;
+
+	if (!boot_rom_global.boot_info_set_done) {
+		boot_rom_status_update(PROGRESS_BOOT_INFO_IMG_DESC_INFO_SET_START);
+
+		/* TODO: save boot info required for other FW modules */
+
+		cold_rst_data.data = regmap_readl(IO0_H0_PSOC_BOOT(mmPSOC_BOOT_CONF_COLD_RST_REG_2));
+		cold_rst_data.boot_info_set_done = 1;
+		regmap_writel(cold_rst_data.data, IO0_H0_PSOC_BOOT(mmPSOC_BOOT_CONF_COLD_RST_REG_2));
+		boot_rom_status_update(PROGRESS_BOOT_INFO_IMG_DESC_INFO_SET_FINISH);
+	} else {
+		/* Read and make sure databackup after any reset */
+		boot_rom_status_update(PROGRESS_BOOT_INFO_IMG_DESC_INFO_GET_START);
+
+		/* TODO: retrieve boot info and verify */
+
+		boot_rom_status_update(PROGRESS_BOOT_INFO_IMG_DESC_INFO_GET_FINISH);
+	}
+
+	return;
+}
