[SW-30456] boot info: set fit boot failure

In case that uboot failed to load the linux FIT, it will store the
reason for the failure in boot info.

This commit reads this info and sets CPU_BOOT_ERR0_x accordingly.

+int boot_info_get(u32 type, void *data, u32 size)
+{
+	struct hl_boot_info_ptrs ptrs;
+	struct hl_boot_info_desc desc;
+	u32 *offsets, *src, *dst, i, j;
+	u8 chksum[HL_CHKSUM_SIZE];
+
+	/* Size must be 32bit aligned */
+	if (size & (sizeof(u32) - 1))
+		return 1;
+
+	dst = data;
+
+	ptrs = *(struct hl_boot_info_ptrs *)offset2ptr(0, BOOT_INFO_BASE);
+	if (ptrs.magic != HL_BOOT_INFO_MAGIC)
+		return 1;
+
+	offsets = (u32 *)offset2ptr(sizeof(struct hl_boot_info_ptrs),
+				    BOOT_INFO_BASE);
+
+	for (i = 0; i < ptrs.count; i++) {
+		desc = *(struct hl_boot_info_desc *)offset2ptr(offsets[i],
+							       BOOT_INFO_BASE);
+		if (type != desc.type)
+			continue;
+
+		if (desc.magic != (HL_BOOT_INFO_DESC_MAGIC | type))
+			return 1;
+
+		if (size < desc.size)
+			return 1;
+
+		src = offset2ptr(desc.offset, BOOT_INFO_BASE);
+		for (j = 0; j < ROUND_UP4(desc.size) / sizeof(u32); j++)
+			dst[j] = src[j];
+
+		sha256_csum((u8 *)data, desc.size, chksum);
+		if (memcmp(chksum, desc.chksum, HL_CHKSUM_SIZE))
+			return 1;
+
+		return 0;
+	}
+
+	return 1;

+static void preboot_process_last_boot_info(void)
+{
+	u32 boot_err = HL_FW_FIT_BOOT_ERR_NONE;
+
+	if (!boot_info_get(HL_BOOT_INFO_FIT_BOOT_ERROR, &boot_err,
+			   sizeof(boot_err))) {
+		switch (boot_err) {
+		case HL_FW_FIT_BOOT_ERR_NONE:
+			break;
+		case HL_FW_FIT_BOOT_ERR_CORRUPTED:
+		case HL_FW_FIT_BOOT_ERR_ADDRESS_VIOLATION:
+		case HL_FW_FIT_BOOT_ERR_NOT_RUNNABLE:
+			cpu_boot_err_clrset(CPU_BOOT_ERR0_FIT_CORRUPTED, true);
+			break;
+		case HL_FW_FIT_BOOT_ERR_REVOKED_KEY:
+		case HL_FW_FIT_BOOT_ERR_SVN:
+		case HL_FW_FIT_BOOT_ERR_SIGNATURE:
+			cpu_boot_err_clrset(CPU_BOOT_ERR0_SECURITY_FAIL, true);
+			break;
+		}
+
+		return;
+	}
