[SW-30993] gaudi: hbm: add soft lane repair

Add soft lane repair to HBM. The repair is done in 2 stages, DC test and
AC test (MCBIST) without the iterative part of the test and without the
combination of row repair.

More information from the design document:

Purpose of HBM Lane Repair is to identify malfunctioning IO on the HBM
bus and attempt recovery by using IO redundancy feature of the HBM.
The IO is tested using two methods:
1. DC test -  this is used to identify disconnections or shorts.
   It uses special test mode of the HBM, in which constant known values
   are driven on all the IOs both in the TX direction and in the RX
   direction. The driven values are compared with the values captured
   by the receiver (HBM or SoC).
2. AC test - this is used to identify transient faults. It uses
   functional operation of the HBM. The SoC is performing Writes with
   pseudo-random data pattern followed by Reads of the same addresses.
   Data is compared. This flow uses the MCBIST mechanism of the MC,
   running in all PCs in parallel.
After each test stage, the results are parsed to check which (if any)
of the IOs malfunctions.

+static int hbm_channel_lane_repair(int device, int ch)
+{
+	struct hbm *hbm = &hbm_data;
+	struct ch_lane_repair_info *hlr = &hlr_info[device][ch];
+	struct dev_fail_info *dev_fail = &dev_fail_info[device];
+	int fails_in_cbus = 0, fails_in_rbus = 0;
+	u8 repair_code[PC_PER_CH][8], repair_code_c, repair_code_r;
+	u32 *dw_code;
+	int pc, byte, bit;
+
+	if (dev_fail->unrep_dw)  /* Failures in unrepairable pins */
+		return 1;
+
+	/* Create repair vector */
+	for (pc = 0; pc < PC_PER_CH; pc++) {
+		dw_code = pc ? &hlr->dw23_code : &hlr->dw01_code;
+
+		/* Convert dw codes nibbles to repair_code array */
+		for (byte = 0; byte < BYTES_PER_PC; byte++)
+			repair_code[pc][byte] = (*dw_code >> (4 * byte)) & 0xf;
+
+		for (byte = 0; byte < BYTES_PER_PC; byte++) {
+			/* repair_code should contains the repair codes that we
+			 * introduce during ATE (hard repair)
+			 */
+			if (repair_code[pc][byte] < REPAIR_CODE_ADJ)
+				hlr->fails_in_byte[pc][byte] = 1;
+			else
+				hlr->fails_in_byte[pc][byte] = 0;
+
+			/* Check result of current byte */
+			for (bit = 0; bit < BITS_PER_BYTE; bit++) {
+				if (!!(hlr->failed_dq[pc] &
+				       BIT(byte * 8 + bit))) {
+					hlr->fails_in_byte[pc][byte]++;
+					repair_code[pc][byte] = bit + REPAIR_CODE_DQ0;
+				}
+			}
+
+			/* In case that DBI (data bit inversion) is enabled and
+			 * we got more than 8 flipped bits, than probably dbi
+			 * lane is failing
+			 */
+			if (hbm->dbi_enable &&
+			    hlr->fails_in_byte[pc][byte] >= 8 &&
+			    !(hlr->failed_dm[pc] & BIT(byte))) {
+				/* DBI fault - detect for AC test */
+				hlr->failed_dbi[pc] |= BIT(byte);
+				hlr->fails_in_byte[pc][byte] -= 8;
+			}
+
+			/* ECC */
+			if (!!(hlr->failed_dm[pc] & BIT(byte))) {
+				hlr->fails_in_byte[pc][byte]++;
+				repair_code[pc][byte] = REPAIR_CODE_DM;
+			}
+
+			/* Data bit inversion */
+			if (!!(hlr->failed_dbi[pc] & BIT(byte))) {
+				hlr->fails_in_byte[pc][byte]++;
+				repair_code[pc][byte] = REPAIR_CODE_DBI;
+			}
+
+			/* more than one lane per byte, cannot repair */
+			if (hlr->fails_in_byte[pc][byte] > REDUNDENT_BYTE_PAIR_LANES)
+				return 1;
+		}
+
+		/* Check result of byte pair */
+		for (byte = 0; byte < BYTES_PER_PC; byte += 2) {
+			/* more than one lane per byte pair */
+			if (hlr->fails_in_byte[pc][byte] +
+			    hlr->fails_in_byte[pc][byte + 1] > REDUNDENT_BYTE_PAIR_LANES)
+				return 1;
+
+			/* mark repair in adjacent byte */
+			if (hlr->fails_in_byte[pc][byte])
+				repair_code[pc][byte + 1] = REPAIR_CODE_ADJ;
+
+			if (hlr->fails_in_byte[pc][byte + 1])
+				repair_code[pc][byte] = REPAIR_CODE_ADJ;
+		}
+	}
+
+	/* AWORD: Check whether repairable and create repair vector */
+	if (dev_fail->unrep_aw)  /* Failures in unrepairable pins */
+		return 1;
+
+	repair_code_c = hlr->aw_code & 0xf;
+	repair_code_r = (hlr->aw_code >> 4) & 0xf;
+
+	/* Check if we already fixed this bus */
+	if (repair_code_c < REPAIR_CODE_NONE)
+		fails_in_cbus = 1;
+
+	/* Check result of current C bus */
+	for (bit = 0; bit < CBUS_LANES_NUM; bit++) {
+		if (!(hlr->failed_c & BIT(bit)))
+			continue;
+
+		fails_in_cbus++;
+		repair_code_c = bit;
+	}
+
+	if (fails_in_cbus > REDUNDENT_CBUS_LANES)
+		return 1;
+
+	/* Check if we already fixed this bus */
+	if (repair_code_r < REPAIR_CODE_NONE)
+		fails_in_rbus = 1;
+
+	/* Check result of current R bus */
+	for (bit = 0; bit < RBUS_LANES_NUM; bit++) {
+		if (!(hlr->failed_r & BIT(bit)))
+			continue;
+
+		fails_in_rbus++;
+		repair_code_r = bit;
+	}
+
+	if (fails_in_rbus > REDUNDENT_RBUS_LANES)
+		return 1;
+
+	/* Convert back the repair code arrays */
+	for (pc = 0; pc < PC_PER_CH; pc++) {
+		dw_code = pc ? &hlr->dw23_code : &hlr->dw01_code;
+
+		*dw_code = 0;
+		for (byte = 0; byte < BYTES_PER_PC; byte++)
+			*dw_code |= repair_code[pc][byte] << (byte * 4);
+	}
+
+	hlr->aw_code = repair_code_c | (repair_code_r << 4);
+
+	lane_update_hbm(device, ch);
+	lane_update_phy(device, ch);
+
+	/* Since soft repair is an iterative process, reset handled failures to
+	 * prevent redudant repair attempts.
+	 */
+	hlr_reset_failures(hlr, false);
+

+{
+	struct ch_lane_repair_info *hlr = &hlr_info[device][ch];
+	u32 base = GAUDI_HBM_CFG_BASE + device * GAUDI_HBM_CFG_OFFSET;
+	u32 status = RREG32(base + ch * 0x1000 + 0x11C);

+					  failed_dq_reg[i].mismatch_reg);

+					  failed_dq_reg[i].expected_reg);
+			flipped = mismatch ^ expected;
+			hlr->failed_dq[pc] |= ((u64)flipped) <<
+				failed_dq_reg[i].shift_res;
+		}

+		expected = RREG32(base + ch * 0x1000 + DM_EXPECTED_REG);

+			hlr->failed_dm[pc] |= calc_dm_bit(flipped, i);
+	}
