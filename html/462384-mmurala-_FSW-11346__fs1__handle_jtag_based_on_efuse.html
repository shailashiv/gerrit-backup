+void security_handle_jtag_if(void)
+{
+	uint32_t val = regmap_readl(IO0_H0_PSOC_BOOT(mmPSOC_BOOT_CONF_JTAG_DISABLE));
+
+	if ((boot_rom_global.config_fuse & EFUSE_SYSCFG_DIS_JTAG) == 0) {
+		/* Enable JTAG */
+		val &= ~(PSOC_BOOT_CONF_JTAG_DISABLE_VAL_M);
+
+		/*
+		 * TODO: RR
+		 * JTAG.MSTR_IF RANGE_EN = 1
+		 * WR_RANGE_SEC & RD_RANGE_SEC = JTAG_SAI
+		 * RANGE_MIN_ADDR & RANGE_MAX_ADDR = 0 to 0xFFFFFFFC
+		 * TODO: PB
+		 * Already all PBs are opened for RW in all pages for all SAI levels.
+		 */
+	} else {
+		val |= PSOC_BOOT_CONF_JTAG_DISABLE_VAL_M;
+
+		/*
+		 * TODO: RR
+		 * JTAG.MSTR_IF RANGE_EN = 0
+		 * WR_RANGE_SEC & RD_RANGE_SEC = 0
+		 * RANGE_MIN_ADDR & RANGE_MAX_ADDR = 0
+		 * TODO: PB
+		 * Blocked from RR level, so no need to change PB.
+		 */
+	}
+
+	/* Clear FSM timer */
+	val &= ~(PSOC_BOOT_CONF_JTAG_DISABLE_COUNTER_EN_M);
+
+	regmap_writel(val, IO0_H0_PSOC_BOOT(mmPSOC_BOOT_CONF_JTAG_DISABLE));
+}
