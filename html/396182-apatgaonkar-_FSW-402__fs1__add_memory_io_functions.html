+#ifndef _MEM_IO_H_
+#define _MEM_IO_H_
+
+#include <stdint.h>
+#include <soc.h>
+
+#define HL_FLASH_BASE				(HL_GEN_MEM_BASE + HL_FLASH_OFFSET)
+#define HL_SBROM_BASE				(HL_FW_MEM_BASE + HL_SBROM_OFFSET)
+#define HL_FWSRAM_BASE				(HL_FW_MEM_BASE + HL_FWSRAM_OFFSET)
+#define HL_SCRATCHPAD_SRAM_BASE			(HL_FW_MEM_BASE + HL_SCRATCHPAD_SRAM_OFFSET)
+
+/* In ARC address manipulation BIT[22:24] are for die id */
+#define HL_ARC_ADDR_DIE_ID_SHIFT		25
+#define HL_ARC_ADDR_DIE_ID_MASK			(0x7 << HL_ARC_ADDR_DIE_ID_SHIFT)
+
+#define BIT(nr)					(1 << (nr))
+extern struct boot_rom_global_s boot_rom_global;
+
+static inline void base_writel(uint32_t val, uint32_t offset)
+{
+	*((volatile uint32_t *)(offset)) = val;
+}
+
+static inline uint32_t base_readl(uint32_t offset)
+{
+	uint32_t reg = *((volatile uint32_t *)(offset));
+
+	return reg;
+}
+
+static inline uint32_t regmap_readl(uint32_t offset)
+{
+	uint32_t reg = base_readl((offset & ~(HL_ARC_ADDR_DIE_ID_MASK)) |
+				(boot_rom_global.die_id << HL_ARC_ADDR_DIE_ID_SHIFT));
+
+	return reg;
+}
+
+static inline void regmap_writel(uint32_t val, uint32_t offset)
+{
+	base_writel(val, (offset & ~(HL_ARC_ADDR_DIE_ID_MASK)) |
+				(boot_rom_global.die_id << HL_ARC_ADDR_DIE_ID_SHIFT));
+}
+
+static inline uint32_t sp_sram_readl(uint32_t offset)
+{
+	uint32_t reg = base_readl(HL_SCRATCHPAD_SRAM_BASE + offset);
+
+	return reg;
+}
+
+static inline void sp_sram_writel(uint32_t val, uint32_t offset)
+{
+	base_writel(val, HL_SCRATCHPAD_SRAM_BASE + offset);
+}
+
+static inline uint8_t regmap_readb(uint32_t offset)
+{
+	uint32_t reg = *((volatile uint32_t *)(HL_CFG_BASE + offset));
+
+	return (uint8_t)(reg & 0xFF);
+}
+
+static inline void regmap_writeb(uint8_t val, uint32_t offset)
+{
+	*((volatile uint32_t *)(HL_CFG_BASE + offset)) = (uint32_t)val;
+}
+
+#endif /* _MEM_IO_H_ */
