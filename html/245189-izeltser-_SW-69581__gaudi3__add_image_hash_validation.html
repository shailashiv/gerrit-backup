Commit Message

[SW-69581] gaudi3: add image hash validation

In Gaudi3 SPI flash each image descriptor holds sha384 hash value of
it's image. In order to validate the image, the sha384 hash should be
calculated on the actual data, and compared with the hash in the
boot descriptor.

+static unsigned int ppboot_get_num_crc(u32 size)
+{
+	unsigned int num_crc;
+
+	num_crc = size / MAX_DMA_SIZE_CRC;

+		num_crc++;

+bool ppboot_is_valid_img_desc(struct boot_desc *desc)
+{
+	u32 addr, actual_size;
+	unsigned int num_crc;
+	int ret;
+
+	num_crc = ppboot_get_num_crc(desc->img_size);
+	actual_size = desc->img_size - (num_crc * CRC32_SIZE);
+	addr = ppboot_get_arc_addr(desc->loadaddr);
+	if (addr == (u32)(~0)) {
+		preboot_puts_hex64("unsupported img address ", addr, false);
+		return false;
+	}
+
+	ret = secureboot_verify_image_hash384(desc->chksum, (unsigned char *)addr, actual_size);
+
+	return !ret;
