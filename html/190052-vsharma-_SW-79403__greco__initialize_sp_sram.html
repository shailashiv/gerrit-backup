Commit Message

[SW-79403] greco: initialize sp sram

This patch initialize sp sram starting from 4K offset
using memcpy block. First 4K is used by arcboot rom.
PSOC_MEM_ERR interrupt was seen during boot,
which is due to un-initialized sp sram access.

+#define SP_SRAM_60K_SZ_16B_GRANULARITY	0xF00
+void preboot_sp_sram_init(void)
+{
+	u32 mem_cfg, cpy_sts;
+	u64 start;
+
+	/* Use memblock to initialize scratchpad memory from 4K offset
+	 * as first 4K offset is used by arcboot rom
+	 */
+	regmap_writel((u32)(HL_SP_SRAM_BASE + SZ_4K),
+			mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_L);
+	regmap_writel((u32)((HL_SP_SRAM_BASE + SZ_4K) >> 32),
+			mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_H);
+
+	mem_cfg = regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+	mem_cfg &= ~(PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_FLD_SIZE_MASK);
+	/* set size of 60k in granuality of 16 bytes */
+	mem_cfg |= (SP_SRAM_60K_SZ_16B_GRANULARITY & PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_FLD_SIZE_MASK);
+	mem_cfg |= (BIT(PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_USE_CONST_SHIFT));
+
+	regmap_writel(mem_cfg, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+
+	/* Enable copy and wait for completion */
+	regmap_writel(BIT(0), mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL);
+	start = timestamp_get_us();
+
+	while (1) {
+		cpy_sts = regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_STATUS);
+		if (cpy_sts)
+			break;
+
+		/* If copy does not happen in 5 seconds
+		 * timeout and return
+		 */
+		if (timeout_expired(start, MEM_CPY_TIMEOUT_US)) {
+			preboot_puts("Timed-out waiting for mem_cpy\n");
+			break;
+		}
+	}
+}
