+static u64 gaudi2_translate_addr(u64 addr)
+{
+	if (addr >= ARC0_ICCM_FULL_BASE &&
+	    addr < (ARC1_DCCM_FULL_BASE + (u64)ARC1_DCCM_SIZE)) {
+		addr = (addr & ARC_ADDR_MASK) | BIT(29);
+		goto ret_addr;
+	}
+
+	if (addr >= ARM_SDRAM_BASE && addr < (ARM_SDRAM_BASE + (u64)ARM_SDRAM_SIZE)) {
+		addr = (addr & ARM_ADDR_MASK) | BIT(28);
+		goto ret_addr;
+	}
+
+	if (addr >= SDRAM_FULL_BASE && addr < (SDRAM_FULL_BASE + (u64)ARM_SDRAM_SIZE)) {
+		addr = (addr & ARC_ADDR_MASK) | BIT(28);
+		goto ret_addr;
+	}
+
+	preboot_puts_hex64("warn: no address translation: ", addr, true);
+
+ret_addr:
+	return addr;
+}
+
+u64 preboot_translate_addr(u64 addr)
+{
+	return gaudi2_translate_addr(addr);
+}
