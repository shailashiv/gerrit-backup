Commit Message

[SW-107833] gaudi3: update d2d train window

As per recommendation from HW team, in order to get D2D link in low
speed mode as part of rom recovery sequence, the train window default
value needs to be updated (BITSYNC_TRKWIN=0x1F). This patch adds the
support to update train window qualified by fuse bit 25 of row 58.

+#define BITSYNC_TRKWIN_VAL		0x1F
+static void arcboot_rom_d2d_phy_update_train_window(void)
+{
+	uint32_t val, i;
+
+	if (!(arcboot_rom_global.config_fuse & EFUSE_SYSCFG_D2D_UPDATE_TRAIN_WINDOW))
+		return; /* skip d2d train window update */
+
+	for (i = 0; i < NUM_OF_PHY; i++) {
+		val = regmap_readl(mmD0_DPHY0_CH_BCAST_PHYCTRLREG_RXBITSYNCTMR +
+				i * INTER_PHY_DIST);
+		val |= (BITSYNC_TRKWIN_VAL <<
+			D0_DPHY0_CH_BCAST_PHYCTRLREG_RXBITSYNCTMR_BITSYNC_TRKWIN_S);
+		regmap_writel(val, mmD0_DPHY0_CH_BCAST_PHYCTRLREG_RXBITSYNCTMR +
+				i * INTER_PHY_DIST);
+	}
+}
