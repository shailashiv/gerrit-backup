Commit Message

[SW-227605] g2e-pcie: support hbm rate limiters

For G2E PCIe card, there is to support HBM RLs.
Temporarily, the solution is not part of the hl_rl driver.
HBM RLs are configured during initializtion of arcmgmt.
The ticket defined reset_val and max_timeout only, so
there was a need to set max_sat as well - the value is
taken from G3 PID DT which is used by the hl_rl driver.

+	uint32_t rd_rl_mem;
+};

+void hl_rtr_ctrl_init_rl_reg_once(void)
+{
+	int32_t max_sat, reset_val, max_timeout, dcore, rtr;
+	struct hl_rtr_ctrl_rl_regs reg_base_pu;
+	uint32_t cfg = 0;
+
+	max_sat = GAUDI2E_HBM_RL_MAX_SAT;
+	reset_val = GAUDI2E_HBM_RL_RESET_VAL;
+	max_timeout = GAUDI2E_HBM_RL_MAX_TIMEOUT;
+
+	cfg |= (max_sat << GAUDI2E_HBM_RL_SATURATION_SHIFT);
+	cfg |= (reset_val << GAUDI2E_HBM_RL_RST_TOKEN_SHIFT);
+	cfg |= (max_timeout << GAUDI2E_HBM_RL_TIMEOUT_SHIFT);
+
+	LOG_INF("Setting HBM RLs: max_sat = %d, reset_val = %d, max_timeout = %d, cfg = 0x%x",
+		max_sat, reset_val, max_timeout, cfg);
+
+	/* Set HBM RL */
+	for (dcore = 0; dcore < GAUDI2E_HBM_RL_DCORE_NUM; dcore++) {
+		for (rtr = 0; rtr < GAUDI2E_HBM_RL_RTR_NUM; rtr++) {
+			reg_base_pu.wr_rl_mem = mmDCORE0_RTR0_CTRL_BASE +
+				(dcore * DCORE_N_OFFSET) + (rtr * RTR_N_OFFSET) + WR_RL_MEM_OFFSET;
+			reg_base_pu.rd_rl_mem = mmDCORE0_RTR0_CTRL_BASE +
+				(dcore * DCORE_N_OFFSET) + (rtr * RTR_N_OFFSET) + RD_RL_MEM_OFFSET;
+			sys_write32(cfg, (mem_addr_t)reg_base_pu.wr_rl_mem);
+			sys_write32(cfg, (mem_addr_t)reg_base_pu.rd_rl_mem);
+		}
+	}
+
+	/* Enable HBM RL */
+	for (dcore = 0; dcore < GAUDI2E_HBM_RL_DCORE_NUM; dcore++) {
+		for (rtr = 0; rtr < GAUDI2E_HBM_RL_RTR_NUM; rtr++) {
+			reg_base_pu.wr_rl_mem = mmDCORE0_RTR0_CTRL_BASE +
+				(dcore * DCORE_N_OFFSET) + (rtr * RTR_N_OFFSET) + WR_RL_MEM_OFFSET;
+			reg_base_pu.rd_rl_mem = mmDCORE0_RTR0_CTRL_BASE +
+				(dcore * DCORE_N_OFFSET) + (rtr * RTR_N_OFFSET) + RD_RL_MEM_OFFSET;
+			sys_rmwrite32((mem_addr_t)reg_base_pu.wr_rl_mem, 1, GAUDI2E_HBM_RL_EN_MASK);
+			sys_rmwrite32((mem_addr_t)reg_base_pu.rd_rl_mem, 1, GAUDI2E_HBM_RL_EN_MASK);
+		}
+	}
