+static int security_get_first_valid_key(uint8_t customer)
+{
+	uint8_t end_index, key_revoke;
+
+	/* customer keys are from index 1 - 4
+	 * habana key is only at index 0
+	 */
+	if (customer)
+		end_index = SECURITY_KEY_HASH_COUNT;
+	else
+		end_index = SECURITY_KEY_HASH_HABANA_COUNT;
+
+	key_revoke = regmap_readl(mmPSOC_SECURITY_KEY_REVOKE);
+
+	/* return offset of first non-revoked key
+	 * habana will scan index 0 only
+	 * customer will scan index 1 - 5
+	 */
+	for (uint8_t i = customer; i < end_index; i++) {
+		if ((key_revoke & BIT(i)) == 0)
+			return i;
+	}
+
+	/* no valid key found */
+	return -1;
+}
+
+int security_get_key_hash(uint32_t *hash, uint8_t customer)
+{
+	int valid_key_offset = security_get_first_valid_key(customer);

+	} else {
+		valid_key_offset = mmPSOC_SECURITY_HABANA_KEY_0;
+	}
+
+	for (int i = 0; i < SECURITY_KEY_HASH_COUNT; i++)
+		*hash = regmap_readl((valid_key_offset++) * 4);
+
+	return 1;
