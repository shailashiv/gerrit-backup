[SW-69281] gaudi2: add fit support via elbi interface

This change adds FIT download support via ELBI interface.
Gaudi uses spsram  as backend storage for fit to be
downloaded via elbi interface and gaudi2 uses arc farm
arc0 dccm as backend storage. This change generalize
elbi changes as backend storage agnostic

+static int halt_arc_farm_arc0(void)
+{
+	int ret = 0;
+	u32 val;
+	u32 retry = ARC_FARM_ARC0_HALT_RETRIES;
+
+	val = regmap_readl(mmARC_FARM_ARC0_AUX_RUN_HALT_ACK);
+	if (!(val & ARC_FARM_ARC0_RUN_HALT_ACK_SYS_HALT_R_MASK)) {
+		/* Halt ARC incase if it is not already halted */
+		preboot_puts("Halting ARC farm ARC0\n");
+		regmap_writel(BIT(1), mmARC_FARM_ARC0_AUX_RUN_HALT_REQ);
+		while (1) {
+			val = regmap_readl(mmARC_FARM_ARC0_AUX_RUN_HALT_ACK);
+			if (val & ARC_FARM_ARC0_RUN_HALT_ACK_HALT_ACK_MASK)
+				break;
+
+			udelay(1);
+			retry--;
+			if (!retry) {
+				preboot_puts("No ack from ARC farm ARC0\n");
+				ret = 1;
+				break;
+			}
+		}
+		/* Clear HALT_REQ bit */
+		regmap_writel(0x0, mmARC_FARM_ARC0_AUX_RUN_HALT_REQ);
+	}
+
+	return ret;
+}

+static void reset_arc_farm_arc0(void)
+{
+	/* Reset ARC */
+	regmap_writel(BIT(0), mmARC_FARM_ARC0_AUX_ARC_RST);
+}
+
+static void arc_farm_arc0_out_of_reset(void)
+{
+	/*
+	 * Bring ARC out of reset
+	 * Clear CORE BIT(0) and set PRESETDBGN BIT(4)
+	 */
+	regmap_writel(BIT(4), mmARC_FARM_ARC0_AUX_ARC_RST);
+
+	/* Disable DCCM_TERMINATE_EN to enable writes to DCCM memory */
+	regmap_writel(0x0, mmARC_FARM_ARC0_AUX_CFG_DCCM_TERMINATE_EN);
+}
+
+static int init_arc_farm_arc0_dccm0(void)
+{
+	if (halt_arc_farm_arc0())
+		return 1;
+
+	reset_arc_farm_arc0();
+	mdelay(1);
+	arc_farm_arc0_out_of_reset();
+
+	return 0;
+}
+
+uintptr_t preboot_get_elbi_reg_base(void)

+void preboot_security_dynamic_img_elbi_reg_open(void)

+

+}
+
+void preboot_security_dynamic_img_elbi_reg_close(void)
+{
+	u64 min, max;
+	struct hl_rr_eng_select rr_sel;
+
+	if (!preboot_security_is_enabled())
+		return;
+
+	memset(&rr_sel, 0, sizeof(struct hl_rr_eng_select));
+
+	rr_sel.noc = true;
+	rr_sel.pcie = true;
+
+	/*
+	 * ARC_FARM_ARC0_DCCM0_DCCM_BASE_FULL to
+	 * (ARC_FARM_ARC0_DCCM0_DCCM_BASE_FULL + FIT_IMG_ARC_FARM_ARC0_DCCM0_SIZE)
+	 * 0x1000007FFCE90000 - 0x1000007FFCE94000
+	 */
+	min = CFG_BASE + mmARC_FARM_ARC0_DCCM0_DCCM_BASE_ADDR;
+	max = CFG_BASE + mmARC_FARM_ARC0_DCCM0_DCCM_BASE_ADDR + FIT_IMG_ARC_FARM_ARC0_DCCM0_SIZE;
+	hl_secure_lbw_rr(HL_SEC_LBW_ELBI_FIT_REGION, min, max, rr_sel);
+}
+
+void preboot_dynamic_img_elbi_reg_cleanup(void *map_addr, u32 size)
