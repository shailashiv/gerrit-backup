+static int fit_get_key_prop(const void *pub_key_fit, const char *key_name,
+			    struct key_prop *prop)
+{
+	int pub_key_node = 0;
+	int length;
+
+	if (fdt_check_header(pub_key_fit))
+		return 1;
+
+	pub_key_node = get_pub_key_node(pub_key_fit, key_name);
+	if (pub_key_node < 0)
+		return 1;
+
+	prop->num_bits =
+		fit_get_int(pub_key_fit, pub_key_node, "rsa,num-bits", 0);
+	prop->n0inv =
+		fit_get_int(pub_key_fit, pub_key_node, "rsa,n0-inverse", 0);
+	prop->public_exponent =
+		fdt_getprop(pub_key_fit, pub_key_node, "rsa,exponent", &length);
+	if (!prop->public_exponent || length < (int) sizeof(uint64_t))
+		prop->public_exponent = NULL;
+	prop->modulus =
+		fdt_getprop(pub_key_fit, pub_key_node, "rsa,modulus", NULL);
+	prop->rr =
+		fdt_getprop(pub_key_fit, pub_key_node, "rsa,r-squared", NULL);
+
+	if (!prop->num_bits || !prop->modulus)
+		return 1;

+static int fit_get_pub_key(const void *fit, int noffset, int sig_sub_node,
+			   const void **pub_key_fit, enum fit_sec_failure *res)
+{
+	int node, key_len, pub_key_node;
+	const void *key_data;
+	const char *key_name;
+
+	node = fit_pub_key_node_get(fit, noffset);
+	if (node < 0)
+		return 1;
+
+	key_data = fdt_getprop(fit, node, FIT_DATA_PROP, &key_len);
+	if (!key_data)
+		return 1;
+
+	arcboot_key_idx = fit_get_int(fit, 0, FIT_SEC_KEY_IDX, -1);
+	if (arcboot_key_idx < 0)
+		return 1;
+
+	*pub_key_fit = key_data;
+
+	/* Get modulus from public key */
+	if (fdt_check_header(*pub_key_fit))
+		return 1;
+
+	if (get_key_name(fit, sig_sub_node, &key_name))
+		return 1;
+
+	pub_key_node = get_pub_key_node(*pub_key_fit, key_name);
+	if (pub_key_node < 0)
+		return 1;
+
+	/* Update key data and key length with 384 bytes modulus */
+	key_data = fdt_getprop(*pub_key_fit, pub_key_node,
+			"rsa,modulus", &key_len);
+
+	if (security_check_key_revoke_status(arcboot_key_idx) != 1) {
+		*res = FIT_SEC_KEY_REVOKED;
+		return 1;
+	}
+
+	if (secureboot_verify_key_hash(arcboot_key_idx, (uint8_t *)key_data, key_len)) {
+		*res = FIT_SEC_SIGN_FAILED;
+		return 1;
+	}
+
+	return 0;
+}

+/* Checks the signed FIT and validate it againts the public key */
+static int fit_verify_config(const void *fit, int noffset,
+			     enum fit_sec_failure *res)
+{
+	int sig_sub_node;
+	const char *hashed_node_prop;
+	int count, err;
+	char *node_inc[IMAGE_MAX_HASHED_NODES];
+	char * const exc_prop[] = { FIT_DATA_PROP };
+	char path[FIT_MAX_PATH];
+	const uint32_t *strings;
+	const void *pub_key_fit;
+
+	sig_sub_node = fdt_subnode_offset(fit, noffset, FIT_SIG_NODENAME);
+	if (sig_sub_node < 0)
+		return 1;
+
+	if (fit_get_pub_key(fit, noffset, sig_sub_node, &pub_key_fit, res))
+		return 1;
+
+	hashed_node_prop = fit_get_hashed_nodes(fit, sig_sub_node, &count,
+						node_inc);
+	if (!hashed_node_prop)
+		return 1;
+
+	/* Get a list of regions to hash */
+	count = fdt_find_regions(fit, node_inc, count, exc_prop,
+				 ARRAY_SIZE(exc_prop), fdt_regions,
+				 MAX_FDT_REGIONS - 1, path, sizeof(path), 0);
+
+	if ((count <= 0) || (count >= MAX_FDT_REGIONS - 1))
+		return 1;
+
+	/* Add the strings */
+	strings = fdt_getprop(fit, sig_sub_node, "hashed-strings", NULL);
+	if (strings) {
+		fdt_regions[count].offset = fdt_off_dt_strings(fit);
+		fdt_regions[count].size = fdt32_to_cpu(strings[1]);
+		count++;
+	}
+
+	err = fit_verify_config_regions(fit, fdt_regions, count, sig_sub_node,
+					pub_key_fit);
+	if (err) {
+		*res = FIT_SEC_SIGN_FAILED;
+		return err;
+	}
+
+	return 0;
+}

+static int fit_verify_security(const void *fit, int noffset,
+			       enum fit_sec_failure *res)
+{
+	if (!security_efuse_image_validation_en())

+static int fit_getprop_u64(const void *fit, int node, const char *name,
+			   uint64_t *val64)
+{
+	const uint32_t *val32;
+	int len;
+
+	val32 = fdt_getprop(fit, node, name, &len);
+	if (!val32 || len != sizeof(uint64_t))
+		return 1;
+
+	*val64 = ((uint64_t)fdt32_to_cpu(val32[0])) << 32;
+	*val64 |= fdt32_to_cpu(val32[1]);
+	return 0;
+}

+static bool fit_load_is_runnable(const void *fit, int conf)
+{
+	const uint32_t *cell;
+	uint32_t type;
+	int len;
+
+	cell = fdt_getprop(fit, conf, FIT_LOAD_TYPE_PROP, &len);
+
+	/* If load type does not exists, assume it's u-boot */
+	if (!cell)
+		return true;
+
+	if (len != sizeof(*cell))
+		return false;
+
+	type = fdt32_to_cpu(*cell);
+	if (type >= FIT_LOAD_TYPE_COUNT)
+		return false;
+
+	return type == FIT_LOAD_TYPE_RUN;
+}

+static int fit_node_check_hash(const void *fit, int node, const uint8_t *buf,
+			       int len)
+{
+	const uint8_t *node_hash = fit_node_get_hash(fit, node);
+	uint8_t calc_hash[SHA_SUM_LEN];
+
+	if (!node_hash)
+		return 1;
+
+	sha_csum(buf, len, calc_hash);
+	if (memcmp(node_hash, calc_hash, SHA_SUM_LEN))
+		return 1;
+
+	return 0;
+}

+static enum fit_load_result fit_copy_images(const void *fit,
+					    enum fit_sec_failure *res)
+{
+	int node, len;
+	int conf, num_images, i;
+	const uint8_t *buf;
+	uint64_t load, start;
+	const char *cmpr;
+	size_t fit_size = fdt_totalsize(fit);
+
+	/* Check if image supports a minimum FDT version or newer */
+	if (fdt_version(fit) < FDT_MIN_SUPPORTED_VERSION)
+		return FIT_LOAD_SEC_FAILED;
+
+	/* check that the calculated memory range are legal */
+	if (!arcboot_is_valid_address_range((uintptr_t)fit, fit_size))
+		return FIT_LOAD_SEC_FAILED;
+
+	/* Get number of images in preboot configuration */
+	conf = fit_find_config_node(fit);
+	if (conf < 0)
+		return FIT_LOAD_CORRUPTED;
+
+	if (!fit_load_is_runnable(fit, conf))
+		return FIT_LOAD_CORRUPTED;
+
+	if (fit_verify_image(fit, fit_size, conf, res))
+		return FIT_LOAD_SEC_FAILED;
+
+	num_images = fdt_stringlist_count(fit, conf, FIT_LOADABLE_PROP);
+	if (num_images > FIT_MAX_IMAGES)
+		return FIT_LOAD_CORRUPTED;
+	else if (num_images < 0)
+		return FIT_LOAD_CORRUPTED;
+
+	for (i = 0; i < num_images; i++) {
+		node = fit_loadables_node_get(fit, i);
+		if (node < 0)
+			return FIT_LOAD_NOT_RUNNABLE;
+
+		if (fit_getprop_u64(fit, node, FIT_LOADADDR_PROP, &load))
+			return FIT_LOAD_BAD_LOAD_ADDR;
+
+		if (fit_getprop_u64(fit, node, FIT_STARTADDR_PROP, &start))
+			start = load;
+
+		buf = fdt_getprop(fit, node, FIT_DATA_PROP, &len);
+		if (!buf)
+			return FIT_LOAD_CORRUPTED;
+
+		if (fit_node_check_hash(fit, node, buf, len))
+			return FIT_LOAD_CORRUPTED;
+
+		cmpr = (const char *)fdt_getprop(fit, node,
+						 FIT_COMPRESSION_PROP,
+						 NULL);
+		if (!cmpr) {
+			/*
+			 * mkimage is adding compression, if it's missing then
+			 * something is wrong.
+			 */
+			return FIT_LOAD_CORRUPTED;
+		}
+
+		if (!arcboot_is_valid_address_range(load, len))
+			return FIT_LOAD_BAD_LOAD_ADDR;
+
+		images[i].arch = (const char *)fdt_getprop(fit, node,
+							FIT_ARCH_PROP, NULL);
+		if (!images[i].arch)
+			return FIT_LOAD_CORRUPTED;
+
+		images[i].loadaddr = load;
+		images[i].startaddr = start;
+
+		if (fit_copy_image(load, buf, len, images[i].arch, cmpr))
+			return FIT_LOAD_CORRUPTED;
+	}

+	return FIT_LOAD_SUCCESS;

+		     struct fdt_region region[], int max_regions,
+		     char *path, int path_len, int add_string_tab)

+		if (add_string_tab)

+	}
+	count++;

+	return count;
