[SW-56233] security: gaudi2: protection bits init

PSOC global conf address space will be opened using
range registers in preboot. Prior to that unrelevant
register spaces are blocked using protection bits
configurations.
Only following registers are opened,

•	BTM_FSM
•	SCRATCHPAD
•	CPU_BOOT_STATUS
•	KMD_MSG_TO_CPU
•	TRACE_ADDR
•	TRACE_AWUSER
•	TRACE_ARUSER

+static int init_prot_bits_block(u8 major, u8 minor, u8 sub_minor)
+{
+	struct hl_prot_bits_cfg *cfg = &gaudi2_prot_bits_cfg[0];
+	u8 b_idx = 0, d_idx = 0;
+	u32 val;
+	u32 addr, reg_addr;
+
+	addr = cfg->addr.base_addr + HL_GLBL_PRIV_REG_OFFSET + major * cfg->addr.major_offset +
+			  minor * cfg->addr.minor_offset + sub_minor * cfg->addr.sub_minor_offset;
+
+	while (b_idx != HL_PROT_BITS_REGS) {
+		reg_addr = addr + (b_idx * 4);
+		if (cfg->data_map & BIT(b_idx)) {
+			if (!cfg->data) {
+				preboot_puts("No data for prot bit cfg\n");
+				return 1;
+			}
+			if (d_idx >= cfg->data_size) {
+				preboot_puts("bad data idx for prot bit cfg\n");
+				return 1;
+			}
+			val = cfg->data[d_idx];
+			d_idx++;
+		} else {
+			val = (cfg->prot_map & BIT(b_idx)) ?
+						HL_CNF_ALL_NON_PROT : HL_CNF_ALL_PROT;
+		}
+		regmap_writel(val, reg_addr);
+		b_idx++;
+	}
+
+	return 0;
+}

+static void hl_dev_prot_bits_init(void)
+{
+	u8 major_idx, minor_idx, sub_minor_idx;
+	struct hl_prot_bits_addr block_addr;
+
+	block_addr = gaudi2_prot_bits_cfg[0].addr;
+	for (major_idx = 0; major_idx < block_addr.major; major_idx++) {
+		minor_idx = 0;
+		do {
+			sub_minor_idx = 0;
+			do {
+				init_prot_bits_block(major_idx, minor_idx, sub_minor_idx);
+			} while (sub_minor_idx < block_addr.sub_minor);
+			minor_idx++;
+		} while (minor_idx < block_addr.minor);
+	}
+
+	preboot_puts("gaudi2 protection bits configured\n");
+}
