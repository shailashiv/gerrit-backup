+void boot_rom_persistent_dccm_init(uint8_t die_id)
+{
+	volatile uint32_t *dccm_addr = (uint32_t *) HL_DCCM_BASE;
+	uint32_t dccm_dwords = HL_ARC1_DCCM_PERSIST_SECTION_SIZE >> 2;
+	struct fs1_cold_rst_data cold_rst_data;
+	uint32_t i;
+	uint32_t die_cold_rst_flop_2 = HL_CFG_BASE +
+				       IO0_HALF0_PSOC_BOOT(mmPSOC_BOOT_CONF_COLD_RST_REG_2);
+
+	/* Replace correct die id in register offset*/
+	die_cold_rst_flop_2 = (die_cold_rst_flop_2 & ~(HL_ARC_ADDR_DIE_ID_MASK)) |
+			      (die_id << HL_ARC_ADDR_DIE_ID_SHIFT);
+	cold_rst_data.data = base_readl(die_cold_rst_flop_2);
+
+	if (cold_rst_data.arc1_persist_dccm_init_done)
+		goto non_zero_init;
+
+	for (i = 0; i < dccm_dwords; i++)
+		*dccm_addr++ = 0;
+
+	cold_rst_data.arc1_persist_dccm_init_done = 1;
+	base_writel(cold_rst_data.data, die_cold_rst_flop_2);
+
+non_zero_init:
+	/* Non zero initializers only required for master(die 0) */
+	if (die_id == IO_DIE_0)
+		boot_rom_persist_data_non_zero_init();
+}
