+#!/bin/bash
+
+host=$1
+shift
+params=$*
+remove_pkgs=$(echo $params | grep -ow remove | uniq)
+install_pkgs=$(echo $params | grep -ow install | uniq)
+program_flash=$(echo $params | grep -ow program | uniq)
+
+result=~/$(basename $0).result
+fw_dir=/lib/firmware/habanalabs/dali
+local_scr_dir=/var/lib/tftpboot/tests
+scr_name=(local_run local_remote_common)
+fw_ldr=fw_loader
+remote_scr_dir="."
+
+source $local_scr_dir/local_remote_common
+
+rm -f $result
+
+if [[ -z "$remove_pkgs" && -z "$install_pkgs" && -z "$program_flash" ]]; then
+	print_msg "Error: no arguments provided" | tee -a $result
+	exit 10;
+fi
+
+function remote_prepare_work_dirs() {
+	ssh -t $host "sudo mkdir -p $fw_dir;" | tee -a $result
+
+	local pipestat=${PIPESTATUS[0]}
+	print_msg "ssh mkdir $fw_dir status: $pipestat" | tee -a $result
+
+	return $pipestat;
+}
+
+function remote_copy_fw_scripts() {
+	local scripts=(${scr_name[*]/local_/${local_scr_dir}/local_})
+	# sync the latest built binaries
+	for ((j=0; j<3; j++)); do
+		sshpass -p123456 scp ${fw_dir}/* root@${host}:${fw_dir}/	|| break;
+		sshpass -p123456 scp ${local_scr_dir}/${fw_ldr} ${host}:${remote_scr_dir}/	|| break;
+		sshpass -p123456 scp ${scripts[*]} ${host}:${remote_scr_dir}/	|| break;
+		print_msg "Finished copying the firmware and scripts" | tee -a $result
+		return 0;
+	done
+
+	print_msg "Failed copying the firmware and scripts" | tee -a $result
+	return 1;
+}
+
+function remote_run_script() {
+	print_msg "Starting ${scr_name[0]}, args: $remove_pkgs $install_pkgs $program_flash" | tee -a $result
+
+	ssh -t $host "${remote_scr_dir}/${scr_name[0]} $remove_pkgs $install_pkgs $program_flash;" | tee -a $result
+
+	local pipestat=${PIPESTATUS[0]}
+	case $pipestat in
+	0)
+		print_msg "${scr_name[0]}: returned success" | tee -a $result
+		;;
+	1)
+		print_msg "${scr_name[0]}: Failed installing packages" | tee -a $result
+		;;
+	2)
+		print_msg "${scr_name[0]}: Failed programming SPI flash" | tee -a $result
+		;;
+	*)
+		print_msg "${scr_name[0]}: Unknown error" | tee -a $result
+		;;
+	esac
+
+	print_msg "${scr_name[0]} script status: $pipestat" | tee -a $result
+	return $pipestat;
+}
+
+for ((i=100; i>0; i--)); do
+	echo $i
+	sleep 1;
+	ping -c 1 -W 1 $host
+	if [ $? -eq 0 ]; then
+		rm -f $result
+		remote_prepare_work_dirs	|| continue;
+		remote_copy_fw_scripts		|| break;
+		remote_run_script		|| break;
+		echo done;
+		exit 0;
+	fi
+done
+
+echo error
+exit 1;
