Commit Message

[SW-4185] gaudi: pll: add hbw clk sw glitch w/a

Since the clock switch has a glitch while switching between clocks
different by a factor of about 10, we W/A the problem by performing
the switch in a "gated" manner. That is, we gate the clock output,
perform the switch, and then ungate the clock output.
This way we are able to W/A the clock switch glitch as we wait enough
time for the clock to stabilize before ungating.

+#define HL_PLL_GATER_OFFSET	0x300
+#define HL_PLL_DIVSEL_OFFSET	0x280

+static void hl_pll_hbw_gated_clk_sw(struct hl_pll_cfg *cfg)
+{
+	regmap_writel(0xE, cfg->base + HL_PLL_GATER_OFFSET);
+	hl_pll_udelay(1);
+	regmap_writel(HL_PLL_PLL_CLK, cfg->base + HL_PLL_DIVSEL_OFFSET);
+	hl_pll_udelay(10);
+	regmap_writel(0xF, cfg->base + HL_PLL_GATER_OFFSET);
+	hl_pll_udelay(10);
+}
+
+void hl_pll_wa(void)
+{
+	int i;
+
+	for (i = 0; i < ARRAY_SIZE(gaudi_pll_cfg); i++) {
+		switch (gaudi_pll_cfg[i].base) {

+		case mmPSOC_HBM_PLL_BASE:
+		case mmNIC2_HBM_PLL_BASE:
+			continue;
+		default:
+			break;
+		}
