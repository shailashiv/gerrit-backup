[FSW-9707] utils: add cross die read and write apis

This patch adds CFG_ADDR_MNPL_WREG32, CFG_ADDR_MNPL_RREG32
and CFG_HBW_ADDR_MNPL apis which will be used when accessing
other die config space. These apis disables the address manipulation
control and then perform read/write operations and renables
this bit.

+/*
+ * Note: It is the responsiblity of the caller to do arch_irq_lock/unlock such that
+ * the caller is not preempted which otherwise might result in undefined behaviour
+ */
+void fs1_addr_mnpl_ctrl(bool en, bool hbw_mem)
+{
+	uint32_t reg = hbw_mem ? mmARC_LOCAL_CONF_MEM_ADDR_MNPL_CTRL :
+				mmARC_LOCAL_CONF_CFG_ADDR_MNPL_CTRL;
+	if (fs1_die_id == FS1_INVALID_DIE_ID)
+		fs1_die_id = dev_read_die_id();
+	if (core_id == FS1_INVALID_CORE_ID)
+		core_id = FS_ARC_V2_CORE_ID();
+
+	if (offset)
+		goto addr_cntrl;
+
+	switch (fs1_die_id) {
+	case FS1_IO_DIE_0:
+		if (core_id == IO0_PCIE_ARC_CXL_ID)
+			return;
+		offset = io_base_addr[core_id];
+	break;
+	case FS1_COMPUTE_DIE_0:
+		offset = cmpt_base_addr[core_id];
+	break;
+	case FS1_COMPUTE_DIE_1:
+		offset = COMPUTE_DIST + cmpt_base_addr[core_id];
+	break;
+	default:
+		LOG_DBG("Invalid die id %u", fs1_die_id);
+	return;
+	}
+
+addr_cntrl:
+	CFG_WREG32((offset + reg), en ? 1 : 0);
+}
