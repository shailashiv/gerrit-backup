+// SPDX-License-Identifier:     GPL-2.0+

+ * All Rights Reserved.

+
+#include <efuse.h>
+#include <boot_if.h>
+#include <tpm2.h>
+#include <uart.h>
+#include "lzo/unaligned.h"
+
+#define TPM_NV_DEV_SN_ADDRESS	0x01c10100
+
+struct tpm_dev_id {
+	union {
+		u32 efuse_data[2];
+		u64 tpm_data;
+	};
+};
+
+static int preboot_get_efuse_dev_id(struct tpm_dev_id *efuse_dev_id_ret)
+{
+	int ret;
+
+	ret = preboot_fuse_read(EFUSE_DEV_ID_BACKUP_OFFSET_1, &efuse_dev_id_ret->efuse_data[0]);
+	if (ret)
+		return ret;
+
+	ret = preboot_fuse_read(EFUSE_DEV_ID_BACKUP_OFFSET_2, &efuse_dev_id_ret->efuse_data[1]);
+	if (ret)
+		return ret;
+
+	return 0;
+}
+
+void preboot_tpm_auth(void)
+{
+	int ret;
+	u64 tpm_dev_id = 0;
+	struct tpm_dev_id efuse_dev_id;
+
+	efuse_dev_id.tpm_data = 0x0;
+
+	ret = tpm2_init();
+	if (ret)
+		cpu_boot_err_set(CPU_BOOT_ERR_TPM_FAIL);
+
+	ret = tpm2_startup(TPM2_SU_CLEAR);
+	if (ret) {
+		preboot_puts_hex("TPM: startup failed. ret = ", ret, true);
+		cpu_boot_err_set(CPU_BOOT_ERR_TPM_FAIL);
+	}
+
+	ret = tpm2_nv_read(TPM_NV_DEV_SN_ADDRESS, &tpm_dev_id, sizeof(tpm_dev_id));
+	if (ret) {
+		preboot_puts_hex("TPM: NV_READ failed. ret = ", ret, true);
+		cpu_boot_err_set(CPU_BOOT_ERR_TPM_FAIL);
+	}
+
+	/* tpm device id returns as Big Endian, need to convert */
+	tpm_dev_id = get_unaligned_be64(&tpm_dev_id);
+	preboot_puts_hex64("TPM device ID: ", tpm_dev_id, true);
+
+	ret = preboot_get_efuse_dev_id(&efuse_dev_id);
+	if (ret) {
+		preboot_puts_hex("TPM flow: efuse read failed. ret = ", ret, true);
+		cpu_boot_err_set(CPU_BOOT_ERR_TPM_FAIL);
+	} else {
+		preboot_puts_hex64("EFUSE device ID: ", efuse_dev_id.tpm_data, true);
+	}
+
+	if (efuse_dev_id.tpm_data == 0) {
+		preboot_puts("TPM flow: error: efuse device ID not programed\n");
+		cpu_boot_err_set(CPU_BOOT_ERR_TPM_FAIL);
+	} else if (efuse_dev_id.tpm_data != tpm_dev_id) {
+		preboot_puts("TPM flow: error: efuse device ID is different than TPM device ID\n");
+		cpu_boot_err_set(CPU_BOOT_ERR_TPM_FAIL);
+	}
+}
