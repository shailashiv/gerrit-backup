+static bool is_arc_in_reset(int core_id)
+{
+	uint32_t reg_offset;
+
+	switch (core_id) {
+	case FIT_CORE_ID_D0_PSOC_ARC0:
+		reg_offset = mmD0_PARC_ARC0_CFG_ARC_RST;
+		break;
+	case FIT_CORE_ID_D0_PSOC_ARC1:
+		reg_offset = mmD0_PARC_ARC1_CFG_ARC_RST;
+		break;
+	case FIT_CORE_ID_D0_PSOC_ARC2:
+		reg_offset = mmD0_PARC_ARC2_CFG_ARC_RST;
+		break;
+	default:
+		return false;
+	}
+
+	if (regmap_readl(reg_offset) & D0_PARC_ARC0_CFG_ARC_RST_CORE_M)
+		return true;
+

+#define ARC_RST_VEC_ADDR_S				0xA
+
+void arcboot_rom_bring_up_arc(int core_id, uint64_t load_addr)
+{
+
+	uint32_t arc_dst, base_iccm_addr, arc_rst, arc_vec, val;
+
+	if (!is_arc_in_reset(core_id))
+		run_halt_arc(core_id, 0);
+
+	switch (core_id) {
+	case FIT_CORE_ID_D0_PSOC_ARC0:
+		arc_rst = mmD0_PARC_ARC0_CFG_ARC_RST;
+		arc_vec = mmD0_PARC_ARC0_CFG_RST_VEC_ADDR;
+		base_iccm_addr = HL_ARC0_ICCM_BASE;
+		break;
+	case FIT_CORE_ID_D0_PSOC_ARC2:
+		arc_rst = mmD0_PARC_ARC2_CFG_ARC_RST;
+		arc_vec = mmD0_PARC_ARC2_CFG_RST_VEC_ADDR;
+		base_iccm_addr = HL_ARC2_ICCM_BASE;
+		break;
+	default:
+		return;
+	}
+
+	/*
+	 * Assumption made here is that deassert core will be called
+	 * other than arc1 which is already running.
+	 */
+	arc_dst = (uint32_t)arcboot_rom_get_arc_addr(load_addr);
+	if (((load_addr >= HL_ARC0_ICCM_FULL_BASE) &&
+		(load_addr <= HL_ARC0_ICCM_FULL_BASE + ARC0_ICCM_SIZE)) ||
+		((load_addr >= HL_ARC2_ICCM_FULL_BASE) &&
+		(load_addr <= HL_ARC2_ICCM_FULL_BASE + ARC2_ICCM_SIZE))) {
+		arc_dst = arc_dst - base_iccm_addr;
+	}
+
+	/* put core in reset */
+	regmap_writel(D0_PARC_ARC0_CFG_ARC_RST_CORE_M, arc_rst);
+
+	/* change reset vector based on load address */
+	regmap_writel((arc_dst >> ARC_RST_VEC_ADDR_S), arc_vec);
+
+	/* bring core out of reset */
+	val = (D0_PARC_ARC0_CFG_ARC_RST_PRESETDBGN_M | 0x0);
+	regmap_writel(val, arc_rst);
+}
