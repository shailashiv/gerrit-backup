+	case HL_FW_FIT_BOOT_ERR_REVOKED_KEY:
+		sig_fail->revoked_pub_key = 1;
+		is_update_sig_os = true;
+		break;
+	case HL_FW_FIT_BOOT_ERR_SVN:
+		sig_fail->svn_failure = 1;
+		is_update_sig_os = true;
+		break;
