+	int err = 0;
+
+	if (state->log_requests)

+
+	for (int idx = 0; idx < comp_sensor_count; idx++) {
+		if (req->op_field[idx].event_msg_enable) {

+			err = PLDM_PMC_SET_STATE_SENSOR_ENABLES_STATUS_INVALID_OP_STATE;

+	err = pldm_resp_prepare(pldm, msg, 0);

+static void oobi_decode_state_sensor_state(uint8_t *op_states, uint16_t comp_op_state)

+	for (int i = 0; i < PLDM_PDR_MAX_COMPOSITE_SENSORS; i++) {

+static void sensord_persist_state_sensor_enable(struct sensord_sensor_entry *sensor)
+{
+	struct hl_oobi_pldm_state_sensor_op_state *item = NULL;
+	struct hl_oobi_pldm_state_sensor_op_state *empty = NULL;
+	struct hl_oobi_pldm_sensors *oobi = &sensord_state.oobi;
+	uint16_t sensor_id = get_sensor_id(sensor);
+
+	for (size_t i = 0; i < ARRAY_SIZE(oobi->state_op_state); i++) {
+		if (oobi->state_op_state[i].sensor_id == sensor_id) {
+			item = &oobi->state_op_state[i];
+			break;
+		}
+		if (!empty && oobi->state_op_state[i].sensor_id == 0)
+			empty = &oobi->state_op_state[i];
+	}
+	if (!item)
+		item = empty;
+	if (!item) {
+		LOG_WRN("%s: Unable to persist sensor op_states  %d: table_full", __func__,
+			sensor_id);
+		return;
+	}
+
+	item->sensor_id = sensor_id;
+	item->comp_op_state = 0; /* ENABLE all composite sensors */
+	for (int idx = 0; idx < sensor->state->value.composite.count; idx++) {
+		uint8_t current_op_state =
+			sensor->state->value.composite.element[idx].operational_state;
+
+		if (current_op_state == PLDM_SENSOR_OP_STATE_UNAVAILABLE)
+			item->comp_op_state |= HL_OOBI_PLDM_STATE_SENSOR_OP_STATE_UNAVAILABLE(idx);
+		else if (current_op_state == PLDM_SENSOR_OP_STATE_DISABLED)
+			item->comp_op_state |= HL_OOBI_PLDM_STATE_SENSOR_OP_STATE_DISABLED(idx);
+	}
+
+	/* No composite sensor is disabled/unavailable remove from persistence */
+	if (!item->comp_op_state)
+		item->sensor_id = 0;
+
+	hl_oobi_pldm_sensor_persist_data(oobi);
+}

+int sensord_state_sensor_enables_set(
+	uint16_t sensor_id,
+	uint8_t op_state_count,
+	uint8_t *op_states)
+{
+	struct sensord_sensor_entry *sensor;
+	struct sensord_sensor_state *state;
+	uint8_t composite_sensor_count;
+	k_spinlock_key_t key;
+
+	sensor = find_sensor(sensor_id, STATE_SENSOR_TYPE_MASK);
+	if (!sensor)
+		return PLDM_PMC_SET_STATE_SENSOR_ENABLES_STATUS_INVALID_SENSOR_ID;
+
+	state = sensor->state;
+
+	composite_sensor_count = MIN(op_state_count, state->value.composite.count);
+
+	key = k_spin_lock(&sensord_state.lock);
+
+	for (int idx = 0; idx < composite_sensor_count; idx++) {
+		struct sensord_sensor_value_composite_element *element =
+			&state->value.composite.element[idx];
+		uint8_t current_op_state = element->operational_state;
+		uint8_t desired_op_state = op_states[idx];
+
+		if (current_op_state != desired_op_state) {
+			element->operational_state = desired_op_state;
+			element->present_state = PLDM_SENSOR_THRESHOLD_UNKNOWN;
+			element->previous_state = PLDM_SENSOR_THRESHOLD_UNKNOWN;
+			element->event_state = PLDM_SENSOR_THRESHOLD_UNKNOWN;
+
+			if (desired_op_state == PLDM_SENSOR_OP_STATE_ENABLED) {
+				state->polled = false; /* reset poll-once tracking */
+
+				/* trigger daemon poll immediately after enabling a sensor */
+				state->timestamp_next_poll = 0;
+				sensord_state.timestamp_next_poll = 0;
+			}
+		}
+	}
+
+	k_spin_unlock(&sensord_state.lock, key);
+
+	sensord_persist_state_sensor_enable(sensor);
+
+	return MCTP_STATUS_SUCCESS;
+}
