Commit Message

[SW-36216] greco: security: hbw rr preparation

Implementation of base apis for hbw range register configuration.

There are 10(0..9 rr idx) RRs for HBW
NOTE: All the RRs for HBW are of 4KB granularity

1) 6(0..5 rr idx) Regions with 39 bit addressing (0x7F_Fx_xxxx)
API: preboot_secure_hbw_rr_short(rr idx, min addr, max addr)

2) 2(6..7 rr idx) Regions with 50 bit addressing.
API: preboot_secure_hbw_rr_full(rr idx, min addr, max addr)

3) 2(8..9 rr idx) Regions with 50 bit addressing, to block only write.
API: preboot_secure_hbw_rr_full(rr idx, min addr, max addr)

+#define MAX_HBW_SHORT_RR	6
+#define MAX_HBW_READ_RRS	7
+#define BITS_IN_REG		32

+static void preboot_secure_hbw_rr_full_if(int rr_idx, u64 min64, u64 max64,
+		enum preboot_sec_engine_type type)
+{
+	u32 min_l = (u32)(min64);
+	u32 min_h = (u32)(min64 >> BITS_IN_REG);
+	u32 max_l = (u32)(max64);
+	u32 max_h = (u32)(max64 >> BITS_IN_REG);
+	u32 rr_base, rr_dest, rr_calc;
+	int i;
+	struct preboot_sec_rrs_cfg *cfg = &preboot_sec_hbw_rrs_tbl[type];
+
+	rr_calc = rr_idx - MAX_HBW_SHORT_RR;
+	for (i = 0; i < cfg->instances_num; i++) {
+		rr_base = rr_calc * sizeof(u32) + cfg->instances_dist * i;
+		rr_dest = rr_base + cfg->min_full_breg;
+		regmap_writel(min_l, rr_dest);
+		rr_dest = rr_base + cfg->min_full_breg_hi;
+		regmap_writel(min_h, rr_dest);
+		rr_dest = rr_base + cfg->max_full_breg;
+		regmap_writel(max_l, rr_dest);
+		rr_dest = rr_base + cfg->max_full_breg_hi;
+		regmap_writel(max_h, rr_dest);
+
+		/*
+		 * RRs 8..9 only blocks write operations
+		 */
+		if (rr_idx > MAX_HBW_READ_RRS)
+			preboot_secure_update_hit_reg(rr_idx, cfg, i, true);
+		else
+			preboot_secure_update_hit_reg(rr_idx, cfg, i, false);
+	}
+}
+

+		preboot_secure_update_rr_reg(rr_idx, cfg->instances_dist * i, cfg->min_short_breg,

+		preboot_secure_update_hit_reg(rr_idx, cfg, i, false);
