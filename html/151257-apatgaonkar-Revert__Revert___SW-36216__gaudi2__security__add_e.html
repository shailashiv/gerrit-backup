+/*
+ * The routine shall return true in case of security enable in strap.
+ * Otherwise, shall return false (The polarity of strap is inverted:
+ * meaning '0' is for security enable).
+ */
+static bool hl_is_secuirty_strap_enabled(void)
+{
+	u32 boot_strap;
+
+	boot_strap = regmap_readl(mmPSOC_GLOBAL_CONF_BOOT_STRAP_PINS_H);
+
+	if (boot_strap & BOOT_STRAP_SECURITY_ENABLE_BIT)
+		return false;
+
+	return true;
+}
+
+bool preboot_security_is_enabled(void)
+{
+	u32 efuse_word;
+
+	/* If security is enabled via boot strap pins */
+	if (hl_is_secuirty_strap_enabled())
+		return true;
+
+	/* If eFuse read failed, assume an attacker sabotage, security in on */
+	if (preboot_fuse_read(EFUSE_SECURITY_WORD_OFFSET, &efuse_word))
+		return true;
+
+	return !!(efuse_word & EFUSE_SECURITY_ENABLE_BIT);
