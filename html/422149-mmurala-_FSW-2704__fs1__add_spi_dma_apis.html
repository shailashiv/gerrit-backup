+
+#include <boot_rom.h>
+#include <mem_io.h>
+#include <fs1_blocks_linux_driver.h>
+#include <parc_global_conf_masks.h>
+#include <parc_global_conf_regs.h>
+#include <fs1_reg_map.h>
+#include <dma.h>
+
+#define DMA_ADDRESS_LOW(x)	((x) & MASK_32_BIT)
+#define DMA_ADDRESS_HIGH(x)	(((x) >> 32) & MASK_32_BIT)
+#define DMA_SRC_BLOCK_SIZE	1024
+#define DMA_DST_BLOCK_SIZE	4
+
+void dma_copy_image_internal(uint16_t image_size, uint32_t flash_offset,
+		uint64_t dest_offset, uint8_t fw_sram_dst, uint8_t restart_crc,
+		uint8_t mask_crc)
+{
+	uint32_t reg_spi_dma_ctrl_s =  0;
+	uint32_t spi_dma_dst_addr_l = DMA_ADDRESS_LOW(dest_offset);
+	uint32_t spi_dma_dst_addr_h = DMA_ADDRESS_HIGH(dest_offset);
+	uint32_t flash_offset_1k = flash_offset / DMA_SRC_BLOCK_SIZE;
+	uint32_t image_size_4b = (image_size / DMA_DST_BLOCK_SIZE);
+
+	regmap_writel(spi_dma_dst_addr_l,
+		      IO0_H0_PSOC_PARC_CFG_MISC_BASE(mmPARC_GLOBAL_CONF_SPI_DMA_DST_ADDR_L));
+	regmap_writel(spi_dma_dst_addr_h,
+		      IO0_H0_PSOC_PARC_CFG_MISC_BASE(mmPARC_GLOBAL_CONF_SPI_DMA_DST_ADDR_H));
+
+	/* enable DMA transfer */
+	reg_spi_dma_ctrl_s = PARC_GLOBAL_CONF_SPI_DMA_CTRL_EN_M;
+
+	/* sram=1 hbw=0 as destination */
+	reg_spi_dma_ctrl_s |= fw_sram_dst << PARC_GLOBAL_CONF_SPI_DMA_CTRL_DST_SRAM_S;
+
+	/* 0: continue crc calc 1: restart crc*/
+	reg_spi_dma_ctrl_s |= restart_crc << PARC_GLOBAL_CONF_SPI_DMA_CTRL_CRC_RESTART_S;
+
+	/* 0: pass crc error to status 1: mask crc error */
+	reg_spi_dma_ctrl_s |= mask_crc << PARC_GLOBAL_CONF_SPI_DMA_CTRL_MASK_CRC_ERR_S;
+
+	/* # of 4bytes to DMA copy */
+	reg_spi_dma_ctrl_s |= image_size_4b << PARC_GLOBAL_CONF_SPI_DMA_CTRL_MEM_SIZE_S;
+
+	/* spi offset in 1k chunks */
+	reg_spi_dma_ctrl_s |= flash_offset_1k << PARC_GLOBAL_CONF_SPI_DMA_CTRL_ADDR_SHIFT_S;
+
+	regmap_writel(reg_spi_dma_ctrl_s,
+		      IO0_H0_PSOC_PARC_CFG_MISC_BASE(mmPARC_GLOBAL_CONF_SPI_DMA_CTRL));
+}
+
+int dma_wait_finish(void)
+{
+	uint32_t reg_spi_dma_status_s = 0;
+
+	do {
+		reg_spi_dma_status_s =
+		  regmap_readl(IO0_H0_PSOC_PARC_CFG_MISC_BASE(mmPARC_GLOBAL_CONF_SPI_DMA_STATUS));
+	} while ((reg_spi_dma_status_s & PARC_GLOBAL_CONF_SPI_DMA_STATUS_DONE_M) == 0);
+
+	return (reg_spi_dma_status_s & PARC_GLOBAL_CONF_SPI_DMA_STATUS_ERROR_M);
+}
+
+enum dma_error dma_crc_check(uint32_t expected_crc)
+{
+	uint32_t reg_spi_dma_crc_s = 0;
+
+	reg_spi_dma_crc_s =
+		regmap_readl(IO0_H0_PSOC_PARC_CFG_MISC_BASE(mmPARC_GLOBAL_CONF_SPI_DMA_CRC));
+
+	if (reg_spi_dma_crc_s != expected_crc)
+		return DMA_INVALID_CRC_ERROR;
+
+	return DMA_SUCCESS;
+}
