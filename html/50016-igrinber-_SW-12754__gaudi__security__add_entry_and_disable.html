+	regmap_writel(0, mmPCIE_WRAP_RR_RD_PRV_RANGE_REG_MIN_0);
+	regmap_writel(0, mmPCIE_WRAP_RR_RD_PRV_RANGE_REG_MAX_0);
+	regmap_writel(0, mmPCIE_WRAP_RR_RD_PRV_RANGE_REG_MIN_1);
+	regmap_writel(0, mmPCIE_WRAP_RR_RD_PRV_RANGE_REG_MAX_1);
+	preboot_puts("Disabled PCIe RRs\n");
+
+	/* PCIe RTR RR - HBM WP 0x0 - 0x07FFFFFFFF */

+	regmap_writel(0, mmSIF_RTR_CTRL_7_RANGE_PRIV_BASE_LOW_AR_0);
