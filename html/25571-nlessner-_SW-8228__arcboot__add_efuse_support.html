+#include <unistd.h>
+#include "arcboot.h"
+
+#define FUSE_TIMEOUT_US		100000
+#define FUSE_DELAY_US		20
+#define FUSE_RETRIES		(FUSE_TIMEOUT_US / FUSE_DELAY_US)
+#define FUSE_WORDS_PER_BANK	64
+
+/*
+ * fuse box is devided into half for redundancy
+ * each half is bank of FUSE_WORDS_PER_BANK words
+ * NOTE: bank1 is a complete duplication of bank0
+ */
+int fuse_read(u32 word, u32 *val)
+{
+	int i, offset, delay;
+
+	if (word > 64) {
+		return -1;
+	}
+
+	regmap_writel(1, mmPSOC_EFUSE_READ);
+
+	for (i=0; i < FUSE_RETRIES; i++)
+		if (regmap_readl(mmPSOC_EFUSE_READ))
+			for(delay=FUSE_DELAY_US; delay > 0; delay--) asm("nop");
+		else
+			break;
+
+	if (i == FUSE_RETRIES) {
+		return -2;
+	}
+
+	// read value from bank0
+	offset = (word * sizeof(u32));
+	*val = regmap_readl(mmPSOC_EFUSE_DATA_0 + offset);
+
+	// read value from bank1 and OR with value from bank0
+	offset = ((FUSE_WORDS_PER_BANK + word) * sizeof(u32));
+	*val |= regmap_readl(mmPSOC_EFUSE_DATA_0 + offset);
+
+	return 0;
+}
