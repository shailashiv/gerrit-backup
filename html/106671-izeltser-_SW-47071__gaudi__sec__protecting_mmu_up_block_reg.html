+		PROT_BIT(mmMMU_UP_LOG2_DDR_SIZE) |

+		PROT_BIT(mmMMU_UP_DBG_MEM_WRAP_RM) |
+		PROT_BIT(mmMMU_UP_PIPE_CREDIT) |
+		PROT_BIT(mmMMU_UP_SLICE_CREDIT) |
+		PROT_BIT(mmMMU_UP_MMU_BYPASS));
