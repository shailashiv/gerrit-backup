[SW-156330] hl: tpm: extract ppboot img hash

In gaudi2 ppboot image hash is extracted from
the ppboot image, while in gaudi3 it is extracted
from the image descriptors. Added platform specific
functions to get the ppboot img hash. Extracted hash
will be used to extend the TPM PCR of ppboot.

+	ppboot_pcie_size -= SIG_SIZE;

+	ppboot_pcie_size -= TOTAL_PUB_KEY_PROPS_SZ;
+	ppboot_pcie_size -= PPBOOT_SHA384_FOOTER;
+	ppboot_pcie_size -= PPBOOT_HASH_MAGIC_SIZE;
+
+	return (uintptr_t)(HL_FWSRAM_BASE + ppboot_pcie_size);
+}
+
+bool get_ppboot_img_hash(u8 *hash)
+{
+	uintptr_t data;
+	uint32_t magic;
+
+	data = hl_get_ppboot_data();
+
+	memcpy(&magic, (void *)data, PPBOOT_HASH_MAGIC_SIZE);
+	if (magic != PPBOOT_SHA_MAGIC) {
+		preboot_puts("no ppboot sha magic found\n");
+		return false;
+	}
+
+	memcpy(hash, (void *)(data + PPBOOT_HASH_MAGIC_SIZE), SHA384_SUM_LEN);
+	return true;
