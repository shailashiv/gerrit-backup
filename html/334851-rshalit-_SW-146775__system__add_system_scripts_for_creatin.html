+#!/usr/bin/env bash
+
+set -e          # stop on any error in this script
+set -u          # stop on any reference to undefine variable
+set -o pipefail # stop on any error in pipeline
+
+
+MKIMAGE_FILE=mkimage
+CPLD_ITS_TEMPLATE=cpld.its
+FIT_LOAD_TYPE_CPLD=8
+HL_DEV=${1}
+CPLD_ITS_FILE=${HL_DEV}-cpld.its
+CPLD_ITB_FILE=${HL_DEV}-cpld.itb
+
+function cpld_usage {
+	echo "Usage: $0 <greco/gaudi/gaudi2/gaudi3> [input vme file] [output itb file]"
+	echo "For example:"
+	echo "$0 gaudi ~/input.txt ~/output.itb"
+	echo "  shall create .itb file for gaudi using input ~/input.vme, and output itb will be created in ~/output.itb"
+	echo "This file can then be used with hl-fw-loader to program cpld"
+	echo ""
+	echo ""
+	exit
+}
+
+if [ "$#" -lt 1 ]; then
+	cpld_usage
+fi
+
+if [ "$1" = "gaudi" ]; then
+	echo "Chosen device: gaudi"
+elif [ "$1" = "gaudi2" ]; then
+	echo "Chosen device: gaudi2"
+elif [ "$1" = "gaudi3" ]; then
+	echo "Chosen device: gaudi3"
+elif [ "$1" = "greco" ]; then
+	echo "Chosen device: greco"
+else
+	cpld_usage
+fi
+
+if [ "$#" -gt 1 ] && [ -f $2 ]; then
+	INPUT_FILE=${2}
+else
+	cpld_usage
+fi
+
+if [ "$#" -gt 2 ] && [ -f $2 ]; then
+	OUTPUT_FILE=${3}
+else
+	echo "No output path, so create output in current folder."
+	OUTPUT_FILE=output.itb
+fi
+
+if [ "$HL_DEV" = "gaudi" ]; then
+	FIT_TARGET_DEV=0x1000
+	SHA_VERSION=sha256
+elif [ "$HL_DEV" = "gaudi2" ]; then
+	FIT_TARGET_DEV=0x1020
+	SHA_VERSION=sha384
+elif [ "$HL_DEV" = "gaudi3" ]; then
+	FIT_TARGET_DEV=0x1060
+	SHA_VERSION=sha384
+elif [ "$HL_DEV" = "greco" ]; then
+	FIT_TARGET_DEV=0x0020
+	SHA_VERSION=sha256
+else
+	echo "Wrong input"
+	exit
+fi
+
+ITS_SED_PARAMS+="s|@HL_DEV@|${HL_DEV}|g;"
+ITS_SED_PARAMS+="s/@TARGET_DEVICE_CPLD@/${FIT_TARGET_DEV}/g;"
+ITS_SED_PARAMS+="s|@CPLD_BIN@|${INPUT_FILE}|g;"
+ITS_SED_PARAMS+="s/@SHA_VERSION@/${SHA_VERSION}/g;"
+ITS_SED_PARAMS+="s|@FIT_LOAD_TYPE_CPLD@|${FIT_LOAD_TYPE_CPLD}|g;"
+
+sed ${ITS_SED_PARAMS} ${CPLD_ITS_TEMPLATE} > ${CPLD_ITS_FILE}
+$MKIMAGE_FILE -D "-W no-unit_address_vs_reg" -f ${CPLD_ITS_FILE} ${OUTPUT_FILE}
+
+rm ${CPLD_ITS_FILE}

+#!/usr/bin/env bash
+
+set -e          # stop on any error in this script
+set -u          # stop on any reference to undefine variable
+set -o pipefail # stop on any error in pipeline
+
+function eeprom_usage {
+	echo "Usage: $0 <greco/gaudi/gaudi2/gaudi3> [input text file] [output itb file]"
+	echo "For example:"
+	echo "$0 gaudi ~/input.txt ~/output.itb"
+	echo "  shall create .itb file for gaudi using input ~/input.txt, and output itb will be created in ~/output.itb"
+	echo "This file can then be used with hl-fw-loader to program eeprom"
+	echo ""
+	echo ""
+	exit
+}
+
+if [ "$#" -lt 1 ]; then
+	eeprom_usage
+fi
+
+if [ "$1" = "gaudi" ]; then
+	echo "Chosen device: gaudi"
+elif [ "$1" = "gaudi2" ]; then
+	echo "Chosen device: gaudi2"
+elif [ "$1" = "gaudi3" ]; then
+	echo "Chosen device: gaudi3"
+elif [ "$1" = "greco" ]; then
+	echo "Chosen device: greco"
+else
+	eeprom_usage
+fi
+
+if [ "$#" -gt 1 ] && [ -f $2 ]; then
+	INPUT_FILE=${2}
+else
+	eeprom_usage
+fi
+
+if [ "$#" -gt 2 ] && [ -f $2 ]; then
+	OUTPUT_FILE=${3}
+else
+	echo "No output path, so create output in current folder."
+	OUTPUT_FILE=output.itb
+fi
+
+echo ${INPUT_FILE}
+
+HL_DEV=${1}
+
+FIT_LOAD_TYPE_EEPROM=7
+MKIMAGE_FILE=mkimage
+EEPROM_ITS_TEMPLATE=eeprom.its
+
+if [ "$HL_DEV" = "gaudi" ] || [ "$HL_DEV" = "greco" ]; then
+	SHA_VERSION=sha256
+elif [ "$HL_DEV" = "gaudi2" ] || [ "$HL_DEV" = "gaudi3" ]; then
+	SHA_VERSION=sha384
+else
+	echo "Wrong input"
+	exit
+fi
+
+EEPROM_ITS_FILE=${HL_DEV}-eeprom.its
+EEPROM_ITB_FILE=${OUTPUT_FILE}
+ITS_SED_PARAMS+="s|@HL_DEV@|${HL_DEV}|g;"
+ITS_SED_PARAMS+="s|@FIT_LOAD_TYPE_EEPROM@|${FIT_LOAD_TYPE_EEPROM}|g;"
+ITS_SED_PARAMS+="s|@EEPROM_TXT@|${INPUT_FILE}|g;"
+ITS_SED_PARAMS+="s|@SHA_VERSION@|${SHA_VERSION}|g;"
+
+function eeprom_check_input
+{
+	FILE=$(file -b ${INPUT_FILE})
+	if [[ ${FILE} != "ASCII text" ]];then
+		echo "Error: File encoding not supported.Please use UNIX encoding supported os/editor"
+		exit 1
+	fi
+
+	INV_CHAR=$({ grep -o -P '[^a-z,A-Z,0-9,:,\x5b,\x5d,\x0a,\x3d,\x20]' ${INPUT_FILE} || test $? = 1;} | wc -w)
+	if [ ${INV_CHAR} -gt 0 ];then
+		echo "Error: Invalid characters in the input file"
+		exit 1
+	fi
+}
+
+function align_4B()
+{
+	local TMP_FILE=$(mktemp)
+
+	# LKD like this to be 4B aligned in size
+	dd status=none conv=sync bs=4 if=${1} > ${TMP_FILE}
+	mv ${TMP_FILE} ${1}
+}
+
+eeprom_check_input
+sed  ${ITS_SED_PARAMS} ${EEPROM_ITS_TEMPLATE} > ${EEPROM_ITS_FILE}
+$MKIMAGE_FILE -D "-W no-unit_address_vs_reg" -f ${EEPROM_ITS_FILE} ${EEPROM_ITB_FILE}
+
+align_4B ${EEPROM_ITB_FILE}
+rm ${EEPROM_ITS_FILE}
