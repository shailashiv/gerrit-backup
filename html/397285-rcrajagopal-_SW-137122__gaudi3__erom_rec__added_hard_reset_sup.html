+#define DELAY_1K_USEC	1000
+
+static void hl_rst_sw_all_reset(void)
+{
+	preboot_puts("Issuing hard reset\n");
+
+	regmap_rmwritel(D0_PSOC_GLOBAL_CONF_REG(mmGLOBAL_CONF_DIE_READY_IN), 0x0,
+					GLOBAL_CONF_DIE_READY_IN_IND_M);
+	/* Issue reset to slave die */
+	regmap_writel(0x1, D1_PSOC_RESET_CONF_REG(mmPSOC_RESET_CONF_SW_ALL_RST));
+	/* wait for slave die to reset no indication is required */
+	udelay(DELAY_1K_USEC);
+	/* Issue master die0 reset */
+	regmap_writel(0x1, D0_PSOC_RESET_CONF_REG(mmPSOC_RESET_CONF_SW_ALL_RST));
+}
