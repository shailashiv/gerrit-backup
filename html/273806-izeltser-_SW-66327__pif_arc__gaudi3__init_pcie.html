Commit Message

[SW-66327] pif_arc: gaudi3: init pcie

In previous devices the task of configuring and monitoring PciE link
was offloaded to psoc ARC0 or another core of the ARM CPU.
In Gaudi3, since we have a dedicated ARC CPU in the PIF, it shall
handle this task.

Every time the PIF ARC is booting (cold/warm reboot), it waits for
prst_n signal to deassert, configures the required parameters of the
PHY and PIF, upgrades the PHY FW, and enables the link state machine.

Once the PciE link is up, PIF ARC proceeds to monitoring the incoming
FLR requests.

+static void pif_arc_init_pcie_phy_fw_update(void)
+{
+	/*
+	 * Configure FW update size.
+	 * PHY_FW_FSM_SIZE takes few clocks to propagate. It is being sampled by HW few
+	 * clock cycles after the rising edge of PCIE_WRAP ➔ PHY_FW_FSM ➔ DIRECT_WRITE_EN bit.
+	 */
+	regmap_writel(0, mmD0_PCIE_WRAP_PHY_FW_FSM);
+	regmap_writel(FW_WORD_NUM, mmD0_PCIE_WRAP_PHY_FW_FSM_SIZE);
+
+	/* let PHY_FW_FSM_SIZE propagate */
+	pif_ts_delay(1);
+
+	/*
+	 * PHY FW FSM EN and Direct FW write mechanism enable
+	 * FW FSM assumes that the data is present in the SRAM
+	 * The FSM will wait for SRAM init done before updating
+	 * the SRAM contents
+	 */
+	regmap_writel(PHY_FW_FSM_ENA_MASK, mmD0_PCIE_WRAP_PHY_FW_FSM);
+
+	/* Take the PHY out of reset */
+	regmap_writel(1, mmD0_PSOC_GLOBAL_CONF_PCIE_PHY_RST_N);
+}

+static void pcie_ref_clk_mode_init(void)
+{
+	u32 strap_h = regmap_readl(mmD0_PSOC_BOOT_CONF_BOOT_STRAP_PINS_H);
+	bool srns, sris;
+
+	srns = strap_h & D0_PSOC_BOOT_CONF_BOOT_STRAP_PINS_H_SRNS_M;
+	sris = strap_h & D0_PSOC_BOOT_CONF_BOOT_STRAP_PINS_H_SRIS_M;
+
+	/* if common clk mode, no configs are required */
+	if (!srns && !sris)
+		return;
+
+	/* common configs for both srns and sris clk */
+	regmap_writel(0, mmD0_PCIE_PHY_PIPE_LANE_CMN_REFCLK_MODE);
+	regmap_writel(0, mmD0_PCIE_PHY_PHYX_MPLL_SHORT_LOCK_EN_0);
+	regmap_writel(0, mmD0_PCIE_PHY_PHYX_MPLL_SHORT_LOCK_EN_1);
+	regmap_writel(0, mmD0_PCIE_PHY_PHYX_MPLL_SHORT_LOCK_EN_2);
+	regmap_writel(0, mmD0_PCIE_PHY_PHYX_MPLL_SHORT_LOCK_EN_3);
+
+	if (sris) {
+		regmap_writel(1, mmD0_PCIE_AUX_APP_SRIS_MODE);
+		regmap_writel(1, mmD0_PCIE_PHY_PIPE_RX_SRIS_MODE_EN);

+void pif_arc_pcie_monitoring(void)
+{
+	/* Wait for PRSTn */
+	while (!regmap_readl(mmD0_PSOC_GLOBAL_CONF_PRSTN))
+		;
+
+	/* Only after PRSTn is de-asserted */
+	pif_arc_init_pcie();
+
+	hl_pif_report_sts(PIF_STAT_WAITING_FOR_LINK);
+	while (1) {
+		unsigned int prstn, ltssm, smlh, rdlh;
+
+		prstn = regmap_readl(mmD0_PSOC_GLOBAL_CONF_PRSTN);
+		ltssm = regmap_readl(mmD0_PSOC_GLOBAL_CONF_PRSTN);
+		smlh = regmap_readl(mmD0_PCIE_AUX_SMLH_LINK_UP);
+		rdlh = regmap_readl(mmD0_PCIE_AUX_RDLH_LINK_UP);
+
+		if (prstn && smlh && rdlh)
+			break;
+

+		 * Some host assert prstn during link up while moving to GEN3
+		 * This cause BAR size to be reset
+		 * We stop link up process by disabling ltssm
+		 * till BAR is reconfigured
+		 */

+		}

+			continue;
+		}
+	}

+	hl_pif_report_sts(PIF_STAT_PCIE_INIT_DONE);
