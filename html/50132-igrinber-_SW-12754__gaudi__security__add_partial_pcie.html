+static void preboot_secure_pci_wrap(void)
+{
+	u32 mask, prot_base, base = mmPCIE_WRAP_BASE & ~HL_CFG_BASE;
+
+	/* Set PCI level to secured */
+	mask = PCIE_WRAP_LBW_PROT_OVR_WR_EN_MASK |
+		PCIE_WRAP_LBW_PROT_OVR_RD_EN_MASK;
+	regmap_writel(mask, mmPCIE_WRAP_LBW_PROT_OVR);
+
+	prot_base = base;	/* offsets: 0 - 0x200 */
+	set_prot_bits_range_all(prot_base, SNGL_REG_PROT_SIZE * 4);
+
+	prot_base += SNGL_REG_PROT_SIZE * 4;	/* offsets: 0x200 - 0x280 */
+	mask = ~(PROT_BIT(mmPCIE_WRAP_RR_ELBI_WR_SEC_REG_CTRL) &
+		PROT_BIT(mmPCIE_WRAP_RR_ELBI_RD_SEC_REG_CTRL));
+	regmap_writel(mask, PRIV_PROT_REG(prot_base));
+
+	prot_base += SNGL_REG_PROT_SIZE;	/* offsets: 0x280 - 0xB00 */
+	set_prot_bits_range_all(prot_base, SNGL_REG_PROT_SIZE * 17);
+
+	prot_base += SNGL_REG_PROT_SIZE * 17;	/* offsets: 0xB00 - 0xB80 */
+	mask = ~PROT_BIT(mmPCIE_WRAP_MAX_OUTSTAND);
+	regmap_writel(mask, PRIV_PROT_REG(prot_base));
+
+	prot_base += SNGL_REG_PROT_SIZE;	/* offsets: 0xB80 - 0xD00 */
+	set_prot_bits_range_all(prot_base, SNGL_REG_PROT_SIZE * 3);
+
+	prot_base += SNGL_REG_PROT_SIZE;	/* offsets: 0xD00 - 0xD80 */
+	mask = ~(PROT_BIT(mmPCIE_WRAP_HBW_DRAIN_TIMEOUT) &
+		PROT_BIT(mmPCIE_WRAP_HBW_DRAIN_CFG) &
+		PROT_BIT(mmPCIE_WRAP_LBW_DRAIN_TIMEOUT) &
+		PROT_BIT(mmPCIE_WRAP_LBW_DRAIN_CFG));
+	regmap_writel(mask, PRIV_PROT_REG(prot_base));
+
+	prot_base += SNGL_REG_PROT_SIZE;	/* offsets: 0xD80 - 0xF80 */
+	set_prot_bits_range_all(prot_base, PROT_SIZE_MAX - (prot_base - base));
+}
