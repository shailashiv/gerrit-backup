+#define SP_SRAM_INIT_TIMEOUT_US		(5 * USEC_PER_SEC)
+
+/*
+ * Initialize scratch SRAM to avoid ECC exceptions.
+ * Normally bootrom takes care of that, but in pldm when running without
+ * bootrom, let preboot initialize the spsram memory
+ */
+void preboot_sp_sram_init(void)
+{
+	unsigned int rst_src = hl_rst_src_get();
+	u32 init_sts;
+	u64 start;
+
+	if (rst_src != HL_COLD_RST)
+		return;
+
+	regmap_writel(D0_PSOC_GLOBAL_CONF_SCRATCHPAD_INIT_CTRL_START_M,
+		      mmD0_PSOC_GLOBAL_CONF_SCRATCHPAD_INIT_CTRL);
+	start = timestamp_get_us();
+
+	while (1) {
+		init_sts = regmap_readl(mmD0_PSOC_GLOBAL_CONF_SCRATCHPAD_INIT_CTRL);
+		if (init_sts & D0_PSOC_GLOBAL_CONF_SCRATCHPAD_INIT_CTRL_DONE_M)
+			break;
+
+		/*
+		 * If init does not happen in 5 seconds
+		 * timeout and return
+		 */
+		if (timeout_expired(start, SP_SRAM_INIT_TIMEOUT_US)) {
+			preboot_puts("Timed-out waiting for sp sram init\n");
+			break;
+		}
+	}
