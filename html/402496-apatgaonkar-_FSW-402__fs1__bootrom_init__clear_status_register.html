+void boot_rom_clear_status(void)
+{
+	/* As per Architecture recommendation, clear status reg */
+	regmap_writel(0, IO0_HALF0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_STATUS));
+
+	regmap_writel(0, IO0_HALF0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_SW_STATUS_0));
+	regmap_writel(0, IO0_HALF0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_SW_STATUS_1));
+	regmap_writel(0, IO0_HALF0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_SW_STATUS_2));
+	regmap_writel(0, IO0_HALF0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_SW_STATUS_3));
+	regmap_writel(0, IO0_HALF0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_SW_STATUS_4));
+	regmap_writel(0, IO0_HALF0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_SW_STATUS_5));
+	regmap_writel(0, IO0_HALF0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_SW_STATUS_6));
+	regmap_writel(0, IO0_HALF0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_SW_STATUS_7));
+
+	/* Die indication register is on io die slave and compute */
+	if (boot_rom_global.die_type != IO_DIE_MASTER)
+		regmap_writel(0, IO0_H0_PSOC_GLOBAL_CONF(mmGLOBAL_CONF_SCRATCHPAD_2));
