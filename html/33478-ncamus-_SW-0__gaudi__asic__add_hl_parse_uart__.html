+#define GAUDI_MEZZ_ID_MASK	0x7 /* We use only 3 bits out of 5 */
+#define GAUDI_MEZZ_ID_SHIFT	22
+#define ESC_KEY			0x1b
+void hl_parse_uart(int val)
+{
+	u8 mezz_id;
+	u32 strap;
+
+	strap = regmap_readl(mmPSOC_GLOBAL_CONF_BOOT_STRAP_PINS);
+	mezz_id = (strap >> GAUDI_MEZZ_ID_SHIFT) & GAUDI_MEZZ_ID_MASK;
+
+	switch (val) {
+	case ESC_KEY:
+		preboot_puts_raw("OAM[");
+		preboot_puth_raw(mezz_id, false);
+		preboot_puts("] Gaudi!\n");
+		break;
+	default:
+		break;
+	}
