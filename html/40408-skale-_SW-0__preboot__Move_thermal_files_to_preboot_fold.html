+static bool hl_tmp_is_sdif_free(u32 addr)

+/*
+ * convert 12-bit HL_TMP register value to milliCelsius
+ *
+ * Mode 1: output conversion (calibrated)
+ * Temp = A + B * Eqbs
+ *
+ * Mode 2: output conversion (uncalibrated)
+ * Temp = G + H * Eqbs
+ *
+ * Eqbs = Nbs / cal5 - 0.5
+ *
+ */
+
+long hl_tmp_temp_reg_to_mC(int val, int ts)
+{
+	long tmp = (val & HL_TS_00_SDIF_DATA_SMPL_DATA_MASK);
+	u32 coef_a_g, coef_b_h, coef_div;
+
+	coef_a_g = hl_ts_data[ts].hl_tmp_data.coef_a_g;
+	coef_b_h = hl_ts_data[ts].hl_tmp_data.coef_b_h;
+	coef_div = hl_ts_data[ts].hl_tmp_data.coef_div;
+
+	tmp = coef_a_g + ((coef_b_h * (((tmp * 1000) / coef_div) -
+					500)) / 1000);
+
+	/*
+	 * result of equation has to be multiplied
+	 * by 1000, since coefficients are multiplied by 100 an extra
+	 * multiplication of 10 is required.
+	 */
+	tmp *= 10;
+
+	return tmp;
+}
+
+/* convert milliCelsius to 12-bit HL_TMP register value */
+u32 hl_tmp_mC_to_temp_reg(long val, int ts)
+{
+	u32 reg_val;
+	u32 coef_a_g, coef_b_h, coef_div;
+
+	coef_a_g = hl_ts_data[ts].hl_tmp_data.coef_a_g;
+	coef_b_h = hl_ts_data[ts].hl_tmp_data.coef_b_h;
+	coef_div = hl_ts_data[ts].hl_tmp_data.coef_div;
+
+	reg_val = ((coef_div * ((val / 10) -
+					coef_a_g + (coef_b_h / 2))) / coef_b_h);
+
+	return reg_val & HL_TS_00_SDIF_DATA_SMPL_DATA_MASK;
+}
