+static void i2a_bridge_work_handler(struct k_work *work)
+{
+	struct mgmt_compute_config *config = mgmt_get_compute_cfg();
+	struct bridge_simple_msg *msg;
+	uint32_t size = config->bridge_wq_data.size;
+
+	if ((size < BRIDGE_SIZE_MIN) || (size > BRIDGE_SIZE_MAX)) {
+		LOG_ERR("Invalid size received from driver, clear interrupt");
+		ipm_clear(config->bridge_wq_data.dev, 0);
+		return;
+	}
+
+	/* decode header */
+	msg = (struct bridge_simple_msg *) config->bridge_wq_data.ptr;
+	LOG_INF("Received OPCODE: 0x%x from PORT_ID: 0x%x size: %d from I2A bridge",
+			msg->opcode, msg->source_port_id, config->bridge_wq_data.size);
+
+	/*
+	 * TODO: add processing of opcodes
+	 * some opcodes will need CR write as action to request and some
+	 * opcodes will need new response via I2C bridge in which case
+	 * create the header + data required and perform ipm_send.
+	 * CR details yet to be updated by ARCH team.
+	 */
+	switch (msg->opcode) {
+	case BRIDGE_OPCODE_CST_WAKE:
+		i2a_bridge_process_cst_wake(config);
+		break;
+	case BRIDGE_OPCODE_IP_READY:
+		i2a_bridge_process_ip_ready(config);
+		break;
+	case BRIDGE_OPCODE_FUSE_PULL:
+		i2a_bridge_process_fuse_pull(config);
+		break;
+	default:
+		LOG_ERR("OPCODE: 0x%x not supported", msg->opcode);
+		break;
+	}
+
+	/* clear interrupt */
+	ipm_clear(config->bridge_wq_data.dev, 0);
+}

+
+	config->bridge_wq_data.size = size;
+	config->bridge_wq_data.ptr = (void *) data;
+	config->bridge_wq_data.dev = dev;
+	k_work_submit(&config->bridge_work);
