+/* Read RMA header table */
+static void preboot_process_rma_header(void)
+{
+	u32 size = MAX_RMA_HEADER_DATA;
+	int ret;
+	/* Buffer to store RMA info from flash */
+	u8 rma_buf[MAX_RMA_HEADER_DATA];
+	struct rma_entry *rma_header;
+
+	ret = rma_info_get(RMA_HEADER_TABLE, rma_buf, &size);
+	if (ret)
+		return;
+
+	rma_header = (struct rma_entry *) &rma_buf[RMA_TABLE_HEADER_SIZE];
+	/* Update ERR_STS if any error */
+	if (rma_header->rma_failure.sram_ecc ||
+		rma_header->rma_failure.hbm_ecc ||
+		rma_header->rma_failure.hbm_parity ||
+		rma_header->rma_failure.row_replacement ||
+		rma_header->rma_failure.hbm_init) {
+		err_sts_reg_val |= HL_ERROR_STATUS_FATAL_ERROR;
+		preboot_puts_hex("Fatal error:", rma_header->rma_failure.data, 0);
+	}
+}
