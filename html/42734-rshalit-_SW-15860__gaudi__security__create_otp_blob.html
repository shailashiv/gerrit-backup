+#!/usr/bin/env bash
+
+OTP_MAGIC_NUMBER=0x444F4C48
+OTP_HEADER_VERSION=0x1
+OTP_SHA_CRYPTO_TYPE=0x1
+OTP_KEYS_COUNT=0x5
+OTP_RESERVED=0x0
+
+BOARD_BASE=${BR2_EXTERNAL}/board/habanalabs/${HL_DEV}
+# MKIMAGE_FILE=${BR2_EXTERNAL}/output/${HL_DEV}/host/bin/mkimage
+KEY_DIR=${BR2_EXTERNAL}/board/habanalabs/common/key/u-boot
+KEY_FILE1_NAME=hl_rsa3072.key
+KEY_FILE2_NAME=hl_rsa3072.key
+KEY_FILE3_NAME=hl_rsa3072.key
+KEY_FILE4_NAME=hl_rsa3072.key
+KEY_FILE5_NAME=hl_rsa3072.key
+OTP_IMAGE_NAME=otp.bin
+
+OTP_IMAGE=${KEY_DIR}/${OTP_IMAGE_NAME}
+
+#TBD check if rev_bytes_stdin required !
+echo "======================"
+echo $OTP_IMAGE
+#write u32 magic number
+rm $OTP_IMAGE
+printf "%08x" $OTP_MAGIC_NUMBER 	| xxd -r -p - $OTP_IMAGE
+#write u8 header version
+printf "%02x" $OTP_HEADER_VERSION	| xxd -r -p -s 4 - $OTP_IMAGE
+printf "%02x" $OTP_SHA_CRYPTO_TYPE 	| xxd -r -p -s 5 - $OTP_IMAGE
+printf "%02x" $OTP_KEYS_COUNT 		| xxd -r -p -s 6 - $OTP_IMAGE
+printf "%02x" $OTP_RESERVED 		| xxd -r -p -s 7 - $OTP_IMAGE
+
+#write keys #1 to #5 into file
+
+sha256sum -b ${KEY_DIR}/${KEY_FILE1_NAME} | cut -d' ' -f1 | xxd -r -p -s 8 - $OTP_IMAGE	
+sha256sum -b ${KEY_DIR}/${KEY_FILE2_NAME} | cut -d' ' -f1 | xxd -r -p -s 40 - $OTP_IMAGE	
+sha256sum -b ${KEY_DIR}/${KEY_FILE3_NAME} | cut -d' ' -f1 | xxd -r -p -s 72 - $OTP_IMAGE		
+sha256sum -b ${KEY_DIR}/${KEY_FILE4_NAME} | cut -d' ' -f1 | xxd -r -p -s 104 - $OTP_IMAGE		
+sha256sum -b ${KEY_DIR}/${KEY_FILE5_NAME} | cut -d' ' -f1 | xxd -r -p -s 136 - $OTP_IMAGE
