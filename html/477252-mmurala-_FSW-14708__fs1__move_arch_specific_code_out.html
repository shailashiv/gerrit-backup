+		/* read value from bank1 and OR with value from bank0 */
+		bank_offset = ((EFUSE_WORDS_PER_BANK + IO_EFUSE_SYSCFG_OFFSET) * sizeof(uint32_t));
+		syscfg1_data |= spi_flash_readl(spi_offset + bank_offset);
