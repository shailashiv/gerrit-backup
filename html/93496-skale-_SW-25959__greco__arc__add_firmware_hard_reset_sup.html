Commit Message

[SW-25959] greco: arc: add firmware hard reset support

Add FW hard reset sequence flow to fix the rare bug when a transaction
is stuck in the PCIE_IF and prevents the chip from normal execution
after the reset.

As a part of the new reset flow the PCIe is closed for accesses in both,
the memory space access and the ELBI access. Therefore the new reset
sequence depends on the fact that after the reset Ppboot knows to open
the PCIe for requests.

Note that hard reset by firmware is not enabled, it will be enabled
once LKD adds hard reset by firmware support for greco.

+/*
+ * This function makes sure no access is done from host to the SoC's PCIe IF.
+ * The inbound memory requests are disabled. The ELBI access gets "retry"
+ * response. All channels provide responses to the host and PCIe
+ * link will not break.
+ */
+void hl_pcie_disable_access(void)
+{
+	u32 reg;
+
+	/* Disable the ELBI access */
+	regmap_writel(1, mmPCIE_CORE_REQ_RETRY_EN);
+
+	/* Disable the PCIe inbound memory requests */
+	reg = regmap_readl(mmPCIE_DBI_STATUS_COMMAND_REG);
+	reg &= ~BIT(PCIE_DBI_STATUS_COMMAND_REG_PCI_TYPE0_MEM_SPACE_EN_SHIFT);
+	regmap_writel(reg, mmPCIE_DBI_STATUS_COMMAND_REG);
+}

+{
+	u32 reg;
+

+

+	reg |= BIT(PCIE_DBI_STATUS_COMMAND_REG_PCI_TYPE0_MEM_SPACE_EN_SHIFT);

+#define SW_ALL_RST_L_MSK		0xC00FF8F2
+#define SW_ALL_RST_M_MSK		0xFFFFFFFF
+#define SW_ALL_RST_H_MSK		0x7FFFFF
+
+static void greco_hw_fini(void)
+{
+	/* Restart BTL/BLR upon hard-reset */
+	regmap_writel(0x770, mmPSOC_GLOBAL_CONF_BOOT_SEQ_RE_START);
+
+	udelay(1000);
+
+	regmap_writel(SW_ALL_RST_L_MSK, mmPSOC_GLOBAL_CONF_SW_ALL_RST_CFG_L);
+	regmap_writel(SW_ALL_RST_M_MSK, mmPSOC_GLOBAL_CONF_SW_ALL_RST_CFG_M);
+	regmap_writel(SW_ALL_RST_H_MSK, mmPSOC_GLOBAL_CONF_SW_ALL_RST_CFG_H);
+
+	regmap_writel(1, mmPSOC_GLOBAL_CONF_SW_ALL_RST);
+}
+
+void hl_dev_reset(void)
+{
+	hl_pcie_disable_access();
+
+	greco_hw_fini();
+	wfe();
+}
