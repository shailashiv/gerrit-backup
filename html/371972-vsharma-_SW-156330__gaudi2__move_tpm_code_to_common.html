+	if (ret < 0)
+		return ret;
+
+	start = timestamp_get_us();
+	stop = tpm_tis_calc_ordinal_duration(ordinal);
+	do {
+		ret = tpm_spi_recv(tpm_xfer_buf, sizeof(tpm_xfer_buf));
+		if (ret >= 0) {
+			if (ret > *recv_size)
+				return -1;
+			memcpy(recvbuf, tpm_xfer_buf, ret);
+			*recv_size = ret;
+			ret = 0;
+			break;
+		} else if (ret != -1) {
+			return ret;
+		}
+
+		mdelay(TPM_TIMEOUT_MS);
+		if (timeout_expired(start, MS_TO_US(stop))) {
+			preboot_puts("TPM2: timeout expired\n");
+			ret = -1;
+			break;
+		}
+	} while (ret);
+
+	if (ret)
+		tpm_spi_cleanup();
+
+	return 0;
+}

+static int tpm2_unpack_byte_string(const u8 *str, size_t size,
+					const char *format, ...)
+{
+	va_list args;
+	size_t offset = 0, length = 0;
+	u8 *ptr8 = NULL;
+	u16 *ptr16 = NULL;
+	u32 *ptr32 = NULL;
+
+	va_start(args, format);
+	for (; *format; format++) {
+		switch (*format) {
+		case 'b':
+			offset = va_arg(args, size_t);
+			ptr8 = va_arg(args, u8 *);
+			length = 1;
+			break;
+		case 'w':
+			offset = va_arg(args, size_t);
+			ptr16 = va_arg(args, u16 *);
+			length = 2;
+			break;
+		case 'd':
+			offset = va_arg(args, size_t);
+			ptr32 = va_arg(args, u32 *);
+			length = 4;
+			break;
+		case 's':
+			offset = va_arg(args, size_t);
+			ptr8 = va_arg(args, u8 *);
+			length = va_arg(args, u32);
+			break;
+		default:
+			va_end(args);
+			preboot_puts("TPM2: Couldn't recognize format string\n");
+			return -1;
+		}
+
+		if (offset + length > size) {
+			va_end(args);
+			preboot_puts("TPM2: Failed to read\n");
+			return -1;
+		}
+
+		switch (*format) {
+		case 'b':
+			*ptr8 = str[offset];
+			break;
+		case 'w':
+			*ptr16 = get_unaligned_be16(str + offset);
+			break;
+		case 'd':
+			*ptr32 = get_unaligned_be32(str + offset);
+			break;
+		case 's':
+			memcpy(ptr8, str + offset, length);
+			break;
+		}
+	}
+	va_end(args);
+
+	return 0;
+}

+static int tpm2_pack_byte_string(u8 *str, size_t size, const char *format, ...)

+	u8 *data = NULL;
+	u32 value = 0;
+
+	va_start(args, format);
+	for (; *format; format++) {
+		switch (*format) {
+		case 'b':
+			offset = va_arg(args, size_t);
+			value = va_arg(args, int);
+			length = 1;
+			break;
+		case 'w':
+			offset = va_arg(args, size_t);
+			value = va_arg(args, int);
+			length = 2;
+			break;
+		case 'd':
+			offset = va_arg(args, size_t);
+			value = va_arg(args, u32);
+			length = 4;
+			break;
+		case 's':
+			offset = va_arg(args, size_t);
+			data = va_arg(args, u8 *);
+			length = va_arg(args, u32);
+			break;
+		default:
+			preboot_puts("TPM2: Couldn't recognize format string\n");
+			va_end(args);
+			return -1;
+		}
+
+		if (offset + length > size) {
+			va_end(args);
+			return -1;
+		}
+
+		switch (*format) {
+		case 'b':
+			str[offset] = value;
+			break;
+		case 'w':
+			put_unaligned_be16(value, str + offset);
+			break;
+		case 'd':
