+static struct hl_fw_boot_ptrs_ctx *p_img_ptr_ctx;
+static struct hl_fw_boot_ptrs *p_boot_ptr = &boot_rom_global.spi_manifest_hdr;
+static uint32_t erom_img_descs_offset = 0;

+static bool boot_desc_img_desc_is_valid(struct hl_fw_img_desc *img_desc)
+{
+	if ((img_desc->magic.val & DESC_MAGIC_MASK) != HL_FW_IMG_DESC_MAGIC)
+		return false;
+
+	if ((img_desc->desc_ver == INV_VER1) ||
+	    (img_desc->desc_ver == INV_VER2))
+		return false;
+
+	if (img_desc->idx >= INVALID_DESC_ID)
+		return false;
+
+	return true;
+}

+static void boot_desc_update_ptrs(enum hl_fw_boot_ptr_imgs img)
+{
+	p_img_ptr_ctx = &boot_rom_manifest.boot_ptr_ctx[img];
+	if (img == HL_FW_BOOT_PTR_RECOVERY) {
+		p_boot_ptr = (struct hl_fw_boot_ptrs *)(BOOT_PTR_REC_DMA_DEST_ARC);
+	} else {
+		p_boot_ptr = &boot_rom_global.spi_manifest_hdr;
+	}
+}

+int boot_desc_init(enum hl_fw_boot_ptr_imgs img)
+{
+	uint32_t i, img_ptrs_size, misc_descs = 0;
+	uint32_t *p_offset = NULL;
+	enum boot_rom_progress dma_sts_base = PROGRESS_INIT_SPI_BOOT_DESC_EXCEEDS_MAX_SIZE;
+	int err;
+	static struct hl_fw_img_desc *p_img_desc;
+
+	if (img >= HL_FW_BOOT_PTR_MAX_IMG)
+		return 0;
+
+	boot_desc_update_ptrs(img);
+
+	/* Initialize boot descriptor once */
+	if (p_img_ptr_ctx->boot_desc_flag)
+		return p_boot_ptr->count;
+
+	if (img == HL_FW_BOOT_PTR_RECOVERY) {
+		dma_sts_base = PROGRESS_INIT_RECOVERY_IMG_BOOT_DESC_EXCEEDS_MAX_SIZE;
+	} else {
+		if (boot_rom_global.validation_pass)
+			goto skip_boot_desc_process;
+	}
+
+	img_ptrs_size = spi_flash_readl(p_img_ptr_ctx->img_ptr_spi_offset);
+	if (img_ptrs_size > HL_IMG_PTRS_MAX_SIZE) {
+		boot_rom_status_update_bringup(dma_sts_base,
+				PROGRESS_BOOT_DESC_EXCEEDS_MAX_SIZE);
+		return 0;
+	}
+
+	boot_rom_status_update_bringup(dma_sts_base, PROGRESS_DMA_START);
+
+	/* Fetch boot desc using DMA into dest offset */
+	err = boot_rom_spi_dma(img_ptrs_size, p_img_ptr_ctx->img_ptr_spi_offset,
+			p_img_ptr_ctx->dest_addr_dma, false, false, 0);
+
+	if (boot_rom_process_dma_err(err, dma_sts_base))
+		return 0;
+
+	boot_rom_status_update_bringup(dma_sts_base, PROGRESS_DMA_FINISH);
+
+	if (img == HL_FW_BOOT_PTR_SPI)
+		memcpy(p_boot_ptr, (uint32_t *)(p_img_ptr_ctx->dest_addr_arc), sizeof(struct hl_fw_boot_ptrs));
+
+	if (p_boot_ptr->magic.val != HL_FW_BOOT_MAGIC) {
+		boot_rom_status_update_bringup(dma_sts_base, PROGRESS_BOOT_DESC_HDR_MAGIC_FAIL);
+		return 0;
+	}
+
+	p_img_ptr_ctx->desc_info.desc_size = p_boot_ptr->img_desc_sz;
+	p_img_ptr_ctx->desc_info.count = p_boot_ptr->count;
+
+	p_offset = (uint32_t *)(p_img_ptr_ctx->dest_addr_arc + p_boot_ptr->offset2array);
+
+	/* Validate all descriptors */
+	for (i = 0; i < p_boot_ptr->count; i++) {
+		p_img_desc = (struct hl_fw_img_desc *)(p_img_ptr_ctx->dest_addr_arc + p_offset[i]);
+
+		/* Range check for p_img_desc */
+		if (!(((uint32_t) p_img_desc >= p_img_ptr_ctx->dest_addr_arc) &&
+			((uint32_t) p_img_desc < p_img_ptr_ctx->dest_addr_arc + img_ptrs_size))) {
+			boot_rom_status_update_bringup(dma_sts_base,
+				PROGRESS_BOOT_DESC_INVALID_IMG_OFFSET);
+			return 0;
+		}
+
+		if (!boot_desc_img_desc_is_valid(p_img_desc)) {
+			boot_rom_status_update_bringup(dma_sts_base, PROGRESS_BOOT_DESC_VERIFICATION_FAIL);
+			return 0;
+		}
+
+		if (img != HL_FW_BOOT_PTR_SPI)
+		     continue;
+
+		if (p_img_desc->magic.type == HL_FW_IMG_PPBOOT) {
+			memcpy(&boot_rom_manifest.spi_img_desc[ROM_DESC_PPBOOT],
+				p_img_desc, sizeof(struct hl_fw_img_desc));
+		} else if (p_img_desc->magic.type == HL_FW_IMG_FAKE_EFUSE) {
+			memcpy(&boot_rom_manifest.spi_img_desc[ROM_DESC_FAKE_FUSE],
+				p_img_desc, sizeof(struct hl_fw_img_desc));
+		} else if (p_img_desc->magic.type == HL_FW_IMG_TPM_SPI) {
+			memcpy(&boot_rom_manifest.spi_img_desc[ROM_DESC_TPM],
+				p_img_desc, sizeof(struct hl_fw_img_desc));
+		} else {
+			/* Store upto 5 misc images(indifferent to ROM boot flow) */
+			if (misc_descs < NUM_OF_MISC_DESCS ) {
+				memcpy(&boot_rom_manifest.spi_img_desc[ROM_DESC_MISC_1 +
+				misc_descs], p_img_desc, sizeof(struct hl_fw_img_desc));
+				misc_descs++;
+			}
+		}
+	}
+
+skip_boot_desc_process:
+	if (img == HL_FW_BOOT_PTR_RECOVERY) {
+		/* Store the erom image descriptor offset to re-use in get api */
+		erom_img_descs_offset = p_offset[0];
+	}
+
+	p_img_ptr_ctx->boot_desc_flag = 1;
+
+	return p_boot_ptr->count;
+}
+
+static uint32_t boot_desc_validate_ptrs()
+{
+	/* Image descriptor pointer validity check */
+	if (!p_img_ptr_ctx)
+		return 1;
+
+	if (p_img_ptr_ctx->boot_desc_flag != 1)
+		return 1;
+
+	return 0;
+}
+
+static uint32_t boot_desc_copy_desc(struct boot_desc *desc,
+				    struct hl_fw_img_desc *local_img_desc,
+				    enum fw_post_reset_cmd upd_status)
+{
+	uint32_t idx;
+
+	idx = local_img_desc->idx;
+	if (upd_status == FLOPS_UPD_PENDING)
+		idx = (idx + 1) % 2;
+
+	desc->spi_offset = local_img_desc->offset[idx];
+	desc->img_size = local_img_desc->size[idx];
+	desc->loadaddr = ((uint64_t) local_img_desc->load_high[idx] << 32) |
+		local_img_desc->load_low[idx];
+	desc->desc_ver = local_img_desc->desc_ver;
+	desc->start_addr = ((uint64_t) local_img_desc->start_high[idx] << 32) |
+		local_img_desc->start_low[idx];
+
+	memcpy(desc->chksum, local_img_desc->chksum2[idx], ARRAY_SIZE(desc->chksum));
+
+	return idx;
+}
+
+static struct hl_fw_img_desc * boot_desc_img_desc_addr_calc(enum hl_fw_boot_ptr_imgs img,
+		uint32_t index)
+{
+	if (img == HL_FW_BOOT_PTR_SPI) {
+		if (index < ROM_DESC_NUM_MAX)
+			return &boot_rom_manifest.spi_img_desc[index];
+	} else {
+		return (struct hl_fw_img_desc *)(p_img_ptr_ctx->dest_addr_arc
+					+ erom_img_descs_offset
+					+ p_img_ptr_ctx->desc_info.desc_size * index);
+	}
+
+	return NULL;
+}
+
+uint32_t boot_desc_get_img_desc(enum hl_fw_boot_ptr_imgs boot_ptr_type, int boot_desc_type,
+				struct boot_desc *desc, enum fw_post_reset_cmd upd_status)
+{
+	uint32_t i;
+	struct hl_fw_img_desc *p_loc_img_desc = NULL;
+
+	if (boot_ptr_type >= HL_FW_BOOT_PTR_MAX_IMG)
+		return BOOT_ROM_IMAGE_NOT_FOUND;
+
+	boot_desc_update_ptrs(boot_ptr_type);
+
+	if (boot_desc_validate_ptrs())
+		return BOOT_ROM_IMAGE_NOT_FOUND;
+
+	for (i = 0; i < p_img_ptr_ctx->desc_info.count ; i++) {
+		p_loc_img_desc = boot_desc_img_desc_addr_calc(boot_ptr_type, i);
+
+		if (!p_loc_img_desc)
+			continue;
+
+		if (p_loc_img_desc->magic.type != boot_desc_type)
+			continue;
+
+		return boot_desc_copy_desc(desc, p_loc_img_desc, upd_status);
+	}
+
+	return BOOT_ROM_IMAGE_NOT_FOUND;
+}
