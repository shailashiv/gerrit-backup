+/**
+ * strstr - function finds the first occurrence of string2 in string1
+ * @str: The string to be searched
+ * @substr: The string to search for
+ */
+char *strstr(const char *str, const char *substr)
+{
+	while (*str) {
+		const char *s1 = str;
+		const char *s2 = substr;
+
+		while (*s1 && *s2 && *s1 == *s2) {
+			s1++;
+			s2++;
+		}
+
+		if (*s2 == '\0')
+			return (char *)str;
+
+		str++;
+	}
+
+	return NULL;
+}
