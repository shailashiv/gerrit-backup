+/*
+ * Copyright (C) 2017 HabanaLabs Ltd.
+ *   Author: Noam Camus <noamikong@gmail.com>

+
+#ifndef _ARCBOOT_H_
+#define _ARCBOOT_H_
+
+#define BIT(nr)					(1 << (nr))
+#define ARRAY_SIZE(x)				(sizeof(x) / sizeof((x)[0]))
+
+#define FW_SIZE					65535
+#define SIG_SIZE				384
+#define KEY_SIZE				384
+#define BLOB_START				4
+
+#define SPI_FLASH_BLOCK_SIZE			1024
+
+#define SDRAM_BASE				0x30000000
+#define HL_FLASH_OFFSET				0xA000000
+#define HL_SBROM_OFFSET				0xB040000
+#define HL_FWSRAM_OFFSET			0xB050000
+
+
+#define HL_FLASH_BASE				(SDRAM_BASE + HL_FLASH_OFFSET)
+#define HL_SBROM_BASE				(SDRAM_BASE + HL_SBROM_OFFSET)
+#define HL_FWSRAM_BASE				(SDRAM_BASE + HL_FWSRAM_OFFSET)
+#define HL_CFG_BASE				0xFC000000
+
+//SPI FLASH CONST
+#define FLASH_IMAGE_1_OFFSET			0x00000
+#define FLASH_IMAGE_2_OFFSET			0x20000
+
+#define FLASH_IMAGE_PREBOOT_OFFSET		0x10000
+#define FLASH_IMAGE_PCIE_FW_OFFSET		0x18000
+
+
+typedef unsigned char u8;
+typedef unsigned int u32;
+typedef unsigned long long u64;
+
+static inline void base_writeb(u8 val, u32 offset)
+{
+	*((volatile u32 *)(offset)) = (u32)val;
+}
+
+static inline u32 base_readl(u32 offset)
+{
+	u32 reg = *((volatile u32 *)(HL_CFG_BASE + offset));
+
+	return reg;
+}
+
+static inline u8 regmap_readb(u32 offset)
+{
+	u32 reg = base_readl(HL_CFG_BASE + offset);
+
+	return (u8)(reg & 0xFF);
+}
+
+static inline void regmap_writeb(u8 val, u32 offset)
+{
+	base_writeb((u32)val, HL_CFG_BASE + offset);
+}
+
+static inline u32 regmap_readl(u32 offset)
+{
+	u32 reg = base_readl(HL_CFG_BASE + offset);
+
+	return reg;
+}
+
+static inline void regmap_writel(u32 val, u32 offset)
+{
+	base_writeb(val, HL_CFG_BASE + offset);
+}
+
+static inline u32 rom_readl(u32 offset)
+{
+	u32 reg = base_readl(HL_SBROM_BASE + offset);
+
+	return reg;
+}
+
+static inline u32 flash_readl(u32 offset)
+{
+	u32 reg = base_readl(HL_FLASH_BASE + offset);
+
+	return reg;
+}
+
+static inline u32 fw_sram_readl(u32 offset)
+{
+	u32 reg = base_readl(HL_FWSRAM_BASE + offset);
+
+	return reg;
+}
+
+static inline void fw_sram_writel(u32 val, u32 offset)
+{
+	base_writeb(val, HL_FWSRAM_BASE + offset);
+}
+
+#endif /* _ARCBOOT_H_ */
