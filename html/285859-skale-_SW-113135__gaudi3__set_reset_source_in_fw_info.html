Since ppboot reads the rst source from BTM FIFO, it needs
to propagate the information to other FW components. This patch
add changes to set reset source read from BTM FIFO as part of
fw info and store it in boot info.

+void ppboot_init_fw_info(void)
+{
+	struct mem_desc mem_region_desc;
+	struct hl_fw_info_compat *fw_info;
+	u32 copy_sz, fw_info_sz;
+
+	if (preboot_get_mem_region(MEM_GP, &mem_region_desc)) {
+		preboot_puts("Failed to get mem region MEM_GP\n");
+		return;
+	}
+
+	fw_info = (struct hl_fw_info_compat *)mem_region_desc.base;
+	fw_info_sz = sizeof(struct hl_fw_info_compat);
+	memset((void *)fw_info, 0x0, fw_info_sz);
+
+	if (rst_src.src == HL_COLD_RST)
+		goto fw_info_set;
+
+	if (boot_info_get_no_merge(HL_BOOT_INFO_FW_INFO, fw_info, mem_region_desc.size, &copy_sz))
+		preboot_puts("BOOT_INFO: Failed to get fw_info\n");
+	else
+		fw_info_sz = copy_sz;
+
+fw_info_set:
+	fw_info->last_reset_source = rst_src.src;
+	if (ppboot_boot_info_set(HL_BOOT_INFO_FW_INFO, fw_info, fw_info_sz))
+		preboot_puts("BOOT_INFO: Failed to set last reset cause in fw_info\n");
