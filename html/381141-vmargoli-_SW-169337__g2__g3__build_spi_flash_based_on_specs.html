+function hl_build_spi_flash_file {
+	local -n region_to_file=$1
+	local -n region_to_offset=$2
+	local -n region_to_size=$3
+	local -n out_file=$4
+	local file_length=0
+	local input_file
+	local file_length
+	local offset
+	local size
+	local kb
+	local end_of_region
+	local current_region_offset=0
+	local region_found
+	local red=$(tput setaf 1)
+	local normal=$(tput sgr0)
+
+	for region_name in "${!region_to_offset[@]}"; do
+		offset="${region_to_offset[${region_name}]}"
+		size="${region_to_size[${region_name}]}"
+		end_of_region=$(expr $offset + $size)
+		if [ $end_of_region -gt $file_length ]; then
+			file_length=$end_of_region
+		fi
+	done
+
+	printf "\nLength of flash image file %s: 0x%x bytes \n\n" "$out_file" "$file_length"
+	rm -f "$out_file"
+	touch "$out_file"
+
+	while [ $current_region_offset -lt $file_length ]; do
+		region_found=false
+		for region_name in "${!region_to_offset[@]}"; do
+			offset="${region_to_offset[${region_name}]}"
+			size="${region_to_size[${region_name}]}"
+			kb=$(expr "$size" \/ 1024)
+			if [ $offset -eq $current_region_offset ]; then
+				region_found=true
+				input_file="${region_to_file[${region_name}]}"
+				if [ "$input_file" = "FF" ]; then
+					printf "Writing FF to offset 0x%x  %sK \n" "$offset" "$kb"
+					dd status=none conv=sync bs=${kb}K count=1 if=/dev/zero | tr "\000" "\377" >> ${out_file}
+				elif [ "$input_file" = "00" ]; then
+					printf "Writing 00 to offset 0x%x  %sK \n" "$offset" "$kb"
+					dd status=none conv=sync bs=${kb}K count=1 if=/dev/zero >> ${out_file}
+				else
+					printf "Writing %s from %s to offset 0x%x  %sK \n" "$region_name" "$input_file" "$offset" "$kb"
+					dd status=none conv=sync bs=${kb}K if=${input_file} >> ${out_file}
+				fi
+				current_region_offset=$(expr $offset + $size)
+				break
+			fi
+		done
+		if [ "$region_found" = false ]; then
+			printf "$red SPI flash region with offset 0x%x not defined $normal \n" "$current_region_offset"
+			exit 1
+		fi
+	done
+}
