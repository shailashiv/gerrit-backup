+int boot_rom_verify_image(enum hl_fw_boot_ptr_imgs img_type)
+{
+	enum boot_rom_progress boot_desc_verif_base = PROGRESS_SPI_BOOT_DESC_VERIFY_START;
+	struct hl_fw_boot_ptrs boot_ptr;
+	uint32_t boot_ptr_addr = 0;
+	int ret;
+	uint32_t img_plus_key_len;
+
+	if (!security_efuse_image_validation_en())
+		return BOOT_ROM_SUCCESS;
+
+	if ((img_type == HL_FW_BOOT_PTR_SPI) && (boot_rom_global.validation_pass))
+		return BOOT_ROM_SUCCESS;
+
+	if (img_type == HL_FW_BOOT_PTR_RECOVERY)
+		boot_desc_verif_base = PROGRESS_RECOVERY_IMG_BOOT_DESC_VERIFY_START;
+
+	boot_ptr_addr = boot_desc_param_get(img_type, &boot_ptr);
+
+	if (!boot_ptr_addr)
+		return BOOT_ROM_FAIL;
+
+	boot_rom_status_update_bringup(boot_desc_verif_base, PROGRESS_VERIFY_START);
+
+	/*
+	 * TODO: verify it once new SPI DMA changes finalised and working.
+	 * Now we have PPBOOT is less than 64KB, so this changes expected to work.
+	*/
+	img_plus_key_len = boot_ptr.boot_ptrs_img_sz - boot_ptr.sig_sz - CRC32_SIZE;
+
+	ret = secure_boot_verify_image_signature((uint8_t *)boot_ptr_addr, img_plus_key_len,
+			(uint8_t *)(boot_ptr_addr + img_plus_key_len - boot_ptr.pub_key_sz),
+			boot_ptr.pub_key_sz, (uint8_t *)(boot_ptr_addr + img_plus_key_len), boot_ptr.sig_sz,
+			boot_ptr.sec_ver, boot_ptr.key_idx);
+
+	if (ret == SECUREBOOT_VERIFY_SUCCESS) {
+		boot_rom_status_update_bringup(boot_desc_verif_base, PROGRESS_VERIFY_SUCCESS);
+	} else {
+		if (ret == SECUREBOOT_VERIFY_VER_FAILED)
+			boot_rom_status_update_bringup(boot_desc_verif_base, PROGRESS_VERIFY_VERSION_FAIL);
+		else if (ret == SECUREBOOT_VERIFY_KEY_REVOKED)
+			boot_rom_status_update_bringup(boot_desc_verif_base, PROGRESS_VERIFY_KEY_REVOKED);
+		else if (ret == SECUREBOOT_VERIFY_KEY_FAILED)
+			boot_rom_status_update_bringup(boot_desc_verif_base, PROGRESS_VERIFY_KEY_HASH_FAIL);
+		else
+			boot_rom_status_update_bringup(boot_desc_verif_base, PROGRESS_VERIFY_AUTH_FAIL);
+
+		return BOOT_ROM_FAIL;
+	}
+
+	return BOOT_ROM_SUCCESS;
