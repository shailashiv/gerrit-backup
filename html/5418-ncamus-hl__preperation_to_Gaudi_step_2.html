+#!/usr/bin/env bash
+
+pushd ${TARGET_DIR}/etc/init.d/
+	if [ "${HL_TARGET}" == "pldm" ] || [ "${HL_TARGET}" == "asic" ]; then
+		if [ -e S40network ]; then
+			mv S40network .S40network
+		fi
+	elif [ "${HL_TARGET}" == "onix" ]; then
+		if [ -e .S40network ]; then
+			mv .S40network S40network
+		fi
+	else
+		echo "Unknown ${HL_DEV} target: ${HL_TARGET}"
+		exit 1
+	fi
+popd
+
+# Add armcpd to hl device
+if [ -e ${TARGET_DIR}/etc/inittab ]; then
+    grep -qE "^${HL_DEV}::" ${TARGET_DIR}/etc/inittab || \
+	sed -i "/GENERIC_SERIAL/a# Put armcpd on Dali device/\n\
+${HL_DEV}::respawn:/usr/bin/armcpd # ARM CP" ${TARGET_DIR}/etc/inittab
+fi

+#!/usr/bin/env bash
+
+BOARD_BASE=${BR2_EXTERNAL_HL_DALI_PATH}/board/habanalabs/${HL_DEV}
+SPL_BINFILE=${SPL_BINFILE:-u-boot-spl.bin}
+UBOOT_CMDFILE=${UBOOT_CMDFILE:-${BOARD_BASE}/uboot-eeprom.cmd}
+UBOOT_FITFILE=u-boot.itb
+UBOOT_ITSFILE=${UBOOT_ITSFILE:-${BOARD_BASE}/../common/hl_fit_spl.its}
+UBOOT_BINFILE=${UBOOT_BINFILE:-${UBOOT_FITFILE}}
+LINUX_FITFILE=fit.itb
+
+function swap_file {
+	INF=$1
+	OUTF=$2
+
+	# SWAP needs this to be 4B aligned in size
+	dd status=none conv=sync bs=4 if=${INF} > ${OUTF}.tmp
+	# Sadly --endian=big is not swapping for -tx1 so I used some gawk magic
+	od -Anone -v -tx1 -w4 ${OUTF}.tmp | gawk '{ print $4 $3 $2 $1 }' | xxd -r -p > ${OUTF}
+}
+
+pushd ${BINARIES_DIR}
+	cp ${BOARD_BASE}/${HL_DEV}_${HL_TARGET}.its .
+	cp ${BASE_DIR}/build/linux-custom/arch/arm64/boot/Image .
+	${HOST_DIR}/usr/bin/mkimage -D "-W no-unit_address_vs_reg" -f ${HL_DEV}_${HL_TARGET}.its ${LINUX_FITFILE}
+
+	cp ${UBOOT_ITSFILE} ${HL_DEV}_fit_spl.its
+	# In case ITSFILE points to locations under public directory /tools
+	ln -nfs /tools
+	# SPI IOMAP likes these to be 8B aligned in size
+	dd status=none conv=sync bs=8 if=${BASE_DIR}/build/uboot-custom/u-boot-nodtb.bin	 > u-boot-nodtb.bin
+	dd status=none conv=sync bs=8 if=${BASE_DIR}/build/uboot-custom/u-boot.dtb		 > u-boot.dtb
+	${HOST_DIR}/usr/bin/mkimage -f ${HL_DEV}_fit_spl.its -E ${UBOOT_FITFILE}
+	${HOST_DIR}/usr/bin/mkimage -C none -A arm64 -T script -d ${UBOOT_CMDFILE} boot.scr
+
+	dd status=none conv=sync bs=8 if=${SPL_BINFILE}						 > ${HL_DEV}-u-boot-spl.bin
+	# Swap SPL
+	swap_file ${HL_DEV}-u-boot-spl.bin ${HL_DEV}-u-boot-spl.bin.be
+	SPL_BINFILE=${HL_DEV}-u-boot-spl.bin.be
+
+	# KMD like this to be 4B aligned in size
+	dd status=none conv=sync bs=4 if=${LINUX_FITFILE}					 > ${HL_DEV}-fit.itb
+	# Used by TRACE32 (Lauterbach JTAG)
+	dd status=none conv=sync bs=4 if=u-boot.bin						 > ${HL_DEV}-u-boot.bin
+
+	### Construct bin file
+	# BTL likes these to be 16B aligned in size
+	dd status=none conv=sync bs=16 if=preboot.bin						 > ${HL_DEV}-preboot.bin
+	# prepare BTL image
+	BTL_CMD_ARGS="${BTL_CMD_ARGS} SPL_SIZE=$(printf "0x%X\n" `stat -c "%s" ${SPL_BINFILE}`)"
+	BTL_CMD_ARGS="${BTL_CMD_ARGS} UBOOT_SIZE=$(printf "0x%X\n" `stat -c "%s" ${UBOOT_BINFILE}`)"
+	cp -a ${BOARD_BASE}/btl .
+	make -C btl clean all ${BTL_CMD_ARGS}
+	cp btl/sectors-be.bin .
+
+	# Metadata
+	dd status=none conv=sync bs=64K count=1 if=/dev/zero				 > ${HL_DEV}-pkey-hash.bin
+	dd status=none conv=sync bs=64K count=1 if=/dev/zero				 > ${HL_DEV}-ver.bin
+	echo "11110000" > ${HL_DEV}-next-image.txt
+	dd status=none conv=sync bs=64K if=${HL_DEV}-next-image.txt			 > ${HL_DEV}-next-image.bin
+	dd status=none conv=sync bs=64K count=1 if=/dev/zero				 > ${HL_DEV}-images-pointers.bin
+
+	# BTL
+	dd status=none conv=sync bs=256K	if=sectors-be.bin			 > ${HL_DEV}-spi-flash.bin.nocrc
+	# Metadata
+	dd status=none conv=sync bs=64K		if=${HL_DEV}-pkey-hash.bin		>> ${HL_DEV}-spi-flash.bin.nocrc
+	dd status=none conv=sync bs=64K		if=${HL_DEV}-ver.bin			>> ${HL_DEV}-spi-flash.bin.nocrc
+	dd status=none conv=sync bs=64K		if=${HL_DEV}-next-image.bin		>> ${HL_DEV}-spi-flash.bin.nocrc
+	dd status=none conv=sync bs=64K		if=${HL_DEV}-images-pointers.bin	>> ${HL_DEV}-spi-flash.bin.nocrc
+	# U-Boot ENV
+	dd status=none conv=sync bs=64K		if=uboot-env.bin			>> ${HL_DEV}-spi-flash.bin.nocrc
+	# U-Boot ENV Redundand
+	dd status=none conv=sync bs=64K		if=uboot-env.bin			>> ${HL_DEV}-spi-flash.bin.nocrc
+	# SPL
+	dd status=none conv=sync bs=64K		if=${SPL_BINFILE}			>> ${HL_DEV}-spi-flash.bin.nocrc
+	# U-Boot
+	dd status=none conv=sync bs=1M		if=${UBOOT_BINFILE}			>> ${HL_DEV}-spi-flash.bin.nocrc
+
+	# Prepare checksum
+	CKSUM=`cksum ${HL_DEV}-spi-flash.bin.nocrc | cut -d' ' -f1`
+	CKSUM_BE=`printf "%08x" ${CKSUM}`
+	CKSUM_LE=${CKSUM_BE:6:2}${CKSUM_BE:4:2}${CKSUM_BE:2:2}${CKSUM_BE:0:2}
+	echo ${CKSUM_LE} | xxd -p -r							 > ${HL_DEV}-spi-flash.crc
+	dd status=none conv=sync bs=64K		if=${HL_DEV}-spi-flash.bin.nocrc	 > ${HL_DEV}-spi-flash.bin
+	dd status=none conv=sync bs=4		if=${HL_DEV}-spi-flash.crc		>> ${HL_DEV}-spi-flash.bin
+
+	# Convert bin file into Palladium hex format
+	xxd -p -c1 -g1				${HL_DEV}-spi-flash.bin	 		 > ${HL_DEV}-spi-flash.hex
+popd
