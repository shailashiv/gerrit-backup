Commit Message

[SW-20141] gaudi: add fw hard reset support

Add FW hard reset sequence flow to fix the rare bug when a transaction
is stuck in the PCIE_IF and prevents the chip from normal execution
after the reset.
As a part of the new reset flow the PCIe is closed for accesses in both,
the memory space access and the ELBI access. Therefore the new reset
sequence depends on the fact that after the reset Ppboot knows to open
the PCIe for requests.

+	int i, j;

+static void gaudi_nic_qpc_disable(unsigned int reg, int nic, int qpc)
+{
+	unsigned int off, mask = BIT(6);

+	off = reg + nic * 0x40000 + qpc * 0x1000;
+	reg = regmap_readl(off);
+	reg |= mask;

+static void gaudi_mme_stall(void)
+{
+	/* both ACC and SBAB stall bits shifts are the same */
+	u32 stall = BIT(MME_ACC_ACC_STALL_R_SHIFT);
+	u32 acc_offset = mmMME0_ACC_ACC_STALL;
+	u32 sbab_offset = mmMME0_SBAB_SB_STALL;
+	/* MME offsets are the same between ACCs and SBABs */
+	u32 mmes_off_delta = mmMME1_ACC_BASE - mmMME0_ACC_BASE;
+	int i;
+
+	for (i = 0; i < GAUDI_MMES_NUM; i++) {
+		/* WA for H3-1800 bug: do ACC and SBAB writes twice */
+		regmap_writel(stall, acc_offset + i * mmes_off_delta);
+		regmap_writel(stall, acc_offset + i * mmes_off_delta);
+		regmap_writel(stall, sbab_offset + i * mmes_off_delta);
+		regmap_writel(stall, sbab_offset + i * mmes_off_delta);
+	}
+}

+/*
+ * This function makes sure no access is done from host to the SoC units during
+ * the reset. All incoming downstream traffic is routed to RAZWI, so responses
+ * will be sent to the host and PCIe link will not break.
+ *
+ * Note: since we close all the access, the boot FW must enable the access back.
+ */
+static void gaudi_close_pcie_rr(void)
+{
+	/* RRs min/max HIT is not inclusive: min < hit < max */
+	u32 min = 0xF0000000 - 1, max = 0xFFFFFFFF;
+	u32 hmsk = 0x3FF80;
+
+	/* PCIe RR - CFG WP 0x7FF0000000 - 0x7FFFFFFFFF */
+	regmap_writel(min, mmPCIE_WRAP_RR_WR_PRV_RANGE_REG_MIN_0);
+	regmap_writel(0xFCC01F64, mmPCIE_WRAP_RR_WR_PRV_RANGE_REG_MAX_0);
+	/* Hole for VUART TX status 0x7FFCC01F64 - 0x7FFCC01F67 */
+	regmap_writel(0xFCC01F67, mmPCIE_WRAP_RR_WR_PRV_RANGE_REG_MIN_1);
+	regmap_writel(max, mmPCIE_WRAP_RR_WR_PRV_RANGE_REG_MAX_1);
+	regmap_writel(min, mmPCIE_WRAP_RR_RD_PRV_RANGE_REG_MIN_0);
+	regmap_writel(0xFCC01F60, mmPCIE_WRAP_RR_RD_PRV_RANGE_REG_MAX_0);
+	/* Hole for VUART TX 0x7FFCC01F60 - 0x7FFCC01F67 */
+	regmap_writel(0xFCC01F67, mmPCIE_WRAP_RR_RD_PRV_RANGE_REG_MIN_1);
+	regmap_writel(max, mmPCIE_WRAP_RR_RD_PRV_RANGE_REG_MAX_1);
+
