+	return hl_pll_get_freq(mmD0_G4_PCI_PLL_CTRL_CFG, HL_PLL_DIV3);

+void hl_uart_init(void)
+{
+	/* As per Gaudi3 Frequencies.xlsx, PCI PLL Div[3] drives uart */
+	u32 uart_clk_freq = hl_pll_get_freq(mmD0_G4_PCI_PLL_CTRL_CFG, HL_PLL_DIV3);
+
+	regmap_writel(1, mmD0_PSOC_GLOBAL_CONF_PAD_SEL_48);	/* UART0 RXD */
+	regmap_writel(1, mmD0_PSOC_GLOBAL_CONF_PAD_SEL_49);	/* UART0 TXD */
+	preboot_uart_init(uart_clk_freq / HL_UART_BAUDRATE);
+}
+
+u32 hl_spi_baudr_get(void)
+{
+	u32 factor, pcie_div3_freq;
+
+	pcie_div3_freq = hl_pll_get_freq(mmD0_G4_PCI_PLL_CTRL_CFG, HL_PLL_DIV3);
+	factor = pcie_div3_freq / HL_SPI_FLASH_MAX_FREQ;
+
+	/* baud rate doesn't support odd number and must not be 0 */
+	factor = (factor < 2) ? 2 : factor;
+
+	return (factor + 1) & 0xFFFE;
