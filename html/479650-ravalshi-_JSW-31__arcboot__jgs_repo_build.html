+uint32_t boot_rom_pll_get_freq(enum boot_rom_pll_div div)
+{
+	uint32_t div_sel, factor, offset;
+	uint32_t reg, fbdiv, refdiv, postdiv1, postdiv2;
+	uint64_t out_freq = SYS_REF_CLOCK;
+
+	offset = div * sizeof(uint32_t);
+	// 2'd0 - ref_clk, 2'd1 - pll_clk, 2'd2 - divided ref_clk, 2'd3 - divided pll_clk
+	div_sel = regmap_readl((IO0_H0_SOC_0_PLL_0_PSOC_PLL(mmPLL_CONTROL_DIV_SEL_0)) + offset);
+
+	if (div_sel == PLL_PLL_CLK || div_sel == PLL_DIV_PLL) {
+		// If 2'd1 - pll_clk [or] 2'd3 - divided pll_clk
+		reg = regmap_readl(IO0_H0_SOC_0_PLL_0_PSOC_PLL(mmPLL_CONTROL_CFG_DIV));
+		/* fbdiv : bits 0-11 */
+		fbdiv = EXTRACT_BITS(reg, 0, 12);
+		/* postdiv1 : bits 16-18 */
+		postdiv1 = EXTRACT_BITS(reg, 16, 3);
+		/* postdiv2 : bits 20-22 */
+		postdiv2 = EXTRACT_BITS(reg, 20, 3);
+		/* refdiv : bits 24-29 */
+		refdiv  = EXTRACT_BITS(reg, 24, 6);
+		// SPECS: Fout = Fin * FBDIV / (POSTDIV1+1) / (POSTDIV2+1) / REFDIV
+		out_freq = SYS_REF_CLOCK * fbdiv;
+		out_freq /= ((uint64_t)refdiv) * (postdiv1 + 1) * (postdiv2 + 1);
+	}
+
+	if (div_sel == PLL_DIV_REF || div_sel == PLL_DIV_PLL) {
+		// If 2'd2 - divided ref_clk [or] 2'd3 - divided pll_clk
+		factor = regmap_readl((IO0_H0_SOC_0_PLL_0_PSOC_PLL(mmPLL_CONTROL_DIV_FACTOR_0)) + offset);
+		out_freq /= factor + 1;
+	}
+
+	return (uint32_t)out_freq;
+}

+void boot_rom_eom_overrides()
+{
+	uint32_t key_revoke;
+
+	if (boot_rom_global.config_fuse & EFUSE_SYSCFG_EOM_BIT)
+		boot_rom_global.config_fuse |= EFUSE_SYSCFG_PRE_PRODUCTION_FLAGS;
+
+	key_revoke = boot_rom_global.config_fuse & EFUSE_SYSCFG_KEY_REVOCATION_BITS;
+
+	/*
+	 * Validate key revokation bits. Expected values from fuse are
+	 * 0, 1, 3, 7, 15. If any other values are read from fuse override
+	 * them accordingly.
+	 */
+	if (key_revoke == 2)
+		boot_rom_global.config_fuse |=  EFUSE_SYSCFG_REVOKE_KEY_0_1;
+	else if ((key_revoke < 7) && (key_revoke > 3))
+		boot_rom_global.config_fuse |=  EFUSE_SYSCFG_REVOKE_KEY_0_1_2;
+	else if ((key_revoke < 17) && (key_revoke > 7))
+		boot_rom_global.config_fuse |=  EFUSE_SYSCFG_REVOKE_KEY_0_1_2_3;
+
+	boot_rom_global.config_fuse &= ~EFUSE_SYSCFG_REVOKE_KEY_4;
+}

+void boot_rom_report_image_status(enum hl_fw_manifest_type man_type, enum security_img_sts error)
+{
+	uint32_t val;
+	uint32_t reg_addr;

+	if (man_type == HL_FW_SPI_MANIFEST_1)

+	else
+		reg_addr = IO0_H0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_PRI_IMG_STATUS);

+	val = regmap_readl(reg_addr);
+	switch(error) {
+		case FW_VER_LOWER:
+			val |= PSOC_SECURITY_PRI_IMG_STATUS_FW_VER_LOWER_M;
+			break;
+		case AUTHENTICATE_FAIL:
+			val |= PSOC_SECURITY_SEC_IMG_STATUS_AUTHENTICATE_FAIL_M;
+			break;
+		case CRC_FAIL:
+			val |= PSOC_SECURITY_SEC_IMG_STATUS_CRC_FAIL_M;
+			break;
+		default:
+			return;
+	}
+
+	regmap_writel(val, reg_addr);
+}

+void boot_rom_report_security_status(enum security_status status_bit, uint8_t index)
+{
+	uint32_t reg = regmap_readl(IO0_H0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_STATUS));
+
+	switch(status_bit)
+	{
+		case STATUS_IMG_VALIDATED:
+			reg |= PSOC_SECURITY_STATUS_IMG_VALIDATED_M;
+			break;
+		case STATUS_FW_UPGRADE:
+			reg |= PSOC_SECURITY_STATUS_FW_UPGRADE_M;
+			break;
+		case STATUS_KEY_REVOKED:
+			reg |= PSOC_SECURITY_STATUS_KEY_REVOKED_M;
+			reg |= (index << PSOC_SECURITY_STATUS_KEY_REVOKED_IND_S);
+			break;
+		case STATUS_MIN_FW_MODIFIED:
+			reg |= PSOC_SECURITY_STATUS_MIN_FW_MODIFIED_M;
+			break;
+		case STATUS_FW_VER_INC:
+			reg |= PSOC_SECURITY_STATUS_FW_VER_INC_M;
+			break;
+		case STATUS_SPI_IMG_MODIFIES:
+			if (index == 0)
+				reg &= ~(PSOC_SECURITY_STATUS_SPI_IMG_MODIFIES_M);
+			else if (index == 1)
+				reg |= PSOC_SECURITY_STATUS_SPI_IMG_MODIFIES_M;
+			break;
+		case STATUS_FW_AUTH_FAIL:
+			reg |= PSOC_SECURITY_STATUS_FW_AUTHENTICAION_FAIL_M;
+			break;
+		default:
+			reg = 0;
+	}
+
+	regmap_writel(reg, IO0_H0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_STATUS));
+}

+void security_rr_lbw_config_en(struct security_initiator_ctx initr_data,
+	const struct security_rr_ctx *rr_ip_ctx)
+{
+	uint32_t range_mask, rr_dist, initr_sai_val, initr_rd_sai, initr_wr_sai;
+	uint32_t wr_sai_val, rd_sai_val, rr_en_val;
+
+	if ((initr_data.base_addr == 0) || (initr_data.sai_base == 0)) {
+		return;
+	}
+
+	rr_dist = rr_ip_ctx->rr_idx * sizeof(uint32_t);
+
+	initr_sai_val = regmap_readl(initr_data.sai_base + mmAXUSER_LBW_LB_SAI);
+	initr_rd_sai = (initr_sai_val & AXUSER_LBW_LB_SAI_RD_M);
+	initr_wr_sai = ((initr_sai_val & AXUSER_LBW_LB_SAI_WR_M) >> AXUSER_LBW_LB_SAI_WR_S);
+
+	wr_sai_val = regmap_readl(initr_data.base_addr + mmRANGE_REG_LBW_WR_RANGE_SEC_0 + rr_dist);
+	rd_sai_val = regmap_readl(initr_data.base_addr + mmRANGE_REG_LBW_RD_RANGE_SEC_0 + rr_dist);
+
+	/*
+	 * TODO
+	 * 1. When efuse BLOCK_RR is 1/0, all range_en=0x11, rd/wr_range_sec=0x7fffffff?
+	 * 2. Before we use RR index which is already used for open_all, we need to clear?
+	 */
+	rr_en_val = regmap_readl(initr_data.base_addr + mmRANGE_REG_LBW_RANGE_EN_0 + rr_dist);
+
+	if ((rr_ip_ctx->acc_type == SEC_ACCESS_WR_ONLY) || (rr_ip_ctx->acc_type == SEC_ACCESS_RD_WR)) {
+		rr_en_val |= RANGE_REG_LBW_RANGE_EN_0_WR_EN_M;
+		wr_sai_val |= (1 << initr_wr_sai);
+	}
+	if ((rr_ip_ctx->acc_type == SEC_ACCESS_RD_ONLY) || (rr_ip_ctx->acc_type == SEC_ACCESS_RD_WR)) {
+		rr_en_val |= RANGE_REG_LBW_RANGE_EN_0_RD_EN_M;
+		rd_sai_val |= (1 << initr_rd_sai);
+	}
+
+	range_mask = regmap_readl(initr_data.base_addr + mmRANGE_REG_LBW_RANGE_MASK_0 + rr_dist);
+	regmap_writel(rr_ip_ctx->data_min & range_mask,
+		initr_data.base_addr + mmRANGE_REG_LBW_RANGE_MIN_ADDR_0 + rr_dist);
+	regmap_writel(rr_ip_ctx->data_max & range_mask,
+		initr_data.base_addr + mmRANGE_REG_LBW_RANGE_MAX_ADDR_0 + rr_dist);
+
+	regmap_writel(wr_sai_val & RANGE_REG_LBW_WR_RANGE_SEC_0_VAL_M,
+		initr_data.base_addr + mmRANGE_REG_LBW_WR_RANGE_SEC_0 + rr_dist);
+	regmap_writel(rd_sai_val & RANGE_REG_LBW_RD_RANGE_SEC_0_VAL_M,
+		initr_data.base_addr + mmRANGE_REG_LBW_RD_RANGE_SEC_0 + rr_dist);
+
+	regmap_writel(rr_en_val, initr_data.base_addr + mmRANGE_REG_LBW_RANGE_EN_0 + rr_dist);
+}

+static uint32_t prepare_mask(uint8_t min, uint8_t max)
+{
+	uint32_t mask = PB_CONF_ALL_PROTECTED;
+
+	mask = mask >> min << min;
+	mask = mask << (31 - max) >> (31 - max);
+
+	return mask;
+}

+static void prepare_pb_mask_array(uint32_t min_bit, uint32_t max_bit, uint32_t mask_array[], uint8_t length)
+{
+	uint32_t array_index, cur_min, msb, i;
+
+	cur_min = min_bit;
+	for (i = 0; i < length; i++) {
+		array_index = cur_min / 32;
+		msb = (32 * (array_index + 1)) - 1;
+
+		if (max_bit <= msb) {
+			mask_array[array_index] = prepare_mask(cur_min, max_bit);
+			return;
+		} else {
+			mask_array[array_index] = prepare_mask(cur_min, msb);
+		}
+		cur_min = msb + 1;
+	}
+}

+static uint32_t set_bit_index(uint32_t filter)
+{
+	uint32_t i, count = 0;
+
+	for (i = 0; i < (sizeof(uint32_t) * CHAR_BIT); i++) {
+		if ((filter >> i) & 1)
+			count += 1;
+	}
+
+	return count;

+	regmap_writel(grp_cfg.sai_mask, spl_base + pb_sai_reg[grp]);

+{
+	regmap_writel(data, spl_base + pb_reg[grp] + (set_idx * sizeof(uint32_t)));

+static void boot_rom_program_prot_bits(struct pb_page_ctx *pb_cfg_data, uint32_t pb_cfg_size)
+{
+	uint32_t mask_array[PB_MAX_REG_COPY_PER_GROUP];
+	struct pb_grp_ctx grp_cfg;
+	uint8_t i, group, set_idx;
+	uint32_t page, spl_base, filter_bit_ind;
+	uint32_t pb_mask_len, pb_mask_filter;
+
+	for (page = 0; page < pb_cfg_size; page++) {
+		spl_base = pb_cfg_data[page].spl_base;
+
+		for(group = 0; group < PB_GROUP_MAX; group++) {
+			grp_cfg = pb_cfg_data[page].pb_grp_cfg[group];
+
+			if (grp_cfg.close_all != 0) {
+				program_prot_bit_reg(spl_base, group, grp_cfg);
+
+			} else if (grp_cfg.win_len == 0) {
+				program_prot_bit_reg(spl_base, group, grp_cfg);
+
+				pb_mask_filter = grp_cfg.pb_mask_filter;
+				pb_mask_len = grp_cfg.pb_mask_len;
+
+				if (pb_mask_filter == 0) {
+					/* Write pb_mask as it is, no bit sequence filter */
+					for (set_idx = 0; set_idx < pb_mask_len; set_idx++) {
+						write_protection_bits(spl_base, group, set_idx, grp_cfg.pb_mask[set_idx]);
+					}
+				} else {
+					/* Use pb_mask_filter bit sequence filter */
+					if (grp_cfg.pb_mask_set == 0) {
+						/* Write data from pb_mask based on pb_mask_filter bit sequence */
+						if (set_bit_index(pb_mask_filter) == pb_mask_len) {
+							filter_bit_ind = 0;
+							for (set_idx = 0; (set_idx < PB_MAX_REG_COPY_PER_GROUP) && (pb_mask_filter >> set_idx); set_idx++) {
+								if ((pb_mask_filter >> set_idx) & 1) {
+									write_protection_bits(spl_base, group, set_idx, grp_cfg.pb_mask[filter_bit_ind]);
+									filter_bit_ind += 1;
+								}
+							}
+						} else {
+							/* TODO: report if mask len not equal to bit set count in filter */
+						}
+					} else {
+						/* Write 32bit mask based on pb_mask_filter bit sequence */
+						for (set_idx = 0; (set_idx < PB_MAX_REG_COPY_PER_GROUP) && (pb_mask_filter >> set_idx); set_idx++) {
+							if ((pb_mask_filter >> set_idx) & 1) {
+								write_protection_bits(spl_base, group, set_idx, PB_CONF_ALL_PROTECTED);
+							}
+						}
+					}
+				}
+			} else if (grp_cfg.win_len != 0) {
+				for (i = 0; i < PB_MAX_REG_COPY_PER_GROUP; i++)
+					mask_array[i] = 0;
+
+				program_prot_bit_reg(spl_base, group, grp_cfg);
+
+				for (i = 0; i < grp_cfg.win_len; i++)
+					prepare_pb_mask_array(grp_cfg.win_strt[i], grp_cfg.win_stop[i], mask_array, PB_MAX_REG_COPY_PER_GROUP);
+
+				for (set_idx = 0; set_idx < PB_MAX_REG_COPY_PER_GROUP; set_idx++)
+					write_protection_bits(spl_base, group, set_idx, mask_array[set_idx]);
+			}
+		} /* group for loop */
+	} /* page for loop */
+}
