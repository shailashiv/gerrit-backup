Commit Message

[SW-50880] gaudi2: arc: run pcie initialization on arc0

This patch add support to perform PCIe initialization on secondary
CPU ARC0 and let primary CPU ARC1 continue with preboot. Upon successful
initialization of PCIe, secondary CPU ARC0 will go in sleep mode or halt
mode which will be indication in preboot and PCIe EROM for PCIe is
initialized.

+static void preboot_pcie_monitoring(void)
+{
+	/* Wait for PRSTn */
+	while (!regmap_readl(mmPSOC_GLOBAL_CONF_PRSTN))
+		;
+
+	/* Only after PRSTn is de-asserted */
+	preboot_pcie_enable();
+
+	while (1) {
+		unsigned int prstn, ltssm, smlh, rdlh;
+
+		prstn = regmap_readl(mmPSOC_GLOBAL_CONF_PRSTN);
+		ltssm = regmap_readl(mmPCIE_AUX_LTSSM_EN);
+		smlh = regmap_readl(mmPCIE_AUX_SMLH_LINK_UP);
+		rdlh = regmap_readl(mmPCIE_AUX_RDLH_LINK_UP);
+
+		if (prstn && smlh && rdlh)
+			break;
+
+		/*
+		 * Some host assert prstn during link up while moving to GEN3
+		 * This cause BAR size to be reset
+		 * We stop link up process by disabling ltssm
+		 * till BAR is reconfigured
+		 */
+		if (!prstn && ltssm) {
+			regmap_writel(0, mmPCIE_AUX_LTSSM_EN);
+			continue;
+		}
+
+		if (prstn && !ltssm) {
+			regmap_writel(3, mmPCIE_AUX_LTSSM_EN);
+			continue;
+		}
+	}
+
+	/* W/A H/W bug that may lead to have a wrong mask (0x1) in LTSSM */
+	regmap_writel(3, mmPCIE_AUX_LTSSM_EN);
+}

+void secondary_main(void)
+{
+	preboot_pcie_monitoring();
+
+	/* Put secondary CPU in sleep state until further notice */
+	while (1)
+		__asm__ __volatile__("sleep\n\t" : : );

+void preboot_pcie_wait_for_link(void)
+{
+	u32 *secondary_iccm_addr = (u32 *) ARC0_ICCM_BASE;
+
+	/* init dev id as soon as possible, while waiting for prstn */
+	preboot_device_id_init();
+
+	/*
+	 * update secondary entry address to 0th location of ICCM
+	 * ARC once put in running state will jump to address
+	 * specified in 0th location of ICCM
+	 */
+	*secondary_iccm_addr = (u32) &secondary_entry;
+	run_arc(0);
+}
+
+bool is_pcie_initialized(void)
+{
+	u32 secondary_sts;
+
+	secondary_sts = regmap_readl(mmPSOC_ARC0_CFG_RUN_HALT_ACK);
+
+	/*
+	 * Secondary CPU is put to sleep once PCIe initialization
+	 * is done. So notify PCIe is initialized if secondary CPU
+	 * (ARC0) is in sleep state or in halt state.
+	 */
+	if ((secondary_sts & PSOC_ARC0_CFG_RUN_HALT_ACK_SYS_SLEEP_R_MASK) ||
+		(secondary_sts & PSOC_ARC0_CFG_RUN_HALT_ACK_SYS_HALT_R_MASK))
+		return true;
+	else
+		return false;
+}
+
+void preboot_secondary_cpu_stop(void)
+{
+	u32 *secondary_iccm_addr = (u32 *) ARC0_ICCM_BASE;
+
+	/*
+	 * Secondary CPU will be in sleep once pcie initialization
+	 * is done, halt and reset ARC0 so that it can ready
+	 * to load and run zephyr images received via FIT.
+	 */
+	*secondary_iccm_addr = 0x0;
+
+	halt_n_reset_arc(0);
+
+	regmap_writel(ARC_RST_VEC_ADDR, mmPSOC_ARC0_CFG_RST_VEC_ADDR);
+	regmap_writel(ARC_RST_VALUE, mmPSOC_ARC0_CFG_ARC_RST);
+}

+/*
+ * Set ARC in running state
+ * 1. Set run bit in RUN_HALT_REQ register
+ * 2. Wait for ACK from core in RUN_HALT_ACK register
+ * 3. Clear run bit in RUN_HALT_REQ register
+ */
+void run_arc(u8 id)
+{
+	u32 run_ack_offset, run_req_offset;
+	u32 run_ack_mask;
+	u64 start;
+	u32 cpu_boot_err_val;
+
+	if (id) {
+		run_ack_offset = mmPSOC_ARC1_CFG_RUN_HALT_ACK;
+		run_req_offset = mmPSOC_ARC1_CFG_RUN_HALT_REQ;
+		run_ack_mask = PSOC_ARC1_CFG_RUN_HALT_ACK_RUN_ACK_MASK;
+
+	} else {
+		run_ack_offset = mmPSOC_ARC0_CFG_RUN_HALT_ACK;
+		run_req_offset = mmPSOC_ARC0_CFG_RUN_HALT_REQ;
+		run_ack_mask = PSOC_ARC0_CFG_RUN_HALT_ACK_RUN_ACK_MASK;
+
+	}
+
+	/* Set ARC in running state */
+	regmap_writel(BIT(0), run_req_offset);
+
+	start = timestamp_get_us();
+	while (!(regmap_readl(run_ack_offset) & run_ack_mask)) {
+
+		if (timeout_expired(start, ARC_RUN_ACK_TIMEOUT_US)) {
+			preboot_puts_hex("No running state ack from ARC ",
+					id, false);
+			cpu_boot_err_val = (!id) ?
+				CPU_BOOT_ERR0_ARC0_RUN_ACK_NOT_RCVD :
+				CPU_BOOT_ERR0_ARC1_RUN_ACK_NOT_RCVD;
+			cpu_boot_err_clrset(cpu_boot_err_val, true);
+			break;
+		}
+	}
+
+	regmap_writel(0x0, run_req_offset);
+}
