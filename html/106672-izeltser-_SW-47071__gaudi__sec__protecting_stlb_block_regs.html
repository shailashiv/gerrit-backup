+void preboot_secure_stlb(void)
+{
+	u32 mask, prot_base, base = mmSTLB_BASE & ~HL_CFG_BASE;
+
+	/* blocking regs within offsets: 0 - 0x80 */
+	prot_base = base;
+	mask = (PROT_BIT(mmSTLB_STLB_FEATURE_EN) |
+		PROT_BIT(mmSTLB_STLB_AXI_CACHE) |
+		PROT_BIT(mmSTLB_HOP_CONFIGURATION) |
+		PROT_BIT(mmSTLB_LINK_LIST) |
+		PROT_BIT(mmSTLB_MEM_CACHE_CONFIG) |
+		PROT_BIT(mmSTLB_SET_THRESHOLD_HOP0) |
+		PROT_BIT(mmSTLB_SET_THRESHOLD_HOP1) |
+		PROT_BIT(mmSTLB_SET_THRESHOLD_HOP2) |
+		PROT_BIT(mmSTLB_SET_THRESHOLD_HOP3) |
+		PROT_BIT(mmSTLB_SET_THRESHOLD_HOP4));
+	regmap_writel(mask, PRIV_PROT_REG(prot_base));
+
+	/* leaving 1 reg unblocked within offsets: 0x80 - 0x100 */
+	prot_base += SNGL_REG_PROT_SIZE;
+	mask = ~(PROT_BIT(mmSTLB_MEM_READ_ARPROT));
+	regmap_writel(mask, PRIV_PROT_REG(prot_base));
+
+	prot_base += SNGL_REG_PROT_SIZE;	/* offsets: 0x100 - 0xF80 */
+	set_prot_bits_range_all(prot_base, PROT_SIZE_MAX - (prot_base - base));
+}
