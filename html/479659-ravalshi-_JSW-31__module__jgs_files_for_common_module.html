+int fit_process_core_id(enum fit_core_id core_id, uint64_t loadaddr, uint8_t *config)
+{
+	switch (core_id) {
+	case FIT_CORE_ID_IO0_MGMT:
+		/* Nothing needs to be done for psoc mgmt arc as its already running */
+		*config = true;
+		return 0;
+
+	case FIT_CORE_ID_IO0_PID:
+	case FIT_CORE_ID_IO0_RST:
+	case FIT_CORE_ID_IO0_MISC:
+	case FIT_CORE_ID_IO0_NMS:
+	case FIT_CORE_ID_IO0_MUC:
+		/*
+		 * It is expected to receive two images for single core (iccm/dccm),
+		 * so assumption is made here that 1st image will be iccm followed
+		 * by dccm.
+		 */
+		if (!(*config)) {
+			/* halt and reset arc along with desassert */
+			preboot_configure_arc(core_id, loadaddr);
+			*config = true;
+		}
+		return 0;
+
+	default:
+		EMB_LOG_HEX_ERR("Unsupported core id: ", core_id);
+		return 1;
+	}
+}

+/* based on lkd_dev_spec_get_img_addr from preboot */
+u64 get_64bit_addr_from_32bit_addr(uintptr_t base)
+{
+	/* ARC.MNG DCCM and ICCM regions */
+	if (base >= LOCAL_ICCM_BASE && base < IO_MNG_ICCM_SIZE)
+		return (base & ~IO0_MNG_ICCM_BASE) + IO0_MNG_ICCM_FULL_BASE;
+
+	if (base >= LOCAL_DCCM_BASE && base < IO_MNG_DCCM_SIZE)
+		return (base & ~IO0_MNG_DCCM_BASE) + IO0_MNG_DCCM_FULL_BASE;
+
+	/* TODO Other regions better be added as well (GT, HBW, DBG) */
+
+	/* CFG region (15) */
+	if (base >= mmIO_0_CFG_BASE) /* all the top 256MB region is CFG */ {
+		return (base & ~mmIO_0_CFG_BASE) + IO0_CFG_SPACE;
+	}
+
+	EMB_LOG_HEX_ERR("no address translation for base ", (u32)base);
+
+	return 0;
+}

+u32 fs1_get_arc_addr_from_64bit(u64 addr)
+{
+	/* ARC.MNG DCCM and ICCM regions */
+	if (addr >= IO0_MNG_ICCM_FULL_BASE && addr < IO0_MNG_ICCM_FULL_BASE + (u64)IO_MNG_ICCM_SIZE)
+		return (u32)((addr & ADDR_MASK(IO_MNG_ICCM)) + LOCAL_ICCM_BASE);
+
+	if (addr >= IO0_MNG_DCCM_FULL_BASE && addr < IO0_MNG_DCCM_FULL_BASE + (u64)IO_MNG_DCCM_SIZE)
+		return (u32)((addr & ADDR_MASK(IO_MNG_DCCM)) + LOCAL_DCCM_BASE);
+
+	/* TODO Other regions better be added as well (GT, HBW, DBG) */
+
+	/* CFG region (15) */
+	return (addr & 0xffffffff);

+/* currently eligible to run from IO arc mgmt only, because of HL_MNG_W_BARRIER() macro */
+int hw_mem_copy(uint64_t src, uint64_t dst, uint32_t size, bool is_hbw, bool const_data)
+{
+	u64 start = timer_get_us();
+	u32 cpy_sts, copy_sz;
+	u32 cp_err, wr_err, rd_err;
+	u32 sz_limit, sz_shift, sz_mask;
+
+	if (is_hbw) {
+		sz_limit = MEM_CPY_HBW_SIZE_LIMIT;
+		sz_shift = MEM_CPY_HBW_SIZE_SHIFT;
+		sz_mask = MEM_CPY_HBW_SIZE_SHIFT_MASK;
+	} else {
+		sz_limit = MEM_CPY_LBW_SIZE_LIMIT;
+		sz_shift = MEM_CPY_LBW_SIZE_SHIFT;
+		sz_mask = MEM_CPY_LBW_SIZE_SHIFT_MASK;
+	}
+
+	if (size > sz_limit) {
+		EMB_LOG_RAW_ERR("size exceeds MEM_CPY engine limit\n");
+		return -1;
+	}
+
+	regmap_writel((uint32_t)src & MEM_COPY_SRC_ADDR_LSB_BITS_31_4_M,
+		      mmIO_0_CFG_PSOC_PSOC_CFG_MEM_COPY_BASE + mmMEM_COPY_SRC_ADDR_LSB);
+	regmap_writel((uint32_t)(src >> 32), mmIO_0_CFG_PSOC_PSOC_CFG_MEM_COPY_BASE +
+					     mmMEM_COPY_SRC_ADDR_MSB);
+
+	regmap_writel((uint32_t)dst & MEM_COPY_DEST_ADDR_LSB_BITS_31_4_M,
+		      mmIO_0_CFG_PSOC_PSOC_CFG_MEM_COPY_BASE + mmMEM_COPY_DEST_ADDR_LSB);
+	regmap_writel((uint32_t)(dst >> 32), mmIO_0_CFG_PSOC_PSOC_CFG_MEM_COPY_BASE +
+					     mmMEM_COPY_DEST_ADDR_MSB);
+
+	/* set size of in granuality of 4 bytes */
+	copy_sz = size >> sz_shift;
+	if (size & sz_mask) {
+		EMB_LOG_RAW_WRN("MEM_CPY size not aligned, rounding up\n");
+		copy_sz++;
+	}
+
+	regmap_writel(copy_sz, mmIO_0_CFG_PSOC_PSOC_CFG_MEM_COPY_BASE +
+			       mmMEM_COPY_COPY_SIZE);
+
+	/* use constant value write mode for memory init */
+	if (const_data)
+		regmap_writel(MEM_COPY_CONST_VAL_WR_MODE_VAL_M,
+			      mmIO_0_CFG_PSOC_PSOC_CFG_MEM_COPY_BASE +
+			      mmMEM_COPY_CONST_VAL_WR_MODE);
+	else
+		regmap_writel(0, mmIO_0_CFG_PSOC_PSOC_CFG_MEM_COPY_BASE +
+				 mmMEM_COPY_CONST_VAL_WR_MODE);
+
+	/* disable LBW size check per H15-255 */
+	regmap_rmwritel(mmIO_0_CFG_PSOC_PSOC_CFG_MEM_COPY_MSTR_IF_LBW_TSRL_T_BASE +
+			mmTRAN_CHK_LBW_CHECK_EN, 0, TRAN_CHK_LBW_CHECK_EN_SIZE_CHK_M);
+
+	/* Enable copy and wait for completion */
+	regmap_writel(MEM_COPY_COPY_EN_VAL_M, mmIO_0_CFG_PSOC_PSOC_CFG_MEM_COPY_BASE +
+					      mmMEM_COPY_COPY_EN);
+
+	/* let 'enable' command propagate before polling on status */
+	HL_MNG_W_BARRIER();
+
+	while (1) {
+		cpy_sts = regmap_readl(mmIO_0_CFG_PSOC_PSOC_CFG_MEM_COPY_BASE +
+				       mmMEM_COPY_COPY_EN);
+		/* COPY_EN is set by FW, and cleared by HW when copy ends */
+		if (!(cpy_sts & MEM_COPY_COPY_EN_VAL_M))
+			break;
+
+		if (timer_expired(start, HL_MEM_CPY_TIMEOUT_US)) {
+			EMB_LOG_RAW_ERR("error: MEM_CPY engine timeout\n");
+			return -1;
+		}
+	}
+
+	cp_err = regmap_readl(mmIO_0_CFG_PSOC_PSOC_CFG_MEM_COPY_BASE +
+			      mmMEM_COPY_COPY_ERR_STS);
+	wr_err = regmap_readl(mmIO_0_CFG_PSOC_PSOC_CFG_MEM_COPY_BASE +
+			      mmMEM_COPY_WR_ERR_STS);
+	rd_err = regmap_readl(mmIO_0_CFG_PSOC_PSOC_CFG_MEM_COPY_BASE +
+			      mmMEM_COPY_RD_ERR_STS);
+
+	if (cp_err | wr_err | rd_err) {
+		EMB_LOG_RAW_ERR("MEM_CPY finished with errors\n");
+		return -1;
+	}
+
+	return 0;
+}

+static void fs1_arc_halt_cpu(u32 cpu_base, int type)
+{
+	u64 start;
+	bool ack;
+
+	if (type == CPU_TYPE_PSOC)
+		regmap_writel(PSOC_CPU_HALT_MASK, cpu_base + mmARC_RUN_HALT_REQ);
+	else if (type == CPU_TYPE_MUC)
+		regmap_writel(1, cpu_base + mmMUC_ARC_AUX_HALT_REQ);
+
+	start = timer_get_us();
+	while (1) {
+		if (type == CPU_TYPE_PSOC)
+			ack = !!(regmap_readl(cpu_base + mmARC_RUN_HALT_ACK) &
+				 ARC_RUN_HALT_ACK_HALT_ACK_M);
+		else if (type == CPU_TYPE_MUC)
+			ack = !!(regmap_readl(cpu_base + mmMUC_ARC_AUX_RUN_HALT_ACK) &
+				MUC_ARC_AUX_RUN_HALT_ACK_HALT_ACK_M);
+
+		if (ack)
+			break;
+
+		if (timer_expired(start, ARC_RUN_HALT_TIMEOUT_US)) {
+			EMB_LOG_HEX_ERR("timeout waiting for arc halt: ", cpu_base);
+			return;
+		}
+	}
+
+	if (type == CPU_TYPE_PSOC)
+		regmap_writel(0, cpu_base + mmARC_RUN_HALT_REQ);
+	else if (type == CPU_TYPE_MUC)
+		regmap_writel(0, cpu_base + mmMUC_ARC_AUX_HALT_REQ);
+}

+int arch_stop_and_reset_cpu(uint64_t base, int type, uint64_t start)
+{
+	u32 cpu_base = fs1_get_arc_reg_offset(base);
+	bool halted = false;
+	u32 rst_addr = (u32)start >> ARC_RST_ADDR_SHIFT;
+
+	if (type == CPU_TYPE_PSOC) {
+		halted = !!(regmap_readl(cpu_base + mmARC_RUN_HALT_ACK) &
+				 ARC_RUN_HALT_ACK_SYS_HALT_R_M);
+	} else if (type == CPU_TYPE_MUC) {
+		halted = !!(regmap_readl(cpu_base + mmMUC_ARC_AUX_RUN_HALT_ACK) &
+				MUC_ARC_AUX_RUN_HALT_ACK_SYS_HALT_R_M);
+	}
+	/* if arc is not halted halt the arc */
+	if (!halted)
+		fs1_arc_halt_cpu(cpu_base, type);
+
+	/* reset, set reset vector, and release reset */
+	if (type == CPU_TYPE_PSOC) {
+		regmap_writel(1, cpu_base + mmARC_ARC_RST);
+		regmap_writel(rst_addr, cpu_base + mmARC_RST_VEC_ADDR);
+		regmap_writel(0, cpu_base + mmARC_ARC_RST);
+	} else if (type == CPU_TYPE_MUC) {
+		regmap_writel(1, cpu_base + mmMUC_ARC_AUX_ARC_RST);
+		regmap_writel(rst_addr, cpu_base + mmMUC_ARC_AUX_RST_VEC_ADDR);
+		regmap_writel(0, cpu_base + mmMUC_ARC_AUX_ARC_RST);
+	}
+
+	return 0;
+}
