+#!/bin/bash
+
+pdu_uri=http://192.168.10.80
+pdu_cred="--user=snmp --password=1234"
+
+if [ $# -eq 2 ]; then
+	board=$1
+	action=$2
+else
+	echo "error - usage: $0 <board name> <-l | on | off>"
+	exit 1
+fi
+
+case "$board" in
+*onix)
+	pwr_id=1
+	;;
+*)
+	echo "error - Unknown board name: $board"
+	exit 2
+	;;
+esac
+
+get_id=`echo $((10+pwr_id))`
+set_id=`for i in $(seq $((pwr_id-1))); do echo -n 0; done; echo -n 1`
+case "$action" in
+-l)
+	status=`wget -q -O - $pdu_cred ${pdu_uri}/status.xml | grep pot0 | cut -d, -f$get_id`
+	case "$status" in
+	0)
+		echo off
+		;;
+	1)
+		echo ON
+		;;
+	*)
+		echo "error - Unknown state: $status"
+		exit 3
+		;;
+	esac
+	;;
+on | off)
+	wget -q -O - $pdu_cred ${pdu_uri}/${action}s.cgi?led=$set_id
+	status=3 # D_ON=2/D_OFF=3
+	while [ $status -gt 1 ]; do
+		sleep 1
+		status=`wget -q -O - $pdu_cred ${pdu_uri}/status.xml | grep pot0 | cut -d, -f$get_id`
+	done
+	;;
+*)
+	echo "error - Unknown action: $action"
+	exit 4
+	;;
+esac

+#!/bin/sh
+#
+# Start misc....
+#
+
+case "$1" in
+  start)
+	echo "Starting misc..."
+	echo powr1220 0x25 > /sys/bus/i2c/devices/i2c-0/new_device
+	echo ir35221 0x40 > /sys/bus/i2c/devices/i2c-0/new_device
+	echo 0 > /sys/module/rcupdate/parameters/rcu_cpu_stall_suppress
+	for tz in /sys/class/thermal/thermal_zone*;
+	do
+		echo enabled > ${tz}/mode
+	done
+	mount -t debugfs none /sys/kernel/debug
+	;;
+  stop)
+	;;
+  restart|reload)
+	;;
+  *)
+	echo "Usage: $0 {start|stop|restart}"
+	exit 1
+esac
+
+exit $?
