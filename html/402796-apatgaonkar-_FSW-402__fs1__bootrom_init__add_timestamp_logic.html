+void boot_rom_clear_ts_counter(void)
+{
+	regmap_writel(0x0, IO0_HALF0_PSOC_TIMESTAMP(mmTIMESTAMP_CNTCR));
+}
+
+void boot_rom_start_timer(uint32_t timestamp_freq)
+{
+	regmap_writel(timestamp_freq, IO0_HALF0_PSOC_TIMESTAMP(mmTIMESTAMP_CNTFID0));
+	regmap_writel((TIMESTAMP_CNTCR_CNTCR_EN_M | TIMESTAMP_CNTCR_CNTCR_HDBG_M),
+		      IO0_HALF0_PSOC_TIMESTAMP(mmTIMESTAMP_CNTCR)); /* EN | HDBG */
+}
+
+uint64_t boot_rom_get_timestamp_freq(void)
+{
+	/* TODO: Confirm timestamp frequency is always constant */
+	return SYS_REF_CLOCK / 2;
+}
+
+uint64_t boot_rom_udelay_get_ts(void)
+{
+	uint32_t low, up1, up2;
+
+	up1 = regmap_readl(IO0_HALF0_PSOC_TIMESTAMP(mmTIMESTAMP_CNTCVU));
+	low = regmap_readl(IO0_HALF0_PSOC_TIMESTAMP(mmTIMESTAMP_CNTCVL));
+	up2 = regmap_readl(IO0_HALF0_PSOC_TIMESTAMP(mmTIMESTAMP_CNTCVU));
+	/* In case of overlap read low part again */
+	if (up1 != up2)
+		low = regmap_readl(IO0_HALF0_PSOC_TIMESTAMP(mmTIMESTAMP_CNTCVL));
+
+	return ((uint64_t)up2 << 32) + low;
+}

+// SPDX-License-Identifier:     GPL-2.0+
+/*
+ * Copyright (C) 2022 HabanaLabs Ltd.
+ * All Rights Reserved.

+
+#include <boot_rom.h>
+#include <delay.h>
+
+#define USEC_PER_SEC		1000000
+#define BYPASS_CYCLES_PER_USEC	(SYS_REF_CLOCK / USEC_PER_SEC)
+
+static uint64_t cycles_per_usec = BYPASS_CYCLES_PER_USEC;
+
+uint64_t boot_rom_get_ts_us(void)
+{
+	return boot_rom_udelay_get_ts() / cycles_per_usec;
+}
+
+void boot_rom_init_ts(void)
+{
+	uint32_t timestamp_freq = boot_rom_get_timestamp_freq();
+
+	cycles_per_usec = timestamp_freq / USEC_PER_SEC;
+
+	/* stop counter */
+	boot_rom_clear_ts_counter();
+
+	/* timer init */
+	boot_rom_start_timer(timestamp_freq);
+}
+
+bool boot_rom_timeout_expired(uint64_t start_tv, uint32_t delay)
+{
+	uint64_t current, timeout_tv = start_tv + delay;
+	bool overlap = false;
+
+	if (timeout_tv < start_tv)
+		overlap = true;
+
+	current = boot_rom_get_ts_us();
+	if (!overlap) {
+		if (current > timeout_tv || current <= start_tv)
+			return true;
+	} else {
+		if (current > timeout_tv && current < start_tv)
+			return true;
+	}
+
+	return false;
+}
+
+void udelay(unsigned int usec)
+{
+	uint64_t start, current, end = cycles_per_usec * usec;
+	bool overlap = false;
+
+	start = boot_rom_udelay_get_ts();
+	end += start;
+
+	if (end < start)
+		overlap = true;
+
+	while (1) {
+		current = boot_rom_udelay_get_ts();
+		if (!overlap) {
+			/* Check also for unplanned overlap */
+			if (current > end || current <= start)
+				break;
+		} else {
+			if (current > end && current < start)
+				break;
+		}
+	}
+}
