Commit Message

[SW-82716] common: add timestamp apis

In order to take care of synchronization issues related to i2c OOB
communication, this patch add timestamp api which will be used
by i2c OOB comms. The timestamp used is PSOC timestamp which take
clock from div3 of PCIE/PSOC PLL @ 50MHz.

Register map and mask are added for greco and gaudi2 to avoid
compilation issues.

+#include <arcboot.h>
+
+#define SYS_REF_CLOCK		50000000ULL
+#define USEC_PER_SEC		1000000
+#define BYPASS_CYCLES_PER_USEC	(SYS_REF_CLOCK / USEC_PER_SEC)
+
+static uint64_t cycles_per_usec = BYPASS_CYCLES_PER_USEC;
+
+__weak uint64_t arcboot_rom_get_timestamp_freq(void)
+{
+	return SYS_REF_CLOCK;
+}
+
+void arcboot_rom_init_ts(void)
+{
+	uint32_t timestamp_freq = arcboot_rom_get_timestamp_freq();
+
+	cycles_per_usec = timestamp_freq / USEC_PER_SEC;
+
+	/* timer init */
+	regmap_writel(timestamp_freq, mmTIMESTAMP_CNTFID0);
+	regmap_writel((TIMESTAMP_CNTCR_CNTCR_EN_M | TIMESTAMP_CNTCR_CNTCR_HDBG_M),
+			mmTIMESTAMP_CNTCR); /* EN | HDBG */
+}
+
+static uint64_t udelay_get_ts(void)
+{
+	uint32_t low, up1, up2;
+
+	up1 = regmap_readl(mmTIMESTAMP_CNTCVU);
+	low = regmap_readl(mmTIMESTAMP_CNTCVL);
+	up2 = regmap_readl(mmTIMESTAMP_CNTCVU);
+	/* In case of overlap read low part again */
+	if (up1 != up2)
+		low = regmap_readl(mmTIMESTAMP_CNTCVL);
+
+	return ((uint64_t)up2 << 32) + low;
+}
+
+uint64_t arcboot_rom_get_ts_us(void)
+{
+	return udelay_get_ts() / cycles_per_usec;
+}
+
+bool arcboot_rom_timeout_expired(uint64_t start_tv, uint32_t delay)
+{
+	uint64_t current, timeout_tv = start_tv + delay;
+	bool overlap = false;
+
+	if (timeout_tv < start_tv)
+		overlap = true;
+
+	current = arcboot_rom_get_ts_us();
+	if (!overlap) {
+		if (current > timeout_tv || current <= start_tv)
+			return true;
+	} else {
+		if (current > timeout_tv && current < start_tv)
+			return true;
+	}
+
+	return false;
+}
