+void preboot_start_image(u64 startaddr)
+{
+	void (*image_entry)(void);
+	u32 lsb, msb;
+
+	if (startaddr) {
+		lsb = (u32)startaddr;
+		msb = (u32)(startaddr >> 32);
+		/* Set CPU0 Reset address to U-Boot */
+		regmap_writel(lsb, mmCPU_CA53_CFG_RST_ADDR_LSB_0);
+		regmap_writel(msb, mmCPU_CA53_CFG_RST_ADDR_MSB_0);
+
+		/* Wake up CPU0, after we put all CPUs in reset */
+		regmap_writel(0x0, mmCPU_CA53_CFG_ARM_RST_CONTROL);
+		regmap_writel(0x10111, mmCPU_CA53_CFG_ARM_RST_CONTROL);
+	}
+
+	/*
+	 * Set ARC0 in running state
+	 * 1. Set run bit in RUN_HALT_REQ register
+	 * 2. Wait for ACK from core in RUN_HALT_ACK register
+	 * 3. Clear run bit in RUN_HALT_REQ register
+	 */
+	regmap_writel(BIT(0), mmPSOC_ARC0_CFG_RUN_HALT_REQ);
+	while (!(regmap_readl(mmPSOC_ARC0_CFG_RUN_HALT_ACK) &
+				PSOC_ARC0_CFG_RUN_HALT_ACK_RUN_ACK_MASK))
+		;
+	regmap_writel(0x0, mmPSOC_ARC0_CFG_RUN_HALT_REQ);
+
+	/*
+	 * Jump to __reset of zephyr. After image_entry, control is not
+	 * expected to return here and continue.
+	 */
+	image_entry = (void *) (*(u32 *)ARC_ICCM_BASE);
+	image_entry();
+}

+u32 hl_pll_get_freq(u32 pll_base, enum hl_pll_div_out div)
+{
+	u32 div_sel, factor, offset;
+	u32 reg, fbdiv, refdiv, postdiv1, postdiv2;
+	u64 out_freq = SYS_REF_CLOCK;
+
+	offset = div * sizeof(u32);
+	div_sel = regmap_readl(pll_base + HL_PLL_DIV_SEL + offset);
+
+	if (div_sel == HL_PLL_PLL_CLK || div_sel == HL_PLL_DIV_PLL) {
+		reg = regmap_readl(pll_base + GRECO_PLL_CFG_DIV);
+		/* fbdiv : bits 0-11 */
+		fbdiv = EXTRACT_BITS(reg, 0, 12);
+		/* postdiv1 : bits 16-18 */
+		postdiv1 = EXTRACT_BITS(reg, 16, 3);
+		/* postdiv2 : bits 20-22 */
+		postdiv2 = EXTRACT_BITS(reg, 20, 3);
+		/* refdiv : bits 24-29 */
+		refdiv  = EXTRACT_BITS(reg, 24, 6);
+		out_freq = SYS_REF_CLOCK * fbdiv;
+		out_freq /= refdiv * postdiv1 * postdiv2;
+	}
+
+	if (div_sel == HL_PLL_DIV_REF || div_sel == HL_PLL_DIV_PLL) {
+		factor = regmap_readl(pll_base + HL_PLL_DIV_FACTOR + offset);
+		out_freq /= factor + 1;
+	}
+
+	/*
+	 * TODO : The clock relaxation is not accounted here.

+	 * POSTDIV1 - bit 16
+	 * POSTDIV2 - bit 20
+	 * REFDIV - bit 24

+	reg = cfg->nr << GRECO_PLL_REFDIV_SHIFT |

+	/* after updating the coefficients we need to wait >=1us */
+	udelay(GRECO_PLL_MAX_WAIT_TIME);
+
+	/* enable the PLL by writing a value of 1 to CTRL_CFG.EN*/
+	hl_pll_reset(cfg, true);

+/*
+ * Copyright 2020 HabanaLabs, Ltd.
+ * All Rights Reserved.

+
+#include <bitops.h>
+#include <preboot.h>
+#include <string.h>
+#include <uart.h>
+#include <sha256.h>
+#include <greco/greco_security.h>
+
+#define EFUSE_SYSCFG_ENABLE_VALIDATION_MASK	BIT(10)
+
+static u32 sp_sram_readl(u32 offset)
+{
+	u32 reg = *((volatile u32 *) (HL_SCRATCHPAD_SRAM_BASE + offset));
+	return reg;
+}
+
+bool preboot_security_is_enabled(void)
+{
+	u32 config_efuse;
+
+	config_efuse = sp_sram_readl(SECURITY_EFUSE_SYS_CFG_OFFSET);
+	return (config_efuse & EFUSE_SYSCFG_ENABLE_VALIDATION_MASK);
+}
+
+bool preboot_security_key_is_valid(int key_idx, const void *pub_key,
+			 int pub_key_len, const void *pub_key_ext,
+			 int pub_key_ext_len)
+{
+	u32 key_revoke, key_base, hash_calculated, hash_stored;
+	u8 bytes_per_dword = sizeof(u32);
+	u8 value[SHA256_SUM_LEN];
+	u8 key[KEY_SIZE];
+	u8 i;
+
+	if (key_idx < 0 || key_idx >= SECURITY_KEY_HASH_COUNT)
+		return false;
+
+	key_revoke = sp_sram_readl(SECURITY_EFUSE_KEYS_REVOKE_OFFSET);
+	if (key_revoke & BIT(key_idx)) {
+		preboot_puts_hex("Key revoked: ", key_idx, 0);
+		return false;
+	}
+
+	if (pub_key_len != (KEY_SIZE - KEY_EXT_SIZE)) {
+		preboot_puts_hex("Invalid pub key length: ", pub_key_len, 0);
+		return false;
+	}
+
+	if (pub_key_ext_len != KEY_EXT_SIZE) {
+		preboot_puts_hex("Invalid pub key ext length: ",
+				pub_key_ext_len, 0);
+		return false;
+	}
+
+	/* Concanate public key and public key extension */
+	memcpy(key, pub_key, pub_key_len);
+	memcpy((key + pub_key_len), pub_key_ext, pub_key_ext_len);
+
+	sha256_csum(key, pub_key_len + pub_key_ext_len, value);
+
+	key_base = (key_idx * bytes_per_dword * SECURITY_DWORD_PER_KEY)
+			+ SECURITY_EFUSE_KEYS_OFFSET;
+
+	for (i = 0; i < SECURITY_DWORD_PER_KEY; i++) {
+		memcpy(&hash_calculated, value + (i * bytes_per_dword),
+			bytes_per_dword);
+		hash_stored = sp_sram_readl(key_base + (i * bytes_per_dword));
+		if (hash_calculated != hash_stored)
+			return false;
+	}
+
+	return true;
+}
+
+bool preboot_security_svn_is_valid(int sec_ver)
+{
+	u8 sec_fw_ver, sec_valid_fw_ver;
+	u32 valid_fw_ver;
+
+	sec_fw_ver = SECURITY_FW_VERSION_TO_SECURITY_BYTE(sec_ver);
+	valid_fw_ver = sp_sram_readl(SECURITY_EFUSE_FW_VERSION_OFFSET);
+	sec_valid_fw_ver = SECURITY_FW_VERSION_TO_SECURITY_BYTE(valid_fw_ver);
+
+	if (sec_fw_ver < sec_valid_fw_ver)
+		preboot_puts_hex("Security version too low: ", sec_ver, 0);
+
+	return sec_fw_ver >= sec_valid_fw_ver;
+}

+{
+	return true;
