Commit Message

[SW-20810] i2c: add indirect protocol

This protocol will be used in order to transfer a firmware to preboot.

The indirect protocol defines a request port can be used for large data
transfters.

This review implements the write functionality of the indirect protocol
or WDATA.

This code is optimized in order to avaoid HW queue overflow. This means
indirect protocol functionality is only called when all the needed data
have received. After the indirect protocol is being processed, more data
can be requested by the command procedure. When the requested data is
available, the same command procedure is being called. This goes on
until no more data is needed by the command procedure.

+static inline u32 i2csi_req_wdata_handle_chunk(u8 **data, u8 cnt)
+{
+	register u8 *buf;
+
+	if (cnt == 2) { /* Chunk (partial payload) is expected */
+		buf = i2csi_req_opcode_alloc_buffer();
+		if (buf)
+			buf += i2csi_req.chunk.offset;
+
+		*data = buf;
+
+		return i2csi_req.chunk.hdr.rqcl;
+	}
+
+	/* Chuck received */
+	i2csi_req.chunk.offset += i2csi_req.chunk.hdr.rqcl;
+
+	/* Payload fully received */
+	if (i2csi_req.chunk.offset == i2csi_req.hdr.rqdl) {
+		i2csi_req.chunk.offset = 0;
+		i2csi_req.sts.sm_sts = HL_I2CI_REQ_ST_PROCESS;
+		i2csi_req_process_opcode();
+		i2csi_req.sts.sm_sts = HL_I2CI_REQ_ST_IDLE;
+	}
+
+	return 0;
+}
+
+static inline u32 i2csi_req_wdata_handle_payload(u8 **data, u8 cnt)
+{
+	if (cnt == 1) { /* Payload is expected */
+		*data = i2csi_req_opcode_alloc_buffer();
+		return i2csi_req.hdr.rqdl;
+	}
+
+	/* Payload received */
+	i2csi_req.sts.sm_sts = HL_I2CI_REQ_ST_PROCESS;
+	i2csi_req_process_opcode();
+	i2csi_req.sts.sm_sts = HL_I2CI_REQ_ST_IDLE;
+	return 0;
+}
+
+/*
+ * Handle request port WDATA command. This command holds a data buffer which
+ * is being transferred by the master. The request may be divided into chunks
+ * and this function also handles this possibility.
+ *
+ * This function is being called 3-4 time per WDATA command. Depends whether
+ * the request is divided into chunks or not.
+ * - 1. [cnt=0] WDATA was recorded, wait for header.
+ * - 2. [cnt=1] Header arrived, wait for chunk header or payload
+ * - 3. [cnt=2,no chunks] - payload was written, done this WDATA command
+ *      [cnt=2,chunks]    - chunk header was written, wait for payload
+ * - 4. [cnt=3,chunks]    - payload was written, done this WDATA command
+ *
+ * Arguments:
+ * - data: needs to be set to a designated buffer on which a new data will be
+ *         written into.
+ * - cnt:  the number of time this function is called per current command
+ *         context.
+ *
+ * Returns:
+ *   The number of bytes the lower layer should write into supplied buffer
+ *   before calling this function again. Returning 0 means the command handling
+ *   was finished.
+ */
+u32 i2csi_req_handle_wdata(u8 **data, u8 cnt)
+{
+	switch (cnt) {
+	case 0: /* Header is expected */
+		i2csi_req.sts.sm_sts = HL_I2CI_REQ_ST_RECEIVE;
+		i2csi_req.sts.err = HL_I2CI_REQ_ERR_NONE;
+		i2csi_req.chunk.hdr.rqcl = 0;
+		*data = i2csi_req.hdr.data;
+		return HL_I2CI_REQ_WDATA_LEN;
+	case 1: /* Chunk header or payload are expected */
+		if (!i2csi_req.hdr.write) {
+			i2csi_req.sts.err = HL_I2CI_REQ_ERR_P_WRITE_PERM;
+			return 0;
+		}
+
+		if (HL_I2CI_REQ_RQCL_IS_EXISTS(i2csi_req.hdr)) {
+			*data = i2csi_req.chunk.hdr.data;
+			return HL_I2CI_REQ_CHUNK_LEN;
+		}
+		/* Fall through */
+	default:
+		break;
+	}
+
+	if (i2csi_req.chunk.hdr.rqcl)
+		return i2csi_req_wdata_handle_chunk(data, cnt);
+	else
+		return i2csi_req_wdata_handle_payload(data, cnt);
+}
+
+/*
+ * Handle request port status reading by the master. Update the request status
+ * header with the latest WDATA opcode data and send it on its way.
+ */
+u8 i2csi_req_handle_sts(u8 cmd, u32 read_seq)
+{
+	int idx = cmd + read_seq - HL_INDIRECT_REQ_STS_DATA;
+
+	if (idx >= HL_I2CI_REQ_STS_LEN)
+		return 0;
+
+	i2csi_req.sts.opcode = i2csi_req.hdr.opcode;
+	i2csi_req.sts.write = i2csi_req.hdr.write;
+	i2csi_req.sts.ic = i2csi_req.hdr.ic;
+
+	return i2csi_req.sts.data[idx];
+}
+
+/*
+ * Lower layer reports sync error. This means it expected the master to write
+ * bytes but instead is got a read request.
+ * In this case we zero out the state machine and buffer offsets.
+ */
+void i2csi_req_handle_sync_error(void)
+{
+	if (i2csi_req.sts.sm_sts == HL_I2CI_REQ_ST_IDLE)
+		return;
+
+	i2csi_req.chunk.offset = 0;
+	i2csi_req.sts.sm_sts = HL_I2CI_REQ_ST_IDLE;
+	i2csi_req.sts.err = HL_I2CI_REQ_ERR_P_DATA_SIZE;
+}
+
+/*
+ * Handle request port configuration register. Set the init bit in request
+ * status reply.
+ * This function is called when the lower layer needs a buffer to write the
+ * data to and the number of bytes it whould write before calling this function
+ * again.
+ *
+ * Arguments:
+ * - data: needs to be set to a designated buffer on which a new data will be
+ *         written into.
+ * - cnt:  the number of time this function is called per current command
+ *         context.
+ *
+ * Returns:
+ *   The number of bytes the lower layer should write into supplied buffer
+ *   before calling this function again. Returning 0 means the command handling
+ *   was finished.
+ */
+u32 i2csi_req_handle_cfg(u8 **data, u8 cnt)
+{
+	if (!cnt) { /* Header is expected */
+		*data = i2csi_req.cfg.data;
+		return HL_I2CI_REQ_CFG_LEN;
+	}
+
+	/* Request config received */
+	i2csi_req.sts.init = i2csi_req.cfg.init;
+	i2csi_req.sts.sm_sts = HL_I2CI_REQ_ST_IDLE;
+	i2csi_req.sts.err = HL_I2CI_REQ_ERR_NONE;
+
+	return 0;
+}
