+#include "preboot.h"
+#include "uart.h"
+#include "pll.h"

+static enum hl_pll_bypass_val hl_pll_bypass_get(struct hl_pll_cfg *cfg,
+							enum hl_pll_div_out out)
+{
+	int offset = out * sizeof(u32);
+
+	return regmap_readl(cfg->base + HL_PLL_DIV_SEL + offset);
+}
+
+static void hl_pll_udelay(unsigned int usec)
+{
+	int pll_num;
+	long cnt;
+	struct hl_pll_cfg *hl_pll_cfg;
+	enum hl_pll_bypass_val bval;
+
+	pll_num = hl_pll_get_cfg(&hl_pll_cfg);
+	if (pll_num) {
+		/*
+		 * Assumptions:
+		 * 1) the CPU PLL offset in the array is 0
+		 * 2) DIV1 is the PLL output to timestamp counter.
+		 */
+		bval = hl_pll_bypass_get(hl_pll_cfg, HL_PLL_DIV1);
+	}
+
+	if (bval == HL_PLL_REF_CLK || bval == HL_PLL_DIV_REF)
+		cnt = BYPASS_CYCLES_PER_USEC;
+	else
+		cnt = CPU_CYCELS_PER_USEC;
+
+	cnt = (cnt * usec) / MIN_CYCLES_PER_ITER;
+
+	while (cnt--)
+		;
+}
+
+static void hl_pll_reset(struct hl_pll_cfg *cfg, bool reset)
+{
+	regmap_writel(reset, cfg->base + HL_PLL_RST);
+}
+
+static inline int hl_pll_is_locked(struct hl_pll_cfg *cfg)
+{
+	return regmap_readl(cfg->base + HL_PLL_CFG) & HL_PLL_CFG_LOCK;
+}
+
+static inline void hl_pll_clear_lock_status(struct hl_pll_cfg *cfg)
+{
+	u32 reg = regmap_readl(cfg->base + HL_PLL_CFG);
+
+	regmap_writel(reg & ~HL_PLL_CFG_LOCK, cfg->base + HL_PLL_CFG);
+}
+
+static int hl_pll_cfg(struct hl_pll_cfg *cfg)
+{
+	int tout = HL_PLL_LOCK_TOUT;
+
+	hl_pll_reset(cfg, true);
+
+	regmap_writel(cfg->nr, cfg->base + HL_PLL_NR);
+	regmap_writel(cfg->nf, cfg->base + HL_PLL_NF);
+	regmap_writel(cfg->od, cfg->base + HL_PLL_OD);
+	regmap_writel(cfg->nb, cfg->base + HL_PLL_NB);
+	regmap_writel(0x11, cfg->base + HL_PLL_DATA_CHNG);
+
+	/* after updating the coefficients we need to wait >=5us */
+	hl_pll_udelay(HL_PLL_MAX_WAIT_TIME);
+
+	hl_pll_reset(cfg, false);
+
+	/* Wait >500 (10us) reference cycles until PLL re-locks. */
+	hl_pll_udelay(HL_PLL_MAX_WAIT_TIME);
+
+	/* to be on a safe side, poll on the PLL lock status bit */
+	while (!hl_pll_is_locked(cfg) && tout--)
+		hl_pll_udelay(1);
+
+	if (!tout) {
+		preboot_puts_raw("PLL: Timeout waiting for ");
+		preboot_puts_raw(cfg->name);
+		preboot_puts_raw(" PLL lock...\n\r");
+		return 1;
+	}
+
+	/* w/a the h/w erratum: lock status bit isn't getting cleared */
+	hl_pll_clear_lock_status(cfg);
+
+	return 0;
+}
+
+static void hl_pll_bypass(struct hl_pll_cfg *cfg, bool bypass)
+{
+	int sel;
+	u64 reg;
+
+	for (sel = 0; sel < HL_PLL_DIV_SIZE; sel++) {
+		reg = cfg->base + HL_PLL_DIV_SEL + (sel * sizeof(u32));
+
+		/* -1 means always bypass */
+		if (cfg->div_factor[sel] == -1 || bypass)
+			regmap_writel(HL_PLL_REF_CLK, reg);
+		else
+			regmap_writel(cfg->div_factor[sel] ?
+					HL_PLL_DIV_PLL : HL_PLL_PLL_CLK, reg);
+	}
+
+	/* just to be on a safe side, wait for bypass to propagate */
+	hl_pll_udelay(1);
+}
+
+static void hl_pll_div_init(struct hl_pll_cfg *cfg)
+{
+	int sel;
+	u64 div_base;
+
+	for (sel = 0; sel < HL_PLL_DIV_SIZE; sel++) {
+		if (cfg->div_factor[sel] == -1)
+			continue;
+
+		div_base = cfg->base + (sel * sizeof(u32));
+
+		regmap_writel(cfg->div_factor[sel],
+				div_base + HL_PLL_DIV_FACTOR);
+		regmap_writel(1, div_base + HL_PLL_DIV_FACTOR_CMD);
+		regmap_writel(1, div_base + HL_PLL_DIV_EN);
+	}
+
+	/* just to be on a safe side, wait for DIV init to propagate */
+	hl_pll_udelay(1);
+}
+
+static void hl_pll_ack_int(struct hl_pll_cfg *cfg)
+{
+	regmap_writel(1, cfg->base + HL_PLL_LOCK_INTR);
+}
+
+static int hl_pll_set(struct hl_pll_cfg *cfg)
+{
+	int ret;
+
+	hl_pll_bypass(cfg, true);
+
+	ret = hl_pll_cfg(cfg);
+
+	hl_pll_div_init(cfg);
+
+	hl_pll_bypass(cfg, false);
+
+	hl_pll_ack_int(cfg);
+
+	return ret;
+}
+
+/**
+ * hl_pll_init - Initialize PLL registers
+ */
+void hl_pll_init(void)
+{
+	int i, pll_num;
+	struct hl_pll_cfg *hl_pll_cfg;
+	enum hl_pll_bypass_val bval;
+
+	pll_num = hl_pll_get_cfg(&hl_pll_cfg);
+
+	for (i = 0; i < pll_num; i++) {
+		/* DIV0 is the undivided PLL output */
+		bval = hl_pll_bypass_get(hl_pll_cfg + i, HL_PLL_DIV0);
+		if (hl_pll_cfg[i].base == CPU_PLL_BASE &&
+				bval != HL_PLL_REF_CLK &&
+				bval != HL_PLL_DIV_REF)
+			continue;
+
+		hl_pll_set(hl_pll_cfg + i);
+	}
+}
