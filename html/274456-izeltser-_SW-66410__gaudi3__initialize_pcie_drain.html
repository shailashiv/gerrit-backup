+#define HBW_DRAIN_TIMEOUT	0x1000
+#define LBW_DRAIN_TIMEOUT	0xF000
+
+void ppboot_pcie_drain_init(void)
+{
+	u32 val;
+
+	/* If FW skip enabled return */
+	if (hl_skip_fw_cfg())
+		return;
+
+	regmap_writel(HBW_DRAIN_TIMEOUT, mmD0_PCIE_WRAP_HBW_DRAIN_TIMEOUT);
+	regmap_writel(D0_PCIE_WRAP_HBW_RSP_ERR_DRAIN_STAMP_EN_M,
+		      mmD0_PCIE_WRAP_HBW_RSP_ERR_DRAIN_STAMP);
+	val = regmap_readl(mmD0_PCIE_WRAP_HBW_DRAIN_CFG);
+	val |= D0_PCIE_WRAP_HBW_DRAIN_CFG_EN_M;
+	regmap_writel(val, mmD0_PCIE_WRAP_HBW_DRAIN_CFG);
+
+	regmap_writel(LBW_DRAIN_TIMEOUT, mmD0_PCIE_WRAP_LBW_DRAIN_TIMEOUT);
+	regmap_writel(D0_PCIE_WRAP_LBW_RSP_ERR_DRAIN_STAMP_EN_M,
+		      mmD0_PCIE_WRAP_LBW_RSP_ERR_DRAIN_STAMP);
+	val = regmap_readl(mmD0_PCIE_WRAP_LBW_DRAIN_CFG);
+	val |= D0_PCIE_WRAP_LBW_DRAIN_CFG_EN_M;
+	regmap_writel(val, mmD0_PCIE_WRAP_LBW_DRAIN_CFG);
+}
