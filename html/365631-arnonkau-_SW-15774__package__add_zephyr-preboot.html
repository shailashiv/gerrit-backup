+ZEPHYR_PREBOOT_APP_VERSION = $(shell cut -d- -f2- $(@D)/localversion-habanalabs)

+ZEPHYR_PREBOOT_CONF_ENV += ZEPHYR_TOOLCHAIN_VARIANT=cross-compile

+
+ZEPHYR_PREBOOT_CONF_OPTS = -UCMAKE_TOOLCHAIN_FILE
+ZEPHYR_PREBOOT_CONF_OPTS += -DUSER_CACHE_DIR=$(@D)/.cache

+ZEPHYR_PREBOOT_CONF_OPTS += -DTARGET=$(HL_TARGET)
+ZEPHYR_PREBOOT_CONF_OPTS += -DEXTRA_CFLAGS="\

+		$(ZEPHYR_PREBOOT_CONF_APP_EXTRA_CFLAGS)"
+ZEPHYR_PREBOOT_CONF_OPTS += $(ZEPHYR_PREBOOT_CONF_APP_OPTS)

+ZEPHYR_PREBOOT_CONF_OPTS += -DBUILD_MACHINE="$(shell hostname -s)"

+
+ZEPHYR_PREBOOT_CONF_OPTS += -DDTS_EXTRA_CPPFLAGS="-I$(SPECS_EMBEDDED_DIR)"
+ZEPHYR_PREBOOT_CONF_OPTS += -DBOARD=hl_$(ZEPHYR_PREBOOT_DEVICE_NAME)_$(ZEPHYR_PREBOOT_APP_NAME)_$(HL_TARGET)
+ZEPHYR_PREBOOT_CONF_OPTS += -DDEVICE=$(ZEPHYR_PREBOOT_DEVICE_NAME)
+ZEPHYR_PREBOOT_CONF_OPTS += -DAPP_NAME=$(ZEPHYR_PREBOOT_APP_PROD_NAME)
+ZEPHYR_PREBOOT_CONF_OPTS += -DAPP_VERSION=$(ZEPHYR_PREBOOT_APP_VERSION)
+
+ZEPHYR_PREBOOT_WEST_SRC_DIR :=
+ZEPHYR_PREBOOT_WEST_DEST_DIR :=
+
+define ZEPHYR_PREBOOT_COPY_ZEPHYR_SRC
+	rsync -au --chmod=u=rwX,go=rX  --exclude .svn --exclude .git --exclude .hg --exclude .bzr --exclude CVS $(ZEPHYR_PREBOOT_ZEPHYR_SRCDIR) $(ZEPHYR_PREBOOT_BUILD_DIR)
+endef
+
+# Arguments:
+# - module name
+# - component dir
+# - component name
+#
+define ZEPHYR_PREBOOT_DEFINE_MODULE_ENVVARS
+	$(eval MODULE_NAME=$(strip $1))
+	$(eval MODULE_NAME_UPPER_CASE=$(shell echo $(MODULE_NAME) | tr a-z A-Z))
+	$(eval MODULE_NAME_UPPER_CASE=$(strip $(MODULE_NAME_UPPER_CASE)))
+	$(eval $(MODULE_NAME_UPPER_CASE)_SRCDIR=$(EMBEDDED_DIR)/zephyr-modules/$(MODULE_NAME))
+	$(eval $(MODULE_NAME_UPPER_CASE)_DIR=$(@D)/build/modules/$(MODULE_NAME))
+	$(eval $(MODULE_NAME_UPPER_CASE)_NAME=$2_$(MODULE_NAME))
+endef
+
+$(foreach module,$(ZEPHYR_PREBOOT_APP_MODULES),$(call ZEPHYR_PREBOOT_DEFINE_MODULE_ENVVARS,$(module),$(ZEPHYR_PREBOOT_APP_PROD_NAME)))
+
+# Arguments:
+# - module (mbedtls, eeprom, etc.)
+define ZEPHYR_PREBOOT_MODULE_VERSION_HOOK
+	$(eval MODULE_NAME_UPPER_CASE=$(shell echo $1 | tr a-z A-Z))
+	$(eval MODULE_NAME_UPPER_CASE=$(strip $(MODULE_NAME_UPPER_CASE)))
+	$(call HL_VERSION_GEN, ${${MODULE_NAME_UPPER_CASE}_SRCDIR}, ${${MODULE_NAME_UPPER_CASE}_DIR}, ${${MODULE_NAME_UPPER_CASE}_NAME})
+endef
+
+define ZEPHYR_PREBOOT_APP_VERSION_GEN
+	@$(call HL_VERSION_GEN, ${ZEPHYR_PREBOOT_SRCDIR},.,${ZEPHYR_PREBOOT_APP_PROD_NAME})
+	@$(call HL_VERSION_GEN, ${ZEPHYR_PREBOOT_SRCDIR},build,${ZEPHYR_PREBOOT_APP_PROD_NAME})
+	@$(foreach module,$(ZEPHYR_PREBOOT_APP_MODULES),$(call ZEPHYR_PREBOOT_MODULE_VERSION_HOOK,$(module)))
+endef
+
+define ZEPHYR_PREBOOT_CREATE_HL_VERSION
+	$(call MESSAGE, " [$(ZEPHYR_PREBOOT_BUILD_DIR)/include/hl_version.h]")
+
+	@mkdir -p $(ZEPHYR_PREBOOT_BUILD_DIR)/include
+	@echo "#ifndef _HL_VERSION_" >  $(ZEPHYR_PREBOOT_BUILD_DIR)/include/hl_version.h
+	@echo "#define _HL_VERSION_" >>  $(ZEPHYR_PREBOOT_BUILD_DIR)/include/hl_version.h
+	@echo "#define HL_VERSION_MAJOR $$(echo $(ZEPHYR_PREBOOT_LOCAL_VERSION) | cut -d. -f1)" >> $(ZEPHYR_PREBOOT_BUILD_DIR)/include/hl_version.h
+	@echo "#define HL_VERSION_MINOR $$(echo $(ZEPHYR_PREBOOT_LOCAL_VERSION) | cut -d. -f2)" >> $(ZEPHYR_PREBOOT_BUILD_DIR)/include/hl_version.h
+	@echo "#define HL_VERSION_PATCH $$(echo $(ZEPHYR_PREBOOT_LOCAL_VERSION) | cut -d. -f3)" >> $(ZEPHYR_PREBOOT_BUILD_DIR)/include/hl_version.h
+	@echo "#endif" >>  $(ZEPHYR_PREBOOT_BUILD_DIR)/include/hl_version.h
+endef
+
+define ZEPHYR_PREBOOT_CONFIGURE_CMDS
+#	$(INSTALL) -D -m 0755 $(ZEPHYR_PREBOOT_WEST_SRC_DIR)/west.config $(ZEPHYR_PREBOOT_WEST_DEST_DIR)/config
+	@$(call ZEPHYR_PREBOOT_CREATE_HL_VERSION)
+	@$(ZEPHYR_PREBOOT_CONF_ENV) $(HOST_DIR)/bin/cmake $(ZEPHYR_PREBOOT_CONF_OPTS) -S $(ZEPHYR_PREBOOT_BUILD_DIR) -B $(ZEPHYR_PREBOOT_BUILD_DIR)/build
+endef
+
+ZEPHYR_PREBOOT_POST_RSYNC_HOOKS += HL_VERSION_GEN

+
+define ZEPHYR_PREBOOT_BUILD_CMDS
+	$(TARGET_MAKE_ENV) $(MAKE) -C $(ZEPHYR_PREBOOT_BUILD_DIR)/build all
+endef
+
+define ZEPHYR_PREBOOT_INSTALL_TARGET_CMDS
+	@$(INSTALL) -D -m 0644 $(@D)/build/zephyr/zephyr.elf $(BINARIES_DIR)/zephyr_$(ZEPHYR_PREBOOT_APP_NAME).elf
+	@$(INSTALL) -D -m 0644 $(@D)/build/zephyr/zephyr.bin $(BINARIES_DIR)/$(ZEPHYR_PREBOOT_APP_NAME).bin
+endef
