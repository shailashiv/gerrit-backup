[SW-11867] preboot: add asic thermal sensors reading capability

this patch adds the infrastructure changes required to get on-chip
thermal sensors readings from pre-boot using i2c interfaces

As part of the SLIM flash layout changes it is decided to service
I2C slave requests from pre-boot and hence ASIC temperature sensors
readings should be supported from preboot.

+/* twos complement to u32 without changing result sign
+ * since the temperature formula uses the positive value of
+ * the conversion.
+ */
+static inline s32 twos_complement_to_s32(u32 val)
+{
+	return  val & 0x8000 ? ((~val & 0xFFFF) + 1) * (-1) : val;
+}
+
+/* Extract A and B coefficients from efuse
+ *
+ * A:
+ * N_bin_twos_comp = efuse_read & 0x0000FFFF
+ * N_bin = twos_comp(N_bin_twos_comp, 16) #twos complement to int
+ * A_North = (N_bin * (-1)) / 1000.0
+ * A = A + A_North
+ *
+ * B:
+ * N_bin_twos_comp = (efuse_read >> (2) * 8) & 0x0000FFFF
+ * N_bin = twos_comp(N_bin_twos_comp, 16) #twos complement to int
+ * B_North = (N_bin * (-1)) / 1000.0
+ * B = B + B_North
+ */
+static inline void hl_tmp_extract_A_B(u32 calib_data, int ts)
+{
+	s32 efuse_a_coeff = calib_data & 0xFFFF;
+	s32 efuse_b_coeff = (calib_data >> 16) & 0xFFFF;
+
+	/* convert from two's complement to int */
+	efuse_a_coeff = twos_complement_to_s32(efuse_a_coeff);
+	efuse_b_coeff = twos_complement_to_s32(efuse_b_coeff);
+
+	/* A and B extracted has to be divided by 1000,
+	 * but since defined coefficients are multiplied by 100
+	 * for accuracy purposes, the coeff extracted are divided
+	 * only by 10 to match defined coefficients resolution.
+	 */
+	hl_ts_data[ts].hl_tmp_data.coef_a_g += efuse_a_coeff / 10;
+	hl_ts_data[ts].hl_tmp_data.coef_b_h += efuse_b_coeff / 10;
+}
+

+void hl_ts_init(void)
+{
+	u32 efuse_word, ts;
+
+	hl_ts_data_init();
+
+	for (ts = 0; ts < hl_ts_get_count(); ts++) {
+
+		if (hl_ts_pvt_init(ts))
+			continue;
+
+		if (!hl_ts_data[ts].use_calib)
+			hl_set_calib_coeff(ts, false,
+						&hl_ts_data[ts].hl_tmp_data);
+
+		else {
+			preboot_fuse_read(hl_ts_data[ts].calib_offset,
+						&efuse_word);
+
+			/* Calculate new A and B coeff using efuse offsets */
+			hl_tmp_extract_A_B(efuse_word, ts);
+			hl_set_calib_coeff(ts, true,
+						&hl_ts_data[ts].hl_tmp_data);
+		}
+
+		hl_ts_alarm_set(ts);
+		hl_ts_clk_set(hl_ts_data[ts].base, hl_ts_data[ts].freq);
+		hl_ts_sdif_set(hl_ts_data[ts].base, hl_ts_data[ts].use_calib);
+
+	}
+}
+
+__attribute__((weak)) int get_sdif_data_offset(void)
+{
+	return HL_TS_SDIF_DATA_OFFSET;
+}
+
+int hl_ts_get_max_temp(void)
+{
+	int tmp, max_tmp = 0;
+	int ts_num, ts;
+
+	ts_num = hl_ts_get_count();
+	for (ts = 0; ts < ts_num; ts++) {
+		tmp = regmap_readl(hl_ts_data[ts].base +
+					get_sdif_data_offset());
+		tmp = hl_tmp_temp_reg_to_mC(tmp, ts);
+		if (max_tmp < tmp)
+			max_tmp = tmp;
+	}
+
+	return max_tmp;
+}
