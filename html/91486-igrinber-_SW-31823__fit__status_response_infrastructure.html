+static int fwupd_create_resp(void *res_fit, const int load_type,
+		const char *prop_name, const void *prop_data,
+		int *prop_data_sz)
+{
+	int err;
+	unsigned int sz = 4 * 1024;
+
+	sz = ((*prop_data_sz + FIT_RSP_MARGIN_SIZE_IN_KB) > FIT_DFL_RSP_SIZE_IN_KB) ?
+		(*prop_data_sz + FIT_RSP_MARGIN_SIZE_IN_KB) : FIT_DFL_RSP_SIZE_IN_KB;
+
+	err = fit_init_resp(res_fit, sz);
+	if (err)
+		goto fit_init_err;
+
+	err = fdt_property_u32(res_fit, HL_FIT_LOAD_TYPE_PROP, load_type);
+	if (err)
+		goto fit_init_err;
+
+	err = fdt_property(res_fit, prop_name, prop_data, *prop_data_sz);
+	if (err)
+		goto fit_init_err;
+
+	err = fit_fin_resp(res_fit, &sz);
+	if (err)
+		goto fit_init_err;
+
+	*prop_data_sz = sz;
+
+fit_init_err:
+	return err;
+}

+static void fwupd_get_proto_status(struct hl_upd_proto_status *status, int err, int serr)
+{
+	status->version = HL_UPD_PROTO_VERSION;
+	status->compat_mask = cpu_boot_dev_sts_get_all();
+	status->fw_comp = HL_FW_COMP_PREBOOT;
+	status->err = err;
+	status->sub_err = serr;
+}
+
+static int fwupd_prepare_proto_status(void *res_fit, int rsp_err, int rsp_sub_err)
+{
+	struct hl_upd_proto_status status;
+	int size = sizeof(status);
+
+	fwupd_get_proto_status(&status, rsp_err, rsp_sub_err);
+
+	return fwupd_create_resp(res_fit, FIT_LOAD_TYPE_PROTO_STATUS,
+		HL_FIT_PROTO_STATUS_PROP, &status, &size);
+}
+
+static void module_fwupd_proto_status(void)
+{
+	int err;
+	enum hl_upd_sts sts;
+	void *offset;
+
+	module_fwupd_send_status(HL_UPD_STS_ACK, 0);
+	offset = (void *)preboot_shm_get_offset();
+	err = fwupd_prepare_proto_status(offset, 0, 0);
+	if (err)
+		goto fwupd_proto_sts_err;
+	err = module_fwupd_wfc(WFC_TIMEOUT_MS);
+	if (err)
+		goto fwupd_proto_sts_wfc;
+
+	module_fwupd_send_status(HL_UPD_STS_OK, offset);
+	return;
+
+fwupd_proto_sts_wfc:
+	sts = HL_UPD_STS_TIMEOUT_ERR;
+	preboot_puts("Timeout error\n");
+
+fwupd_proto_sts_err:
+	module_fwupd_send_status(sts, 0);
+}
