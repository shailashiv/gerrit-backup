+		-I$(EMBEDDED_DIR)/zephyr-modules/fdt/libfdt \
+		-I$(EMBEDDED_DIR)/zephyr-modules/boot_info
+
+###################################################################
+
+PREBOOT_D1_APP_PROD_NAME = zephyr_arc_$(PREBOOT_D1_APP_NAME)
+PREBOOT_D1_DEVICE_NAME = $(firstword $(subst _, ,$(HL_DEV)))
+PREBOOT_D1_APP_VERSION = $(shell cut -d- -f2- $(@D)/localversion-habanalabs)
+PREBOOT_D1_LOCAL_VERSION = $$(sed "s/^.*fw-\\([0-9]\\+\\.[0-9]\\+\\.[0-9]\\+\\).*$$/\\1/" $(@D)/localversion-habanalabs)

+PREBOOT_D1_BUILD_DIR = $(@D)

+
+PREBOOT_D1_CONF_OPTS  = $(ZEPHYR_CONF_OPTS)

+PREBOOT_D1_CONF_OPTS += -DCMAKE_EXE_LINKER_FLAGS_INIT=-Wl,-e__start
+PREBOOT_D1_CONF_OPTS += -DTARGET=$(HL_TARGET)
+PREBOOT_D1_CONF_OPTS += -DEXTRA_CFLAGS="\
+		-Werror \
+		-Wno-misleading-indentation \
+		$(PREBOOT_D1_CONF_APP_EXTRA_CFLAGS)"
+PREBOOT_D1_CONF_OPTS += $(PREBOOT_D1_CONF_APP_OPTS)

+PREBOOT_D1_CONF_OPTS += -DBUILD_MACHINE="$(shell hostname -s)"
+PREBOOT_D1_CONF_OPTS += -DCMAKE_C_FLAGS="\
+		$($(call UPPERCASE,$(DEV_PREFIX))_CONF_DEV_EXTRA_CFLAGS) \
+		$(PREBOOT_D1_CONF_APP_CFLAGS)"
+
+PREBOOT_D1_CONF_OPTS += -DDTS_EXTRA_CPPFLAGS="-I$(SPECS_EMBEDDED_DIR)"
+PREBOOT_D1_CONF_OPTS += -DBOARD=hl_$(PREBOOT_D1_DEVICE_NAME)_$(PREBOOT_D1_BOARD_NAME)_$(HL_TARGET)
+PREBOOT_D1_CONF_OPTS += -DDEVICE=$(PREBOOT_D1_DEVICE_NAME)
+PREBOOT_D1_CONF_OPTS += -DAPP_NAME=$(PREBOOT_D1_APP_PROD_NAME)
+PREBOOT_D1_CONF_OPTS += -DAPP_VERSION=$(PREBOOT_D1_APP_VERSION)
+
+PREBOOT_D1_WEST_SRC_DIR :=
+PREBOOT_D1_WEST_DEST_DIR :=
+
+define PREBOOT_D1_COPY_ZEPHYR_SRC
+	rsync -au --chmod=u=rwX,go=rX  --exclude .svn --exclude .git --exclude .hg --exclude .bzr --exclude CVS $(PREBOOT_D1_ZEPHYR_SRCDIR) $(PREBOOT_D1_BUILD_DIR)
+endef
+
+# Arguments:
+# - module name
+# - component dir
+# - component name
+#
+define PREBOOT_D1_DEFINE_MODULE_ENVVARS
+	$(eval MODULE_NAME=$(strip $1))
+	$(eval MODULE_NAME_UPPER_CASE=$(shell echo $(MODULE_NAME) | tr a-z A-Z))
+	$(eval MODULE_NAME_UPPER_CASE=$(strip $(MODULE_NAME_UPPER_CASE)))
+	$(eval $(MODULE_NAME_UPPER_CASE)_SRCDIR=$(EMBEDDED_DIR)/zephyr-modules/$(MODULE_NAME))
+	$(eval $(MODULE_NAME_UPPER_CASE)_DIR=$(@D)/build/modules/$(MODULE_NAME))
+	$(eval $(MODULE_NAME_UPPER_CASE)_NAME=$2_$(MODULE_NAME))
+endef
+
+$(foreach module,$(PREBOOT_D1_APP_MODULES),$(call PREBOOT_D1_DEFINE_MODULE_ENVVARS,$(module),$(PREBOOT_D1_APP_PROD_NAME)))
+
+# Arguments:
+# - module (mbedtls, eeprom, etc.)
+define PREBOOT_D1_MODULE_VERSION_HOOK
+	$(eval MODULE_NAME_UPPER_CASE=$(shell echo $1 | tr a-z A-Z))
+	$(eval MODULE_NAME_UPPER_CASE=$(strip $(MODULE_NAME_UPPER_CASE)))
+	$(call HL_VERSION_GEN, ${${MODULE_NAME_UPPER_CASE}_SRCDIR}, ${${MODULE_NAME_UPPER_CASE}_DIR}, ${${MODULE_NAME_UPPER_CASE}_NAME})
+endef
+
+define PREBOOT_D1_APP_VERSION_GEN
+	@$(call HL_VERSION_GEN, ${PREBOOT_D1_SRCDIR},.,${PREBOOT_D1_APP_PROD_NAME})
+	@$(call HL_VERSION_GEN, ${PREBOOT_D1_SRCDIR},build,${PREBOOT_D1_APP_PROD_NAME})
+	@$(foreach module,$(PREBOOT_D1_APP_MODULES),$(call PREBOOT_D1_MODULE_VERSION_HOOK,$(module)))
+endef
+
+define PREBOOT_D1_CREATE_HL_VERSION
+	@mkdir -p $(PREBOOT_D1_BUILD_DIR)/include
+	@echo "#ifndef _HL_VERSION_" >  $(PREBOOT_D1_BUILD_DIR)/include/hl_version.h
+	@echo "#define _HL_VERSION_" >>  $(PREBOOT_D1_BUILD_DIR)/include/hl_version.h
+	@echo "#define HL_VERSION_MAJOR $$(echo $(PREBOOT_D1_LOCAL_VERSION) | cut -d. -f1)" >> $(PREBOOT_D1_BUILD_DIR)/include/hl_version.h
+	@echo "#define HL_VERSION_MINOR $$(echo $(PREBOOT_D1_LOCAL_VERSION) | cut -d. -f2)" >> $(PREBOOT_D1_BUILD_DIR)/include/hl_version.h
+	@echo "#define HL_VERSION_PATCH $$(echo $(PREBOOT_D1_LOCAL_VERSION) | cut -d. -f3)" >> $(PREBOOT_D1_BUILD_DIR)/include/hl_version.h
+	@echo "#endif" >>  $(PREBOOT_D1_BUILD_DIR)/include/hl_version.h
+endef
+
+define PREBOOT_D1_CONFIGURE_CMDS
+#	$(INSTALL) -D -m 0755 $(PREBOOT_D1_WEST_SRC_DIR)/west.config $(PREBOOT_D1_WEST_DEST_DIR)/config
+	@$(call PREBOOT_D1_CREATE_HL_VERSION)
+	@$(PREBOOT_D1_CONF_ENV) $(HOST_DIR)/bin/cmake $(PREBOOT_D1_CONF_OPTS) -S $(PREBOOT_D1_BUILD_DIR) -B $(PREBOOT_D1_BUILD_DIR)/build
+endef
+
+PREBOOT_D1_POST_RSYNC_HOOKS += HL_VERSION_GEN
+PREBOOT_D1_POST_RSYNC_HOOKS += PREBOOT_D1_COPY_ZEPHYR_SRC
+PREBOOT_D1_POST_RSYNC_HOOKS += PREBOOT_D1_APP_VERSION_GEN
+
+define PREBOOT_D1_BUILD_CMDS
+	$(TARGET_MAKE_ENV) $(MAKE) -C $(PREBOOT_D1_BUILD_DIR)/build all
+endef
+
+define PREBOOT_D1_INSTALL_TARGET_CMDS
+	@$(INSTALL) -D -m 0644 $(@D)/build/zephyr/zephyr.elf $(BINARIES_DIR)/zephyr_$(PREBOOT_D1_APP_NAME).elf
+	@$(INSTALL) -D -m 0644 $(@D)/build/zephyr/zephyr.bin $(BINARIES_DIR)/$(PREBOOT_D1_APP_NAME).bin
+endef

+	rm -f $(PREBOOT_D1_BUILD_DIR)/.stamp_{configured,built}

+$(eval $(cmake-package))
