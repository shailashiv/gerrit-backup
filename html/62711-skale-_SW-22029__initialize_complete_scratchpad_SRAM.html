Commit Message

[SW-22029] initialize complete scratchpad SRAM

Uninitialized reads to scratchpad SRAM causes fake ECC error when byte
reads are performed to scratchpad SRAM which turns out to be
read-modify-write transactions. Since arcboot uses scratchpad SRAM to
store and retrieve key hash, revoke bits, firmware version and efuse
values, this patch add changes to initialize complete scratchpad SRAM
region of 64KB using memcopy block as part of arcboot init to avoid any
fake ECC errors.

The total cycles taken by memcopy block to initialize complete
scratchpad SRAM are ~53350 cycles (which is ~105 microseconds with ARC
running @ 500 MHz).

+void arcboot_rom_init_sp_sram(void)
+{
+	uint32_t mem_cfg, cpy_sts;
+	uint8_t retry = HL_INIT_SP_SRAM_RETRIES;
+
+	/* Use memblock to initialize scratchpad memory */
+	regmap_writel((uint32_t)HL_SP_SRAM_BASE,
+			mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_L);
+	regmap_writel((uint32_t)(HL_SP_SRAM_BASE >> 32),
+			mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_H);
+
+	mem_cfg = regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+	mem_cfg &= ~(PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_FLD_SIZE_MASK);
+	/* set size of 64k in granularity of 16 bytes */
+	mem_cfg |= (SP_SRAM_SIZE_16B_GRANULARITY &
+			PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_FLD_SIZE_MASK);
+	mem_cfg |= (0x1 << PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_USE_CONST_SHIFT);
+
+	regmap_writel(mem_cfg, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+
+	/* Enable copy and wait for completion */
+	regmap_writel(0x1, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL);
+	while (1) {
+		cpy_sts = regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_STATUS);
+		if (cpy_sts)
+			break;
+		/*
+		 * Do not wait indefinitely on copy status
+		 * Wait for 1 milliseconds (100 retries with 10 micro-
+		 * seconds sleep), else bailout.
+		 */
+		retry--;
+		if (!retry)
+			break;
+
+		udelay(HL_INIT_SP_SRAM_UDELAY);
+	}
+}
