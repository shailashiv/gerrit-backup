+function hl_pldm_hex {
+	NAME=$(echo "$1" | cut -f 1 -d '.')
+	FILE=$2
+	BYTES_PER_LINE=$3
+	# pad file to 128bit (16Byte) as pladium emulates 128bit memory, so xxd wont trunc lines
+	dd status=none conv=sync bs=16 if=${FILE} > ${FILE}.tmp
+	xxd -e -g ${BYTES_PER_LINE} ${FILE}.tmp | xxd -r | xxd -ps -c${BYTES_PER_LINE} > ${DEV_PREFIX}-${HL_TARGET}-${NAME}.hex
+	rm ${FILE}.tmp
+}
+
+function hl_pldm_flash_hex {
+	NAME=$(echo "$1" | cut -f 1 -d '.')
+	FILE=$2
+	# save hex as a 1 byte list of values
+	xxd -e -g 4 ${FILE} | xxd -r | xxd -ps -c1			> ${DEV_PREFIX}-${HL_TARGET}-${NAME}.hex
