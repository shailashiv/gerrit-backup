Commit Message

[SW-17444] preboot: goya2_arc: add svn validation

This patch adds support to verify minimum allowed security version
number as part of secure boot flow.

As part of ARC based secure boot flow, arcboot collects all the
information such as svn, revoke bits and key hash from efuse, otp
or flash and copies it at start of scratchpad SRAM. preboot while
performing image verification re-use the above information stored
in scratchpad SRAM by using proper offset shared between arcboot
and preboot

+static u32 sp_sram_readl(u32 offset)
+{
+	u32 reg = *((volatile u32 *) (HL_SCRATCHPAD_SRAM_BASE + offset));
+	return reg;
+}
+
+bool preboot_security_svn_is_valid(int sec_ver)
+{
+	u8 sec_fw_ver, sec_valid_fw_ver;
+	u32 valid_fw_ver;
+
+	sec_fw_ver = SECURITY_FW_VERSION_TO_SECURITY_BYTE(sec_ver);
+	valid_fw_ver = sp_sram_readl(SECURITY_EFUSE_FW_VERSION_OFFSET);
+	sec_valid_fw_ver = SECURITY_FW_VERSION_TO_SECURITY_BYTE(valid_fw_ver);
+
+	if (sec_fw_ver < sec_valid_fw_ver)
+		preboot_puts_hex("Security version too low: ", sec_ver, 0);
+
+	return sec_fw_ver >= sec_valid_fw_ver;
+}
