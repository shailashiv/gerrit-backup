+void sha_csum(const unsigned char *input, unsigned int ilen,
+		unsigned char *output)
+{
+	sha384_csum(input, ilen, output, 0);
+}
+
+void sha_csum_regions(const struct image_region region[], int region_count,
+			 u8 *output)
+{
+	struct sha512_context ctx;
+	int i;
+
+	sha384_starts(&ctx);
+	for (i = 0; i < region_count; i++)
+		sha384_update(&ctx, region[i].data, region[i].size);
+	sha384_finish(&ctx, output);
+}
