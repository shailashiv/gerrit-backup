+#include <pcie_aux_regs.h>
+#include <pcie_core_regs.h>
+
+#include <preboot.h>
+
+#include "cmd_module.h"
+#include "uart.h"
+
+struct pci_status {
+	u32 flr_reg_val;
+	u32 hot_rst_reg_val;
+};
+
+static struct pci_status pci_sts = {
+	.flr_reg_val = ~1,
+	.hot_rst_reg_val = ~1,
+};
+
+enum cmd_check_res module_pci_check(void)
+{
+	pci_sts.flr_reg_val = regmap_readl(mmPCIE_AUX_FLR_INT);
+	pci_sts.hot_rst_reg_val = regmap_readl(mmPCIE_CORE_HOT_RST) & BIT(0);
+
+	if (pci_sts.flr_reg_val == 1 || pci_sts.hot_rst_reg_val == 1)
+		return FOUND_FAST_RETURN;
+
+	return NOT_FOUND;
+}
+
+void module_pci_run(void)
+{
+	if (pci_sts.flr_reg_val == 1) {
+		regmap_writel(1, mmPCIE_AUX_PF_FLR_ACTIVE);
+		regmap_writel(1, mmPCIE_AUX_FLR_INT);
+		regmap_writel(1, mmPCIE_AUX_PF_FLR_DONE);
+		preboot_puts("FLR received\n");
+	}
+
+	if (pci_sts.hot_rst_reg_val == 1) {
+		regmap_writel(0x10, mmPCIE_CORE_HOT_RST);
+		regmap_writel(1, mmPCIE_AUX_LTSSM_EN);
+		preboot_puts("HOT RST received\n");
+	}
+}
