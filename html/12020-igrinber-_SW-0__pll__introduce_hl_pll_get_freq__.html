[SW-0] pll: introduce hl_pll_get_freq()

The hl_pll_get_freq() function calculates and returns the output
frequency of the requested PLL divider output.
The calculation takes into an account the DIV_SEL configuration
(both the reference clock and other options) as well as the PLL
coefficients.

+u32 hl_pll_get_freq(u64 pll_base, enum hl_pll_div_out div)
+{
+	u32 nr, nf, od, div_sel, factor, offset;
+	u64 out_freq = SYS_REF_CLOCK;
+
+	offset = div * sizeof(u32);
+	div_sel = regmap_readl(pll_base + HL_PLL_DIV_SEL + offset);
+
+	if (div_sel == HL_PLL_PLL_CLK || div_sel == HL_PLL_DIV_PLL) {
+		nr = regmap_readl(pll_base + HL_PLL_NR) + 1;
+		nf = regmap_readl(pll_base + HL_PLL_NF) + 1;
+		od = regmap_readl(pll_base + HL_PLL_OD) + 1;
+		out_freq = (SYS_REF_CLOCK * nf) / (nr * od);
+	}
+
+	if (div_sel == HL_PLL_DIV_REF || div_sel == HL_PLL_DIV_PLL) {
+		factor = regmap_readl(pll_base + HL_PLL_DIV_FACTOR + offset);
+		out_freq /= factor + 1;
+	}
+
+	/*
+	 * The clock relaxation is not accounted here.
+	 * We might need to deal with it in the future.
+	 */
+
+	return out_freq;
+}
