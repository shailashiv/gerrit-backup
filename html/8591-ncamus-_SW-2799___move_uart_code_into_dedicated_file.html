+/*

+#include <linux/io.h>
+#include <mach/hl_regs.h>
+
+#define mmPSOC_UART_0_DATA		(HL_CFG_BASE + 0xC45000)
+#define mmPSOC_UART_0_STATUS		(HL_CFG_BASE + 0xC45004)
+#define UART_SR_TXFULL			(1 << 0) /* TX Buffer Full */
+#define STR_UINT_SZ			8
+
+#ifdef DEV_IS_GOYA
+static void preboot_putc_virt_uart(const char c)
+{
+	/*
+	 * update vUART to HOST
+	 * DATA: mmPCIE_DB_RSV_RSV_0
+	 * STATUS: mmPCIE_DB_RSV_RSV_1
+	 */
+	writel(c, mmPCIE_DB_RSV_RSV_0);
+	writel(1, mmPCIE_DB_RSV_RSV_1);
+}
+#else
+static inline void preboot_putc_virt_uart(const char c) {}
+#endif
+
+void preboot_puts_raw(const char* str)
+{
+	int i;
+
+	for (i=0; str[i] && (i < 128); i++) {
+		while (readl(mmPSOC_UART_0_STATUS) & UART_SR_TXFULL) ;
+		writel(str[i], mmPSOC_UART_0_DATA);
+		preboot_putc_virt_uart(str[i]);
+	}
+}
+
+void preboot_puts(const char* str)
+{
+	preboot_puts_raw(str);
+	preboot_puts_raw("\r");
+}
+
+/*
+ * preboot_byte2char	- convert half byte to char.
+ *
+ * This function takes a byte number and converts its value to
+ * a character representing the value (e.g. 0x8c can be converted to
+ * '8' or 'c').
+ */
+static inline char preboot_byte2char(u8 byte, bool upper)
+{
+	/* upper or lower 4 bits of the byte */
+	byte = upper ? byte >> 4 : byte & 0xf;
+
+	/*
+	 * convert: if half byte is a number [0..9] => use '0' as an offset
+	 * 	    if half byte is a letter [0xa..0xf] => use 'a' as a base,
+	 * 	    but since 'a' is not 0, we need to subtract
+	 * 	    its base offset - 0xa.
+	 */
+	return byte + ((byte >= 0xa) ? 'a' - 0xa : '0');
+}
+
+/*
+ * preboot_uint2hexstring	- convert uint to a hex values string.
+ *
+ * Take a number and provide its hexadecimal string representation.
+ */
+int preboot_uint2hexstring(char *str, unsigned int num, bool pad)
+{
+	int j = 0;
+	char tmp;
+
+	/*
+	 * Here we walk on each byte of uint num in the reverse order, so
+	 * we build the string in the correct order.
+	 */
+	for (int i = 3; i >= 0; i--) {
+		u8 byte = (u8)((num >> (i * 8)) & 0xff);
+		/* here we run the loop on 2 parts of byte: upper and lower */
+		for (int k = 0; k < 2; k++) {
+			tmp = preboot_byte2char(byte, !k);
+			if (tmp == '0')	/* care of the leading zeroes (pad) */
+				pad = true;
+			if (pad)
+				str[j++] = tmp;
+		}
+	}
+
+	/* for !pad, return how many characters were actually written */
+	return j;
+}
+
+void preboot_puth_raw(unsigned int val, bool pad)
+{
+	char str[STR_UINT_SZ + 1];
+	int val_sz;
+
+	val_sz = preboot_uint2hexstring(str, val, pad);
+	val_sz = val_sz > STR_UINT_SZ ? STR_UINT_SZ : val_sz;
+	str[val_sz] = '\0';
+	preboot_puts_raw(str);
+}
+
+void preboot_puth(unsigned int val, bool pad)
+{
+	preboot_puth_raw(val, pad);
+	preboot_puts("\n");
+}
+
+static inline void preboot_puts_hex(const char* log, unsigned int val, bool pad)
+{
+	preboot_puts_raw(log);
+	preboot_puts_raw("0x");
+	preboot_puth(val, pad);
+}
+
+static inline void preboot_puts_char(const char* str, unsigned char val)
+{
+	preboot_puts_hex(str, val, false);
+}
