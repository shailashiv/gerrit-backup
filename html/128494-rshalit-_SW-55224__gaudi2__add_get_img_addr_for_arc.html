Commit Message

[SW-55224] gaudi2: add get_img_addr for arc

Currently, ARC is using lkd_dev_spec_get_img_addr from the weak
function, which is actually not a correct implementation for ARC.
This patch fix it by adding lkd_dev_spec_get_img_addr specific
to ARC. This function is converting the base address
from ARC view address to AXI (absolute) address. Base address
can reside in spsram or in sram (default option).

+#define ARC_REL_ADDR_MASK	0x000000000FFFFFFFULL
+#define HL_AXI_GENERAL_BASE	0x1000007FF0000000ULL
+
+u64 lkd_dev_spec_get_img_addr(uintptr_t base)
+{
+	uintptr_t base_offset = base & ARC_REL_ADDR_MASK; /* offset in general region */
+
+	/* SPSRAM fit can only be in general region */
+	if ((base >= SP_SRAM_BASE) && (base < SP_SRAM_BASE + HL_SP_SRAM_SIZE))
+		return base_offset | HL_AXI_GENERAL_BASE;
+
+	return base_offset | SDRAM_BASE_FULL;
+}
