+#define MEM_CPY_16B_DIVIDER		4
+void init_arc_mem(u8 id, u64 addr, u32 size)
+{
+	u32 mem_cfg;
+	u64 start;
+
+	/* Use memblock to initialize dccm memory */
+	regmap_writel((u32)addr, mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_L);
+	regmap_writel((u32)(addr >> 32), mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_H);
+
+	mem_cfg = regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+	mem_cfg &= ~(PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_FLD_SIZE_MASK);
+
+	/* set size in granuality of 16 bytes */
+	size = size >> MEM_CPY_16B_DIVIDER;
+	mem_cfg |= (size & PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_FLD_SIZE_MASK);
+	mem_cfg |= (0x1 << PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_USE_CONST_SHIFT);
+
+	regmap_writel(mem_cfg, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+
+	/* Enable copy and wait for completion */
+	regmap_writel(0x1, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL);
+
+	start = timestamp_get_us();
+	while (!(regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_STATUS))) {
+		if (timeout_expired(start, ARC_MEM_INIT_TIMEOUT_US)) {
+			preboot_puts_hex("Init memory timed-out for arc", id, false);
+			return;
+		}
+	}
+}
