+/* Number of DWORDS available in PSOC registers to store security key hash */
+#define NUM_OF_DW_AVLBL_IN_PSOC_FOR_SEC_HASH	8
+static void preboot_sec_copy_otp_from_rom(struct hl_sec_otp *otp)
+{
+	u32 *dword;
+	u32 key_soc_location_base;
+	int i, j;
+
+	for (i = 0; i < SEC_KEY_HASH_COUNT_IN_ROM; i++) {
+		key_soc_location_base = sec_key_address[i];
+		for (j = 0; j < NUM_OF_DW_AVLBL_IN_PSOC_FOR_SEC_HASH; j++) {
+			dword = (u32 *)&otp->keys[j * sizeof(*dword) + i * SHA_SUM_LEN];
+			*dword = regmap_readl(key_soc_location_base);
+			key_soc_location_base += sizeof(*dword);
+		}
+	}
+}
+
+static int preboot_sec_copy_otp_from_efuse(struct hl_sec_otp *otp)
+{
+	u32 *dword;
+	int i, j;
+
+	for (i = 0; i < SEC_KEY_HASH_COUNT_IN_EFUSE; i++) {
+		for (j = 0; j < SECURITY_DWORD_PER_KEY; j++) {
+			dword = (u32 *)&otp->keys[j * sizeof(*dword) + i * SHA_SUM_LEN];
+			if (preboot_fuse_read(EFUSE_SEC_KEYS_WORD_OFFSET + j +
+						i * SECURITY_DWORD_PER_KEY, dword)) {
+				preboot_puts("fuse read failed\n");
+				return 1;
+			}
+		}
+	}
+	return 0;
+}
+
+static void preboot_security_copy_otp(struct hl_sec_otp *otp)
+{
+	const struct hl_sec_efuse *fuse;
+	int num_of_keys;
+
+	fuse = preboot_get_hl_sec_real_efuse();
+	if (fuse == NULL)
+		return;
+
+	if (fuse->keys_loc_selector == SEC_KEY_HASH_LOCATION_EFUSE) {
+		num_of_keys = SEC_KEY_HASH_COUNT_IN_EFUSE;
+		if (preboot_sec_copy_otp_from_efuse(otp))
+			return;
+
+	} else {
+		num_of_keys = SEC_KEY_HASH_COUNT_IN_ROM;
+		preboot_sec_copy_otp_from_rom(otp);
+	}
+
+	otp->magic_number = OTP_SEC_MAGIC;
+	otp->num_of_keys = num_of_keys;
+	otp->sha_rsa_type = SHA_384_RSA_3072;
+}
+
+static const struct hl_sec_otp *preboot_security_get_otp(void)
+{
+	struct hl_sec_otp *otp = &preboot_sec_otp_cache.otp;
+
+	if (preboot_sec_otp_cache.is_invalid) {
+		if (preboot_fake_security_is_enabled())
+			preboot_security_copy_fake_otp(otp);
+		else
+			preboot_security_copy_otp(otp);
+
+		if (otp->magic_number != OTP_SEC_MAGIC)
+			return NULL;
+
+		preboot_sec_otp_cache.is_invalid = false;
+	}
+
+	return otp;
+}
