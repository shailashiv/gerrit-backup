+#define FUSE_VER_PREFIX		"Fuse version "
+#define FUSE_VER_UNKNOWN	"unknown"
+static char preboot_fuse_ver_str[VERSION_MAX_LEN] = FUSE_VER_PREFIX;
+
+__weak int hl_fuse_get_devid(char *devid)
+{
+	return 1;
+}
+
+static void fuse_version(void)
+{
+	unsigned long offset = (unsigned long)preboot_fuse_ver_str;
+	char *devid = preboot_fuse_ver_str + sizeof(FUSE_VER_PREFIX) - 1;
+	const char *unknown = FUSE_VER_UNKNOWN;
+	int err, i;
+
+	err = hl_fuse_get_devid(devid);
+	if (err)
+		for (i = 0; i < sizeof(FUSE_VER_UNKNOWN); i++)
+			devid[i] = unknown[i];
+
+	preboot_puts_raw(preboot_fuse_ver_str);
+	preboot_puts("\n");
+
+	regmap_writel((u32)(offset & ~SDRAM_BASE), mmFUSE_VER_OFFSET);
+}
