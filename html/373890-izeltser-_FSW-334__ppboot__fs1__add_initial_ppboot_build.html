+u32 hl_spi_baudr_get(void)
+{
+	u32 factor, psoc_div3_freq;
+
+	psoc_div3_freq = hl_pll_get_freq(mmIO_0_CFG_HALF_0_SOC_PLL_PCIE_CTRL_BASE, HL_PLL_DIV3);
+	factor = psoc_div3_freq / HL_SPI_FLASH_MAX_FREQ;
+
+	/* baud rate doesn't support odd number and must not be 0 */
+	factor = (factor < 2) ? 2 : factor;
+
+	return (factor + 1) & 0xFFFE;
+}
+
+u32 hl_skip_pcie_init(void)
+{
+	return 0;
+}
+
+void hl_uart_init(void)
+{
+	/* As per Gaudi3 Frequencies.xlsx, PSOC PLL Div[3] drives uart */
+	u32 uart_clk_freq = hl_pll_get_freq(mmIO_0_CFG_HALF_0_SOC_PLL_PSOC_CTRL_BASE, HL_PLL_DIV3);
+
+	regmap_writel(1, mmIO_0_CFG_HALF_0_PSOC_PSOC_CFG_PAD_CONF_BASE +
+			 mmPSOC_CFG_PAD_CONF_PAD_SEL_49);			/* UART0 RXD */
+	regmap_writel(1, mmIO_0_CFG_HALF_0_PSOC_PSOC_CFG_PAD_CONF_BASE +
+			 mmPSOC_CFG_PAD_CONF_PAD_SEL_50);			/* UART0 TXD */
+	preboot_uart_init(uart_clk_freq / HL_UART_BAUDRATE);
