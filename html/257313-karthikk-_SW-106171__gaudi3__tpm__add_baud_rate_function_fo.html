+#define HL_SPI_FLASH_MAX_FREQ		(SYS_REF_CLOCK / 2) /* 50MHz/2 = 25MHz*/
+
+uint64_t hl_spi_baudr_get(void)
+{
+	uint32_t factor;
+	uint64_t pcie_div3_freq;
+
+	pcie_div3_freq = get_pcie_pll_freq(PLL_DIV3);
+	factor = pcie_div3_freq / HL_SPI_FLASH_MAX_FREQ;
+
+	/* baud rate doesn't support odd number and must not be 0 */
+	factor = (factor < 2) ? 2 : factor;
+
+	return (factor + 1) & 0xFFFE;
+}
