+#include <preboot.h>
+#include <hl.h>
+#include <d0_pcie_wrap_regs.h>
+#include <d0_pcie_core_regs.h>
+#include <d0_pcie_aux_regs.h>
+#include <d0_pcie_elbi_rr_mstr_if_rr_lbw_regs.h>
+
+void hl_pcie_enable_access(void)
+{
+	unsigned int rst_src;
+
+	/* FSM blocks all range hence openning here */
+	regmap_writel(0, mmD0_PCIE_ELBI_RR_MSTR_IF_RR_LBW_PRIV_RANGE_EN_SHORT_13);
+	rst_src = hl_rst_src_get();
+
+	if ((rst_src == HL_FLR_RST) || (rst_src == HL_PRSTN_RST))
+		regmap_writel(0, mmD0_PCIE_CORE_REQ_RETRY_EN);
+
+	/* Enable downstream access */
+	regmap_writel(0, mmD0_PCIE_AUX_HOST_TERM_CTRL);
+	/* Enable PCIe outbound memory access  */
+	regmap_writel(0, mmD0_PCIE_WRAP_MASK_REQ);
+}
