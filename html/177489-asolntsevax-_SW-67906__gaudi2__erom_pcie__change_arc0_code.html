+static void pcie_config(void)
+{
+	uint32_t boot_strap, link_ctrl;
+	uint8_t ltssm_en;
+	u32 config_fuse = sp_sram_readl(EFUSE_SYSCFG_OFFSET);
+
+	pcie_loop_while_prstn(PCIE_PRSTN_ASSERT);
+	ltssm_en = regmap_readl(mmPCIE_AUX_LTSSM_EN) &
+				PCIE_AUX_LTSSM_EN_IND_MASK;
+
+	if (!ltssm_en) {
+		regmap_writel(0, mmPSOC_GLOBAL_CONF_PCIE_PHY_RST_N);
+
+		pcie_phy_para_sel_set();
+
+		/* gpio bootstrap pin gpio17 = 0 */
+		boot_strap = regmap_readl(mmPSOC_GLOBAL_CONF_BOOT_STRAP_PINS_L);
+
+		if (get_sris_mode_from_strap()) {
+			regmap_writel(1, mmPCIE_PHY_PIPE_RX_SRIS_MODE_EN);
+			regmap_writel(1, mmPCIE_AUX_APP_SRIS_MODE);
+		}
+
+		pcie_wa_snps_case_01038891(boot_strap);
+		regmap_writel(1, mmPSOC_GLOBAL_CONF_PCIE_PHY_RST_N);
+
+		pcie_wait_for_sram_init_done();
+
+		pcie_phy_set_sram_ext_ld_done();
+
+		if ((config_fuse & EFUSE_SYSCFG_PCIE_GEN_LIMIT) == 0) { /* this if */
+			/* limit PCIe link gen to 1 */
+			link_ctrl = regmap_readl(mmPCIE_DBI_LINK_CAPABILITIES_REG);
+			link_ctrl &=
+				~PCIE_DBI_LINK_CAPABILITIES_REG_PCIE_CAP_MAX_LINK_SPEED_MASK;
+			link_ctrl |=
+				(1 << PCIE_DBI_LINK_CAPABILITIES_REG_PCIE_CAP_MAX_LINK_SPEED_SHIFT);
+			regmap_writel(link_ctrl, mmPCIE_DBI_LINK_CAPABILITIES_REG);
+
+			link_ctrl = regmap_readl(mmPCIE_DBI_LINK_CONTROL2_LINK_STATUS2_REG);
+			link_ctrl &=
+				~PCIE_DBI_LINK_CONTROL2_LINK_STATUS2_REG_PCIE_CAP_TARGET_LINK_SPEED_MASK;
+			link_ctrl |= (1 <<
+				PCIE_DBI_LINK_CONTROL2_LINK_STATUS2_REG_PCIE_CAP_TARGET_LINK_SPEED_SHIFT);
+			regmap_writel(link_ctrl, mmPCIE_DBI_LINK_CONTROL2_LINK_STATUS2_REG);
+		}
+
+		/* Configure IATU */
+		pcie_config_iatu();
+
+		/* enable flr handling by HW state machine */
+		enable_flr_handling_fsm();
+
+		/* disable HOT reset handling by HW state machine */
+		regmap_writel(PCIE_WRAP_CPU_HOT_RST_CTRL_MASK,
+			      mmPCIE_WRAP_CPU_HOT_RST);
+
+		regmap_writel(0x3, mmPCIE_AUX_LTSSM_EN);
+	}
