spsram memory will be temporary used as storage.
first 4k of the spsram are left for the boot info.
Considering overall limited functionality, at the moment no data
overlapping is expected.
After SRAM memory init is implemented, memory regions should be
placed in the SRAM.

+/*
+ * due to the complexity of SRAM initialization, for phase1 of testing
+ * spsram memory will be temporary used as storage.
+ * first 4k of the spsram are left for the boot info.
+ * due to overall limited functionality, at the moment no data overlapping
+ * is expected.
+ * TODO: after SRAM memory init is implemented, return the memory regions to
+ * SRAM.
+ */
+
+#define GAUDI3_TMP_MEM		(SP_SRAM_BASE + SZ_4K)
+#define GAUDI3_TMP_MEM_SZ	(HL_SP_SRAM_SIZE - SZ_4K)
+
+static struct mem_desc mem_desc[MEM_NOENT] = {
+	[MEM_GP] = {
+		.base = GAUDI3_TMP_MEM,
+		.size = GAUDI3_TMP_MEM_SZ,
+	},
+	[MEM_LKD_PCI] = {
+		.base = GAUDI3_TMP_MEM,
+		.size = GAUDI3_TMP_MEM_SZ,
+	},
+	[MEM_FWUPD_PCI] = {
+		.base = GAUDI3_TMP_MEM,
+		.size = GAUDI3_TMP_MEM_SZ,
+	},
+	[MEM_I2C] = {
+		.base = GAUDI3_TMP_MEM,
+		.size = GAUDI3_TMP_MEM_SZ,
+	},
+};
+
+int preboot_get_dev_spec_mem_region(enum mem_region region, struct mem_desc *desc)
+{
+	if (region >= MEM_NOENT) {
+		preboot_puts_hex("unknown region = ", region, true);
+		return 1;
+	}
+
+	if (!desc)
+		return 1;
+
+	desc->base = mem_desc[region].base;
+	desc->size = mem_desc[region].size;
+
+	return 0;
+}
+
+bool preboot_is_valid_address_range(u64 addr, int len)
+{
+	if (addr >= SP_SRAM_BASE &&
+	    addr + len <= SP_SRAM_BASE + HL_SP_SRAM_SIZE)
+		return true;
+
+	if (addr >= HL_SP_SRAM_BASE &&
+	    addr + len <= HL_SP_SRAM_BASE + HL_SP_SRAM_SIZE)
+		return true;
+
+	if (addr >= ARC0_ICCM_FULL_BASE &&
+	    addr + len <= ARC0_ICCM_FULL_BASE + ARC0_MEM_SIZE)
+		return true;
+
+	if (addr >= ARC1_ICCM_FULL_BASE &&
+	    addr + len <= ARC1_ICCM_FULL_BASE + ARC1_MEM_SIZE)
+		return true;
+
+	preboot_puts_hex64("ERROR: invalid address range addr = ", addr, true);
+
+	return false;
+}
