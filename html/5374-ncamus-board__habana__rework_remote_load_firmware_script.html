Commit Message

board: habana: rework remote_load_firmware script

Current script brings the whole npu-stack repositories and builds the kmd out of it.
This is time consuming and prone to repo corruptions.

Rework the remote_load_firmware script to use the prebuilt deb packages which are
the ones later released to customers.
Bring a new script (local_install_packages) to the system which is copied to
the host machine and takes care of the packages installation.

+#!/bin/bash -l
+
+pkg_dir="$HOME/habana-pkgs"
+dl_host_base="http://jenkins:8080/view/deployment/job"
+dl_suff="lastSuccessfulBuild/artifact/*zip*"	# the '*' are not for expansion
+dl_file="archive.zip"
+pre_pkg_lst="protobuf-compiler libprotoc-dev"
+habana_pkg_lst="habanalabs habanatools libhabana"
+
+function local_msg() {
+	local msg=$1
+	echo -e   "********************************************************"
+	echo -e   "$msg"
+	echo -e   "********************************************************"
+}
+
+function local_prepare_work_dirs() {
+	local_msg "****** Preparing directories for packages download *****"
+	mkdir -p $pkg_dir;
+	sudo umount -lrf $pkg_dir && sleep 1;
+	sudo mount -t tmpfs -o uid=$UID,dirsync,mode=0755 tmpfs $pkg_dir	|| return 1;
+}
+
+function local_dl_prep_packages() {
+	local_msg "********** Downloading and preparing packages **********"
+	cd $pkg_dir
+	wget -N ${dl_host_base}/build-habana-py/${dl_suff}/${dl_file}	|| return 1;
+	unzip -j ${dl_file}						|| return 1;
+	cd -
+	return 0;
+}
+
+function local_remove_packages() {
+	local_msg "************* Removing old habana packages *************"
+	sudo apt-get purge --auto-remove habanalabs habanatools libhabana -y
+	sudo apt-get clean
+}
+
+function local_install_packages() {
+	local_msg "******** Installing habana python prerequisites ********"
+	sudo apt-get install ${pre_pkg_lst} -y	|| return 1;
+	local_msg "************ Installing new habana packages ************"
+	sudo apt update				|| return 1;
+	sudo apt install ${habana_pkg_lst} -y	|| return 1;
+}
+
+local_prepare_work_dirs
+if [ $? -ne 0 ]; then
+	local_msg "********* Failed preparing tmpfs for packages **********"
+	exit 1;
+fi
+
+local_dl_prep_packages
+if [ $? -ne 0 ]; then
+	local_msg "************* Failed downloading packages **************"
+	exit 2;
+fi
+
+local_remove_packages
+local_install_packages
+if [ $? -ne 0 ]; then
+	local_msg "************* Failed installing packages ***************"
+	local_remove_packages
+	exit 3;
+fi
