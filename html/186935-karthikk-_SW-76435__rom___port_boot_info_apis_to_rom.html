+// SPDX-License-Identifier:     GPL-2.0+

+#include <string.h>
+#include <stdbool.h>
+#include <arcboot.h>
+#include <hl_boot_desc.h>
+#include <common.h>
+#include <crc32.h>
+
+#include "boot_info.h"
+
+#define BOOT_INFO_BASE		(HL_GEN_MEM_BASE + HL_SCRATCHPAD_SRAM_OFFSET)
+#define ROUND_UP4(n)		(((n) + sizeof(uint32_t) - 1) & ~(sizeof(uint32_t) - 1))
+
+static struct boot_info boot_info = { .ptrs = { .count = 0 } };
+
+/* API to export for ppboot/preboot to read boot info base address */
+uint32_t boot_info_get_base_addr(void)
+{
+	return BOOT_INFO_BASE;
+}
+
+static inline void *offset2ptr(uint32_t offset, uintptr_t base)
+{
+	return (void *)(base + offset);
+}
+
+static inline uint32_t ptr2offset(const void *ptr, uintptr_t base)
+{
+	return (uint32_t)((uintptr_t)ptr) - base;
+}
+
+static struct hl_boot_info_desc *boot_info_get_desc(uint32_t type)
+{
+	uint32_t i;
+
+	for (i = 0; i < boot_info.ptrs.count; i++)
+		if (boot_info.descs[i].type == type)
+			return &boot_info.descs[i];
+
+	return NULL;
+}
+
+static void create_crc32_csum(const unsigned char *input, unsigned int ilen,
+			    struct hl_boot_info_desc *desc)
+{
+	uint32_t crc32_res;
+
+	crc32_res = crc32(0, input, ilen);
+	memcpy(&desc->crc32, &crc32_res, sizeof(crc32_res));
+}
+
+static int validate_crc32_csum(const unsigned char *input,
+			     unsigned int ilen,
+			     struct hl_boot_info_desc *desc)
+{
+	uint32_t crc32_res;
+
+	crc32_res = crc32(0, input, ilen);
+	return memcmp(&crc32_res, &desc->crc32, CRC32_NUM_BYTES);
+}
+
+void boot_info_finalize(void)
+{
+	struct hl_boot_info_desc *desc;
+	uint32_t offsets[MAX_INFO_DESCS];
+	uint32_t offset, copy_sz;
+	const uint32_t *src;
+	uint32_t *dst;
+	uint32_t i, j;
+	struct hl_boot_info_desc descs[MAX_INFO_DESCS];
+
+	if (!boot_info.ptrs.count)
+		return;
+
+	boot_info.ptrs.magic = HL_BOOT_INFO_MAGIC;
+	boot_info.ptrs.desc_ver = hl_get_boot_info_ver();
+	boot_info.ptrs.desc_sz = sizeof(struct hl_boot_info_desc);
+
+	/* Data offset */
+	offset = sizeof(struct hl_boot_info_ptrs) +
+		(sizeof(struct hl_boot_info_desc) + sizeof(uint32_t)) *
+		boot_info.ptrs.count;
+
+	dst = offset2ptr(offset, BOOT_INFO_BASE);
+
+	memcpy(descs, boot_info.descs, sizeof(descs));
+	/* Store data */
+	for (i = 0; i < boot_info.ptrs.count; i++) {
+		desc = &descs[i];
+		src = offset2ptr(desc->offset, 0);
+		create_crc32_csum((uint8_t *)src, desc->size, desc);
+		copy_sz = ROUND_UP4(desc->size);
+
+		for (j = 0; j < copy_sz / sizeof(uint32_t); j++)
+			dst[j] = src[j];
+
+		desc->offset = ptr2offset(dst, BOOT_INFO_BASE);
+		dst += j;
+	}
+
+	/* Store descriptors */
+	offset = sizeof(struct hl_boot_info_ptrs) +
+		(sizeof(uint32_t)) * boot_info.ptrs.count;
+	dst = offset2ptr(offset, BOOT_INFO_BASE);
+	for (i = 0; i < boot_info.ptrs.count; i++) {
+		src = (uint32_t *)&descs[i];
+		copy_sz = ROUND_UP4(sizeof(struct hl_boot_info_desc));
+
+		for (j = 0; j < copy_sz / sizeof(uint32_t); j++)
+			dst[j] = src[j];
+
+		offsets[i] = ptr2offset(dst, BOOT_INFO_BASE);
+		dst += j;
+	}
+
+	/* Store offsets array */
+	offset = sizeof(struct hl_boot_info_ptrs);
+	dst = offset2ptr(offset, BOOT_INFO_BASE);
+	for (i = 0; i < boot_info.ptrs.count; i++)
+		dst[i] = offsets[i];
+
+	/* Store the ptr */
+	dst = offset2ptr(0, BOOT_INFO_BASE);
+	for (i = 0; i < sizeof(boot_info.ptrs) / sizeof(uint32_t); i++)
+		dst[i] = boot_info.data[i];
+}
+
+int boot_info_set(uint32_t type, const void *data, uint32_t size)
+{
+	struct hl_boot_info_desc *desc;
+
+	desc = boot_info_get_desc(type);
+	if (!desc) {
+		if (boot_info.ptrs.count >= MAX_INFO_DESCS) {
+			return 1;
+		}
+
+		desc = &boot_info.descs[boot_info.ptrs.count];
+		boot_info.ptrs.count++;
+	}
+
+	desc->type = type;
+	desc->size = size;
+	desc->magic = HL_BOOT_INFO_DESC_MAGIC | type;
+	desc->offset = ptr2offset(data, 0);
+
+	return 0;
+}
+
+/*
+ * Apply memory copy with 32bit access limitation from src, make sure any
+ * reminder is also read using 32bit access.
+ * Both src and dst must be 32bit aligned addresses.
+ */
+static void boot_info_memcpy32(void *dst, void *src, uint32_t size)
+{
+	uint32_t *d, *s, rem;
+	uint32_t i, rem_sz;
+
+	d = dst;
+	s = src;
+
+	for (i = 0; i < size / sizeof(uint32_t); i++)
+		d[i] = s[i];
+
+	rem_sz = size % sizeof(uint32_t);
+	if (!rem_sz)
+		return;
+
+	rem = s[i];
+	memcpy(&d[i], &rem, rem_sz);
+}
+
+int boot_info_get(uint32_t type, void *data, uint32_t max_size, uint32_t *copy_size)
+{
+	struct hl_boot_info_ptrs ptrs;
+	struct hl_boot_info_desc desc;
+	uint32_t *offsets, i, cpy_sz;
+	void *src;
+	int ret;
+
+	ptrs = *(struct hl_boot_info_ptrs *)offset2ptr(0, BOOT_INFO_BASE);
+	if (ptrs.magic != HL_BOOT_INFO_MAGIC)
+		return 1;
+
+	offsets = (uint32_t *)offset2ptr(sizeof(struct hl_boot_info_ptrs),
+				    BOOT_INFO_BASE);
+
+	for (i = 0; i < ptrs.count; i++) {
+		desc = *(struct hl_boot_info_desc *)offset2ptr(offsets[i],
+							       BOOT_INFO_BASE);
+		if (type != desc.type)
+			continue;
+
+		if (desc.magic != (HL_BOOT_INFO_DESC_MAGIC | type))
+			return 1;
+
+		cpy_sz = (max_size > desc.size) ? desc.size : max_size;
+		src = offset2ptr(desc.offset, BOOT_INFO_BASE);
+		boot_info_memcpy32(data, src, cpy_sz);
+
+		ret = validate_crc32_csum((uint8_t *)data, desc.size,
+					&desc);
+		if (ret)
+			return 1;
+
+		*copy_size = cpy_sz;
+
+		return 0;
+	}
+
+	return 1;
+}
