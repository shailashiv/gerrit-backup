+void boot_rom_uart_dev_init(unsigned int val)
+{
+	regmap_writel(0x0, IO0_H0_PSOC_CFG_UART(mmUART_CTRL));
+	regmap_writel(val, IO0_H0_PSOC_CFG_UART(mmUART_BAUDDIV));
+	regmap_writel(UART_TX_ENABLE | UART_RX_ENABLE, IO0_H0_PSOC_CFG_UART(mmUART_CTRL));
+}
+
+void boot_rom_config_uart_gpio()
+{
+	/* TODO: Confirm with system team and remove rx config */
+	/* UART0 RXD */
+	regmap_writel(GPIO_FUNC_1, IO0_H0_PSOC_CFG_PAD_CONF_PAD_SEL(mmPSOC_PAD_CONF_PAD_SEL_49));
+
+	/* UART0 TXD */
+	regmap_writel(GPIO_FUNC_1, IO0_H0_PSOC_CFG_PAD_CONF_PAD_SEL(mmPSOC_PAD_CONF_PAD_SEL_50));
+}
+
+void boot_rom_putc(const char ch)
+{
+	uint8_t loop;
+
+	/*
+	 * TODO: DIE_CHECK
+	 * Confirm with system team, uart required for slave's or not
+	 */
+
+	for (loop = 0; loop < UART_TX_FULL_RETRY_COUNT; loop++) {
+		if ((regmap_readl(IO0_H0_PSOC_CFG_UART(mmUART_STATE)) & UART_SR_TXFULL) == 0) {
+			regmap_writel(ch, IO0_H0_PSOC_CFG_UART(mmUART_DATA));
+			break;
+		}
+	}
+}
