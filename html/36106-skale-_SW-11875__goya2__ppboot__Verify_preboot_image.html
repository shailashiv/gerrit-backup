+		return spi_flash_readl(FLASH_SECONDARY_OFFSET);

+static int ppboot_bringup_preboot(u8 image_ptr)
+{
+	u32 flash_offset;
+	u16 image_size;
+
+	flash_offset = ppboot_flash_image_offset(image_ptr);
+	image_size = (u16) spi_flash_readl(flash_offset);
+
+	/*
+	 * Copy preboot image from flash to Scratchpad SRAM.
+	 * Note that the offset to be provided to perform DMA must be
+	 * in granuality of 4B. ie. In order to write at
+	 * 0x7FFBFE1000ULL, the address to be passed is 0x7FFBFE0400ULL
+	 * instead of 0x7FFBFE1000ULL (right shift offest 0x1000 by 2).
+	 */
+	dma_copy_image(image_size, flash_offset,
+		       (HL_SP_SRAM_BASE + (SP_SRAM_PREBOOT_OFFSET >> 2)),
+		       SPI_DMA_DST_HBW);
+	if (dma_wait_finish() == DMA_CRC_ERROR)
+		return ARCBOOT_BRINGUP_ERROR;
+
+	image_size -= SECUREBOOT_IMG_HEAD_FOOTER;
+	if (secureboot_verify_image(image_size,
+			(SP_SRAM_BASE + SP_SRAM_PREBOOT_OFFSET + IMG_LEN_SIZE))
+			== SECUREBOOT_VERIFY_FAILED)
+		return ARCBOOT_BRINGUP_ERROR;
+
+	return ARCBOOT_BRINGUP_OK;
+}
