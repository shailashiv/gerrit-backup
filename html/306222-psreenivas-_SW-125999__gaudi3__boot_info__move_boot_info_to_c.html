+int boot_info_sync_data(void)
+{
+	struct mem_desc mem_region_desc;
+	struct hl_boot_info_desc *desc;
+	u32 desc_count = boot_info_get_num_descs();
+	uintptr_t mem_base, mem_base_end;
+	u32 idx, actual_size;
+
+	if (preboot_get_mem_region(MEM_BOOT_INFO, &mem_region_desc)) {
+		preboot_puts("boot_info_set: failed to get mem region MEM_BOOT_INFO\n");
+		return BOOT_INFO_MEMERR;
+	}
+
+	mem_base = mem_region_desc.base;
+	mem_base_end = mem_base + mem_region_desc.size;
+	for (idx = 0; idx < desc_count; idx++) {
+		desc = hl_boot_info_get_desc(idx);
+		if (!desc) {
+			preboot_puts_hex("boot_info_set: failed to get pointer to desc index ", idx,
+					 false);
+			continue;
+		}
+
+		/* copy boot info memory into BOOT_INFO mem region */
+		if (mem_base + desc->size > mem_base_end) {
+			preboot_puts("boot_info_set: not enough space for bootinfo\n");
+			return BOOT_INFO_MEMFULL;
+		}
+
+		if (hl_boot_info_get(desc->type, (void *)mem_base, desc->size,
+					   &actual_size)) {
+			preboot_puts_hex("boot_info_set: failed to get type ", desc->type,
+					 false);
+			continue;
+		}
+
+		if (hl_boot_info_set(desc->type, (void *)mem_base, desc->size)) {
+			preboot_puts_hex("boot_info_set: bootinfo set failure for type ",
+					 desc->type, false);
+			continue;
+		}
+
+		mem_base += desc->size;
+	}
+
+	return BOOT_INFO_SUCCESS;
