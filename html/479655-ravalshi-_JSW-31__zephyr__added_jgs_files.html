+#define DIE_ID_MASK		(BIT(27)|BIT(26)|BIT(25))
+#define DIE_ID_BIT_OFFSET	25
+
+static enum fs1_die_id fs1_die_id = FS1_INVALID_DIE_ID;
+
+uint32_t dev_read_die_id(void)
+{
+	uint32_t die_id;
+
+	if (fs1_die_id != FS1_INVALID_DIE_ID)
+		goto ret;
+
+	die_id = CFG_RREG32(IO0_H0_PSOC_ARC_MNG_LOCAL(mmARC_LOCAL_CONF_DIE_CPU_ID));
+
+	fs1_die_id = ((die_id & ARC_LOCAL_CONF_DIE_CPU_ID_DIE_ID_M)
+			>> ARC_LOCAL_CONF_DIE_CPU_ID_DIE_ID_S);
+
+ret:
+	return fs1_die_id;
+}
+
+uint8_t dev_get_die_id(uint32_t addr)
+{
+	return ((addr & DIE_ID_MASK) >> DIE_ID_BIT_OFFSET);
+}
+
+int dev_efuse_get_field(uint8_t die_id, uint32_t word, uint32_t mask, uint32_t *val)
+{
+	int err;
+	uint32_t fuse_val;
+	const struct device *efuse_dev = device_get_binding("EFUSE");
+
+	err = efuse_read_word(efuse_dev, word, &fuse_val);
+	if (err) {
+		LOG_ERR("Error reading fuse, offset=0x%x", word);
+		return err;
+	}
+
+	*val = FIELD_GET(mask, fuse_val);
+	return 0;
+}
+
+bool dev_is_standalone_io(void)
+{
+	int err;
+	uint32_t val;
+
+	err = dev_efuse_get_field(FS1_IO_DIE_0, IO_EFUSE_SYSCFG_OFFSET, EFUSE_SYSCFG_SKU, &val);
+	if (err) {
+		LOG_ERR("can't get SKU, assume standalone mode");
+		return true;
+	}
+
+	return (val == FS1_IO_DIE_SA);
+}
+
+__weak bool dev_is_die_available(uint32_t die_id)
+{
+	if (dev_is_standalone_io() &&
+	    ((die_id >= FS1_COMPUTE_DIE_0) && (die_id <= FS1_COMPUTE_DIE_3)))
+		return false;
+
+	return true;
+}

+bool dev_is_cfg_addr_valid(uint32_t addr, uint8_t current_die_id,

+int fs_iterate_axiovrd(struct iterate_module_ctx *ctx, enum axiovrd_type axiovrd_type)
+{
+	uint32_t axiovrd_idx, die_id, base;
+	uint32_t num_axiovrd;
+	enum axiovrd_sublist subtype;
+	bool is_die_avail;
+
+	for (die_id = ctx->die_id; die_id <= ctx->die_id + 1; die_id++) {
+		is_die_avail = dev_is_die_available(die_id);
+		if (!is_die_avail)
+			continue;
+
+		for (subtype = cmn_axiovrd_list; subtype <= spec_half_axiovrd_list; subtype++) {
+			num_axiovrd = get_axiovrd_list_len(die_id, axiovrd_type, subtype);
+			if (num_axiovrd < 1)
+				continue;
+
+			for (axiovrd_idx = 0; axiovrd_idx < num_axiovrd; axiovrd_idx++) {
+				base = get_axiovrd_base(die_id, axiovrd_idx, axiovrd_type, subtype);
+				ctx->fn(base, ctx->data);
+			}
+		}
+	}
+
+	return 0;
+}
