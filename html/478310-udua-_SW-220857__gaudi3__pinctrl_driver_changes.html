+
+ #include <zephyr/devicetree.h>
+ #include <zephyr/drivers/pinctrl.h>
+ #include <common_utils.h>
+ #include <pinctrl_soc.h>
+ #include <hl_fw_if.h>
+
+int pinctrl_configure_pins(const pinctrl_soc_pin_t *pins, uint8_t pin_cnt, uintptr_t reg)
+{
+	int i;
+
+	uint32_t num_dies = 1, die_id = hl_read_die_id();
+	bool spi_gw_dis = hl_get_spi_gw_disable();
+	uint8_t pin, func;
+	uint32_t pinctrl_base;
+
+	pinctrl_base = (uint32_t)DT_REG_ADDR(DT_NODELABEL(pinctrl));
+
+	if (hl_get_sku_type() == HL_DUAL_DIE_SKU)
+		num_dies = 2;
+
+	if (!hl_is_cfg_addr_valid((uint32_t)pinctrl_base, die_id, num_dies, spi_gw_dis))
+		return -EINVAL;
+
+	for (i = 0; i < pin_cnt; i++) {
+		pin = pins[i].pin;
+		func = pins[i].func;
+		sys_write32(func, pinctrl_base + (pin * sizeof(uint32_t)));
+	}
+
+	return 0;
+}
