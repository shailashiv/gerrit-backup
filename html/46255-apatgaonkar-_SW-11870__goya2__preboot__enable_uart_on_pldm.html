+#ifdef TARGET_IS_PLDM
+/* PLL initializations for PLDM */
+#define HL_PLL_LOCK_TOUT	10	/* 10 us */
+#define GOYA2_PLL_CFG		0x100
+#define GOYA2_PLL_CFG_EN	BIT(0)
+#define GOYA2_PLL_CFG_LOCK	BIT(16)
+#define GOYA2_UNIT_RST_N_L	0x4B0
+#define HL_PLL_DIV_FACTOR_CMD	0x220
+
+static void hl_pll_reset(struct hl_pll_cfg *cfg, bool reset)
+{
+	u32 reg = regmap_readl(cfg->base + GOYA2_PLL_CFG);
+
+	if (true == reset)
+		reg |= GOYA2_PLL_CFG_EN;
+	else
+		reg &= ~GOYA2_PLL_CFG_EN;
+
+	regmap_writel(reg, cfg->base + GOYA2_PLL_CFG);
+}
+
+static int hl_pll_is_locked(struct hl_pll_cfg *cfg)
+{
+	return regmap_readl(cfg->base + GOYA2_PLL_CFG) & GOYA2_PLL_CFG_LOCK;
+}
+
+#define GOYA2_PLL_POSTDIV1_SHIFT	16
+#define GOYA2_PLL_POSTDIV2_SHIFT	20
+#define GOYA2_PLL_REFDIV_SHIFT		24
+#define GOYA2_PLL_MAX_WAIT_TIME		1
+
+static void hl_pll_cfg(struct hl_pll_cfg *cfg)
+{
+	u32 reg = 0;
+
+	/* put the PLL into reset by writing 0 to CTRL_CFG.EN */
+	hl_pll_reset(cfg, false);
+
+	/* FBDIV - bit 0
+	 * POSTDIV1 - bit 16
+	 * POSTDIV2 - bit 20
+	 * REFDIV - bit 24
+	 */
+	reg = cfg->nr << GOYA2_PLL_REFDIV_SHIFT |
+		cfg->nb << GOYA2_PLL_POSTDIV2_SHIFT |
+		cfg->od << GOYA2_PLL_POSTDIV1_SHIFT | cfg->nf;
+
+	regmap_writel(reg, cfg->base + GOYA2_PLL_CFG_DIV);
+
+	/* after updating the coefficients we need to wait >=1us */
+	udelay(GOYA2_PLL_MAX_WAIT_TIME);
+
+	/* enable the PLL by writing a value of 1 to CTRL_CFG.EN*/
+	hl_pll_reset(cfg, true);
+}
+
+static void hl_pll_clk_switch_to_bypass(struct hl_pll_cfg *cfg)
+{
+	int sel;
+	u32 reg;
+
+	for (sel = 0; sel < HL_PLL_DIV_SIZE; sel++) {
+		reg = cfg->base + HL_PLL_DIV_SEL + (sel * sizeof(u32));
+		regmap_writel(HL_PLL_REF_CLK, reg);
+	}
+}
+
+static void hl_pll_clk_switch_to_pll(struct hl_pll_cfg *cfg)
+{
+	int sel;
+	u32 reg;
+
+	for (sel = 0; sel < HL_PLL_DIV_SIZE; sel++) {
+		reg = cfg->base + HL_PLL_DIV_SEL + (sel * sizeof(u32));
+		if (cfg->div_sel[sel] < 0 || cfg->div_sel[sel] > HL_PLL_DIV_PLL)
+			continue;
+		regmap_writel(cfg->div_sel[sel], reg);
+	}
+}
+
+static void hl_pll_wait_for_lock(struct hl_pll_cfg *cfg)
+{
+	int tout = HL_PLL_LOCK_TOUT;
+
+	while (!hl_pll_is_locked(cfg) && tout--)
+		udelay(1);
+
+	if (tout < 0) {
+		preboot_puts_raw("PLL lock timedout for");
+		preboot_puts(cfg->name);
+	}
+}
+
+static void hl_pll_div_init(struct hl_pll_cfg *cfg)
+{
+	int sel;
+	u32 div_base;
+
+	for (sel = 0; sel < HL_PLL_DIV_SIZE; sel++) {
+		if (cfg->div_sel[sel] == HL_PLL_REF_CLK ||
+		    cfg->div_factor[sel] < 0)
+			continue;
+
+		div_base = cfg->base + (sel * sizeof(u32));
+
+		regmap_writel(cfg->div_factor[sel],
+				div_base + HL_PLL_DIV_FACTOR);
+		regmap_writel(1, div_base + HL_PLL_DIV_FACTOR_CMD);
+		regmap_writel(1, div_base + HL_PLL_DIV_EN);
+	}
+
+	/* just to be on a safe side, wait for DIV init to propagate */
+	udelay(1);
+}
+
+static void hl_pll_set(struct hl_pll_cfg *cfg)
+{
+	hl_pll_clk_switch_to_bypass(cfg);
+
+	hl_pll_cfg(cfg);
+
+	/* Polling on PLL lock until lock is cleared or timed out */
+	hl_pll_wait_for_lock(cfg);
+
+	hl_pll_div_init(cfg);
+
+	hl_pll_clk_switch_to_pll(cfg);
+
+}
+
+/**
+ * hl_pll_init - Initialize pll registers
+ */
+void hl_pll_init(struct hl_pll_cfg *hl_pll_cfg, int size)
+{
+	int i;
+
+	for (i = 0; i < size; i++)
+		hl_pll_set(hl_pll_cfg + i);
+}
