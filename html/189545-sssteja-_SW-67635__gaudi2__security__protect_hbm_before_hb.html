Commit Message

[SW-67635] gaudi2: security: protect hbm before hbm init in early ppboot

This patch configures HMMU to block HBM in the ppboot security.
This is needed as host can crash when a malicious user tries to access
HBM after link up and before HBM init.

+static void ppboot_secure_hbm(void)
+{
+	int dcore, hmmu;
+	uint32_t data_min, data_max;
+	uint64_t base, offset;
+
+	/*
+	 * Remove hmmu and xbar isolation before accessing hmmu registers
+	 * Note: Later these isolation registers are re-configured in preboot
+	 * binning and isolation logic based on efuse.
+	 */
+	regmap_writel(0, mmPSOC_GLOBAL_CONF_HIF_HMMU_ISO);
+	regmap_writel(0, mmPSOC_GLOBAL_CONF_XBAR_EDGE_ISO);
+
+	/* Protect max possible range (0x0 - 0xFFFFFFFFFFFFFFFF) by HMMU */
+	data_min = 0;
+	data_max = HL_REGS_DATA_MASK;
+
+	for (dcore = 0; dcore < HL_SEC_NUM_OF_DCORES; dcore++) {
+		base = dcore * HL_SEC_HMMU_DIST;
+		for (hmmu = 0; hmmu < hmmu_cfg.instances_num; hmmu++) {
+			offset = base + (hmmu * hmmu_cfg.instances_dist);
+			regmap_writel(data_min, offset + hmmu_cfg.min_lo);
+			regmap_writel(data_min, offset + hmmu_cfg.min_hi);
+			regmap_writel(data_max, offset + hmmu_cfg.max_lo);
+			regmap_writel(data_max, offset + hmmu_cfg.max_hi);
+			regmap_writel(HMMU_RR_EN, offset + hmmu_cfg.rr_en);
+		}
+	}
+}
