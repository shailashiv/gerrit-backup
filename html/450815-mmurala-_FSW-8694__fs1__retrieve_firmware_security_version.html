+/*
+ * minimal allowed security version is defined by a 128bit map, to find this
+ * we scan from end to beginning of map looking for the first set bit. once
+ * it's found this indicated the minimal version allowed to run on the system
+ */
+static uint8_t security_find_min_fw_version(uint32_t *fw_version)
+{
+	int8_t row, column;
+	uint8_t fw_version_sec;
+	/* init max version */
+	fw_version_sec = SECURITY_MAX_FW_SEC_VERSION;
+
+	/* iterate from end to start, till a set bit is found */
+	for (row = SECURITY_FW_VER_LAST_DWORD; row >= 0; row--) {
+		for (column = SECURITY_FW_VER_LAST_BIT; column >= 0; column--) {
+			/* find highest bit set in valid FW */
+			if (fw_version[row] & BIT(column))
+				return fw_version_sec;
+			fw_version_sec--;
+		}
+	}
+
+	return fw_version_sec;
+}
+
+uint8_t security_get_fw_version(void)
+{
+	uint8_t i;
+	uint32_t fw_version[IO_EFUSE_SVN_FW_DWORD_COUNT];
+	uint8_t fw_version_sec;
+
+	/* read min security version bit map from flash */
+	for (i = 0; i < IO_EFUSE_SVN_FW_DWORD_COUNT; i++)
+		boot_rom_fuse_read(IO_EFUSE_SVN_FW_OFFSET + i, &fw_version[i]);
+
+	fw_version_sec = security_find_min_fw_version(fw_version);
+
+	return fw_version_sec;
+}
