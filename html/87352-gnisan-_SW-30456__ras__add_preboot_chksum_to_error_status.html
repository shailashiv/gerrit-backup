+static void i2csi_err_chksum_validate(void)
+{
+	struct boot_desc desc;
+	enum fw_post_reset_cmd role[] = {
+		FLOPS_UPD_NOOP,		/* Primary */
+		FLOPS_UPD_PENDING,	/* Secondary */
+	};
+	u8 hash[SHA256_SUM_LEN];
+	bool is_valid, is_prime;
+	u32 idx;
+	int i;
+
+	if (!boot_desc_init()) {
+		preboot_puts_raw("Not a valid boot desc.");
+		return;
+	}
+
+	for (i = 0; i < 2; i++) {
+		idx = boot_desc_get_img_desc(role[i], HL_FW_IMG_PREBOOT, &desc);
+		if (idx == HL_FW_IMG_NOT_FOUND)
+			continue;
+
+		is_prime = !i;
+		/* Using fit image base as playground */
+		spi_copy((void *)FIT_IMG_BASE, desc.spi_offset, desc.img_size,
+			 NULL);
+		sha256_csum((void *)FIT_IMG_BASE, desc.img_size, hash);
+		is_valid = !memcmp(hash, desc.chksum, ARRAY_SIZE(hash));
+
+		if (is_prime)
+			op_err_status.checksum.spi.preboot_primary = !is_valid;
+		else
+			op_err_status.checksum.spi.preboot_secondary = !is_valid;
+	}
+
+	err_chksum_avail = true;
+}
