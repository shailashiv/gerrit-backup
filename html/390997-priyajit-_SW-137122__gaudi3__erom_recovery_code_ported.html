+#define GAUDI3_CPLD_WDOG_GPIO		66
+void preboot_clear_wd_gpio(void)
+{
+	uint32_t index = GAUDI3_CPLD_WDOG_GPIO % GPIOS_PER_BANK;
+	uint32_t val, mask = BIT(index);
+
+	/* configure GPIO in input mode */
+	val = regmap_readl(D0_PSOC_GPIO4_REG(mmGPIO_OUTENSET));
+	val = val | mask;
+	regmap_writel(val, D0_PSOC_GPIO4_REG(mmGPIO_OUTENSET));
+
+	/* gpio is active high */
+	val = regmap_readl(D0_PSOC_GPIO4_REG(mmGPIO_DATAOUT));
+	val = val | mask;
