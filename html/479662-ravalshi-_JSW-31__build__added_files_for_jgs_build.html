+function swap_file {
+	local INF=$1
+	local OUTF=$2
+	local DBG=${3:-}
+
+	# SWAP needs this to be 4B aligned in size
+	dd status=none conv=sync bs=4 if=${INF} > ${OUTF}.tmp
+	# Sadly --endian=big is not swapping for -tx1 so I used some gawk magic
+	od -Anone -v -tx1 -w4 ${OUTF}.tmp | gawk '{ print $4 $3 $2 $1 }' | xxd -r -p > ${OUTF}
+
+	if [ -z "$DBG" ]; then
+		rm -f ${OUTF}.tmp
+	fi

+function get_struct_size {
+	header_file="$1"
+	struct_name="$2"
+
+	# Check if a filename is provided
+	if [ -z "$header_file" ]; then
+		echo "Usage: get_struct_size <header_file> <struct_name>"
+		exit 1
+	fi
+
+	# Check if a struct name is provided
+	if [ -z "$struct_name" ]; then
+		echo "Usage: get_struct_size <header_file> <struct_name>"
+		exit 1
+	fi
+
+	# Use gcc to preprocess the header file and calculate the size of the struct
+	size=$(echo "#include \"$header_file\"
+	#include <stdio.h>
+	int main() {
+		printf(\"%zu\", sizeof(struct $struct_name));
+		return 0;
+	}" | gcc -x c - -o tmp - && ./tmp)
+
+	if [ -z "$size" ]; then
+		echo "Error: Struct '$struct_name' not found in '$header_file'"
+		exit 1
+	else
+		echo $size
+	fi
+
+	# Clean up
+	rm -f tmp
+}

+function hl_pack_bin {

+	FILEPATH=$(dirname ${FILEFULLPATH})
+	BSIZE=$2
+	KEY_NUM=$3
+	CRC_ERROR=$4
+	SIGN_ERROR=$5

+	PADFULLBINFILE=${FILEFULLPATH}.pad
+	dd status=none conv=sync bs=16 if=${FILEFULLPATH} of=${PADFULLBINFILE}
+
+	SIZE_WITHOUT_CRC=$(stat --printf="%s" ${PADFULLBINFILE})
+	NUM_SLICES=$((SIZE_WITHOUT_CRC/MAX_SPI_DMA_DATA_TRANSFER))
+	MODULO_SLICES=$((SIZE_WITHOUT_CRC%MAX_SPI_DMA_DATA_TRANSFER))
+	if [ ${MODULO_SLICES} -ne 0 ]; then
+		NUM_SLICES=$((NUM_SLICES+1))
+	fi
+
+	# image size to be computed as part of image descriptor function
+	local leftover=${SIZE_WITHOUT_CRC}
+	local skip_offset=0
+
+
+	# image packing example:
+	# image_size = 200kb + 16 bytes CRC = 0x32010
+	# num_slice: 4
+	# 1st slice: flash offset: 0x0 DST: 0x0 (dma_size: 0xFFF4 left_size: 0x2201C)
+	# 2nd slice: flash offset: 0x10000 DST: 0xFFF0 (dma_size: 0xFFF4 left_size: 0x12028)
+	# 3nd slice: flash offset: 0x20000 DST: 0x1FFE0 (dma_size: 0xFFF4 left_size: 0x2034)
+	# 4th slice: flash offset: 0x30000 DST: 0x2FFD0 (dma_size: 0x2034 left_size: 0x0)
+	#
+	#  =========================================================
+	# | PPBOOT IMG SLICE1 + CRC + PADDING (0xFFF0 + 0x4 + 0x12) |
+	#  ---------------------------------------------------------
+	# | PPBOOT IMG SLICE2 + CRC + PADDING (0xFFF0 + 0x4 + 0x12) |
+	#  ---------------------------------------------------------
+	# | PPBOOT IMG SLICE3 + CRC + PADDING (0xFFF0 + 0x4 + 0x12) |
+	#  ---------------------------------------------------------
+	# | PPBOOT IMG SLICE4 + CRC (0x2030 + 0x4)		    |
+	#  =========================================================
+	#
+	# Note: Padding size is not included in total size as DMA is
+	# not expected to copy padding data.
+	for ((slice_cnt=0; slice_cnt<${NUM_SLICES}; slice_cnt++)); do
+		# bin file: (64KB-16) + CRC
+		local BIN_FILE=${PADFULLBINFILE}.part${slice_cnt}
+		if [ $leftover -lt $MAX_SPI_DMA_DATA_TRANSFER ]; then
+			size_bytes=$leftover
+		else
+			size_bytes=$MAX_SPI_DMA_DATA_TRANSFER
+		fi
+
+		size_words=$((size_bytes/4))
+		dd status=none skip=${skip_offset} conv=sync bs=4 if=${PADFULLBINFILE} count=${size_words} > ${BIN_FILE}
+		hl_calc_crc32 ${BIN_FILE}
+		if [ ${CRC_ERROR} -eq 0 ]; then
+			cat ${BIN_FILE}.crc32 >> ${BIN_FILE}
+		fi
+		flash_offset=$((SPI_DMA_DEST_64KB_OFFS/4))
+		flash_offset=$((flash_offset*slice_cnt))
+		dd status=none conv=sync if=${BIN_FILE} bs=4 seek=$flash_offset of=${STAGE_DIR}/${DEV_PREFIX}-${FILENAME}.le.int
+		leftover=$((leftover-size_bytes))
+		skip_offset=$((skip_offset+size_words))
+	done
+
+	hl_reverse_words ${STAGE_DIR}/${DEV_PREFIX}-${FILENAME}.le.int ${FILEPATH}/${DEV_PREFIX}-${FILENAME}.le 4
