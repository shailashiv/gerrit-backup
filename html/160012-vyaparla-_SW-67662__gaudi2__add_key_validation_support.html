+bool preboot_security_key_is_revoked(int key_idx)
+{
+	const struct hl_sec_efuse *fuse;
+	int max_keys = SEC_KEY_HASH_COUNT_IN_ROM;
+
+	if (preboot_fake_security_is_enabled())
+		fuse = preboot_get_hl_sec_fake_efuse();
+	else
+		fuse = preboot_get_hl_sec_real_efuse();
+
+	if (fuse == NULL)
+		return true;
+
+	if (fuse->keys_loc_selector == SEC_KEY_HASH_LOCATION_EFUSE)
+		max_keys = SEC_KEY_HASH_COUNT_IN_EFUSE;
+
+	if (key_idx < 0 || key_idx >= max_keys)
+		return true;
+
+	if (fuse->key_revoke & BIT(key_idx))
+		return true;
+
+	return false;
+}

+static int preboot_security_get_minimal_svn(void)
+{
+	const struct hl_sec_efuse *fuse;
+	const u32 *svn_bits;
+	int i, svn = 0;
+
+	if (preboot_fake_security_is_enabled())
+		fuse = preboot_get_hl_sec_fake_efuse();
+	else
+		fuse = preboot_get_hl_sec_real_efuse();
+
+	if (fuse == NULL)
+		return -1;
+
+	/* Use 32bit because the data may not be 64bit aligned */
+	svn_bits = (const u32 *)fuse->sec_ver_num;
+
+	/* Count how many consecutive LSB bits out of 128 are set in 4 32bit
+	 * iterations.
+	 */
+	for (i = 0; i < SVN_EFUSE_SIZE/sizeof(u32); i++) {
+		/* No bits are set, stop counting */
+		if (!svn_bits[i])
+			break;
+		/* All bits are set, move to next iteration, bits are counted
+		 * by the iterator at the end. Move to the next iteration.
+		 */
+		if (!~svn_bits[i])
+			continue;
+
+		/* Some bits are set, count them by __builtin_ctz() the bitwise
+		 * NOT value of the SVN since __builtin_ctz() count unset bits.
+		 */
+		svn += __builtin_ctz(~svn_bits[i]);
+		break;
+	}
+	/* Add bits counted by iterator in case that all the bits are set */
+	svn += i * 32;
+
+	preboot_puts_hex("SVN on eFuse: ", svn, 0);
+	return svn;
+}
+
+bool preboot_security_svn_is_valid(int sec_ver)
+{
+	int min_svn = preboot_security_get_minimal_svn();
+
+	if (min_svn < 0)
+		return false;
+
+	if (sec_ver < min_svn || sec_ver > SVN_MAX_VERSIONS) {
+		preboot_puts_hex("Illegal security version: ", sec_ver, false);
+		return false;
+	}
+
+	return true;
+}

+bool preboot_security_key_is_valid(int key_idx, const void *pub_key,
+			 int pub_key_len, const void *pub_key_ext,
+			 int pub_key_ext_len)
+{
+	u8 value[SHA_SUM_LEN];
+	u32 hash_compare_len = SHA_SUM_LEN;
+	const struct hl_sec_otp *otp = preboot_security_get_otp();
+
+	if (otp == NULL)
+		return false;
+
+	if (key_idx < 0 || key_idx >= otp->num_of_keys)
+		return false;
+
+	/*
+	 * In case of real security, if security keys hash location is ROM then only the first
+	 * 256 bits are compared against the hash calculated. Since there are only 256 bits
+	 * available in PSOC registers to store keys hash
+	 */
+	if (!preboot_fake_security_is_enabled() && otp->num_of_keys == SEC_KEY_HASH_COUNT_IN_ROM)
+		hash_compare_len = SHA256_SUM_LEN;
+
+	sha_csum(pub_key, pub_key_len, value);
+
+	return !memcmp(&otp->keys[key_idx * SHA_SUM_LEN], value, hash_compare_len);
+}

+static struct hl_fit_sec_desc fit_sec;
+static u8 signature[FIT_SIGNATURE_SIZE];
+
+int preboot_verify_img_sig(const void *img, size_t img_size, enum fit_sec_failure *res,
+				u32 *key_idx)
+{
+	const void *fit_sec_off;
+	const void *pub_key_fit;
+
+	if (!preboot_security_is_enabled())
+		return 0;
+
+	/* hl fit files are padded so the size is 4 bytes aligned */
+	img_size = ROUND_UP4(img_size);
+
+	/* security descriptor is at the footer of the image */
+	fit_sec_off = (const void *)((u8 *)img + img_size);
+
+	/* check that the calculated memory range are legal */
+	if (!preboot_sram_is_valid_address_range((uintptr_t)fit_sec_off,
+						 sizeof(fit_sec))) {
+		preboot_puts("Invalid address range\n");
+		return 1;
+	}
+
+	memcpy(&fit_sec, fit_sec_off, FIT_SEC_DESC_SIZE);
+
+	if (fit_sec.magic_number != FIT_SEC_MAGIC) {
+		preboot_puts("Bad magic, security descriptor not found\n");
+		return 1;
+	}
+
+	if ((fit_sec.desc_version_maj != FIT_SEC_DESC_VER_MAJOR) ||
+	    (fit_sec.desc_version_min != FIT_SEC_DESC_VER_MINOR)) {
+		preboot_puts("security descriptor version mismatch\n");
+		return 1;
+	}
+
+	if (fit_sec.desc_size < FIT_SEC_DESC_SIZE) {
+		preboot_puts("security descriptor size too small\n");
+		return 1;
+	}
+
+	if (fit_sec.img_size != img_size) {
+		preboot_puts("image size mismatch with security descriptor\n");
+		return 1;
+	}
+
+	if (!preboot_security_svn_is_valid(fit_sec.sec_version)) {
+		preboot_puts("security descriptor svn not valid\n");
+		*res = FIT_SEC_SVN_INVALID;
+		return 1;
+	}
+
+	if (preboot_security_key_is_revoked(fit_sec.key_num)) {
+		preboot_puts_hex("Key revoked: ", fit_sec.key_num, false);
+		*res = FIT_SEC_KEY_REVOKED;
+		return 1;
+	}
+
+	pub_key_fit = &fit_sec.pub_key;
+
+	if (!preboot_security_key_is_valid(fit_sec.key_num, pub_key_fit,
+					   FIT_PUB_KEY_SIZE, NULL, 0)) {
+		preboot_puts("Key validation failed\n");
+		*res = FIT_SEC_SIGN_FAILED;
+		return 1;
+	}
+
+	/*
+	 * signature is calculated on both the fit image and the security
+	 * descriptor with the signature field of the descriptor zeroized.
+	 */
+	memcpy(signature, fit_sec.signature, FIT_SIGNATURE_SIZE);
+	memset(fit_sec.signature, 0, FIT_SIGNATURE_SIZE);
+
+	if (fit_verify_img_sig(img, img_size, &fit_sec, FIT_SEC_DESC_SIZE,
+			       pub_key_fit, signature, FIT_SIGNATURE_SIZE)) {
+		preboot_puts("image signature verification failed\n");
+		*res = FIT_SEC_SIGN_FAILED;
+		return 1;
+	}
+
+	*key_idx = fit_sec.key_num;
+
+	return 0;
+}
