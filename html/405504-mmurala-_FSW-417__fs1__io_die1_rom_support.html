+#define NOTIFY_SLV_DURATION_US		10
+
+void boot_rom_wait_for_recovery_in(void)
+{
+	uint32_t val;
+
+	/* TODO: print uart char to indicate waiting for recovery in */
+
+	while (1) {
+		val = regmap_readl(IO0_H0_PSOC_GLOBAL_CONF(mmGLOBAL_CONF_DIE_RECOVERY_IN));
+		if (val & GLOBAL_CONF_DIE_RECOVERY_IN_IND_M) {
+			/* Master die holds the indication pulse for 10us,
+			 * Hence we need to wait for 10us after detection
+			 * to make sure slave follows the master in terms
+			 * of point of execution
+			 */
+
+			/* TODO: print uart char to indicate, recovery in received */
+
+			udelay(NOTIFY_SLV_DURATION_US);
+			break;
+		}
+	}
+	/*
+	* No timeout added for while(1) due to the usage of this function:
+	* 1) Recovery: It is required for slave to wait for this indication from
+	* master. As slave die in rom is to only execute recovery, it waits forever
+	* for this indication to be set, apart from recovery there is no logic to be
+	* executed on slave in ROM.
+	*
+	* 2) D2D link init: The pcie linkup time on master cannot be predicted, hence
+	* slave needs to wait. However the implication of this while(1) is minimal as
+	* d2d is the last process in the recovery flow which ends also in a while(1).
+	* Hence this would not have any functional impact.
+	*/
+}
+
+void boot_rom_wait_for_signal(void)
+{
+	if (boot_rom_global.die_type == IO_DIE_MASTER)
+		return;
+
+	boot_rom_status_update(PROGRESS_SLV_DIE_WAITING_ON_RECOVERY_IND);
+
+	boot_rom_wait_for_recovery_in();
+
+	/* TODO: recovery */
+}
