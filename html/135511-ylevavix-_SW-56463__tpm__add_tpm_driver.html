+/**
+ * tpm_tis_spi_read() - Read from TPM register
+ *
+ * @addr: register address to read from
+ * @buffer: provided by caller
+ * @len: number of bytes to read
+ *
+ * Read len bytes from TPM register and put them into
+ * buffer (little-endian format, i.e. first byte is put into buffer[0]).
+ *
+ * NOTE: TPM is big-endian for multi-byte values. Multi-byte
+ * values have to be swapped.
+ *
+ * @return -EIO on error, 0 on success.
+ */
+static int tpm_tis_spi_xfer(u32 addr, const u8 *out, u8 *in, u16 len)
+{
+	int transfer_len, ret;
+	u8 tx_buf[MAX_SPI_FRAMESIZE];
+	u8 rx_buf[MAX_SPI_FRAMESIZE];
+
+	if (in && out) {
+		preboot_puts("TPM2: can't do full duplex\n");
+		return -EINVAL;
+	}
+
+	spi_tpm_claim_bus();
+
+	while (len) {
+		/* Request */
+		transfer_len = min_t(u16, len, MAX_SPI_FRAMESIZE);
+		tx_buf[0] = (in ? BIT(7) : 0) | (transfer_len - 1);
+		tx_buf[1] = 0xD4;
+		tx_buf[2] = addr >> 8;
+		tx_buf[3] = addr;
+
+		ret = spi_xfer(&tpm_spi, 4 * 8, tx_buf, rx_buf, SPI_XFER_BEGIN);
+		if (ret < 0) {
+			preboot_puts("TPM2: spi request transfer failed\n");
+			goto release_bus;
+		}
+
+		/* Wait state */
+		if (!(rx_buf[3] & 0x1)) {
+			int i;
+
+			for (i = 0; i < TPM_WAIT_STATES; i++) {
+				ret = spi_xfer(&tpm_spi, 1 * 8, NULL, rx_buf,
+						0);
+				if (ret) {
+					preboot_puts_hex("TPM2: wait state failed: ",
+								ret, true);
+					goto release_bus;
+				}
+
+				if (rx_buf[0] & 0x1)
+					break;
+			}
+
+			if (i == TPM_WAIT_STATES) {
+				preboot_puts("TPM2: timeout on wait state\n");
+				ret = -1;
+				goto release_bus;
+			}
+		}
+
+		/* Read/Write */
+		if (out) {
+			memcpy(tx_buf, out, transfer_len);
+			out += transfer_len;
+		}
+
+		ret = spi_xfer(&tpm_spi, transfer_len * 8,
+			       out ? tx_buf : NULL,
+			       in ? rx_buf : NULL,
+			       SPI_XFER_END);
+		if (ret) {
+			preboot_puts("TPM2: spi read transfer failed\n");
+			goto release_bus;
+		}
+
+		if (in) {
+			memcpy(in, rx_buf, transfer_len);
+			in += transfer_len;
+		}
+
+		len -= transfer_len;
+	}
+
+release_bus:
+	/* If an error occurred, release the chip by deasserting the CS */
+	if (ret < 0)
+		spi_xfer(&tpm_spi, 0, NULL, NULL, SPI_XFER_END);
+
+	spi_tpm_release_bus();
+
+	return ret;
+}

+static int tpm_spi_recv(u8 *buf, size_t count)
+{
+	int size, expected;
+
+	if (count < TPM_HEADER_SIZE) {
+		size = -1;
+		goto out;
+	}
+
+	size = tpm_spi_recv_data(buf, TPM_HEADER_SIZE);
+	if (size < TPM_HEADER_SIZE) {
+		preboot_puts("TPM error, unable to read header\n");
+		goto out;
+	}
+
+	expected = get_unaligned_be32(buf + 2);
+	if (expected > count) {
+		size = -1;
+		goto out;
+	}
+
+	size += tpm_spi_recv_data(&buf[TPM_HEADER_SIZE],
+				   expected - TPM_HEADER_SIZE);
+	if (size < expected) {
+		preboot_puts("TPM error, unable to read remaining bytes of result\n");
+		size = -1;
+		goto out;
+	}
+

+	tpm_tis_spi_cancel();
+	tpm_tis_spi_release_locality(locality, false);

+	return size;

+		return 0;

+		preboot_puts("TPM2: Failed to get locality\n");
+		return ret;

+		preboot_puts("TPM2: Failed to write to TPM\n");

+	stop = TIS_SHORT_TIMEOUT_MS;

+static int tpm_spi_send(const u8 *buf, size_t len)
+{
+	u32 i, size;
+	u8 status;
+	int burstcnt, ret;
+	u8 data;
+
+	if (len > TPM_DEV_BUFSIZE)
+		return -E2BIG;  /* Command is too long for our tpm, sorry */
+
+	ret = tpm_tis_spi_request_locality(0);
+	if (ret < 0)
+		return -EBUSY;
+
+	/*
+	 * Check if the TPM is ready. If not, if not, cancel the pending command
+	 * and poll on the status to be finally ready.
+	 */
+	ret = tpm_spi_status(&status);
+	if (ret)
+		return ret;
+
+	if (!(status & TPM_STS_COMMAND_READY)) {
+		/* Force the transition, usually this will be done at startup */
+		ret = tpm_tis_spi_cancel();
+		if (ret) {
+			preboot_puts("TPM2: Could not cancel previous operation\n");
+			goto out_err;
+		}
+
+		ret = tpm_spi_wait_for_stat(TPM_STS_COMMAND_READY,
+						TIS_LONG_TIMEOUT_MS, &status);
+		if (ret < 0 || !(status & TPM_STS_COMMAND_READY)) {
+			preboot_puts("status not ready after wait for stat returned err\n");
+			goto out_err;
+		}
+	}
+
+	for (i = 0; i < len - 1;) {
+		burstcnt = tpm_spi_get_burstcount();
+		if (burstcnt < 0)
+			return burstcnt;
+
+		size = min_t(int, len - i - 1, burstcnt);
+		ret = tpm_tis_spi_write(TPM_DATA_FIFO(locality),
+					buf + i, size);
+		if (ret < 0)
+			goto out_err;
+
+		i += size;
+	}
+
+	ret = tpm_tis_spi_valid_status(&status);
+	if (ret)
+		goto out_err;
+
+	if ((status & TPM_STS_DATA_EXPECT) == 0) {
+		ret = -EIO;
+		goto out_err;
+	}
+
+	ret = tpm_tis_spi_write(TPM_DATA_FIFO(locality),
+				buf + len - 1, 1);
+	if (ret)
+		goto out_err;
+
+	ret = tpm_tis_spi_valid_status(&status);
+	if (ret)
+		goto out_err;
+
+	if ((status & TPM_STS_DATA_EXPECT) != 0) {
+		ret = -EIO;
+		goto out_err;
+	}
+
+	data = TPM_STS_GO;
+	ret = tpm_tis_spi_write(TPM_STS(locality), &data, 1);
+	if (ret)
+		goto out_err;
+
+	return len;
+
+out_err:
+	tpm_tis_spi_cancel();
+	tpm_tis_spi_release_locality(locality, false);
+
+	return ret;
+}

+static int tpm_spi_cleanup(void)
+{
+	tpm_tis_spi_cancel();
+	/*
+	 * The TPM needs some time to clean up here,
+	 * so we sleep rather than keeping the bus busy
+	 */
+	mdelay(2);
+	tpm_tis_spi_release_locality(locality, false);
+
+	return 0;
+}

+static int tpm_xfer(const uint8_t *sendbuf, size_t send_size,
+	uint8_t *recvbuf, size_t *recv_size)
+{
+	u64 start, stop;
+	unsigned int count, ordinal;
+	int ret = 0;
+	u8 buf[TPM_DEV_BUFSIZE + sizeof(u8)];  /* Max buffer size + addr */
+
+	/* switch endianness: big->little */
+	count = get_unaligned_be32(sendbuf + TPM_CMD_COUNT_BYTE);
+	ordinal = get_unaligned_be32(sendbuf + TPM_CMD_ORDINAL_BYTE);
+
+	if (count == 0) {
+		preboot_puts("TPM2: no data\n");
+		return -1;
+	}
+	if (count > send_size) {
+		preboot_puts("TPM2: invalid count value\n");
+		return -1;
+	}
+
+	ret = tpm_spi_send(sendbuf, send_size);
+	if (ret < 0)
+		return ret;
+
+	start = timestamp_get_us();
+	stop = tpm_tis_calc_ordinal_duration(ordinal);
+	do {
+		ret = tpm_spi_recv(buf, sizeof(buf));
+		if (ret >= 0) {
+			if (ret > *recv_size)
+				return -1;
+			memcpy(recvbuf, buf, ret);
+			*recv_size = ret;
+			ret = 0;
+			break;
+		} else if (ret != -1) {
+			return ret;
+		}
+
+		mdelay(TPM_TIMEOUT_MS);
+		if (timeout_expired(start, MS_TO_US(stop))) {
+			preboot_puts("TPM2: timeout expired\n");
+			ret = -1;
+			break;
+		}
+	} while (ret);
+
+	if (ret)
+		tpm_spi_cleanup();
+
+	return 0;
+}

+{
+	int err, ret;
+	u8 response_buffer[COMMAND_BUFFER_SIZE];
+	size_t response_length;
+

+static int tpm2_unpack_byte_string(const u8 *str, size_t size,
+					const char *format, ...)
+{
+	va_list args;
+	size_t offset = 0, length = 0;
+	u8 *ptr8 = NULL;
+	u16 *ptr16 = NULL;
+	u32 *ptr32 = NULL;
+
+	va_start(args, format);
+	for (; *format; format++) {
+		switch (*format) {
+		case 'b':
+			offset = va_arg(args, size_t);
+			ptr8 = va_arg(args, u8 *);
+			length = 1;
+			break;
+		case 'w':
+			offset = va_arg(args, size_t);
+			ptr16 = va_arg(args, u16 *);
+			length = 2;
+			break;
+		case 'd':
+			offset = va_arg(args, size_t);
+			ptr32 = va_arg(args, u32 *);
+			length = 4;
+			break;
+		case 's':
+			offset = va_arg(args, size_t);
+			ptr8 = va_arg(args, u8 *);
+			length = va_arg(args, u32);
+			break;
+		default:
+			va_end(args);
+			preboot_puts("TPM2: Couldn't recognize format string\n");
+			return -1;
+		}
+
+		if (offset + length > size) {
+			va_end(args);
+			preboot_puts("TPM2: Failed to read\n");
+			return -1;
+		}
+
+		switch (*format) {
+		case 'b':
+			*ptr8 = str[offset];
+			break;
+		case 'w':
+			*ptr16 = get_unaligned_be16(str + offset);
+			break;
+		case 'd':
+			*ptr32 = get_unaligned_be32(str + offset);
+			break;
+		case 's':
+			memcpy(ptr8, str + offset, length);
+			break;
+		}
+	}
+	va_end(args);
+
+	return 0;
+}

+static int tpm2_pack_byte_string(u8 *str, size_t size, const char *format, ...)
+{
+	va_list args;
+	size_t offset = 0, length = 0;
+	u8 *data = NULL;
+	u32 value = 0;
+
+	va_start(args, format);
+	for (; *format; format++) {
+		switch (*format) {
+		case 'b':
+			offset = va_arg(args, size_t);
+			value = va_arg(args, int);
+			length = 1;
+			break;
+		case 'w':
+			offset = va_arg(args, size_t);
+			value = va_arg(args, int);
+			length = 2;
+			break;
+		case 'd':
+			offset = va_arg(args, size_t);
+			value = va_arg(args, u32);
+			length = 4;
+			break;
+		case 's':
+			offset = va_arg(args, size_t);
+			data = va_arg(args, u8 *);
+			length = va_arg(args, u32);
+			break;
+		default:
+			preboot_puts("TPM2: Couldn't recognize format string\n");
+			va_end(args);
+			return -1;
+		}
+
+		if (offset + length > size) {
+			va_end(args);
+			return -1;
+		}
+
+		switch (*format) {
+		case 'b':
+			str[offset] = value;
+			break;
+		case 'w':
+			put_unaligned_be16(value, str + offset);
+			break;
+		case 'd':
+			put_unaligned_be32(value, str + offset);
+			break;
+		case 's':

+		}

+	return 0;
