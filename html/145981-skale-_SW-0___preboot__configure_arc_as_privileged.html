+#include <preboot.h>
+#include <psoc_arc0_mstr_if_core_hbw_regs.h>
+#include <psoc_arc0_mstr_if_core_lbw_regs.h>
+#include <psoc_arc1_mstr_if_core_hbw_regs.h>
+#include <psoc_arc1_mstr_if_core_lbw_regs.h>
+
+
+#define EN_ARC_SEC_PRIV				0x3
+void preboot_secure_arc(void)
+{
+	/* enable security and privilege mode for arc0 and arc1 */
+	regmap_writel(EN_ARC_SEC_PRIV,
+		mmPSOC_ARC0_MSTR_IF_CORE_HBW_AXPROT_OVRD);
+	regmap_writel(EN_ARC_SEC_PRIV,
+		 mmPSOC_ARC0_MSTR_IF_CORE_LBW_AXPROT_OVRD);
+	regmap_writel(EN_ARC_SEC_PRIV,
+		 mmPSOC_ARC1_MSTR_IF_CORE_HBW_AXPROT_OVRD);
+	regmap_writel(EN_ARC_SEC_PRIV,
+		 mmPSOC_ARC1_MSTR_IF_CORE_LBW_AXPROT_OVRD);
+}
+
+void preboot_security_init(void)
+{
+	preboot_secure_arc();
+}
