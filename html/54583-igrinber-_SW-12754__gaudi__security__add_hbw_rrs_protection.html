[SW-12754] gaudi: security: add hbw rrs protection

Add HBW protection with Range Registers.
The HBW RRs protection in Gaudi is implemented by using "base and mask"
concept. Therefore some restrictions are have to be coped with.
Specifically, a "base" implies the maximal protection size (the "mask")
per range register and thus we are forced to use several range registers
in some of contiguous areas (e.g. SRAM, SRAM reduction).

Note: The HBM and SRAM protection is disabled until the dynamic HBW
protection is implemented. This prevents bricking devices.

+static void preboot_secure_hbw_rr(int rr, u64 base, u64 mask)
+{
+	int i;
+	u32 base_l = (u32)base, base_h = (u32)(base >> BITS_IN_REG);
+	u32 mask_l = (u32)mask, mask_h = (u32)(mask >> BITS_IN_REG);
+	u32 rr_base, rr_dest, hit;
+
+	for (i = 0; i < IF_RTRS_NUM; i++) {
+		rr_base = rr * sizeof(u32) + IF_RTRS_DIST * i;
+		rr_dest = rr_base + mmSIF_RTR_CTRL_0_RANGE_PRIV_BASE_LOW_AW_0;
+		regmap_writel(base_l, rr_dest);
+		rr_dest = rr_base + mmSIF_RTR_CTRL_0_RANGE_PRIV_BASE_HIGH_AW_0;
+		regmap_writel(base_h, rr_dest);
+		rr_dest = rr_base + mmSIF_RTR_CTRL_0_RANGE_PRIV_MASK_LOW_AW_0;
+		regmap_writel(mask_l, rr_dest);
+		rr_dest = rr_base + mmSIF_RTR_CTRL_0_RANGE_PRIV_MASK_HIGH_AW_0;
+		regmap_writel(mask_h, rr_dest);
+		rr_dest = rr_base + mmSIF_RTR_CTRL_0_RANGE_PRIV_BASE_LOW_AR_0;
+		regmap_writel(base_l, rr_dest);
+		rr_dest = rr_base + mmSIF_RTR_CTRL_0_RANGE_PRIV_BASE_HIGH_AR_0;
+		regmap_writel(base_h, rr_dest);
+		rr_dest = rr_base + mmSIF_RTR_CTRL_0_RANGE_PRIV_MASK_LOW_AR_0;
+		regmap_writel(mask_l, rr_dest);
+		rr_dest = rr_base + mmSIF_RTR_CTRL_0_RANGE_PRIV_MASK_HIGH_AR_0;
+		regmap_writel(mask_h, rr_dest);
+
+		/* Enable (hit) RR */
+		rr_base = IF_RTRS_DIST * i;
+		rr_dest = rr_base + mmSIF_RTR_CTRL_0_RANGE_PRIV_HIT_AW;
+		hit = regmap_readl(rr_dest);
+		regmap_writel(hit | BIT(rr), rr_dest);
+
+		rr_dest = rr_base + mmSIF_RTR_CTRL_0_RANGE_PRIV_HIT_AR;
+		hit = regmap_readl(rr_dest);
+		regmap_writel(hit | BIT(rr), rr_dest);
+	}
