+static u32 preboot_get_vendor_id_device_id(void)
+{
+	u32 pci_dev_id = PCI_IDS_GAUDI;
+	char devid[5];
+
+	/* Read the second part of the device ID which contains the step */
+	if (!preboot_fuse_read(FUSE_DEV_ID_WORD_OFFSET + 1, (u32 *)devid)) {
+		devid[4] = '\0';
+		/* Check for device type B1 and set the PCI ID accordingly */
+		if (!strcmp(&devid[2], "B1"))
+			pci_dev_id = PCI_IDS_GAUDI_HL2000M;
+	}
+
+	return (pci_dev_id << 16) | PCI_VENDOR_ID_HABANALABS;
+}
