+#include "boot_desc.h"

+#define MAX_IMGS		5
+
+#define BOOT_PTRS_SZ		sizeof(struct hl_fw_boot_ptrs)
+#define IMG_DESC_SZ		sizeof(struct hl_fw_img_desc)
+
+static struct hl_fw_boot_ptrs boot_ptrs;
+static struct hl_fw_img_desc img_desc[MAX_IMGS];
+
+static void boot_desc_print_magic(char *name, struct hl_fw_boot_magic *magic)
+{
+	int i;
+
+	preboot_puts_raw(name);
+	preboot_puts_raw(" magic: ");
+	/* Strings do not obey endianness - print in reverse order */
+	for (i = ARRAY_SIZE(magic->name) - 1; i >= 0; i--)
+		preboot_putc_raw(magic->name[i]);
+
+	preboot_puts_hex_raw(" type: ", magic->type, false);
+	preboot_puts_hex(" value: ", magic->val, true);
+}
+
+static bool boot_desc_img_desc_is_valid(struct hl_fw_img_desc *img_desc)
+{
+	boot_desc_print_magic("FW IMG DESC", &img_desc->magic);
+
+	if ((img_desc->magic.val & 0xFFFFFF00) != HL_FW_IMG_DESC_MAGIC)
+		return false;
+
+	if (img_desc->desc_ver == 0 || img_desc->desc_ver == ~0)
+		return false;
+
+	if (img_desc->idx >= 2)
+		return false;
+
+	return true;
+}
+
+int boot_desc_init(void)
+{
+	unsigned int count, i;
+	u32 spi_off = HL_FW_BOOT_PTRS_BASE;
+	u32 desc_offs[MAX_IMGS] = { 0 };
+
+	/* Read the main boot pointers structure */
+	spi_copy(&boot_ptrs, spi_off, BOOT_PTRS_SZ, "BOOT PTRS");
+	boot_desc_print_magic("FW boot PTRs", &boot_ptrs.magic);
+
+	if (boot_ptrs.magic.val != HL_FW_BOOT_MAGIC)
+		return 0;
+
+	count = boot_ptrs.count;
+
+	/* Read all the descriptors offsets */
+	spi_off += sizeof(struct hl_fw_boot_ptrs);
+	spi_copy(desc_offs, spi_off, sizeof(u32) * count, "DESC OFFS");
+
+	/* Read all descriptors */
+	for (i = 0; i < count; i++) {
+		spi_off = HL_FW_BOOT_PTRS_BASE + desc_offs[i];
+		spi_copy(&img_desc[i], spi_off, IMG_DESC_SZ, "IMG DESC");
+		if (!boot_desc_img_desc_is_valid(&img_desc[i])) {
+			preboot_puts_hex_raw("Bad boot descriptor: ", i, false);
+			return 0;
+		}
+	}
+
+	return count;
+}
+
+u32 boot_desc_get_img_desc(enum fw_post_reset_cmd upd_status,
+			   enum hl_fw_img_type type, struct boot_desc *desc)
+{
+	int i, idx;
+
+	preboot_puts_hex_raw("Looking for image type: ", type, false);
+	for (i = 0; i < boot_ptrs.count; i++) {
+		if (img_desc[i].magic.type != type)
+			continue;
+
+		idx = img_desc[i].idx;
+		if (upd_status == FLOPS_UPD_PENDING)
+			idx = (idx + 1) % 2;
+
+		preboot_puts_hex(" - found! using index: ", idx, false);
+		desc->spi_offset = img_desc[i].offset[idx];
+		desc->img_size = img_desc[i].size[idx];
+		desc->loadaddr = ((u64)img_desc[i].load_high[idx] << 32) |
+					img_desc[i].load_low[idx];
+		desc->desc_ver = img_desc[i].desc_ver;
+
+		/*
+		 * Disable chksum copy temporarily, until alignment issue
+		 * is resolved.
+		 */
+		/* memcpy(desc->chksum, img_desc[i].chksum[idx], */
+		/* ARRAY_SIZE(desc->chksum)); */
+
+		return idx;
+	}
+
+	preboot_puts(" - not found!\n");
+	return HL_FW_IMG_NOT_FOUND;
+}
