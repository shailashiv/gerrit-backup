[SW-10446] preboot: get u-boot over pcie

This is an alternative to previous method which retrieves uboot from
SPI. Now we have slim SPI image without U-Boot on it.
Host pushes U-Boot image to sram in FIT format.
Implement the receiving side.
The implementation is done in generic way.
Each module serving requests from host or any other external entity, is
registered in the common handler and is required to implement the set of
methods defined in struct cmd_module_ops.
Then the main loop walks through the modules and gives them an
opportunity to handle requests they claim to be responsible of.
As a part of the transition, the LKD and FW UPD handling modules are
implemented and also serve as an example for adding more modules as
needed.

+/*

+#ifndef _HL_CMD_MODULE_H_
+#define _HL_CMD_MODULE_H_
+
+/*
+ * enum cmd_check_res	- used to specify various results that check() routine
+ *			  can return following its check.
+ *
+ * NOT_FOUND		- check() routine has not discovered any request which
+ *			  is related to its module.
+ * FOUND_FAST_RETURN	- check() routine has discovered a request that is very
+ *			  fast to handle.
+ * FOUND_SLOW_RETURN	- check() routine has discovered a request that will
+ *			  take relatively large amount time to handle and after
+ *			  it has been handled, the execution will return back
+ * FOUND_NO_RETURN	- check() routine has discovered a request and its
+ *			  handling will never return back.
+ */
+enum cmd_check_res {
+	NOT_FOUND,
+	FOUND_FAST_RETURN,
+	FOUND_SLOW_RETURN,
+	FOUND_NO_RETURN,
+};
+
+/*
+ * struct cmd_module_ops	- operations defined for all registered modules
+ *
+ * setup()		- setup the module and prepare checking for
+ *			  commands/requests and handling them
+ * check()		- check if there is a command/request pending and return
+ *			  the appropriate result
+ * disable()		- temporarily disable the module to enable another
+ *			  module execution of run() routine
+ * run()		- run the handling of the command/request discovered by
+ *			  the check() routine - can be slow since other modules
+ *			  have been disabled by disable() routine
+ * run_fast()		- run the handling of the command/request discovered by
+ *			  the check() routine - must take low amount of time
+ *			  since other modules have not been disabled
+ * teardown()		- the opposite from setup(), the module will no longer
+ *			  be checked or handled - stop the module
+ * run_noreturn()	- all modules have been teared down, so run the handling
+ *			  w/o returning
+ */
+struct cmd_module_ops {
+	void (*setup)(void);
+	enum cmd_check_res (*check)(void);
+	void (*disable)(void);
+	void (*run)(void);
+	void (*enable)(void);
+	void (*run_fast)(void);
+	void (*teardown)(void);
+	void (*run_noreturn)(void);
+};
+
+/* a dummy to fill in irrelevant entries */
+static inline void noop(void) {}
+
+/* FWUPD */
+void module_fwupd_setup(void);
+enum cmd_check_res module_fwupd_check(void);
+void module_fwupd_run(void);
+
+/* LKD */
+void module_lkd_setup(void);
+enum cmd_check_res module_lkd_check(void);
+void module_lkd_disable(void);
+void module_lkd_run(void);
+void module_lkd_run_noret(void);
+
+#endif /* _HL_CMD_MODULE_H_ */

+void module_fwupd_setup(void)
+{
+	struct hl_upd_status status  = { .val = 0 };
+
+	status.status = HL_UPD_STS_RST_OK;
+	regmap_writel(status.val, mmUPD_STS);
+}
+
+enum cmd_check_res module_fwupd_check(void)
+{
+	if (regmap_readl(mmUPD_CMD) != HL_UPD_NOOP)
+		return FOUND_FAST_RETURN;
+
+	return NOT_FOUND;
+}
+
+void module_fwupd_run(void)
+{
+	struct hl_upd_command cmd;
+
+	cmd.val = regmap_readl(mmUPD_CMD);
+	switch (cmd.cmd) {
+	case HL_UPD_CLR_STS:
+		regmap_writel(HL_UPD_STS_NOOP, mmUPD_STS);
+		preboot_puts("FW UPD CLR STS received\n");
+		break;
+	default:
+		break;
+	}
+}

+#include "cmd_module.h"
+#include "uart.h"
+#include "boot_if.h"
+#include "fit.h"
+
+struct lkd_status {
+	int fit_not_checked;	/* negative logic */
+	u32 msg_reg_val;
+};
+
+static struct lkd_status sts = {
+	.fit_not_checked = 1,
+};
+
+void module_lkd_setup(void)
+{
+	cpu_boot_sts_set(CPU_BOOT_STATUS_WAITING_FOR_BOOT_FIT);
+}
+
+enum cmd_check_res module_lkd_check(void)
+{
+	enum cmd_check_res res;
+
+	/* we already have the FIT and it is good */
+	if (sts.fit_not_checked != 1)
+		return FOUND_NO_RETURN;
+
+	sts.msg_reg_val = cpu_boot_cmd_get();
+	switch (sts.msg_reg_val) {
+	case KMD_MSG_GOTO_WFE:
+		res = FOUND_NO_RETURN;
+		/* WFE - we should clear the message/cmd register */
+		cpu_boot_cmd_clr_before_wfe();
+		cpu_boot_rsp_set(CPU_MSG_OK);
+		preboot_puts("LKD WFE received\n");
+		break;
+	case KMD_MSG_FIT_RDY:
+		res = FOUND_SLOW_RETURN;
+		/* Tell LKD we got the message */
+		cpu_boot_rsp_set(CPU_MSG_OK);
+		preboot_puts("LKD FIT RDY received\n");
+		break;
+	case KMD_MSG_SKIP_BMC:
+		cpu_boot_rsp_set(CPU_MSG_ERR);
+		preboot_puts("LKD SKIP BMC received\n");
+		/* no break */
+	default:
+		res = NOT_FOUND;
+	}
+
+	return res;
+}
+
+void module_lkd_disable(void)
+{
+	cpu_boot_sts_set(CPU_BOOT_STATUS_IN_PREBOOT);
+}
+
+void module_lkd_run(void)
+{
+	int err;
+	void *fit = (void *)UBOOT_FIT_BASE;
+	void *dst = (void *)UBOOT_TEXT_BASE;
+
+	err = !fit_check_format(fit);
+	if (!err)
+		err = fit_load_uboot(fit, dst);
+
+	if (err) {/* Report the FIT is bad */
+		cpu_boot_err_clrset(CPU_BOOT_ERR0_FIT_CORRUPTED, true);
+		preboot_puts("FIT loading failed, Retrying...\n");
+		return;
+	}
+
+	sts.fit_not_checked = 0;
+	/* Report the FIT is good */
+	cpu_boot_err_clrset(CPU_BOOT_ERR0_FIT_CORRUPTED, false);
+}
+
+__attribute__((weak)) void uboot_start(void)
+{
+	void (*uboot_entry)(void) = (void *)UBOOT_TEXT_BASE;
+
+	uboot_entry();
+}
+
+void module_lkd_run_noret(void)
+{
+	switch (sts.msg_reg_val) {
+	case KMD_MSG_GOTO_WFE:
+		cpu_boot_sts_set(CPU_BOOT_STATUS_IN_WFE);
+		wfe();
+		break;
+	case KMD_MSG_FIT_RDY:
+		uboot_start();
+		/* no break */
+	default:
+		break;
+	}
+}
