+/*
+ * fuse box is divided into half for redundancy
+ * each half is bank of FUSE_WORDS_PER_BANK words
+ * NOTE: bank1 is a complete duplication of bank0
+ */
+#define FUSE_TIMEOUT_US		100000
+#define FUSE_DELAY_US		20
+#define FUSE_RETRIES		(FUSE_TIMEOUT_US / FUSE_DELAY_US)
+
+static int efuse_init;
+
+int preboot_fuse_init(void)
+{
+	int i;
+
+	if (efuse_init == 1)
+		return -1;
+
+	regmap_writel(1, mmPSOC_EFUSE_READ);
+
+	for (i = 0; i < FUSE_RETRIES; i++)
+		if (regmap_readl(mmPSOC_EFUSE_READ))
+			udelay(FUSE_DELAY_US);
+		else
+			break;
+
+	if (i == FUSE_RETRIES)
+		return -2;
+
+	efuse_init = 1;
+	return 0;
+}
+
+int preboot_fuse_read(u32 word, u32 *val)
+{
+	int offset, ret;
+	u32 read_val;
+
+	if (word >= FUSE_WORDS_PER_BANK) {
+		preboot_puts("Invalid efuse word offset\n");
+		return -1;
+	}
+
+	ret = preboot_fuse_init();
+	if (ret == -2) {
+		preboot_puts("Efuse not initialized\n");
+		return ret;
+	}
+
+	/* read value from bank0 */
+	offset = (word * sizeof(u32));
+	read_val = regmap_readl(mmPSOC_EFUSE_DATA_0 + offset);
+
+	/* read value from bank1 and OR with value from bank0 */
+	offset = ((FUSE_WORDS_PER_BANK + word) * sizeof(u32));
+	read_val |= regmap_readl(mmPSOC_EFUSE_DATA_0 + offset);
+
+	*val = read_val;
+
+	return 0;
+}
