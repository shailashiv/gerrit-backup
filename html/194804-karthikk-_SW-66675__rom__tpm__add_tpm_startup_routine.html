+static int tpm_spi_send(const uint8_t *buf, size_t len)

+	int burstcnt, ret;

+		return -EBUSY;

+	 * Check if the TPM is ready. If not, if not, cancel the pending command
+	 * and poll on the status to be finally ready.
+	 */

+		return ret;

+static int tpm_xfer(const uint8_t *sendbuf, size_t send_size,
+	uint8_t *recvbuf, size_t *recv_size)
+{
+	unsigned int count;
+	int ret = 0;
+
+	/* switch endianness: big->little */
+	count = get_unaligned_be32(sendbuf + TPM_CMD_COUNT_BYTE);
+
+	if (count == 0) {
+		return -1;
+	}
+	if (count > send_size) {
+		return -1;
+	}
+
+	ret = tpm_spi_send(sendbuf, send_size);
+	if (ret < 0)
+		return ret;
+
+	ret = tpm_spi_recv(tpm_buf, sizeof(tpm_buf));
+	if (ret >= 0) {
+		if (ret > (int)*recv_size)
+			return -1;
+		memcpy(recvbuf, tpm_buf, ret);
+		*recv_size = ret;
+		ret = 0;
+	}
+
+	return ret;
+}

+static int tpm_spi_sendrecv_command(const void *command,

+{
+	int err;

+	err = tpm_xfer(command, tpm_command_size(command),
+		       response, &response_length);
+
+	if (err < 0)
+		return err;
+
+	if (size_ptr)
+		*size_ptr = response_length;
+
+	return tpm_return_code(response);
