+secondary_entry:
+	/* stack top set to secondary CPU DCCM 1k offset (internal address) */
+	mov	sp, __stack2_top
+	mov	fp, sp
+
+	/* Disable un-aligned checks on data memory */
+	lr	r0, [STATUS32]
+	bset	r0, r0, 0x13
+	kflag	r0
+	bl	secondary_main
