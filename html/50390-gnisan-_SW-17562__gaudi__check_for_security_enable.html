+/* Enable bit is SOC_STRAP10 in schematics */
+#define SECURITY_ENABLE_SW BIT(10)
+
+/* The routine shall return true in case of security enable in strap.
+ * Otherwise, shall return false (The polarity of strap is inverted:
+ * meaning '0' is for security enable).
+ */
+static bool preboot_security_is_strap_enabled(void)
+{
+	u32 boot_strap;
+
+	boot_strap = regmap_readl(mmPSOC_GLOBAL_CONF_BOOT_STRAP_PINS);
+
+	return !(boot_strap & SECURITY_ENABLE_SW);
+}
+
+/* Security is enabled in the following condition:
+ * Boot strap pin for security is enabled or efuse bit
+ * for "security enabled" is true.
+ */
+static bool preboot_security_is_enabled(void)
+{
+	const struct hl_sec_efuse *fuse = preboot_security_get_fuse();
+
+	if (preboot_security_is_strap_enabled())
+		return true;
+
+	/* If eFuse read failed, assume an attacker sabotage, security in on */
+	if (!fuse)
+		return true;
+
+	return !!fuse->security_enable;
+}
