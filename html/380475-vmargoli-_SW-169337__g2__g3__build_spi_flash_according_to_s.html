+function hl_report_missing_spi_region_and_exit {
+	local red=$(tput setaf 1)
+	local normal=$(tput sgr0)
+
+	printf "$red The definition of SPI flash region %s not found $normal \n" $1
+	exit 1
+}

+function hl_parse_spi_flash_regions {
+	local -n spi_regions=$1
+	local platform_tokens=(${HL_DEV//_/ })
+	local mem_addr_header_file=${BR2_EXTERNAL_HABANALABS_PATH}/../specs/${platform_tokens[0]}/${platform_tokens[0]}_mem_addr_dt.h
+	local regex="\#define(.*)SPI_FLASH_REGION_([A-Z_0-9]+)_SIZE(\s+)0x([0-9A-F]+)"
+	local region_name
+	local region_size
+	local red=$(tput setaf 1)
+	local normal=$(tput sgr0)
+
+	if [ ! -f $mem_addr_header_file ]; then
+		printf "$red File %s not exists $normal \n" "$mem_addr_header_file"
+		exit 1
+	fi
+	printf "%s\n" "SPI flash region sizes"
+	printf "%s\n" "----------------------"
+	IFS=$'\n'
+	for line in $(cat "$mem_addr_header_file")
+	do
+		if [[ $line =~ $regex ]]; then
+			region_name=${BASH_REMATCH[2]}
+			region_size=$(echo "ibase=16; ${BASH_REMATCH[4]}"|bc)
+			spi_regions[${region_name}]=$region_size
+			printf "%s: %s bytes \n" "$region_name" "$region_size"
+		fi
+	done
+}
