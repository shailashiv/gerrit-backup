[FSW-9512] arcboot: enable cpld watchdog timer

This code will start cpld boot watchdog timer before going
to ppboot. If boot WD timer expires CPLD will set
Force Recovery GPIO(GPIO_01). When boot rom check this GPIO
in next reset which is because of ppboot load fail, leads to
enter recovery.

+/*
+ * Request CPLD to start the watchdog, by asserting
+ * a GPIO_66 (Boot_WD).
+ */
+#define GPIOS_PER_BANK			16
+#define FS1_CPLD_WDOG_GPIO		66
+void boot_rom_start_cpld_wdtimer(void)
+{
+	uint32_t index = FS1_CPLD_WDOG_GPIO % GPIOS_PER_BANK;
+	uint32_t val, mask = BIT(index);
+
+	regmap_writel(GPIO_FUNC_1, IO0_H0_PSOC_CFG_PAD_CONF_PAD_SEL(mmPSOC_PAD_CONF_PAD_SEL_66));
+
+	/* configure GPIO in output mode */
+	val = regmap_readl(I0_H0_PSOC_GPIO4(mmGPIO_OUTENSET));
+	val = val | mask;
+	regmap_writel(val, I0_H0_PSOC_GPIO4(mmGPIO_OUTENSET));
+
+	/* gpio is active high */
+	val = regmap_readl(I0_H0_PSOC_GPIO4(mmGPIO_DATAOUT));
+	val = val | mask;
+	regmap_writel(val, I0_H0_PSOC_GPIO4(mmGPIO_DATAOUT));
+
