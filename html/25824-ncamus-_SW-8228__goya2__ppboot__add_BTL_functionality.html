+void init_prstn(void)
+{
+	/* Configure PRSTN to reset entire SoC */
+	regmap_writel(0xFFFFFFFF, mmPSOC_GLOBAL_CONF_PRSTN_RST_CFG_L);
+	regmap_writel(0x3FFFFF, mmPSOC_GLOBAL_CONF_PRSTN_RST_CFG_H);
+
+	/* Clear masking of PRSTN from SW reset */
+	regmap_writel(0x0, mmPSOC_GLOBAL_CONF_PRSTN_MASK);
+
+	/*
+	 * Hold PCIe controller at reset during PRSTn
+	 * Configure Boot Sequencer to wait for PRSTn
+	 * instead of PCLK stable
+	 */
+	regmap_writel(0x1, mmPSOC_GLOBAL_CONF_PHY_STABLE);

+	 * for ext_ld_done) ext_ld_done will be configured during
+	 * the PCIe image as part of the FW update flow
+	 */
