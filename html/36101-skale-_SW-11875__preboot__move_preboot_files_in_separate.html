Commit Message

[SW-11875] preboot: move preboot files in separate directory

This patch adds changes to move files used only by preboot to preboot
directory.

The idea here to compile preboot only with the files preboot binary
uses. In order to achieve that, files in common folders are segregated
where files used by preboot only are moved to preboot folder and files
used by both preboot and ppboot are kept in common folder. This helps
to add single entries in Makefile instead of individual files. Along
with this, preboot files under app folder are also moved in preboot
directory.

Makefile changes are done to accommodate above changes. Also
change is made to copy all binaries in bin directory.

+static int is_public_exponent_bit_set(const struct rsa_public_key *key,

+/**
+ * subtract_modulus() - subtract modulus from the given value
+ *
+ * @key:	Key containing modulus to subtract
+ * @num:	Number to subtract modulus from, as little endian word array
+ */
+static void subtract_modulus(const struct rsa_public_key *key, u32 num[])
+{
+	long long acc = 0;
+	u32 i;
+
+	for (i = 0; i < key->len; i++) {
+		acc += (u64)num[i] - key->modulus[i];
+		num[i] = (u32)acc;
+		acc >>= 32;
+	}

+ * montgomery_mul_add_step() - Perform montgomery multiply-add step
+ *
+ * Operation: montgomery result[] += a * b[] / n0inv % modulus
+ *
+ * @key:	RSA key
+ * @result:	Place to put result, as little endian word array
+ * @a:		Multiplier
+ * @b:		Multiplicand, as little endian word array

+static void montgomery_mul_add_step(const struct rsa_public_key *key,
+				    u32 result[], const u32 a, const u32 b[])
+{
+	u64 acc_a, acc_b;
+	u32 d0;
+	u32 i;

+	acc_a = (u64)a * b[0] + result[0];
+	d0 = (u32)acc_a * key->n0inv;
+	acc_b = (u64)d0 * key->modulus[0] + (u32)acc_a;
+	for (i = 1; i < key->len; i++) {
+		acc_a = (acc_a >> 32) + (u64)a * b[i] + result[i];
+		acc_b = (acc_b >> 32) + (u64)d0 * key->modulus[i] +
+				(u32)acc_a;
+		result[i - 1] = (u32)acc_b;
+	}
+
+	acc_a = (acc_a >> 32) + (acc_b >> 32);

+	result[i - 1] = (u32)acc_a;

+ * @key:	RSA key

+ * @b:		Multiplicand, as little endian word array

+static int pow_mod(const struct rsa_public_key *key, u32 *out, u32 *in)
+{
+	u32 *result, *ptr;
+	u32 i;
+	int j, k;
+
+	u32 val[key->len], acc[key->len], tmp[key->len];
+	u32 a_scaled[key->len];
+
+	result = tmp;	/* Re-use location. */
+
+	/* Convert from big endian byte array to little endian word array. */
+	for (i = 0, ptr = in + key->len - 1; i < key->len; i++, ptr--)
+		val[i] = get_unaligned_be32(ptr);
+
+	if (num_public_exponent_bits(key, &k) != 0) {
+		preboot_puts("num_public_exponent_bits failed!\n");
+		return 1;
+	}
+
+	if (k < 2) {
+		preboot_puts("Public exponent is too short\n");
+		return 1;
+	}
+
+	if (!is_public_exponent_bit_set(key, 0)) {
+		preboot_puts("LSB of RSA public exponent must be set.\n");
+		return 1;
+	}
+
+	/* the bit at e[k-1] is 1 by definition, so start with: C := M */
+	montgomery_mul(key, acc, val, key->rr); /* acc = a * RR / R mod n */
+	/* retain scaled version for intermediate use */
+	memcpy(a_scaled, acc, key->len * sizeof(a_scaled[0]));
+
+	for (j = k - 2; j > 0; --j) {
+		montgomery_mul(key, tmp, acc, acc); /* tmp = acc^2 / R mod n */
+
+		if (is_public_exponent_bit_set(key, j)) {
+			/* acc = tmp * val / R mod n */
+			montgomery_mul(key, acc, tmp, a_scaled);
+		} else {
+			/* e[j] == 0, copy tmp back to acc for next operation */
+			memcpy(acc, tmp, key->len * sizeof(acc[0]));
+		}
+	}
+
+	/* the bit at e[0] is always 1 */
+	montgomery_mul(key, tmp, acc, acc); /* tmp = acc^2 / R mod n */
+	montgomery_mul(key, acc, tmp, val); /* acc = tmp * a / R mod M */
+	memcpy(result, acc, key->len * sizeof(result[0]));
+
+	/* Make sure result < mod; result is at most 1x mod too large. */
+	if (greater_equal_modulus(key, result))
+		subtract_modulus(key, result);
+
+	/* Convert to bigendian byte array */
+	for (i = key->len - 1, ptr = out; (int)i >= 0; i--, ptr++)
+		put_unaligned_be32(result[i], ptr);
+	return 0;
+}

+static int rsa_verify_padding(const u8 *msg, const int pad_len)
+{
+	int ff_len;
+	int ret;
+
+	/* first byte must be 0x00 */
+	ret = *msg++;
+	/* second byte must be 0x01 */
+	ret |= *msg++ ^ 0x01;
+	/* next ff_len bytes must be 0xff */
+	ff_len = pad_len - CHECKSUM_DER_LENGTH - 3;
+	ret |= *msg ^ 0xff;
+	ret |= memcmp(msg, msg+1, ff_len-1);
+	msg += ff_len;
+	/* next byte must be 0x00 */
+	ret |= *msg++;
+	/* next der_len bytes must match der_prefix */
+	ret |= memcmp(msg, CHECKSUM_DER_PREFIX, CHECKSUM_DER_LENGTH);
+
+	return ret;
