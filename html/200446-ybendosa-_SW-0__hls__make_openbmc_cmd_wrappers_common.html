+U_HL_DEV=$(shell echo ${HL_DEV} | tr '[:lower:]' '[:upper:]')
+
+openbmc-help:
+	@echo
+	@echo 'OpenBMC bitbake & devtool wrappers for Habana Labs server build'
+	@echo '---------------------------------------------------------------'
+	@echo
+	@echo 'OpenBMC devtool commands can be used for adding new'
+	@echo ' packages and modifying existing ones, these targets wrap its'
+	@echo ' functionality:'
+	@echo
+	@echo '1. openbmc-help:'
+	@echo '   Print this help menu.'
+	@echo '2. openbmc-add:'
+	@echo '   Adds a new package from source, and generates a recipe for it.'
+	@echo '   Usage*:'
+	@echo '   $ make openbmc-add URI="<git://[...]>"'
+	@echo '   * In case of ssh repos, ";protocol=ssh" should be added as a'
+	@echo '     suffix.'
+	@echo '3. openbmc-<pkg-name>-modify:'
+	@echo '   Deployes a package already managed by OpenBMC and opens a'
+	@echo '   terminal in its sources path.'
+	@echo '   From now on, each change in the sources will trigger'
+	@echo '   a new build when calling 'make $(HL_DEV)-build'.'
+	@echo '   Usage:'
+	@echo '   $ make openbmc-<pkg-name>-modify'
+	@echo '4. openbmc-<pkg-name>-menuconfig:'
+	@echo '   Opens a 'menuconfig' session for the package'
+	@echo '   (if it has a menuconfig target).'
+	@echo '   Usage:'
+	@echo '   $ make openbmc-<pkg-name>-menuconfig'
+	@echo '5. openbmc-<pkg-name>-build:'
+	@echo '   Builds a package that is already deployed via'
+	@echo '   openbmc-<pkg-name>-modify.'
+	@echo '   Usage:'
+	@echo '   $ make openbmc-<pkg-name>-build'
+	@echo '6. openbmc-<pkg-name>-clean:'
+	@echo '   Cleans the build artifacts of the package.'
+	@echo '   Usage:'
+	@echo '   $ make openbmc-<pkg-name>-clean'
+	@echo '7. openbmc-<pkg-name>-remove*:'
+	@echo '   Clears the package files from the local environment*.'
+	@echo '   Usage:'
+	@echo '   $ make openbmc-<pkg-name>-remove'
+	@echo '   * The sources dir will be deleted, so one needs to make'
+	@echo '     sure his work was pushed before running it.'
+	@echo '8. openbmc-<pkg-name>-diffconfig:'
+	@echo '   Opens a 'diffconfig' session for the package'
+	@echo '   (if it has a diffconfig target).'
+	@echo '   Usage:'
+	@echo '   $ make openbmc-<pkg-name>-diffconfig'
+
+inner-openbmc-add:
+	@cd $($(U_HL_DEV)_DEPLOY_DIR);			\
+	$(U_HL_DEV)_ENV $($(U_HL_DEV)_BITBAKE_ENV)	\
+	devtool add "$(URI)"
+
+openbmc-add: $(HL_DEV)-asic-prepare
+	@$(MAKE) inner-$@
+
+inner-openbmc-%-modify:
+	@cd $($(U_HL_DEV)_DEPLOY_DIR);					\
+	$($(U_HL_DEV)_ENV) $($(U_HL_DEV)_BITBAKE_ENV)			\
+	devtool modify $(patsubst inner-openbmc-%-modify,%,$@);		\
+	terminator							\
+	--working-directory="$($(U_HL_DEV)_DEVTOOL_SOURCES)/$(patsubst	\
+		inner-openbmc-%-modify,%,$@)" &
+
+openbmc-%-modify: $(HL_DEV)-asic-prepare
+	@$(MAKE) inner-$@
+
+inner-openbmc-%-menuconfig:
+	@cd $($(U_HL_DEV)_DEPLOY_DIR);					\
+	$($(U_HL_DEV)_ENV $($(U_HL_DEV)_BITBAKE_ENV))			\
+	bitbake -c menuconfig $(patsubst inner-openbmc-%-menuconfig,%,$@)
+
+openbmc-%-menuconfig: $(HL_DEV)-asic-prepare
+	@$(MAKE) inner-$@
+
+inner-openbmc-%-diffconfig:
+	@cd $($(U_HL_DEV)_DEPLOY_DIR);					\
+	$($(U_HL_DEV)_ENV) $($(U_HL_DEV)_BITBAKE_ENV)			\
+	bitbake -c diffconfig $(patsubst inner-openbmc-%-diffconfig,%,$@)
+
+openbmc-%-diffconfig: $(HL_DEV)-asic-prepare
+	@$(MAKE) inner-$@
+
+inner-openbmc-%-build:
+	@cd $($(U_HL_DEV)_DEPLOY_DIR);					\
+	$($(U_HL_DEV)_ENV) $($(U_HL_DEV)_BITBAKE_ENV)			\
+	devtool build $(patsubst inner-openbmc-%-build,%,$@)
+
+openbmc-%-build: $(HL_DEV)-asic-prepare
+	@$(MAKE) inner-$@
+
+inner-openbmc-%-clean:
+	@cd $($(U_HL_DEV)_DEPLOY_DIR);					\
+	$($(U_HL_DEV)_ENV) $($(U_HL_DEV)_BITBAKE_ENV)			\
+	bitbake -c clean $(patsubst inner-openbmc-%-clean,%,$@)
+
+openbmc-%-clean: $(HL_DEV)-asic-prepare
+	@$(MAKE) inner-$@
+
+inner-openbmc-%-cleanall:
+	@cd $($(U_HL_DEV)_DEPLOY_DIR);					\
+	$($(U_HL_DEV)_ENV) $($(U_HL_DEV)_BITBAKE_ENV)			\
+	bitbake -c cleanall $(patsubst inner-openbmc-%-cleanall,%,$@)
+
+openbmc-%-cleanall: $(HL_DEV)-asic-prepare
+	@$(MAKE) inner-$@
+
+openbmc-%-remove:
+	@cd $($(U_HL_DEV)_DEPLOY_DIR);					\
+	rm -rf $($(U_HL_DEV)_WORK_DIR)/workspace/sources/$(patsubst	\
+	openbmc-%-remove,%,$@);						\
+	rm -rf $($(U_HL_DEV)_WORK_DIR)/workspace/recipes/$(patsubst	\
+	openbmc-%-remove,%,$@);						\
+	rm -f $($(U_HL_DEV)_WORK_DIR)/workspace/appends/$(patsubst	\
+	openbmc-%-remove,%,$@)*
