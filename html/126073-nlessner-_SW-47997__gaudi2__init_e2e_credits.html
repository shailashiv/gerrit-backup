+static void gaudi2_init_module_mstr_if(u32 block_idx, u32 instances,
+					   u64 block_base, u64 block_off,
+					   u64 instance_off,
+					   unsigned long enable_mask,
+					   struct block_ctx *cfg_ctx)
+{
+	u64 base = block_base + block_idx * block_off;
+	int i, seq;
+
+	for (i = 0; i < instances; i++) {
+		seq = block_idx * instances + i;
+
+		if (!test_bit(seq, &enable_mask))
+			continue;
+
+		cfg_ctx->instance_cfg_fn(base + i * instance_off,
+								cfg_ctx->data);
+	}
+}

+static void gaudi2_init_e2e_mstr_if_cfg(u64 base, void *data)
+{
+	int i;
+	struct e2e_cfg_data *e2e_data = (struct e2e_cfg_data *)data;
+
+	regmap_writel(0x1, base + MSTR_IF_E2E_CRDT_E2E_EMEM_EN_OFFSET);
+	regmap_writel(0x1, base + MSTR_IF_E2E_CRDT_E2E_PCI_EN_OFFSET);
+	regmap_writel(0x1, base +
+		      MSTR_IF_E2E_CRDT_E2E_EMEM_SB_RELEASE_EN_OFFSET);
+	regmap_writel(0x1, base +
+		      MSTR_IF_E2E_CRDT_E2E_PCI_SB_RELEASE_EN_OFFSET);
+
+	/*
+	 * Actual initialization of the component "credits vector".
+	 * each credits vector has 16 entries in which the indexes groups
+	 * {0-3}, {4-7}, {8-11} and {12-15} has the same value.
+	 */
+	for (i = 0; i <= E2E_PCIE; i++) {
+		regmap_writel(e2e_data->emem_rd_sz.cfg_val_idx_0_3, i * 4 +
+			      base +
+			      MSTR_IF_E2E_CRDT_E2E_EMEM_RD_SIZE_0_OFFSET);
+		regmap_writel(e2e_data->emem_wr_sz.cfg_val_idx_0_3, i * 4 +
+			      base +
+			      MSTR_IF_E2E_CRDT_E2E_EMEM_WR_SIZE_0_OFFSET);
+	}
+	for (i = E2E_HBM_DMA; i <= E2E_HBM_MMU; i++) {
+		regmap_writel(e2e_data->emem_rd_sz.cfg_val_idx_4_7, i * 4 +
+			      base +
+			      MSTR_IF_E2E_CRDT_E2E_EMEM_RD_SIZE_0_OFFSET);
+		regmap_writel(e2e_data->emem_wr_sz.cfg_val_idx_4_7, i * 4 +
+			      base +
+			      MSTR_IF_E2E_CRDT_E2E_EMEM_WR_SIZE_0_OFFSET);
+	}
+	for (i = E2E_PCIE_MMU; i <= E2E_SM; i++) {
+		regmap_writel(e2e_data->emem_rd_sz.cfg_val_idx_8_11, i * 4 +
+			     base +
+			     MSTR_IF_E2E_CRDT_E2E_EMEM_RD_SIZE_0_OFFSET);
+		regmap_writel(e2e_data->emem_wr_sz.cfg_val_idx_8_11, i * 4 +
+			      base +
+			      MSTR_IF_E2E_CRDT_E2E_EMEM_WR_SIZE_0_OFFSET);
+	}
+	for (i = E2E_NIC; i <= E2E_MME; i++) {
+		regmap_writel(e2e_data->emem_rd_sz.cfg_val_idx_12_15, i * 4 +
+			      base +
+			      MSTR_IF_E2E_CRDT_E2E_EMEM_RD_SIZE_0_OFFSET);
+		regmap_writel(e2e_data->emem_wr_sz.cfg_val_idx_12_15, i * 4 +
+			      base +
+			      MSTR_IF_E2E_CRDT_E2E_EMEM_WR_SIZE_0_OFFSET);
+	}
+
+	/* configure the PCIe rs/wr credits */
+	regmap_writel(e2e_data->pcie.rd, base +
+		      MSTR_IF_E2E_CRDT_E2E_PCI_RD_SIZE_OFFSET);
+	regmap_writel(e2e_data->pcie.wr, base +
+		      MSTR_IF_E2E_CRDT_E2E_PCI_WR_SIZE_OFFSET);
+}
