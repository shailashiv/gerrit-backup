[SW-0] gaudi3: upgrade pci phy fw

PCI Phy FW is upgraded by the PHY FSM.
At the moment we copy the entire FW of the PHY. later we can optimize
this by writing trailing zeroes synthetically instead of storing them
in flash and copying them from flash.

+void init_pcie_phy_fw_update(void)
+{
+	/*
+	 * Configure FW update size.
+	 * PHY_FW_FSM_SIZE takes few clocks to propagate. It is being sampled by HW few
+	 * clock cycles after the rising edge of PCIE_WRAP ➔ PHY_FW_FSM ➔ DIRECT_WRITE_EN bit.
+	 */
+	regmap_writel(0, D0_PCIE_WRAP_REG(mmPCIE_WRAP_PHY_FW_FSM));
+	regmap_writel(FW_WORD_NUM, D0_PCIE_WRAP_REG(mmPCIE_WRAP_PHY_FW_FSM_SIZE));
 
+	/* let PHY_FW_FSM_SIZE propagate */
+	udelay(10);
+
+	/*
+	 * PHY FW FSM EN and Direct FW write mechanism enable
+	 * FW FSM assumes that the data is present in the SRAM
+	 * The FSM will wait for SRAM init done before updating
+	 * the SRAM contents
+	 */
+	regmap_writel(PHY_FW_FSM_ENA_MASK, D0_PCIE_WRAP_REG(mmPCIE_WRAP_PHY_FW_FSM));
+}

+

+		}
+	}
