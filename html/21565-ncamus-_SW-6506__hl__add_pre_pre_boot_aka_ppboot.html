+#include <pcie_aux_regs.h>
+#include <cpu_ca53_cfg_regs.h>
+#include <hl_boot_if.h>
+
+#include "common.h"
+#include "preboot.h"
+#include "uart.h"
+#include "spi.h"
+
+/* Taking preboot from where ppboot (btl) is used */
+static u32 preboot_image_offset(void)
+{
+	u32 ppboot_id = regmap_readl(mmPSOC_GLOBAL_CONF_SCRATCHPAD_31);
+
+	if (ppboot_id == '2')
+		return HL_SPI_SECONDARY_BASE;
+
+	return HL_SPI_PRIMARY_BASE;
+}
+
+static void copy_preboot(void)
+{
+	char *name = "PREBOOT";
+	u32 img_offset = preboot_image_offset();
+	u32 preboot_offset = img_offset + HL_SPI_PREBOOT_OFF;
+
+	preboot_puts_raw("Preparing to run Preboot stable");
+	preboot_puts_hex_raw(" (offset: ", img_offset, false);
+	preboot_puts(")\n");
+
+	spi_copy((void *)PREBOOT_TEXT_BASE, preboot_offset, PREBOOT_SIZE, name);
+}
+
+static bool hl_pcie_link_up(void)
+{
+	unsigned int prstn, smlh, rdlh, rst_src;
+
+	/* check source of reset */
+	rst_src = regmap_readl(mmPSOC_GLOBAL_CONF_RST_SRC);
+	if (rst_src < HL_WD_RST)
+		return false;
+
+	prstn = regmap_readl(mmPSOC_GLOBAL_CONF_PRSTN);
+	smlh = regmap_readl(mmPCIE_AUX_SMLH_LINK_UP);
+	rdlh = regmap_readl(mmPCIE_AUX_RDLH_LINK_UP);
+
+	if (prstn && smlh && rdlh)
+		return true;
+
+	return false;
+}
+
+static void ppboot_pcie_init(void)
+{
+	if (hl_pcie_link_up()) {
+		preboot_puts("PCIe link is already UP!\n");
+		return;
+	}
+
+	if (hl_skip_pcie_init()) {
+		preboot_puts("PCIe initialization is skipped!\n");
+		return;
+	}
+
+	//preboot_puts("Waiting for PCIe link...\n");
+	preboot_pcie_wait_for_link();
+	preboot_puts("PCIe link is UP!\n");
+}
+
+static void hl_jtag_init(void)
+{
+	u32 reg, nPRESETDBG = BIT(12);
+
+	/* Enable JTAG attach */
+	reg = regmap_readl(mmCPU_CA53_CFG_ARM_RST_CONTROL);
+	reg |= nPRESETDBG;
+	regmap_writel(reg, mmCPU_CA53_CFG_ARM_RST_CONTROL);
+}
+
+void ppboot_main(void)
+{
+	void (*preboot_start)(void) = (void *)PREBOOT_TEXT_BASE;
+
+	hl_jtag_init();
+	preboot_sync_btl();
+	hl_uart_init();
+	ppboot_pcie_init();
+	preboot_delay();
+	copy_preboot();
+
+	preboot_start();
+}
