+function hl_pldm_hex {
+	local NAME=$(basename "$1" | cut -f 1 -d '.')
+	local FILE=$1
+	local BYTES_PER_LINE=$2
+
+	# pad file to 128bit (16Byte) as pladium emulates 128bit memory, so xxd wont trunc lines
+	dd status=none conv=sync bs=16 if=${FILE} > ${FILE}.tmp
+	xxd -e -g ${BYTES_PER_LINE} ${FILE}.tmp | xxd -r | xxd -ps -c${BYTES_PER_LINE} > ${NAME}.hex
+	rm ${FILE}.tmp
+}
+
+# Function to display help menu
+show_help() {

+    echo
+    echo "Options:"
+    echo "  -h          Show this help message and exit"
+    echo
+    echo "Arguments:"
+    echo "  <file-path> Path to the application binary image"
+    echo "  <bytes-per-line> data with in target hex file"
+}
+
+# Check for the -h flag
+if [[ "$1" == "-h" ]]; then
+    show_help
+    exit 0
+fi
+
+# Check if the correct number of arguments is provided
+if [[ $# -ne 2 ]]; then
+    echo "Error: Invalid number of arguments."
+    show_help

+
+${SCRIPT_DIR}/bin2hex.sh ${FILE_NAME}_be.bin 1
+perl ${TOOLS_DIR}/convert_hex.pl ${FILE_NAME}_be.hex ccm_temp_conv.hex
+${TOOLS_DIR}/eccgen -hex ccm_temp_conv.hex -build ${TOOLS_DIR}/build_configuration.txt -a
+rm -f ccm_temp_conv.hex ${FILE_NAME}_be.hex ${FILE_NAME}_be.bin
+
+echo "Conversion done"
