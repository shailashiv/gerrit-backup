+static int ppboot_validate_secondary_preboot(void)
+{
+	u32 image_size;
+	u32 num_cycles;
+	u32 flash_offset, payload_size, copy_size;
+	int ret, i;
+	struct mem_desc mdesc;
+
+	flash_offset = get_flash_image_offset(ARCBOOT_SECONDARY, PREBOOT);
+	image_size = spi_flash_readl(flash_offset);
+	num_cycles = (image_size / (MAX_DMA_SIZE + CRC_SIZE));
+	if (image_size % (MAX_DMA_SIZE + CRC_SIZE))
+		num_cycles++;
+
+	payload_size = image_size - (num_cycles * CRC_SIZE);
+
+	ret = preboot_get_mem_region(MEM_GP, &mdesc);
+	if (ret) {
+		preboot_puts("Failed to get memory for ppboot fw validation\n");
+		return -1;
+	}
+
+	/*
+	 * when image_size >= 64KB, preboot binary is divided into 2 chunks
+	 * image_size = payload_size + 2*4 (for 2 CRCs)
+	 * bin_part1: (MAX_DMA_SIZE) + CRC              @flash offs 0x0
+	 * bin_part2: (payload_size - MAX_DMA_SIZE) + CRC    @flash offs 0x10000
+	 *
+	 * Approach:
+	 * 1) copy bin_part1 from flash_offs=0 to SRAM_offs=0 (size1 = MAX_DMA_SIZE + CRC)
+	 * 2) copy bin_part2 from flash_offs=0x10000 to ICCM_offs=0..
+	 *   (size2 = payload_size - MAX_DMA_SIZE + CRC)
+	 * 3) memcpy from ICCM_offs = 0 to SRAM_offs=MAX_DMA_SIZE
+	 *   (size3 = payload_size - MAX_DMA_SIZE)
+	 *
+	 * Reasons for this approach:
+	 * 1) SPI DMA transfer destination start and end offset shouldn't
+	 *   include 64KB boundary.
+	 * 2) Flash offset to SPI DMA must be 1KB granuality
+	 * 3) CRC word wont be copied from flash to SRAM
+	 * 4) SPI DMA transfer size (excluding CRC) must be 16 byte aligned.
+	 */
+	copy_size = (num_cycles == 1) ? image_size : (MAX_DMA_SIZE + CRC_SIZE);
+	if (copy_size > mdesc.size) {
+		preboot_puts("preboot fw size exceeds allocated mem region!\n");
+		return -1;
+	}
+
+	ret = ppboot_dma_copy(copy_size, flash_offset,
+			SDRAM_MEM_REGION_FULL_ADDR(mdesc.base),
+			SPI_DMA_DST_HBW);
+	if (ret)
+		return ARCBOOT_BRINGUP_ERROR;
+
+	for (i = 1; i < num_cycles; i++) {
+		copy_size = (i == (num_cycles - 1)) ?
+			(payload_size - MAX_DMA_SIZE * i + CRC_SIZE) :
+				(MAX_DMA_SIZE + CRC_SIZE);
+
+		ret = ppboot_dma_copy(copy_size, (flash_offset + OFFSET_64KB * i),
+					SDRAM_MEM_REGION_FULL_ADDR(mdesc.base),
+					SPI_DMA_DST_HBW);
+		if (ret)
+			return ARCBOOT_BRINGUP_ERROR;
+	}
+
+	return ARCBOOT_BRINGUP_OK;
+}
