[SW-45031] build: arcboot code restructure

In order to avoid continouing 'xxx_arc_gnu' options
(e.g. gaudi2_arc_gnu-asic-build), following build commands
are being changed for GRECO and GAUDI2.

GRECO:
---------
greco_arm -> greco_arm (No change)
greco_arc_gnu -> greco_arc
greco_arc -> greco_arc_elf32
greco-asic-build (builds everything) (No change)

GAUDI2:
-----------
gaudi2_arm -> gaudi2_arm (No change)
gaudi2_arc_gnu -> gaudi2_arc
gaudi2-asic-build (builds everything) (No change)

In this context arcboot arch specific code is aligned accordingly.
This is in line with preboot new dir structure (see https://gerrit.habana-labs.com/#/c/111013/).

Following is the proposed new dir strcture,
├── app
├── arch
│   ├── gaudi2
│   │   └── arc
│   └── greco
│       └── arc_elf32
├── common
│   └── rsa
├── include
└── sdk
    ├── elpsoft
    │   └── src
    ├── sbtool
    └── src
        └── include

+/*
+ * check_load_addr_valid updates 'addr' with the 32bit ARC
+ * view point address used to jump to the code and execute.
+ * If address is valid and within bounds 0 is returned,
+ * else returns 1.
+ */
+static int check_load_addr_valid(struct boot_desc *p_desc, uint32_t *addr)
+{
+	uint32_t addr_low = LOW_DWORD(p_desc->loadaddr);
+	int ret;
+
+	switch (p_desc->loadaddr) {
+	case HL_ICCM_BASE ... HL_ICCM_END:
+		ret = check_boundary(p_desc->loadaddr, p_desc->img_size,
+				HL_ICCM_END);
+		*addr = addr_low & ICCM_OFFS_MASK;
+		break;
+	case HL_SP_SRAM_BASE ... HL_SP_SRAM_END:
+		ret = check_boundary(p_desc->loadaddr, p_desc->img_size,
+				HL_SP_SRAM_END);
+		*addr = HL_SCRATCHPAD_SRAM_BASE +
+			(addr_low & SP_SRAM_OFFS_MASK);
+		/*
+		 * Note that the offset of SP SRAM to be provided to perform
+		 * DMA must be in granuality of 4B. ie.
+		 * (right shift offset 0x1000 by 2).
+		 */
+		p_desc->loadaddr = (HL_SP_SRAM_BASE +
+				((addr_low & SP_SRAM_OFFS_MASK) >> 2));
+		break;
+	case HL_SRAM_PHY_BASE ... HL_SRAM_PHY_END:
+		ret = check_boundary(p_desc->loadaddr, p_desc->img_size,
+				HL_SRAM_PHY_END);
+		*addr = HL_SRAM_LOG_BASE +
+			(addr_low & SRAM_PHYS_OFFS_MASK);
+		break;
+	default:
+		ret = 1;
+		break;
+	}
+	return ret;
+}

+static int img_desc_precheck(struct boot_desc *p_desc, uint32_t *addr)
+{
+	if (p_desc->img_size == 0)
+		goto img_desc_invalid;
+
+	/*
+	 * prevent rogue accesses by checking for-
+	 * 1. flash offset should be <= Max possible offset
+	 * 2. requested size is also less than Max allowed size
+	 */
+	if ((p_desc->spi_offset >=
+		(FLASH_VALID_OFFSET + FLASH_VALID_MAX_SIZE)) ||
+		(p_desc->img_size >= FLASH_VALID_MAX_SIZE))
+		goto img_desc_invalid;
+
+	if (p_desc->spi_offset + p_desc->img_size >=
+			(FLASH_VALID_OFFSET + FLASH_VALID_MAX_SIZE))
+		goto img_desc_invalid;
+
+	if (check_load_addr_valid(p_desc, addr))
+		goto img_desc_invalid;
+
+	return 0;
+
+img_desc_invalid:
+	return 1;
+}

+void execute_pcie_recovery(void)
+{
+	/*
+	 * efuse[14:13]:
+	 * 00: ROM
+	 * 01: eROM
+	 * 10: Reserved
+	 * 11: ROM
+	 */
+	uint8_t pcie_location = ((arcboot_rom_global.config_fuse >>
+			EFUSE_SYSCFG_RECOVERY_CODE_LOCATION) &
+			EFUSE_SYSCFG_RECOVERY_CODE_LOCATION_MASK);

+		/* In all other scenarios, execute from ROM (most secure) */
+		pcie_recovery();
+		return;
+	}
+
+	if (!arcboot_rom_global.erom_boot_desc_flag)
+		if (!boot_desc_init()) {
+			arcboot_rom_status_update(
+					PROGRESS_ARCBOOT_EROM_BOOT_DESC_ERROR);

+		}

+	ext_rom_load_erom_pcie();

+

+	while (1)

+static void security_store_keys_in_psoc(uint8_t max_keys, uint32_t *hash)

+	arcboot_rom_status_update_location(PROGRESS_KEY_LOC_0, key_location);
+
+	switch (key_location) {
+	case KEY_LOCATION_EFUSE:
+		max_keys = NUM_KEYS_IN_EFUSE;
+		efuse_get_hash(key_hash);
+		break;
+	case KEY_LOCATION_ROM:
+	default:
+		rom_get_hash(key_hash);
+		break;
+	}
+
+	/* store hashes in the limited space available in PSOC_SECURITY */
+	security_store_keys_in_psoc(max_keys, key_hash);

+	/* ROM Keys are already in ROM, no need to store them explicitly */
+	if (key_location == KEY_LOCATION_EFUSE) {
+		/* Store the efuse keys in SPSRAM */
+		num_dwords = NUM_KEYS_IN_EFUSE * SECURITY_DWORD_PER_KEY;
+		for (hash_key_index = 0; hash_key_index < num_dwords;
+						hash_key_index++)

+			       SECURITY_EFUSE_KEYS_OFFSET + (hash_key_index *

+/*
+ * arcboot - Signature Verification
+ *
+ * A simple application which call phase0b to validate the signature
+ * of a blob of data.
+ *
+ * Both the protected data and the signature of the protected data can be
+ * changed, but the public key should be locked into the device.
+ *
+ * test_phase0b_data     - blob of protected data
+ * test_phase0b_data_len - size of the blob
+ * test_phase0b_sig      - signature of the blob
+ * test_phase0b_sig_len  - signature size
+ * test_public_key       - public key
+ * test_public_key_len   - public key size
+ */
+
+int secureboot_verify_image(uint32_t image_size, uint32_t image_offset)
+{
+	int verify = 0;
+	int err = 0;
+	int key_scan;
+	unsigned char block_buf[SECUREBOOT_HASH_BLOCK_CALC_BUF];
+	unsigned char test_phase0a_hash[SECURITY_KEY_HASH_ARRAY_BIT_SIZE];
+	uint32_t config_fuse;
+	uint16_t test_phase0b_data_len = image_size;
+	uint32_t test_phase0b_sig_len = SIG_SIZE;
+	uint32_t test_public_key_len = KEY_SIZE;
+
+	unsigned char *test_phase0b_data = (unsigned char *)(image_offset);
+	unsigned char *test_phase0b_sig = test_phase0b_data +
+					  test_phase0b_data_len;
+	unsigned char *test_public_key = test_phase0b_sig -
+					 test_public_key_len;
+	uint32_t fw_version = base_readl((uint32_t)test_phase0b_data);
+
+	/* check efuse for security validation enable */
+	config_fuse = sp_sram_readl(SECURITY_EFUSE_SYS_CFG_OFFSET);
+	if ((config_fuse & EFUSE_SYSCFG_ENABLE_VALIDATION_MASK) == 0)
+		return SECUREBOOT_VERIFY_SUCCESS;
+
+	/* check minimal valid security FW version */
+	if (security_check_valid_fw_version(fw_version) !=
+	    SECURITY_FW_VERSION_SECURE)
+		return SECUREBOOT_VERIFY_VER_FAILED;
+
+	for (key_scan = SECUREBOOT_CUSTOMER_KEY; key_scan >=
+	     SECUREBOOT_HABANA_KEY; key_scan--) {
+		if (security_get_key_hash((uint32_t *)test_phase0a_hash,
+					  key_scan)) {
+			/* check pub key hash versus efuse */
+			err = elp_secureboot_phase0a(sb_proxy_read,
+						     test_public_key,
+						     test_public_key_len,
+						     test_phase0a_hash,
+						     &verify, block_buf,
+						     sizeof(block_buf));
+		}
+		/* if key is verified we can exit the loop */
+		if (verify)
+			break;
+	}
+	/* if for loop doesn't find a valid key hash in fuse, return error */
+	if (!verify)
+		return SECUREBOOT_VERIFY_KEY_FAILED;
+
+	/* Phase0b uses the public key (test_public_key) to validate signature
+	 * (test_phase0b_sig) of the data blob (test_phase0b_data)
+	 */
+	err = elp_secureboot_phase0b(sb_proxy_read, test_phase0b_data,
+				     test_phase0b_data_len, test_public_key,
+				     test_public_key_len, test_phase0b_sig,
+				     test_phase0b_sig_len, &verify,
+				     block_buf, sizeof(block_buf));
+
+	if (err || !verify)
+		return SECUREBOOT_VERIFY_FAILED;
+
+	return verify;
+}

+uint32_t security_key_soc_location_offset(uint32_t hash_key_index)
+{
+	uint32_t ret;
+	uint8_t key_soc_base_index;
+	uint8_t key_soc_location_index;
+	uint32_t key_soc_location[] = {
+		mmPSOC_SECURITY_HABANA_KEY_0,
+		mmPSOC_SECURITY_PUBLIC_KEY_0_0,
+		mmPSOC_SECURITY_PUBLIC_KEY_1_0,
+		mmPSOC_SECURITY_PUBLIC_KEY_2_0,
+		mmPSOC_SECURITY_PUBLIC_KEY_3_0
+	};
+
+	/* select which index to choose (haban, customer 0-4) */
+	key_soc_base_index = hash_key_index / SECURITY_DWORD_PER_KEY;
+	/* key is 256bit, 32bit * 8... which of the 8 location is requested */
+	key_soc_location_index = hash_key_index % SECURITY_DWORD_PER_KEY;
+
+	/* get base address of selected key index */
+	ret = key_soc_location[key_soc_base_index];
+	/* add to base address of key the spicifice location of currect DWORD */
+	ret += (key_soc_location_index * sizeof(uint32_t));
+	return ret;
+}

+void security_load_keys(void)
+{
+	uint8_t hash_key_index;
+	uint32_t key_hash[SECURITY_KEY_HASH_ARRAY_SIZE];
+	enum security_location key_location;
+
+	key_location = ((arcboot_rom_global.config_fuse >>
+			EFUSE_SYSCFG_KEY_LOCATION) &
+			EFUSE_SYSCFG_KEY_LOCATION_MASK);
+
+	arcboot_rom_status_update(PROGRESS_KEY_LOC);
+	arcboot_rom_status_update_location(PROGRESS_KEY_LOC_0, key_location);
+
+	switch (key_location) {
+	case SPI_FLASH:
+		spi_flash_get_hash(key_hash);
