Commit Message

[SW-42060] preboot: gaudi2: add tpm flow

Add TPM flow to Gaudi2 preboot:
	- Send STARTUP command
	- Send NV_READ command to get the Device ID from the TPM
	- Read Device ID from EFUSE.
	- Compare the IDs:
		- equal: continue.
		- different: stop the running.
		- efuse empty: write the ID in the efuse.

+void preboot_tpm_flow(void)
+{
+	int ret;
+	u64 tpm_dev_id = 0;
+	union efuse_dev_id_backup efuse_dev_id;
+	int err = CPU_BOOT_ERR_TPM_FLOW_FAIL;
+
+	efuse_dev_id.l = 0x0;
+
+	ret = tpm2_init();
+	if (ret)
+		cpu_boot_err_set(err);
+
+	ret = tpm2_startup(TPM2_SU_CLEAR);
+	if (ret) {
+		preboot_puts_hex("TPM: startup failed. ret = ", ret, true);
+		cpu_boot_err_set(err);
+	}
+
+	ret = tpm2_nv_read(TPM_NV_DEV_SN_ADDRESS, &tpm_dev_id, sizeof(tpm_dev_id));
+	if (ret) {
+		preboot_puts_hex("TPM: NV_READ failed. ret = ", ret, true);
+		cpu_boot_err_set(err);
+	}
+
+	tpm_dev_id = get_unaligned_be64(&tpm_dev_id);
+	preboot_puts_hex64("TPM device ID: ", tpm_dev_id, true);
+
+	ret = preboot_get_efuse_dev_id(&efuse_dev_id);
+	if (ret) {
+		preboot_puts_hex("TPM flow: efuse read failed. ret = ", ret, true);
+		cpu_boot_err_set(err);
+	}
+
+	preboot_puts_hex64("EFUSE device ID: ", efuse_dev_id.l, true);
+
+	if (efuse_dev_id.l == 0) {
+		preboot_puts("EFUSE device ID not programed, writing efuse\n");
+		ret = preboot_write_efuse_dev_id(tpm_dev_id);
+		if (ret) {
+			preboot_puts("Failed to write EFUSE\n");
+			cpu_boot_err_set(err);
+		}
+
+		ret = preboot_get_efuse_dev_id(&efuse_dev_id);
+		if (ret) {
+			preboot_puts_hex("TPM flow: efuse read failed. ret = ",
+							ret, true);
+			cpu_boot_err_set(err);
+		}
+	} else if (efuse_dev_id.l != tpm_dev_id) {
+		preboot_puts("TPM flow: error: efuse device ID is different than TPM device ID\n");
+		err = CPU_BOOT_ERR_TPM_DEV_ID_FAIL;
+		cpu_boot_err_set(err);
+	}
