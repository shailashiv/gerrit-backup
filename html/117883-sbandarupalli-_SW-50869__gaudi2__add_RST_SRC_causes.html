+void rst_src_dump(void)
+{
+	unsigned int rst_src = regmap_readl(mmPSOC_GLOBAL_CONF_RST_SRC);
+	const char *cause;
+
+	switch (rst_src) {
+	case HL_COLD_RST:
+		cause = "cold";
+		break;
+	case HL_MANUAL_RST:
+		cause = "manual";
+		break;
+	case HL_PRSTN_RST:
+		cause = "prstn";
+		break;
+	case HL_SOFT_RST:
+		cause = "soft";
+		break;
+	case HL_WD_RST:
+		cause = "watch dog";
+		break;
+	case HL_FW_ALL_RST:
+		cause = "firmware all";
+		break;
+	case HL_SW_ALL_RST:
+		cause = "software all";
+		break;
+	case HL_FLR_RST:
+		cause = "functional";
+		break;
+	case HL_ECC_DERR_RST:
+		cause = "ecc derr";
+		break;
+	default:
+		cause = "unknown";
+		break;
+	}
+	preboot_puts_raw("Last reset cause is: ");
+	preboot_puts_raw(cause);
+	preboot_puts(" reset\n");
+}
