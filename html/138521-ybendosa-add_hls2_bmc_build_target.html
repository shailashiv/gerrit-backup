+function create_regs_hdr_file
+{
+	local dest_dir=${1}
+	local project=${2}
+	shift 2
+	local hdr_list=("$@")
+	local regs_hdr_file="${dest_dir}/${project}_regs.h"
+
+	# print license header
+	echo "/* SPDX-License-Identifier: MIT"			> ${regs_hdr_file}
+	echo " *"						>> ${regs_hdr_file}
+	echo " * Copyright 2021 HabanaLabs, Ltd."		>> ${regs_hdr_file}
+	echo " * All Rights Reserved."				>> ${regs_hdr_file}
+	echo " *"						>> ${regs_hdr_file}
+	echo " */"						>> ${regs_hdr_file}
+	echo ""							>> ${regs_hdr_file}
+
+	echo "#ifndef ASIC_REG_${project^^}_REGS_H_"		>> ${regs_hdr_file}
+	echo "#define ASIC_REG_${project^^}_REGS_H_"		>> ${regs_hdr_file}
+	echo ""							>> ${regs_hdr_file}
+
+	for hdr in "${hdr_list[@]}"
+	do
+		echo "#include \"${hdr}\""			>> ${regs_hdr_file}

+	echo ""							>> ${regs_hdr_file}

+# arg1 : dest_dir; arg2 : search_dir; arg3 : lib_name; arg4: hdr_list
+function find_and_copy_headers_from_libs
+{
+	local dest_dir=${1}
+	local search_dir=${2}
+	local lib_name=${3}
+	shift 3
+	local hdr_list=("$@")
+	for file in "${hdr_list[@]}"
+	do
+		file_abs_path=$(find "${search_dir}/${lib_name}" -name ${file})
+		if [ ! -z "$file_abs_path" ]
+		then
+			file_rel_path=$(realpath --relative-to=${search_dir} ${file_abs_path})
+			dir_rel_path=$(dirname ${file_rel_path})
+			mkdir -p ${dest_dir}/${dir_rel_path}
+			cp ${file_abs_path} ${dest_dir}/${file_rel_path}
+		else
+			echo "ERR: Unable to find ${file} in ${search_dir} !!!"
+		fi
+	done
+}
+
+# arg1 : dest_dir; arg2 : project ; arg3 : hdr_list
+function find_and_copy_headers_from_specs
+{
+	local dest_dir=${1}
+	local project=${2}
+	shift 2
+	local hdr_list=("$@")
+	for file in "${hdr_list[@]}"
+	do
+		file_found=0
+		for specs_dir in ${SPECS_SEARCH_DIR[@]}
+		do
+		file_abs_path=$(find "${specs_dir}/${project}" \
+			-not \( -path ${specs_dir}/${project}/asic_reg_structs -prune \) -name ${file})
+		if [ ! -z "$file_abs_path" ]
+		then
+			file_found=1
+			file_rel_path=$(realpath --relative-to=${specs_dir} ${file_abs_path})
+			dir_rel_path=$(dirname ${file_rel_path})
+			mkdir -p ${dest_dir}/${dir_rel_path}
+			cp ${file_abs_path} ${dest_dir}/${file_rel_path}
+			break
+		fi
+		done
+		if [[ $file_found -eq 0 ]]
+		then
+			echo "ERR: Unable to find ${file} in ${SPECS_SEARCH_DIR[@]} /${project} !!!"
+		fi
+	done
+}
+

+	create_regs_hdr_file "${SHARED_SRC}/include/gaudi2/asic_reg" "gaudi2" "${!REGS_HDR_LIST}"
