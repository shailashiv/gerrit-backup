+    f_data_name = debug_habana_str + 'data32'

+        return f_device.read().rstrip('\n\x00')
+
+def clk_gate_set(val):
+    with open(f_clk_gate, 'w') as f_cg:
+        f_cg.write(str(val))
+
+def read_reg(addr):
+    with open(f_addr_name, 'w') as f_addr:

+        f_data = open(f_data_name, 'r')
+        return f_data.read()

+# -*- coding: utf-8 -*-
+import os
+import sys
+import time
+import random
+import common_func
+
+sram_bank_base = 0x7FFC200000
+num_of_mem_blocks = 5120
+
+def print_usage():
+    print("device name (hl0,..)")
+    print("ecc_type (0-serr, 1-derr)")
+    print("sram_block (0-31)")
+    print("sram_block_index (0-1) ")
+    print("sram_addrblock (0-4095)")
+    print("For example: sudo python3.6 sram_ecc.py hl0 0 0 0 0")
+
+if len(sys.argv) < 6:
+    print_usage()
+    sys.exit()
+
+hl_num = sys.argv[1]
+ecc_type = int(sys.argv[2])
+sram_block = int(sys.argv[3])
+sram_block_index = int(sys.argv[4])
+sram_addrblock = int(sys.argv[5]) # 0,0x1000
+
+if ecc_type > 1 or sram_block > 31 or sram_block_index > 1 or sram_addrblock > 4095:
+    print_usage()
+    sys.exit()
+
+print("ecc_type (0-serr, 1-derr)", ecc_type)
+print("sram_block (0-31)", sram_block)
+print("sram_block_index (0-1) ", sram_block_index)
+print("sram_addrblock (0-4095)", hex(sram_addrblock))
+
+sram_bank_block_base = sram_bank_base + sram_block*0x8000
+
+RREG32 = common_func.read_int
+WREG32 = common_func.write_reg
+
+def inject_ecc():
+	if ecc_type == 0:
+		WREG32(sram_bank_block_base + 0xF50, 0x0)
+		WREG32(sram_bank_block_base + 0xF10, sram_block_index)
+	else:
+		# no mask/unmask register for DERR.
+		WREG32(sram_bank_block_base + 0xF14, sram_block_index)
+
+	sram_mem_base = 0x7FF0000000
+
+	for ch in range(num_of_mem_blocks):
+		sram_mem_offset = sram_mem_base + ch*0x1000 + sram_addrblock
+		WREG32(sram_mem_offset, 0xffffffff)
+		RREG32(sram_mem_offset)
+
+
+def main():
+    common_func.prepare_sysfs_nodes(hl_num)
+    common_func.clk_gate_set(0) #Disable clock gating
+    deviceType = common_func.get_device_type()
+
+    if deviceType.lower() == 'gaudi':
+        #inject_ecc()
+        print('device type: {}.'.format(deviceType.lower()))
+    else:
+        print('Unknown device type: {}.'.format(deviceType.lower()))
+
+    inject_ecc()
+
+    common_func.clk_gate_set(1) #Enable clock gating
+
+
+if __name__ == '__main__':
+    main()
