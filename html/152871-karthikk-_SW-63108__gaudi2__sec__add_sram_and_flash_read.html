Commit Message

[SW-63108] gaudi2: sec: add sram and flash read

SRAM and SPI Flash memory is to be opened up for read-only access
when hypervisor sends a fit with FIT_LOAD_TYPE_SRAM_SF_ACCESS load
type. The hbw range registers index 8 and 9 support read-only
functionality.

This patch supports enabling/disabling sram and spi flash read
support.

+void preboot_security_sram_pci_set_ro(bool en)
+{
+	struct hl_rr_eng_select rr_sel;
+
+	memset(&rr_sel, 0, sizeof(struct hl_rr_eng_select));
+	rr_sel.pcie = true;
+
+	if (en) {
+		/*
+		 * Make SRAM Read-only
+		 */
+		hl_secure_hbw_rr(HL_SEC_HBW_SRAM_RO, SRAM_BASE_ADDR, SRAM_END, rr_sel);
+		/*
+		 * Dynamic RR is inclusive of the static SRAM RR, hence the dynamic RRs
+		 * need to be opened in order for SRAM protection to be disabled.
+		 */
+		hl_secure_hbw_rr(HL_SEC_HBW_SRAM_BGN, 0, 0, rr_sel);
+		hl_secure_hbw_rr(HL_SEC_HBW_SRAM_IMG, 0, 0, rr_sel);
+		hl_secure_hbw_rr(HL_SEC_HBW_SRAM_END, 0, 0, rr_sel);
+	} else {
+		/*
+		 * Re-activate SRAM protection using the dynamic init
+		 */
+		preboot_security_dynamic_init();
+		/*
+		 * Disable SRAM Read-only configuration
+		 */
+		hl_secure_hbw_rr(HL_SEC_HBW_SRAM_RO, 0, 0, rr_sel);
+	}
+}
+
+void preboot_security_sf_pci_set_ro(bool en)
+{
+	struct hl_rr_eng_select rr_sel;
+
+	memset(&rr_sel, 0, sizeof(struct hl_rr_eng_select));
+	rr_sel.pcie = true;
+
+	if (en) {
+		/*
+		 * Make SPI Flash Read-only.
+		 */
+		hl_secure_hbw_rr(HL_SEC_HBW_SF_RO, SPI_FLASH_BASE, SPI_FLASH_END, rr_sel);
+		/*
+		 * Disable SPI Flash static protection.
+		 */
+		hl_secure_hbw_rr(HL_SEC_HBW_SPI_FLASH, 0, 0, rr_sel);
+	} else {
+		/*
+		 * Activate SPI Flash static protection.
+		 */
+		hl_secure_hbw_rr(HL_SEC_HBW_SPI_FLASH, SPI_FLASH_BASE, SPI_FLASH_BASE, rr_sel);
+		/*
+		 * Disable SPI Flash Read-only configuration.
+		 */
+		hl_secure_hbw_rr(HL_SEC_HBW_SF_RO, 0, 0, rr_sel);
+	}
