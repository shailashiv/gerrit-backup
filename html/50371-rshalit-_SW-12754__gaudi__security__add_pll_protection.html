+static void preboot_secure_pll(void)
+{
+	u32 base;
+
+	/* DMA_W, IF_E, MESH_W, MESH_E */
+	base = mmDMA_W_PLL_BASE & ~HL_CFG_BASE;
+	set_prot_bits_pages_range_all(base, 0x20000, 4);
+
+	/* IF_W, DMA_E, SRAM_W, SRAM_E */
+	base = mmIF_W_PLL_BASE & ~HL_CFG_BASE;
+	set_prot_bits_pages_range_all(base, 0x20000, 4);
+
+	/* PSOC_CPU, PSOC_MME, PSOC_PCI, PSOC_TPC, PSOC_HBM */
+	base = mmPSOC_CPU_PLL_BASE & ~HL_CFG_BASE;
+	/*
+	 * TODO: the counter should be 5 to cover all 5 PSOC PLLs.
+	 * For the LKD to load successfully before the integration, skip the
+	 * PSOC_HBM_PLL. Once the HBM PLL data passing is implemented in LKD,
+	 * we can return this to 5.
+	 */
+	set_prot_bits_pages_range_all(base, 0x1000, 4);
+
+	/* NIC0, NIC1 */
+	base = mmNIC0_PLL_BASE & ~HL_CFG_BASE;
+	set_prot_bits_pages_range_all(base, 0x40000, 2);
+
+	/* NIC2_HBM, NIC2_MME, NIC2_TPC */
+	base = mmNIC2_HBM_PLL_BASE & ~HL_CFG_BASE;
+	set_prot_bits_pages_range_all(base, 0x1000, 3);
