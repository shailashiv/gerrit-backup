+#!/bin/sh
+
+if [ $# -eq 0 ]; then
+	echo "Usage: ${0} [-s] [-c <cksum>] <image>"
+	echo "       burn <image> into flash sectors"
+	echo "       -s, skip golden flash sectors"
+	echo "       -c <cksum>, crc32 for <image> validation"
+	exit 0
+else
+	while [ $# -gt 0 ]; do
+		case ${1} in
+			-s)
+				IS_SAFE_BURN="yes"
+				shift
+			;;
+			-c)
+				CKSUM=`printf "%d" ${2}`
+				shift
+				shift
+			;;
+			*)
+				INFILE=${1}
+				shift
+			;;
+		esac
+	done
+fi
+
+
+if [ -z "${INFILE}" ]; then
+	echo "Missing image argument"
+	exit 1
+fi
+
+if [ -n "${CKSUM}" ]; then
+	CRC32=`cksum ${INFILE} | cut -d' ' -f1`
+	if ! [ "${CKSUM}" -eq "${CRC32}" ]; then
+		CKSUM_HEX=`printf "0x%x" ${CKSUM}`
+		CRC32_HEX=`printf "0x%x" ${CRC32}`
+		echo "Mismatch with provided checksum [expected ${CKSUM_HEX} while calculated ${CRC32_HEX}]"
+		exit 2
+	fi
+fi
+
+DEVNUM=`mtdinfo				| grep "Count of MTD devices"	| tr -s ' ' | cut -d: -f2	| cut -d' ' -f2`
+FULL_DEV_LIST="$(seq 0 $((${DEVNUM}-1)))"
+
+function burn_image {
+	DEV=/dev/mtd${1}
+	NAME=`mtdinfo ${DEV}		| grep "Name" 			| tr -s ' ' | cut -d: -f2	| cut -d' ' -f2`
+	DEV_IS_SAFE=`mtdinfo ${DEV}	| grep "Name"			| grep " \[safe]"`
+	if [ -n "${IS_SAFE_BURN}" ] && [ -z "${DEV_IS_SAFE}" ]; then
+		return
+	fi
+
+	BLK_SIZE=`mtdinfo ${DEV}	| grep "Eraseblock size"	| tr -s ' ' | cut -d: -f2	| cut -d' ' -f2`
+	BLK_CNT=`mtdinfo ${DEV}		| grep "Amount of eraseblocks"	| tr -s ' ' | cut -d: -f2	| cut -d' ' -f2`
+	DEVID=`mtdinfo ${DEV}		| grep "major/minor"		| tr -s ' ' | cut -d: -f2-3	| cut -d' ' -f2`
+	OFFSET=`cat /sys/dev/char/${DEVID}/offset`
+	OFFSET_HEX=`printf "0x%x" ${OFFSET}`
+	SKIP_CNT=$((${OFFSET}/${BLK_SIZE}))
+	IMG=${INFILE}.mtd${1}
+
+	dd status=none conv=sync bs=${BLK_SIZE} skip=${SKIP_CNT} if=${INFILE} of=${IMG} count=${BLK_CNT}
+	echo -n "    ${NAME} @ ${OFFSET_HEX}... "
+	flash_erase -q -u ${DEV} 0 ${BLK_CNT}
+	flashcp ${IMG} ${DEV}
+
+	dd status=none conv=sync bs=${BLK_SIZE} if=${DEV} of=${IMG}.verify count=${BLK_CNT}
+	VERIFY=`diff -q ${IMG} ${IMG}.verify`
+	if [ -n "${VERIFY}" ]; then
+		echo "FAIL"
+		exit 3
+	else
+		echo "OK"
+	fi
+}
+
+echo "Updating MTD Partitions:"
+for i in ${FULL_DEV_LIST}; do
+	burn_image ${i}
+done
