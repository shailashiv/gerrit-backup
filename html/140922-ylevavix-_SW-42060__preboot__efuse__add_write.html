+static void fuse_grop_word(u32 bank, u32 word, u32 val)
+{
+	uint32_t addr;
+
+	word += (FUSE_WORDS_PER_BANK * bank);
+	regmap_writel(INIT_EFUSE_STEP, mmPSOC_EFUSE_MANUAL_READ);
+	regmap_writel(READ_ENABLE, mmPSOC_EFUSE_MANUAL_READ_EN);
+	regmap_writel(PD_DOWN_STEP, mmPSOC_EFUSE_MANUAL_READ);
+	mdelay(FUSE_WRITE_DELAY_MS);
+	regmap_writel(PS_UP_STEP, mmPSOC_EFUSE_MANUAL_READ);
+	mdelay(FUSE_WRITE_DELAY_MS);
+	regmap_writel(CSB_DOWN_STEP, mmPSOC_EFUSE_MANUAL_READ);
+	regmap_writel(PGENB_DOWN_STEP, mmPSOC_EFUSE_MANUAL_READ);
+	regmap_writel(LOAD_DOWN_STEP, mmPSOC_EFUSE_MANUAL_READ);
+	for (u32 bit = 0; bit < 32; bit++) {
+		if ((1 << bit) & val) {
+			addr = (bit << 7) + word;
+			regmap_writel(LOAD_DOWN_STEP + addr,
+				      mmPSOC_EFUSE_MANUAL_READ);
+			regmap_writel(STROBE_UP_STEP + addr,
+				      mmPSOC_EFUSE_MANUAL_READ);
+			regmap_writel(STROBE_DOWN_STEP + addr,
+				      mmPSOC_EFUSE_MANUAL_READ);
+		}
+	}

+static int fuse_prog(u32 bank, u32 word, u32 val)
+{
+	u32 cur_val;
+
+	if (bank >= FUSE_NUM_BANK) {
+		preboot_puts("Efuse: illegal bank\n");
+		return -1;
+	}
+
+	if (word > FUSE_WORDS_PER_BANK) {
+		preboot_puts("Efuse: illegal address\n");
+		return -1;
+	}
+
+	/* Write only non fused bits, so check first for the fused bits */
+	cur_val = preboot_fuse_read_bank(bank, word);
+
+	val &= ~cur_val;
+	if (!val) /* nothing to write */
+		return 0;
+
+	hl_efuse_write_en(true);
+	fuse_grop_word(bank, word, val);
+	hl_efuse_write_en(false);
+
+	return 0;
+}
+
+__weak int preboot_fuse_write(u32 word, u32 val)
+{
+	int ret0, ret1;
+
+	ret0 = hl_is_efuse_prog_available();
+	if (!ret0) {
+		preboot_puts("Efuse prog is unavailable\n");
+		return -1;
+	}
+
+	if (word >= FUSE_WORDS_PER_BANK) {
+		preboot_puts("Invalid efuse word offset\n");
+		return -1;
+	}
+
+	/* write value to bank0 */
+	ret0 = fuse_prog(0, word, val);
+
+	/* write value to bank1 */
+	ret1 = fuse_prog(1, word, val);
+	if (ret0 || ret1) {
+		preboot_puts("Failed to write to efuse\n");
+		return -1;
+	}
+
+	return 0;
+}
