+/*
+ * arcboot - Signature Verification
+ *
+ * A simple application which call phase0b to validate the signature
+ * of a blob of data.
+ *
+ * Both the protected data and the signature of the protected data can be
+ * changed, but the public key should be locked into the device.
+ *
+ * test_phase0b_data     - blob of protected data
+ * test_phase0b_data_len - size of the blob
+ * test_phase0b_sig      - signature of the blob
+ * test_phase0b_sig_len  - signature size
+ * test_public_key       - public key
+ * test_public_key_len   - public key size
+ */
+
+#include <stdio.h>
+#include "sb.h"
+
+unsigned char __attribute__ ((section(".test_phase0b_data"))) test_phase0b_data[16384];
+uint32_t __attribute__ ((section(".test_phase0b_data_len"))) test_phase0b_data_len;
+unsigned char __attribute__ ((section(".test_phase0b_sig"))) test_phase0b_sig[2048];
+uint32_t __attribute__ ((section(".test_phase0b_sig_len"))) test_phase0b_sig_len;
+unsigned char __attribute__ ((section(".test_public_key"))) test_public_key[2048];
+uint32_t __attribute__ ((section(".test_public_key_len"))) test_public_key_len;
+
+int main(void)
+{
+	int st, err;
+	unsigned char block_buf[512];
+
+	XSTACK_INIT();
+
+	/* Phase0b uses the public key (test_public_key) to validate signature (test_phase0b_sig)
+	 * of the data blob (test_phase0b_data) */
+	err = elp_secureboot_phase0b(sb_proxy_read, test_phase0b_data, test_phase0b_data_len, test_public_key, test_public_key_len,
+			test_phase0b_sig, test_phase0b_sig_len, &st, block_buf, sizeof block_buf);
+
+	printf("SB Phase0b Test: %s\n", (err || !st) ? "FAILED" : "PASSED");
+	if (err || !st) {
+		return -1;
+	}
+	return 0;
+}
