+void security_report_img_status(uint8_t index,
+				enum security_img_status img_stat)
+{
+	uint32_t reg, offset;
+
+	if (index == ARCBOOT_SECONDARY)
+		offset = mmPSOC_SECURITY_SEC_IMG_STATUS;
+	else
+		offset = mmPSOC_SECURITY_PRI_IMG_STATUS;
+
+	reg = regmap_readl(offset);
+	switch (img_stat) {
+	case IMG_STATUS_FW_VERSION_LOW:
+		reg |= 1 << PSOC_SECURITY_PRI_IMG_STATUS_FW_VER_LOWER_SHIFT;
+		break;
+	case IMG_STATUS_SIG_AUTH_FAIL:
+		reg |= 1 <<

+		break;
+	default:
+		reg = 0;
+	}
+	regmap_writel(reg, offset);
