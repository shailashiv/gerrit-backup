Commit Message

[FSW-418] fs1: bootrom: paralellize tpm operations

This patch adds a finite state machine (FSM) to
interleave TPM operations with other functionalities.
Instead of waiting idly for TPM responses, the boot
ROM proceeds with other tasks and periodically keeps
checking for TPM responses. Each state has short
timeouts. If the FSM doesn't complete by the end of
boot ROM execution, a final long timeout-waited
TPM operation is performed.

This patch also adds a new bit to the progress_bit.h
to indicate that the FSM is out of sync.

+static void boot_rom_tpm_init(bool long_timeout_en)
+{
+	int ret = 0;
+
+	/*
+	 * tpm init is attempted with a short timeout repeatedly until it succeeds.
+	 * If all attempts with the short timeout fail, one final attempt with a long
+	 * timeout is performed. If this too fails, abort the TPM init operations
+	 */
+	ret = tpm2_init(long_timeout_en);
+	if (ret != BOOT_ROM_SUCCESS) {
+		if (long_timeout_en) {
+			boot_rom_status_update(PROGRESS_BOOT_ROM_TPM_INIT_FAIL);
+			boot_rom_global.tpm_exec_stage = TPM_EXEC_DONE;
+		}
+	} else {
+		boot_rom_global.tpm_exec_stage = TPM_EXEC_STARTUP_SEND;
+	}
+}
+
+static void boot_rom_send_tpm_startup_cmd(void)
+{
+	int ret = 0;
+
+	ret = tpm2_startup_send_cmd(TPM2_SU_CLEAR);
+	if (ret != BOOT_ROM_SUCCESS) {
+		boot_rom_status_update(PROGRESS_BOOT_ROM_TPM_STARTUP_FAIL);
+		boot_rom_global.tpm_exec_stage = TPM_EXEC_DONE;
+	} else {
+		boot_rom_global.tpm_exec_stage = TPM_EXEC_STARTUP_RECV;
+	}
+}
+
+static void boot_rom_recv_tpm_startup_cmd(bool long_timeout_en)
+{
+	int ret = 0;
+
+	if (tpm2_resp_avail(long_timeout_en)) {
+		ret = tpm2_startup_recv_resp();
+		if (ret != BOOT_ROM_SUCCESS) {
+			boot_rom_status_update(PROGRESS_BOOT_ROM_TPM_STARTUP_FAIL);
+			boot_rom_global.tpm_exec_stage = TPM_EXEC_DONE;
+		} else
+			boot_rom_global.tpm_exec_stage = TPM_EXEC_PCR_EXTEND_SEND;
+	}
+}
+
+static void boot_rom_send_tpm_pcr_extend_cmd(void)
+{
+	int ret = 0;
+	bool is_valid;
+	uint8_t hash[SHA384_SUM_LEN] = {0};
+	enum boot_rom_progress tpm_pcr_extend_fail_prog_bit;
+
+	/* Wait for manifest validation to complete before extending svn & manifest hash*/
+	if (!boot_rom_is_manifest_valid())
+		return;
+
+	is_valid = boot_rom_get_hash_for_tpm_pcr(boot_rom_global.tpm_next_pcr_index, hash);
+	if (!is_valid) {
+		tpm_pcr_extend_fail_prog_bit = boot_rom_get_tpm_extend_fail_prog_bit(
+			boot_rom_global.tpm_next_pcr_index);
+		boot_rom_status_update(tpm_pcr_extend_fail_prog_bit);
+		boot_rom_global.tpm_exec_stage = TPM_EXEC_DONE;
+	}
+
+	ret = tpm2_pcr_extend_send_cmd(boot_rom_global.tpm_next_pcr_index,
+		TPM2_ALG_SHA384, hash, SHA384_SUM_LEN);
+	if (ret != BOOT_ROM_SUCCESS) {
+		tpm_pcr_extend_fail_prog_bit = boot_rom_get_tpm_extend_fail_prog_bit(
+			boot_rom_global.tpm_next_pcr_index);
+		boot_rom_status_update(tpm_pcr_extend_fail_prog_bit);
+		boot_rom_global.tpm_exec_stage = TPM_EXEC_DONE;
+	} else
+		boot_rom_global.tpm_exec_stage = TPM_EXEC_PCR_EXTEND_RECV;
+}
+
+
+static void boot_rom_recv_tpm_pcr_extend_cmd(bool long_timeout_en)
+{
+	int ret = 0;
+	enum boot_rom_progress tpm_pcr_extend_fail_prog_bit;
+
+	if (tpm2_resp_avail(long_timeout_en)) {
+		ret = tpm2_pcr_extend_recv_resp();
+		if (ret != BOOT_ROM_SUCCESS) {
+			tpm_pcr_extend_fail_prog_bit = boot_rom_get_tpm_extend_fail_prog_bit(
+				boot_rom_global.tpm_next_pcr_index);
+			boot_rom_status_update(tpm_pcr_extend_fail_prog_bit);
+			boot_rom_global.tpm_exec_stage = TPM_EXEC_DONE;
+			return;
+		}
+
+		boot_rom_global.tpm_next_pcr_index++;
+
+		if (boot_rom_global.tpm_next_pcr_index == NUM_PCR_TO_EXTEND_IN_BOOTROM) {
+			boot_rom_set_tpm_extend_done();
+			boot_rom_status_update(PROGRESS_BOOT_ROM_TPM_FINISH);
+			boot_rom_global.tpm_exec_stage = TPM_EXEC_DONE;
+		} else {
+			/* Extend remaining PCRs */
+			boot_rom_global.tpm_exec_stage = TPM_EXEC_PCR_EXTEND_SEND;
+		}
+	}
+}
+
+static inline void boot_rom_set_fsm_out_of_sync_err(void)
+{
+	boot_rom_status_update(PROGRESS_BOOT_ROM_TPM_FSM_OUT_OF_SYNC);
+	boot_rom_global.tpm_exec_stage = TPM_EXEC_DONE;
+}
+
+void boot_rom_tpm_ops(bool long_timeout_en)
+{
+
+	if (boot_rom_global.tpm_extend_done)
+		return;
+
+	switch (boot_rom_global.tpm_exec_stage) {
+	case TPM_EXEC_INIT:
+		boot_rom_tpm_init(long_timeout_en);
+		break;
+	case TPM_EXEC_STARTUP_SEND:
+		boot_rom_send_tpm_startup_cmd();
+		break;
+	case TPM_EXEC_STARTUP_RECV:
+		boot_rom_recv_tpm_startup_cmd(long_timeout_en);
+		break;
+	case TPM_EXEC_PCR_EXTEND_SEND:
+		boot_rom_send_tpm_pcr_extend_cmd();
+		break;
+	case TPM_EXEC_PCR_EXTEND_RECV:
+		boot_rom_recv_tpm_pcr_extend_cmd(long_timeout_en);
+		break;
+	case TPM_EXEC_DONE:
+		break;
+	default:
+		boot_rom_set_fsm_out_of_sync_err();
+		break;
+	}
+}
+
+static void boot_rom_set_tpm_ops_final_error(void)
+{
+	enum boot_rom_progress tpm_pcr_extend_fail_prog_bit;
+
+	switch (boot_rom_global.tpm_exec_stage) {
+	case TPM_EXEC_INIT:
+		boot_rom_status_update(PROGRESS_BOOT_ROM_TPM_INIT_FAIL);
+		break;
+	case TPM_EXEC_STARTUP_SEND:
+	case TPM_EXEC_STARTUP_RECV:
+		boot_rom_status_update(PROGRESS_BOOT_ROM_TPM_STARTUP_FAIL);
+		break;
+	case TPM_EXEC_PCR_EXTEND_SEND:
+	case TPM_EXEC_PCR_EXTEND_RECV:
+		tpm_pcr_extend_fail_prog_bit = boot_rom_get_tpm_extend_fail_prog_bit(
+			boot_rom_global.tpm_next_pcr_index);
+		boot_rom_status_update(tpm_pcr_extend_fail_prog_bit);
+		break;
+	default:
+		break;
+	}
+}
+
+void boot_rom_tpm_ops_final(void)
+{
+	uint64_t start;
+
+	start = boot_rom_get_ts_us();
+	do {
+		boot_rom_tpm_ops(true);
+
+		if (boot_rom_global.tpm_exec_stage == TPM_EXEC_DONE)
+			return;
+
+	} while (!boot_rom_timeout_expired(start, MS_TO_US(TPM_LONG_TIMEOUT_MS)));
+
+	boot_rom_set_tpm_ops_final_error();
+}
