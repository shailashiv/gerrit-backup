+static bool boot_desc_img_desc_is_valid(struct hl_fw_img_desc *img_desc)
+{
+	if ((img_desc->magic.val & IMG_DESC_MAGIC_MASK) != HL_FW_IMG_DESC_MAGIC)
+		return false;
+
+	if (img_desc->desc_ver == 0 || img_desc->desc_ver == ~0)
+		return false;
+
+	if (img_desc->idx >= IMG_DESC_IDX_SHIFT)
+		return false;
+
+	return true;

+static int boot_info_desc_init(enum hl_fw_img_type type, struct hl_fw_img_desc *desc)
+{
+	struct hl_fw_desc_info desc_info;
+	uint32_t copy_sz;
+	uint32_t iter;
+	uint8_t *desc_buff;
+	struct hl_fw_img_desc *img_desc;
+	int ret = 0;
+
+	/* Read the main boot pointers structure */
+	if (boot_info_get_no_merge(HL_BOOT_INFO_IMG_DESC_INFO, &desc_info,
+				  sizeof(struct hl_fw_desc_info), &copy_sz)) {
+		LOG_ERR("Error reading desc info from bootinfo\n");
+		return -1;
+	}
+
+	desc_buff = k_malloc(desc_info.desc_size * desc_info.count);
+	if (!desc_buff) {
+		LOG_ERR("Failed to alloc memory for image descriptors");
+		return -1;
+	}
+
+	if (boot_info_get_no_merge(HL_BOOT_INFO_IMG_DESC_DATA, (void *)(desc_buff),
+			(desc_info.desc_size * desc_info.count), &copy_sz)) {
+		LOG_ERR("Failed to copy image desc from bootinfo\n");
+		k_free(desc_buff);
+		return -1;
+	}
+
+	/*
+	 * struct hl_fw_img_desc is used for parsing the descriptor fields, but developer must
+	 * take the descriptor version into account in order to know which fields are available
+	 */
+	for (iter = 0; iter < desc_info.count; iter++) {
+		img_desc = (struct hl_fw_img_desc *)(desc_buff + iter * desc_info.desc_size);
+		if (!boot_desc_img_desc_is_valid(img_desc)) {
+			LOG_ERR("Bad boot descriptor: %u", iter);
+			ret = -1;
+			break;
+		}
+
+		if (img_desc->magic.type == type) {
+			memcpy(desc, img_desc, sizeof(struct hl_fw_img_desc));
+			ret = iter;
+			break;
+		}
+
+	}
+
+	k_free(desc_buff);
+	return ret;
+}
