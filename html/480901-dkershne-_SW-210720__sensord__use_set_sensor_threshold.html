+int sensord_set_thresholds_hbm_global_ecc(const struct sensord_sensor_entry *entry,
+					  struct pldm_pmc_threshold_data *thresholds)
+{
+	uint8_t idx = entry->pdr->common.record_handle - PLDM_RH_SENSOR_HBM_0_GLOBAL_ECC;
+	uint8_t hbm_thresholds[HL_NUM_HBM_COUNT];
+	int ret;
+
+	ret = hl_mem_info_rd_field(hbm_thresholds, hbm_ecc_thresholds);
+	if (ret) {
+		LOG_ERR("%s: failed to read threshold field: %d", __func__, ret);
+		return ret;
+	}
+
+	hbm_thresholds[idx] = MIN(0xFF, thresholds->upper_threshold_warning.u32);
+
+	ret = hl_mem_info_wr_field(hbm_thresholds, hbm_ecc_thresholds, false);
+	if (ret) {
+		LOG_ERR("%s: failed to read threshold field: %d", __func__, ret);
+		return ret;
+	}
+
+	ret = hl_boot_info_finalize();
+	if (ret) {
+		LOG_ERR("%s: Failed in boot_info ret=%d", __func__, ret);
+		return ret;
+	}
+
+	return 0;
