[SW-0] gaudi2: add arc dccm initialization routine

In order to avoid fake DCCM memory exceptions caused due to
uninitialized DCCM reads, this patch add support for DCCM
initialization using memcopy module. In BTL based flow, ARC0 & ARC1
DCCM initialization is done as part of platform init and in secureboot
flow ARC0 DCCM initialization is done as part of init_secondary routine.
Note that in secureboot flow, ARC1 DCCM initialization is done as part
of arcboot ROM init code.

Since udelay is used as part of DCCM init code the init_secondary
function in secureboot flow is moved after udelay_init.

+#define HL_INIT_ARC_DCCM_RETRIES	50
+void init_arc_dccm(u8 id)
+{
+	u32 retry = HL_INIT_ARC_DCCM_RETRIES;
+	u32 addr_h, addr_l, size;
+	u32 mem_cfg, cpy_sts;
+
+	if (id) {
+		addr_l = (u32)ARC1_DCCM_FULL_BASE;
+		addr_h = (u32)(ARC1_DCCM_FULL_BASE >> 32);
+		size = ARC1_DCCM_SIZE;
+	} else {
+		addr_l = (u32)ARC0_DCCM_FULL_BASE;
+		addr_h = (u32)(ARC0_DCCM_FULL_BASE >> 32);
+		size = ARC0_DCCM_SIZE;
+	}
+
+	/* Use memblock to initialize dccm memory */
+	regmap_writel(addr_l, mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_L);
+	regmap_writel(addr_h, mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_H);
+
+	mem_cfg = regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+	mem_cfg &= ~(PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_MEM_SIZE_MASK);
+
+	/* set size of in granuality of 16 bytes */
+	size = size >> 4;
+	mem_cfg |= (size & PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_MEM_SIZE_MASK);
+	mem_cfg |= (0x1 << PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_USE_CONST_SHIFT);
+
+	regmap_writel(mem_cfg, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+
+	/* Enable copy and wait for completion */
+	regmap_writel(0x1, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL);
+	while (1) {
+		cpy_sts = regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_STATUS);
+		if (cpy_sts)
+			break;
+
+		retry--;
+		if (!retry) {
+			preboot_puts_hex("Init DCCM timed-out for arc", id, false);
+			break;
+		}
+		/* TODO: replace udelay with timeout_expired function */
+		udelay(100000);
+	}
+}
