[SW-11867] goya2: preboot: add mr74137 chip programming

This patch adds programming sequences required for initializing and
coniguring mr74137 on-chip tstd sensors(TMP464)

register programming is done as per Moortec Embedded Temperature Sensor
and PVT controller datasheets. TS properties are hardcoded as per Goya2
board since DTS is not available at preboot

Note : patch is not tested on onix platform as temp sensors are not
available

+static bool hl_tmp_is_sdif_free(u32 addr)
+{
+	unsigned int regval;
+	int retry = 5;
+
+	do {
+		regval = regmap_readl(addr + HL_TS_CMN_SDIF_STATUS);
+		if ((regval & HL_TS_CMN_SDIF_STATUS_BUSY_MASK) == 0)
+			return true;
+
+		retry--;
+		udelay(100);
+	} while (retry != 0);
+
+	return false;
+}

+static inline void sdif_write(u32 reg_add, u32 wdata, u8 addr,
+		bool wrn, bool prog)
+{
+
+	u32 regval = (wdata << HL_TS_CMN_SDIF_WDATA_SHIFT)
+		& HL_TS_CMN_SDIF_WDATA_MASK;
+	regval |= (addr << HL_TS_CMN_SDIF_ADDR_SHIFT)
+		& HL_TS_CMN_SDIF_ADDR_MASK;
+	regval |= (wrn << HL_TS_CMN_SDIF_WRN_SHIFT)
+		& HL_TS_CMN_SDIF_WRN_MASK;
+	regval |= (prog << HL_TS_CMN_SDIF_PROG_SHIFT)
+		& HL_TS_CMN_SDIF_PROG_MASK;
+
+	regmap_writel(regval, reg_add + HL_TS_CMN_SDIF);
+}

+void hl_ts_sdif_set(u32 addr, bool use_calib)
+{
+	/* program sensor power up delay */
+	if (hl_tmp_is_sdif_free(addr))
+		sdif_write(addr, HL_TS_IP_TMR_DELAY, HL_TS_IP_TMR, 1, 1);
+
+	/* configure sensor mode */
+	if (hl_tmp_is_sdif_free(addr))
+		sdif_write(addr, use_calib, HL_TS_IP_CFG, 1, 1);
+
+	/* power up sensor in continuous mode */
+	if (hl_tmp_is_sdif_free(addr))
+		sdif_write(addr, HL_TS_IP_CTRL_AUTO_CONT_MODE, HL_TS_IP_CTRL,
+				1, 1);
+}

+void hl_ts_clk_set(u32 addr, unsigned int clk)
+{
+	unsigned int synth_clk;
+	unsigned int val;
+
+	synth_clk = ((clk / HL_TMP_CLK_RATE) / 2) - 1;
+
+	val = (synth_clk << HL_TS_CMN_CLK_SYNTH_LO_SHIFT)
+		& HL_TS_CMN_CLK_SYNTH_LO_MASK;
+	val |= (synth_clk << HL_TS_CMN_CLK_SYNTH_HI_SHIFT)
+		& HL_TS_CMN_CLK_SYNTH_HI_MASK;
+	val |= (1 << HL_TS_CMN_CLK_SYNTH_HOLD_SHIFT)
+		& HL_TS_CMN_CLK_SYNTH_HOLD_MASK;
+	val |= (1 << HL_TS_CMN_CLK_SYNTH_ENA_SHIFT)
+		& HL_TS_CMN_CLK_SYNTH_ENA_MASK;
+
+	regmap_writel(val, addr + + HL_TS_CMN_CLK_SYNTH);
+}

+/*
+ * convert 12-bit HL_TMP register value to milliCelsius
+ *
+ * Mode 1: output conversion (calibrated)
+ * Temp = A + B * Eqbs
+ *
+ * Mode 2: output conversion (uncalibrated)
+ * Temp = G + H * Eqbs
+ *
+ * Eqbs = Nbs / cal5 - 0.5
+ *
+ */
+
+long hl_tmp_temp_reg_to_mC(int val, int ts)
+{
+	long tmp = (val & HL_TS_00_SDIF_DATA_SMPL_DATA_MASK);
+	u32 coef_a_g, coef_b_h, coef_div;
+
+	coef_a_g = hl_ts_data[ts].hl_tmp_data.coef_a_g;
+	coef_b_h = hl_ts_data[ts].hl_tmp_data.coef_b_h;
+	coef_div = hl_ts_data[ts].hl_tmp_data.coef_div;
+
+	tmp = coef_a_g + ((coef_b_h * (((tmp * 1000) / coef_div) -
+					500)) / 1000);
+
+	/*
+	 * result of equation has to be multiplied
+	 * by 1000, since coefficients are multiplied by 100 an extra
+	 * multiplication of 10 is required.
+	 */
+	tmp *= 10;
+
+	return tmp;
+}
+
+/* convert milliCelsius to 12-bit HL_TMP register value */
+u32 hl_tmp_mC_to_temp_reg(long val, int ts)
+{
+	u32 reg_val;
+	u32 coef_a_g, coef_b_h, coef_div;
+
+	coef_a_g = hl_ts_data[ts].hl_tmp_data.coef_a_g;
+	coef_b_h = hl_ts_data[ts].hl_tmp_data.coef_b_h;
+	coef_div = hl_ts_data[ts].hl_tmp_data.coef_div;
+
+	reg_val = ((coef_div * ((val / 10) -
+					coef_a_g + (coef_b_h / 2))) / coef_b_h);
+
+	return reg_val & HL_TS_00_SDIF_DATA_SMPL_DATA_MASK;
+}

+int hl_ts_pvt_init(int ts)

+	unsigned int regval;
+

+	if (regval !=  HL_PVT_TM_SCRATCH_PATTERN)
+		return HL_TS_PVT_INIT_FAILED;
+
+	return 0;
