+.globl	_start
+_start:
+
+/* lf2flt: to avoid compiler (buildroot wrapper) warning during linkage */
+.global	lf2flt
+lf2flt:
+	mov	sp, __stack_top
+	mov	fp, sp
+
+	/* Disable un-aligned checks on data memory */
+	lr r0, [STATUS32]
+	bset r0, r0, 0x13
+	kflag r0
+
+	/* Clear BSS and stack areas */
+	mov	r5, __bss_start
+	sub	r6, __stack_top, r5
+	lsr.f	lp_count, r6, 2
+	lpnz	1f
+	st.ab	0, [r5, 4]
+1:
+	/* jump to main */
+	bl	pcie_pif_main
