+void update_warm_reset_config(void)

+	reg |= PSOC_GLOBAL_CONF_PRSTN_RST_CFG_L_ARC_MASK;
+	reg |= PSOC_GLOBAL_CONF_PRSTN_RST_CFG_L_PSOC_MASK;
+	regmap_writel(reg, mmPSOC_GLOBAL_CONF_PRSTN_RST_CFG_L);
+}

+	uint32_t boot_strap;

+	return !!(boot_strap & PSOC_GLOBAL_CONF_BOOT_STRAP_PINS_L_SRIS_MODE_MASK);

+void enable_flr_handling_fsm(void)

+	regmap_writel(flr_ctrl, mmPCIE_AUX_FLR_CTRL);

+/*
+ * arcboot - Signature Verification
+ *
+ * A simple application which call phase0b to validate the signature
+ * of a blob of data.
+ *
+ * Both the protected data and the signature of the protected data can be
+ * changed, but the public key should be locked into the device.
+ *
+ * test_phase0b_data     - blob of protected data
+ * test_phase0b_data_len - size of the blob
+ * test_phase0b_sig      - signature of the blob
+ * test_phase0b_sig_len  - signature size
+ * test_public_key       - public key
+ * test_public_key_len   - public key size
+ */
+
+int secureboot_verify_image(uint32_t image_size, uint32_t image_offset)
+{
+	int verify = 0;
+	int err = 0;
+	int key_scan;
+	unsigned char block_buf[SECUREBOOT_HASH_BLOCK_CALC_BUF];
+	unsigned char test_phase0a_hash[SECURITY_KEY_HASH_ARRAY_BIT_SIZE];
+	uint32_t config_fuse;
+	uint16_t test_phase0b_data_len = image_size;
+	uint32_t test_phase0b_sig_len = SIG_SIZE;
+	uint32_t test_public_key_len = KEY_SIZE;
+
+	unsigned char *test_phase0b_data = (unsigned char *)(image_offset);
+	unsigned char *test_phase0b_sig = test_phase0b_data +
+					  test_phase0b_data_len;
+	unsigned char *test_public_key = test_phase0b_sig -
+					 test_public_key_len;
+	uint32_t fw_version = base_readl((uint32_t)test_phase0b_data);
+
+	/* check efuse for security validation enable */
+	config_fuse = sp_sram_readl(SECURITY_EFUSE_SYS_CFG_OFFSET);
+	if ((config_fuse & EFUSE_SYSCFG_ENABLE_VALIDATION_MASK) == 0)
+		return SECUREBOOT_VERIFY_SUCCESS;
+
+	/* check minimal valid security FW version */
+	if (security_check_valid_fw_version(fw_version) !=
+	    SECURITY_FW_VERSION_SECURE)
+		return SECUREBOOT_VERIFY_VER_FAILED;
+
+	for (key_scan = SECUREBOOT_CUSTOMER_KEY; key_scan >=
+	     SECUREBOOT_HABANA_KEY; key_scan--) {
+		if (security_get_key_hash((uint32_t *)test_phase0a_hash,
+					  key_scan)) {
+			/* check pub key hash versus efuse */
+			err = elp_secureboot_phase0a(sb_proxy_read,
+						     test_public_key,
+						     test_public_key_len,
+						     test_phase0a_hash,
+						     &verify, block_buf,
+						     sizeof(block_buf));
+		}
+		/* if key is verified we can exit the loop */
+		if (verify)
+			break;
+	}
+	/* if for loop doesn't find a valid key hash in fuse, return error */
+	if (!verify)
+		return SECUREBOOT_VERIFY_KEY_FAILED;
+
+	/* Phase0b uses the public key (test_public_key) to validate signature
+	 * (test_phase0b_sig) of the data blob (test_phase0b_data)
+	 */
+	err = elp_secureboot_phase0b(sb_proxy_read, test_phase0b_data,
+				     test_phase0b_data_len, test_public_key,
+				     test_public_key_len, test_phase0b_sig,
+				     test_phase0b_sig_len, &verify,
+				     block_buf, sizeof(block_buf));
+
+	if (err || !verify)
+		return SECUREBOOT_VERIFY_FAILED;
+
+	return verify;

+uint32_t security_key_soc_location_offset(uint32_t hash_key_index)
+{
+	uint32_t ret;
+	uint8_t key_soc_base_index;
+	uint8_t key_soc_location_index;
+	uint32_t key_soc_location[] = {
+		mmPSOC_SECURITY_HABANA_KEY_0,
+		mmPSOC_SECURITY_PUBLIC_KEY_0_0,
+		mmPSOC_SECURITY_PUBLIC_KEY_1_0,
+		mmPSOC_SECURITY_PUBLIC_KEY_2_0,
+		mmPSOC_SECURITY_PUBLIC_KEY_3_0
+	};
+
+	/* select which index to choose (haban, customer 0-4) */
+	key_soc_base_index = hash_key_index / SECURITY_DWORD_PER_KEY;
+	/* key is 256bit, 32bit * 8... which of the 8 location is requested */
+	key_soc_location_index = hash_key_index % SECURITY_DWORD_PER_KEY;
+
+	/* get base address of selected key index */
+	ret = key_soc_location[key_soc_base_index];
+	/* add to base address of key the spicifice location of currect DWORD */
+	ret += (key_soc_location_index * sizeof(uint32_t));

+	case SPI_OTP:

+		spi_flash_otp_access(FLASH_OTP_DISABLE);
