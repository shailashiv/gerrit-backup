+int arcboot_rom_tpm_pcr_extend(void)
+{
+	struct boot_desc desc;
+	uint32_t idx;
+	int ret;
+
+	if (!arcboot_rom_boot_desc_init_done(HL_FW_BOOT_PTR_SPI))
+		return 1;
+
+	idx = boot_desc_get_img_desc(HL_FW_BOOT_PTR_SPI, HL_FW_IMG_PPBOOT, &desc, FLOPS_UPD_NOOP);
+	if (HL_FW_IMG_NOT_FOUND == idx)
+		idx = boot_desc_get_img_desc(HL_FW_BOOT_PTR_SPI, HL_FW_IMG_PPBOOT, &desc, FLOPS_UPD_PENDING);
+
+	if (HL_FW_IMG_NOT_FOUND == idx)
+		return 1;
+
+	ret = tpm2_pcr_extend(TPM2_PPBOOT_PCR_IDX, TPM2_ALG_SHA384, desc.chksum, SHA384_SUM_LEN);
+	return ret;
+}

+int arcboot_rom_tpm_rom(void)
+{
+	if (tpm2_init()) {
+		arcboot_rom_status_update(PROGRESS_ARCBOOT_ROM_TPM_INIT_FAIL);
+		return 1;
+	}
+
+	if (tpm2_startup(TPM2_SU_CLEAR)) {
+		arcboot_rom_status_update(PROGRESS_ARCBOOT_ROM_TPM_STARTUP_FAIL);
+		return 1;
+	}
+
+	if (arcboot_rom_tpm_pcr_extend()) {
+		arcboot_rom_status_update(PROGRESS_ARCBOOT_ROM_TPM_PCR_EXTEND_FAIL);
+		return 1;
+	}
+
+	arcboot_rom_status_update(PROGRESS_ARCBOOT_ROM_TPM_FINISH);
+	return 0;
+}
