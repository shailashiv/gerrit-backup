 #define UART_SR_RXFULL			(1 << 1) /* RX Buffer Full */
 #define STR_UINT_SZ			8
 #define BITS_IN_SINGLE_CHAR		4
+#define RING_BUFF_SIZE			256
+
+struct ring_buf_t {
+	char buf[RING_BUFF_SIZE];
+	unsigned int head;
+	unsigned int tail;
+	bool initialized;
+};
+
+static struct ring_buf_t tx_buf;
+
+void preboot_uart_rbuf_init(void)
+{
+	tx_buf.initialized = true;
+}
+
+/* TODO: make preboot_uart_rbuf_push static in next commit */
+void preboot_uart_rbuf_push(char ch)
+{
+	unsigned int head = tx_buf.head + 1;
+
+	head &= RING_BUFF_SIZE - 1;
+
+	/* if buffer is full, ignoring data */
+	if (head != tx_buf.tail) {
+		tx_buf.buf[tx_buf.head] = ch;
+		tx_buf.head = head;
+	}
+}
+
+void preboot_uart_rbuf_pop(void)
+{
+	unsigned int tail = tx_buf.tail;
+	char ch;
+
+	/* if buffer is empty do nothing */
+	if (tx_buf.head != tail) {
+		ch = tx_buf.buf[tail];
+		tail++;
+		tail &= RING_BUFF_SIZE - 1;
+
+		tx_buf.tail = tail;
+
+		/* transferring data directly to HW */
+		regmap_writel(ch, mmPSOC_UART_0_DATA);
+		preboot_putc_virt_uart(ch);
+	}
+}
+
+void preboot_uart_rbuf_flush(void)
+{
+	while (tx_buf.head != tx_buf.tail) {
+		while (regmap_readl(mmPSOC_UART_0_STATE) & UART_SR_TXFULL)
+			;
+		preboot_uart_rbuf_pop();
+	}
+}
