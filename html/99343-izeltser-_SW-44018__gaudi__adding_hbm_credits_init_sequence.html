+void preboot_hbm_credits_init(void)
+{
+	int i, ret;
+	u32 hbm1_wr, mask, base;
+	char dev_maj, dev_min;
+
+	ret = hl_fuse_get_dev_id(&dev_maj, &dev_min);
+
+	if (!ret) {
+		if ((dev_min == '1') && ((dev_maj == 'A') || (dev_maj == 'B')))
+			hbm1_wr = HBM1_WR_VAL;
+		else
+			hbm1_wr = HBM1_WR_VAL_COMPACT;
+	} else {
+		hbm1_wr = HBM1_WR_VAL_COMPACT;
+	}
+
+	/* The following values were determined by VLSI group. */
+	for (i = 0; i < GAUDI_HBM_NUM_DEVICES; i++) {
+		base = i * GAUDI_HBM_CH_DIST;
+		regmap_writel(HBM0_WR_VAL,
+			      base + mmDMA_IF_W_S_HBM0_WR_CRED_CNT);
+		regmap_writel(hbm1_wr, base + mmDMA_IF_W_S_HBM1_WR_CRED_CNT);
+		regmap_writel(HBM0_RD_VAL,
+			      base + mmDMA_IF_W_S_HBM0_RD_CRED_CNT);
+		regmap_writel(HBM1_RD_VAL,
+			      base + mmDMA_IF_W_S_HBM1_RD_CRED_CNT);
+	}
+
+	mask = BIT(0) | BIT(1);
+
+	for (i = 0; i < GAUDI_HBM_NUM_DEVICES; i++) {
+		base = i * GAUDI_HBM_CH_DIST;
+		regmap_writel(mask, base + mmDMA_IF_W_S_HBM_CRED_EN_0);
+		regmap_writel(mask, base + mmDMA_IF_W_S_HBM_CRED_EN_1);
+	}
+
+	cpu_boot_dev_sts_clrset(CPU_BOOT_DEV_STS0_HBM_CRED_EN, true);
+}
