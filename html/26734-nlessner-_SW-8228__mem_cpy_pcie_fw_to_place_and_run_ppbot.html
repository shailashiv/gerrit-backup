+#include "arcboot.h"
+#include "mem_cpy.h"
+
+void mem_cpy(uint32_t image_size, uint64_t source, uint64_t dest)
+{
+	uint32_t mem_cpy_ctrl2;
+	uint16_t image_block_count_16b;
+
+	/* image_size needs to be in 16 byte blocks count */
+	image_block_count_16b = image_size / 16;
+
+	/* if last block isn't complete 16 bytes add 1 to count */
+	/* so we also copy part of block			*/
+	if ((image_size % 16) > 0)
+		image_block_count_16b++;
+
+	regmap_writel(MEM_CPY_ADDRESS_LOW(source),
+		      mmPSOC_GLOBAL_CONF_MEM_CPY_START_ADDR_L);
+	regmap_writel(MEM_CPY_ADDRESS_HIGH(source),
+		      mmPSOC_GLOBAL_CONF_MEM_CPY_START_ADDR_H);
+
+	regmap_writel(MEM_CPY_ADDRESS_LOW(dest),
+		      mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_L);
+	regmap_writel(MEM_CPY_ADDRESS_HIGH(dest),
+		      mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_H);
+
+	mem_cpy_ctrl2 = image_block_count_16b;
+	mem_cpy_ctrl2 += MEM_CPY_WR_OS_32 <<
+				PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_WR_OS_SHIFT;
+	regmap_writel(mem_cpy_ctrl2,
+		      mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+
+	/* start transfer */
+	regmap_writel(1, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL);
+}
+
+
+int mem_cpy_finish(void)
+{
+	uint32_t mem_cpy_status =
+		 regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_STATUS);
+
+	return mem_cpy_status;
+}
+
+void mem_cpy_wait_finish(void)
+{
+	uint32_t mem_cpy_status;
+
+	do {
+		mem_cpy_status =
+		 regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_STATUS);
+	} while (!mem_cpy_status);
+}
