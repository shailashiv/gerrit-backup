+#define ARC_RUN_HALT_ACK_DELAY_US			100
+#define ARC_RUN_M							BIT(0)
+#define ARC_HALT_M							BIT(1)
+
+__unused static void run_halt_arc(int core_id, bool run)
+{
+	uint32_t run_halt_offset;
+
+	switch (core_id) {
+	case FIT_CORE_ID_D0_PSOC_ARC0:
+		run_halt_offset = mmD0_PARC_ARC0_CFG_RUN_HALT_REQ;
+		break;
+	case FIT_CORE_ID_D0_PSOC_ARC2:
+		run_halt_offset = mmD0_PARC_ARC2_CFG_RUN_HALT_REQ;
+		break;
+	default:
+		return;
+	}
+
+	/* Set ARC in run/halt state */
+	if (run)
+		regmap_writel(ARC_RUN_M, run_halt_offset);
+	else
+		regmap_writel(ARC_HALT_M, run_halt_offset);
+
+	udelay(ARC_RUN_HALT_ACK_DELAY_US);
+
+	/* Clear run/halt request */
+	regmap_writel(0x0, run_halt_offset);
+}
