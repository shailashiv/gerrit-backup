+function convert_arc_to_abs_view {
+	arc_range=$(($1 & $ARC_RANGE_MASK))
+	case $arc_range in
+	$(($ARC_ICCM_RANGE)))
+		if (($2 == $ARC0_CORE)); then
+			addr=$ARC0_ICCM
+		else
+			addr=$ARC1_ICCM
+		fi
+		;;
+	$(($ARC_GENERAL_RANGE)))

+		;;
