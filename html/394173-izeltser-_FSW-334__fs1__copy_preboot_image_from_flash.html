Commit Message

[FSW-334] fs1: copy preboot image from flash

Preboot image is copied from flash using the SPI DMA engine.
Eventually PPBOOT will use the SPI DMA code from BOOTROM, but until
then, for testing and HW validation SPI DMA code is implemented locally.
Later on the SPI DMA implementation can be easily removed as it is
wrapped with "#ifndef USE_BOOTROM_SPI_DMA"

+static void dma_copy_image_internal(uint16_t image_size, uint32_t flash_offset,
+				    uint64_t dest_offset, uint8_t fw_sram_dst, bool first_chunk)
+{
+	uint32_t spi_dma_ctrl =  0;
+	uint32_t spi_dma_status;
+	uint32_t spi_dma_dst_addr_l = DMA_ADDRESS_LOW(dest_offset);
+	uint32_t spi_dma_dst_addr_h = DMA_ADDRESS_HIGH(dest_offset);
+	uint16_t flash_offset_1k = flash_offset / DMA_SRC_BLOCK_SIZE;
+	uint16_t image_size_4b = (image_size / DMA_DST_BLOCK_SIZE);
+
+	regmap_writel(spi_dma_dst_addr_l, mmIO_0_CFG_HALF_0_PSOC_PARC_CFG_MISC_BASE +
+					  mmPARC_GLOBAL_CONF_SPI_DMA_DST_ADDR_L);
+	regmap_writel(spi_dma_dst_addr_h, mmIO_0_CFG_HALF_0_PSOC_PARC_CFG_MISC_BASE +
+					  mmPARC_GLOBAL_CONF_SPI_DMA_DST_ADDR_H);
+
+	/* enable DMA transfer */
+	spi_dma_ctrl = PARC_GLOBAL_CONF_SPI_DMA_CTRL_EN_M;
+
+	/* sram=1 hbw=0 as destination */
+	spi_dma_ctrl |= fw_sram_dst << PARC_GLOBAL_CONF_SPI_DMA_CTRL_DST_SRAM_M;
+
+	/* # of 4bytes to DMA copy */
+	spi_dma_ctrl |= (uint32_t)image_size_4b << PARC_GLOBAL_CONF_SPI_DMA_CTRL_MEM_SIZE_S;
+
+	/* spi offset in 1k chunks */
+	spi_dma_ctrl |= (uint32_t)flash_offset_1k << PARC_GLOBAL_CONF_SPI_DMA_CTRL_ADDR_SHIFT_S;
+
+	/* mask crc error, fw will validate final crc */
+	spi_dma_ctrl |= PARC_GLOBAL_CONF_SPI_DMA_CTRL_MASK_CRC_ERR_M;
+
+	/* restart CRC for first chunk, accumulate for the following chunks */
+	spi_dma_ctrl |= first_chunk ? PARC_GLOBAL_CONF_SPI_DMA_CTRL_CRC_RESTART_M : 0;
+
+	regmap_writel(spi_dma_ctrl, mmIO_0_CFG_HALF_0_PSOC_PARC_CFG_MISC_BASE +

+static enum dma_error dma_copy_image_v3(uint32_t image_size, uint32_t flash_offset,
+					uint64_t dest_addr, enum dma_dest_type dest_type)
+{
+	uint32_t dma_chunks, dma_size, chunk_offset, copy_size = image_size;
+	uint32_t calc_crc, img_crc_addr, img_crc;
+	uint8_t i;
+
+	if (dest_type >= DMA_DEST_MAX)
+		return DMA_INVALID_DEST_TYPE;
+
+	if (dest_addr % ADDR_SIZE_ALIGNMENT)
+		return DMA_UNALIGNED_DEST_ADDR;
+
+	dma_chunks = image_size / MAX_DMA_SIZE;
+	if (image_size % MAX_DMA_SIZE)
+		dma_chunks++;
+
+	if (image_size < MIN_DMA_SIZE_CRC)
+		return DMA_INVALID_IMG_SIZE;
+
+	if ((image_size) % ADDR_SIZE_ALIGNMENT)
+		return DMA_UNALIGNED_IMG_SIZE;
+
+	for (i = 0; i < dma_chunks; i++) {
+		dma_size = copy_size > MAX_DMA_SIZE ? MAX_DMA_SIZE : copy_size;
+		chunk_offset = MAX_DMA_SIZE * i;
+		dma_copy_image_internal(dma_size, (flash_offset + chunk_offset),
+					(dest_addr + chunk_offset), dest_type, !i);
+
+		copy_size -= dma_size;
+	}
+
+	calc_crc = regmap_readl(mmIO_0_CFG_HALF_0_PSOC_PARC_CFG_MISC_BASE +
+				mmPARC_GLOBAL_CONF_SPI_DMA_CRC);
+
+	/* assumption: CRC is included in image_size and copied to dest location */
+	img_crc_addr = fs1_translate_arc_addr(dest_addr + image_size - sizeof(img_crc));
+	img_crc = *(u32 *)img_crc_addr;
+
+	if (calc_crc != img_crc) {
+		preboot_puts("SPI image CRC mismatch");
+		return DMA_INVALID_CRC_ERROR;
+	}
+
+	return DMA_SUCCESS;
+}
