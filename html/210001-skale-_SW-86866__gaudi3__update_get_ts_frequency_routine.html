[SW-86866] gaudi3: update get ts frequency routine

Currently, the frequency required for psoc timestamp is hardcoded
to 35M to onix and 50M to pldm and asic. Since the timestamp block
will be used for udelay implementation, this patch add changes to
get timestamp freq by reading pll registers. psoc ts block is driven
by DIV3 of PCIE PLL clock.

+uint64_t get_pcie_pll_freq(enum pll_div_out div)
+{
+	uint32_t div_sel, factor, offset;
+	uint32_t reg, fbdiv, refdiv, postdiv1, postdiv2;
+	uint64_t out_freq = SYS_REF_CLOCK;
+
+	offset = div * sizeof(uint32_t);
+	div_sel = regmap_readl(mmD0_G4_PCI_PLL_CTRL_DIV_SEL_0 + offset);
+
+	if (div_sel == PLL_PLL_CLK || div_sel == PLL_DIV_PLL) {
+		reg = regmap_readl(mmD0_G4_PCI_PLL_CTRL_CFG_DIV);
+		/* fbdiv : bits 0-11 */
+		fbdiv = EXTRACT_BITS(reg, 0, 12);
+		/* postdiv1 : bits 16-18 */
+		postdiv1 = EXTRACT_BITS(reg, 16, 3);
+		/* postdiv2 : bits 20-22 */
+		postdiv2 = EXTRACT_BITS(reg, 20, 3);
+		/* refdiv : bits 24-29 */
+		refdiv  = EXTRACT_BITS(reg, 24, 6);
+		out_freq = SYS_REF_CLOCK * fbdiv;
+		out_freq /= refdiv * (postdiv1 + 1) * (postdiv2 + 1);
+	}
+
+	if (div_sel == PLL_DIV_REF || div_sel == PLL_DIV_PLL) {
+		factor = regmap_readl(mmD0_G4_PCI_PLL_CTRL_DIV_FACTOR_0 + offset);
+		out_freq /= factor + 1;
+	}
+
+	return out_freq;
+}
