+// SPDX-License-Identifier: Apache-2.0
+/*
+ *
+ * Copyright 2025 Intel, Ltd.
+ * All Rights Reserved.
+ */
+
+#include <intr.h>
+
+#include <zephyr/logging/log.h>
+LOG_MODULE_REGISTER(INTR_DISPATCH);
+
+typedef int (*intr_comp_handlers_t)(struct intr_config *intr_info);
+
+/* Macro to aggregate all group handlers for a given component type */
+#define ASSIGN_ALL_INTR_HANDLER_FOR(comp) { \
+	ASSIGN_INTR_HANDLER(serr, comp), \
+	ASSIGN_INTR_HANDLER(derr, comp), \
+	ASSIGN_INTR_HANDLER(sei, comp), \
+	ASSIGN_INTR_HANDLER(spi, comp), \
+	ASSIGN_INTR_HANDLER(eco, comp) \
+}
+
+/* 2D array of function pointers for each component type and each group type */
+static intr_comp_handlers_t intr_comp_handlers[INTR_COMP_TYPE_MAX][INTR_GRP_TYPE_MAX] = {
+#define INTR_COMP(name) ASSIGN_ALL_INTR_HANDLER_FOR(name),
+	INTR_COMPONENTS_LIST
+#undef INTR_COMP
+};
+
+
+int intr_dispatch_component_handler(struct intr_config *config)
+{
+	enum intr_agg_grp_type grp;
+	enum intr_agg_component_type comp_type;
+
+	if (config->intr_type(config, &grp, &comp_type)) {
+		LOG_ERR("Failed to get interrupt group and component type");
+		return -1;
+	}
+
+	if (grp < 0 || grp >= INTR_GRP_TYPE_MAX) {
+		LOG_ERR("Invalid group type: %d", grp);
+		return -1;
+	}
+
+	if (comp_type < 0 || comp_type >= INTR_COMP_TYPE_MAX) {
+		LOG_ERR("Invalid component type: %d", comp_type);
+		return -1;
+	}
+
+	if (intr_comp_handlers[comp_type][grp])
+		return intr_comp_handlers[comp_type][grp](config);
+
+	LOG_ERR("No handler found for group %d and component %d", grp, comp_type);
+	return -1;
+}
