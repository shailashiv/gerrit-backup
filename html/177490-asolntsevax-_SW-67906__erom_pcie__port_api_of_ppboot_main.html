+void preboot_pcie_enable(void)
+{
+	u32 rst_src;
+
+	rst_src = hl_rst_src_get();
+	switch (rst_src) {
+	case HL_COLD_RST:
+	case HL_PRSTN_RST:
+		/* PCIe is reset, so must initialize PCIe */
+		preboot_init_pcie();
+		break;
+	default:
+		/*
+		 * Currently, keep using the same flow as done before,
+		 * i.e. PCIe should already be initialized by default.
+		 * Except COLD & PRSTN resets, keep PCIe intact.
+		 */
+		break;
+	}
+}
+
+static bool hl_pcie_link_up(void)
+{
+	unsigned int prstn, smlh, rdlh, rst_src;
+
+	/* check source of reset */
+	rst_src = hl_rst_src_get();
+	if ((rst_src == HL_COLD_RST) || (rst_src == HL_PRSTN_RST))
+		return false;
+
+	prstn = regmap_readl(mmPSOC_GLOBAL_CONF_PRSTN);
+	smlh = regmap_readl(mmPCIE_AUX_SMLH_LINK_UP);
+	rdlh = regmap_readl(mmPCIE_AUX_RDLH_LINK_UP);
+
+	if (prstn && smlh && rdlh)
+		return true;
+
+	return false;
+}
+
+void hl_pcie_init(void)
+{
+	if (hl_pcie_link_up()) {
+		hl_pcie_enable_access();
+		preboot_puts("PCIe link is already UP!\n");
+		return;
+	}
+
+	if (hl_skip_pcie_init()) {
+		preboot_puts("PCIe initialization is skipped!\n");
+		return;
+	}
+
+	/* let the log be device specific according to its restrictions */
+	ppboot_log_waiting_for_pcie();
+	preboot_pcie_wait_for_link();
 }
