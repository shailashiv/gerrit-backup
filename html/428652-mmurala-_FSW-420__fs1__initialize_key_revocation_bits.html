+void security_load_revoke_bits(void)
+{
+	uint32_t key_revoke = boot_rom_global.config_fuse & EFUSE_SYSCFG_KEY_REVOCATION_BITS;
+
+	if (boot_rom_global.fake_fuse_en)
+		boot_rom_status_update(PROGRESS_REVOKE_LOC_SPI_FLASH);
+	else
+		boot_rom_status_update(PROGRESS_REVOKE_LOC_ASIC_EFUSE);
+
+	/* store key_revoke map in PSOC */
+	regmap_writel(key_revoke, IO0_H0_PSOC_BOOT_SECURITY(mmPSOC_SECURITY_KEY_REVOKE));
+
+	boot_rom_global.sec_data.key_revoke = key_revoke;
+}

+int security_check_key_revoke_status(uint32_t key_index)
+{
+	uint8_t key_revoke;
+
+	if (key_index >= NUM_KEYS_IN_ROM)
+		return -1;
+
+	key_revoke = boot_rom_global.sec_data.key_revoke;
+
+	/* if the key is not revoked, return 1 */
+	if ((key_revoke & (1 << key_index)) == 0)
+		return 1;
+
+	/* no valid key found */
+	return -1;
+}
