+int fit_load_log_cleanup(const void *fit, int conf)
+{
+	int rc;
+	void *res_fit;
+	uint32_t res_fit_size = HL_FIT_SIMPLE_PROP_SIZE * 2;
+	const char *stage;
+	int err;
+
+	res_fit = fwupd_fit_rel_and_resp_fit_alloc(res_fit_size, &stage, &err);
+	if (!res_fit) {
+		LOG_ERR("Log cleanup %s failed: %d", stage, err);
+		return -EIO;
+	}
+
+	rc = fdt_property_u32(res_fit, HL_FIT_LOAD_TYPE_PROP, FIT_LOAD_TYPE_CLEANUP_LOG);
+	if (rc) {
+		LOG_ERR("Log cleanup failed, set %s failed: %d", HL_FIT_LOAD_TYPE_PROP, rc);
+		return rc;
+	}
+
+	rc = module_lfs_cleanup();
+	if (rc)
+		LOG_ERR("Failed to clean LFS logger partition, RC=%d", rc);
+	else
+		LOG_INF("LFS logger partition was cleaned successfully");
+
+	fdt_property_u32(res_fit, HL_FIT_VALUE_PROP, rc);
+	return rc;
