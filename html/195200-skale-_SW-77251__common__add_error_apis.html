+static struct hl_rsp_err pending_error;
+
+bool arcboot_error_inform(uint32_t major, uint32_t minor)
+{
+	/*
+	 * ensuring sending the first error occurrence
+	 * to reset the error just use the clr_on_read
+	 */
+	if (!pending_error.error) {
+		pending_error.major = major;
+		pending_error.minor = minor;
+		return true;
+	}
+
+	return false;
+}
+
+bool arcboot_error_is_pending(void)
+{
+	return !!pending_error.error;
+}
+
+uint32_t arcboot_error_read_and_clr(uint32_t *major, uint32_t *minor)
+{
+	*major = pending_error.major;
+	*minor = pending_error.minor;
+
+	pending_error.error = 0;
+	return *major;
+}
