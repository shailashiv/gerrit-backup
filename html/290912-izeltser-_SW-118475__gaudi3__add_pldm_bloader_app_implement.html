Commit Message

[SW-118475] gaudi3: add pldm_bloader app implementation

On the pldm emulator a fast lean bootloader is required in order
to run the preboot rtos application (and maybe others in the future).
The pldm-bloader application is making the most necessary preparations
for running the preboot rtos app, and copying a fresh copy of the
preboot rtos binary to the iccm memory.
current flow is:
1. fixup the pci pll to enable prints on the pldm.
2. copy the preboot rtos binary from ARC0/2 ICCMs to ARC1 ICCM.
3. copy the spsram content (bootinfo) from ARC0 DCCM the SPSRAM.
4. run preboot rtos app.

The pldm_bloader is sideloaded to the bootrom memory and executed
automatically after Arc1 reset.

Div3 of PCI PLL drives the SPI GW and the UART.
if the clocks on both dies are not synced the SPI GW can't pass data.
Since the Pci PLL fixup must be performed on both dies and can't be
performed over the SPI GW, an instance of the pldm_bloader is running
on die1 as well.

+;; ************************************************************************************************
+;; -----------------
+;; ARC start up file
+;; -----------------
+;; The startup code for the ARC family of processors does the following
+;; before transferring control to user defined main label :
+;;	1. Set sp to __stack_top (link time variable)
+;;	2. Set fp to zero
+;;	3. Zero out the bss section (for uninitialized globals)
+;; After returning from main, the processor is halted and the pipeline is flushed out
+;;
+;; ************************************************************************************************
+.file "crt0_rom.S"
+
+.macro CPU_EARLY_SETUP
+	; enable I-cache/D-cache
+	lr	r5, [ic_ctrl]
+	bclr	r5, r5, 0	; 0 - Enable, 1 is Disable
+	sr	r5, [ic_ctrl]
+
+	lr	r5, [dc_ctrl]
+	bclr	r5, r5, 6	; Invalidate (discard w/o wback)
+	bclr	r5, r5, 0	; Enable (+Inv)
+	sr	r5, [dc_ctrl]
+
+	; Disable alignment checking
+	lr	r5, [status32]
+	bset	r5, r5, 19
+	kflag	r5
+.endm
+
+.extern main
+
+.section .ivt, "a", @progbits
+
+; Helper macro to define weak symbols to include into interrupt vector table.
+; User code may define those functions in them, so user function will be
+; referenced in the IVT. By default all handlers point to _exit_halt - so they
+; always cause application halt, because if application causes an exception or
+; interrupt, but doesn't set a handler for it - something is wrong in
+; application. Exception is "start" entry of IVT, which points to __start
+; function.
+#define IVT_ENTRY(name) \
+    .word name `\
+    .weak name `\
+    .set name, _exit_halt
+
+; handler's name,		number,	name,			offset in IVT (hex/dec)
+.word __start			; 0	program entry point	0x0		0
+IVT_ENTRY(memory_error)		; 1	memory_error		0x4		4
+IVT_ENTRY(instruction_error)	; 2	instruction_error	0x8		8
+IVT_ENTRY(EV_MachineCheck)	; 3	EV_MachineCheck		0xC		12
+IVT_ENTRY(EV_TLBMissI)		; 4	EV_TLBMissI		0x10		16
+IVT_ENTRY(EV_TLBMissD)		; 5	EV_TLBMissD		0x14		20
+IVT_ENTRY(EV_ProtV)		; 6	EV_ProtV		0x18		24
+IVT_ENTRY(EV_PrivilegeV)	; 7	EV_PrivilegeV		0x1C		28
+IVT_ENTRY(EV_SWI)		; 8	EV_SWI			0x20		32
+IVT_ENTRY(EV_Trap)		; 9	EV_Trap			0x24		36
+IVT_ENTRY(EV_Extension)		; 10	EV_Extension		0x28		40
+IVT_ENTRY(EV_DivZero)		; 11	EV_DivZero		0x2C		44
+IVT_ENTRY(EV_DCError)		; 12	EV_DCError		0x30		48
+IVT_ENTRY(EV_Maligned)		; 13	EV_Maligned		0x34		52
+IVT_ENTRY(EV_Ex14)		; 14	unused			0x38		56
+IVT_ENTRY(EV_Ex15)		; 15	unused			0x3C		60
+IVT_ENTRY(IRQ_Timer0)		; 16	Timer 0			0x40		64
+IVT_ENTRY(IRQ_Timer1)		; 17	Timer 1			0x44		68
+IVT_ENTRY(IRQ_18)		; 18				0x48		72
+IVT_ENTRY(IRQ_19)		; 19				0x4C		76
+IVT_ENTRY(IRQ_20)		; 20				0x50		80
+
+.section .text.__startup, "ax", @progbits
+
+.global __start
+.type __start, @function
+.align 4
+
+; Startup code for the ARC600, ARC700 and ARCv2 processors
+; NOTE:  The following restrictions apply on zero overhead loops (other
+; restrictions are not pertinent to this code)
+; - loop end should be 4 instruction words away from the lp_count setting
+;   instruction
+; - loop body should have at least two instruction words
+__start:
+	CPU_EARLY_SETUP
+	mov gp, @__SDATA_BEGIN__
+
+	mov sp, @__stack_top	; initialize stack pointer
+
+	mov r2, @_rom_non_persistent_dccm_start; r2 = start of the non persistent dccm section
+	sub r3, @_dccm_end, r2	; r3 = size of the dccm section in bytes
+	; set up the loop counter register to the size (in words) of the dccm section
+	asr_s r13, r3
+	asr.f lp_count, r13
+
+	mov r4, 0x0
+	; loop to zero out the dccm.  Enter loop only if lp_count != 0
+	lpnz @.Lend_dccm
+	st.ab r4, [r2, 4]	; zero out the word
+	nop
+.Lend_dccm:
+
+	mov_s r1, @_data_after_persist_sec	; r1 = non persistent section begin in ROM(LMA).
+	mov_s r2, @_rom_non_persistent_dccm_start; r2 = non persistent section begin in DCCM(VMA).
+
+	sub r3, @_edata, r2	; r3 = size of the non persistent data section in bytes
+
+	; set up the loop counter register to the size (in words) of the data section
+	asr_s r13, r3
+	asr.f lp_count, r13
+
+	; loop to copy data from ROM to DCCM.  Enter loop only if lp_count != 0
+	lpnz @.Lend_cdata
+	ld.ab	r5,[r1,4]
+	st.ab	r5,[r2,4]
+.Lend_cdata:
+
+	;; branch to main
+	bl.d @pldm_bloader_main
+	mov fp, 0		; initialize frame pointer
+				; r0 contains exit code
+	j @_exit_halt
+
+.global _exit_halt
+.type _exit_halt, @function
+.align 4
+_exit_halt:
+				; r0 contains exit code
+	flag 0x01
+	nop
+	nop
+	nop
+	j @_exit_halt
+	nop

+#define UART_DIV_ID	3

+/*
+ * pcie pll is initialized by the boot fsm to nominal values (in secured boot mode). In order to
+ * support uart prints on pldm only div3 of the pci pll should be modified. Since div3 also impacts
+ * the baudrate of the d2d spi dma, pci pll of both dies should be carefully reconfigured first,
+ * before using the spi gw for data transfer.
+ */
+void btldr_pll_fixup(uint32_t die_id)
+{
+	gaudi3_cpu_pll_cfg[0].base = die_id ? mmD1_G4_PCI_PLL_CTRL_BASE : mmD0_G4_PCI_PLL_CTRL_BASE;
+
+	hl_pll_reconfig_div(gaudi3_cpu_pll_cfg, UART_DIV_ID);
+}

+void btldr_parc_prepare(void)
+{
+	/* ARC must not be in reset state when accessing the CCM modules */
+	regmap_writel(0, mmD0_PARC_ARC0_CFG_ARC_RST);
+	regmap_writel(0, mmD0_PARC_ARC2_CFG_ARC_RST);
+}

+#define EFUSE_APP_SIZE_OFFSET	55
+#define EFUSE_APP_SIZE_MASK	0x1
+
+static void btldr_copy_preboot_app(void)
+{
+	int err;
+	u32 val;
+
+	/*
+	 * Clean copy of the application is stored in ICCM0. In case the application binary
+	 * exceeds ICCM0 size, the rest is stored in ICCM2.
+	 * An efuse bit is controlling whether the application binary is larger than ICCM0 size.
+	 */
+	gaudi3_hbw_memcpy(ARC0_ICCM_FULL_BASE, ARC1_ICCM_FULL_BASE, ARC0_ICCM_SIZE);
+
+	err = preboot_fuse_read(EFUSE_APP_SIZE_OFFSET, &val);
+	if (err)
+		val = EFUSE_APP_SIZE_MASK;
+
+	val &= EFUSE_APP_SIZE_MASK;
+
+	if (!val)
+		return;
+
+	gaudi3_hbw_memcpy(ARC2_ICCM_FULL_BASE, ARC1_ICCM_FULL_BASE + ARC0_ICCM_SIZE,
+			  ARC1_ICCM_SIZE - ARC0_ICCM_SIZE);
+}
+
+static void btldr_copy_bootinfo(void)
+{
+	/*
+	 * Normally bootinfo is first written to spsram by the arcboot, so preboot rtos expects some
+	 * bootinfo data to be available when it runs. On PLDM, when running standalone preboot,
+	 * pre-recorded boot info data is loaded to the SPSRAM.
+	 * Since SPSRAM content is erazed on cold reboot, fresh copy stored in DCCM0 and copied
+	 * back to the SPSRAM as part of the preparation to run preboot rtos.
+	 */
+	gaudi3_hbw_memcpy(ARC0_DCCM_FULL_BASE, HL_SP_SRAM_BASE, ARC0_DCCM_SIZE);
+}
+
+void btldr_copy_data(void)

+	btldr_copy_bootinfo();
+}
