+#include <psoc_global_conf_regs.h>
+
+#include <hl_boot_if.h>
+#include <preboot.h>
+
+/*
+ * The cpu_boot_cmd_clr_before_wfe() function should NOT be used in general as
+ * the LKD message register should be written by LKD solely.
+ *
+ * There is only one special case when this function can be used is only when
+ * the LKD performs hard reset, which may mean the H/W is not reliable and
+ * writing to PCI might not complete and result in the register not cleared.
+ * So to be on the safe side, when receiving the WFE, clear the message register
+ */
+void cpu_boot_cmd_clr_before_wfe(void)
+{
+	regmap_writel(KMD_MSG_NA, mmPSOC_GLOBAL_CONF_KMD_MSG_TO_CPU);
+}
+
+enum kmd_msg cpu_boot_cmd_get(void)
+{
+	return regmap_readl(mmPSOC_GLOBAL_CONF_KMD_MSG_TO_CPU);
+}
+
+void cpu_boot_sts_set(u32 val)
+{
+	regmap_writel(val, mmPSOC_GLOBAL_CONF_CPU_BOOT_STATUS);
+}
+
+/* We support only ERR0 for now */
+void cpu_boot_err_clrset(u32 mask, bool set)
+{
+	u32 val = regmap_readl(mmCPU_BOOT_ERR0) & ~mask;
+
+	if (set)
+		val |= mask;
+
+	regmap_writel(val, mmCPU_BOOT_ERR0);
+}
+
+void cpu_boot_rsp_set(enum cpu_msg_status sts)
+{
+	regmap_writel(sts, mmCPU_CMD_STATUS_TO_HOST);
+}
