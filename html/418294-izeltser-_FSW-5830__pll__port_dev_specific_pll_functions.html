+#include <stddef.h>
+#include <bitops.h>
+#include <baremetal.h>
+#include <pll.h>
+#include "common.h"
+
+#define FS1_PLL_CFG_EN	(PLL_CONTROL_CFG_FOUTPOSTDIVEN_M | PLL_CONTROL_CFG_EN_M)
+
+void hl_pll_reset(struct hl_pll_cfg *cfg, bool reset)
+{
+	u32 reg = regmap_readl(cfg->base + mmPLL_CONTROL_CFG);
+
+	if (reset)
+		reg &= ~FS1_PLL_CFG_EN;
+	else
+		reg |= FS1_PLL_CFG_EN;
+
+	regmap_writel(reg, cfg->base + mmPLL_CONTROL_CFG);
+}
+
+int hl_pll_is_locked(struct hl_pll_cfg *cfg)
+{
+	return regmap_readl(cfg->base + mmPLL_CONTROL_CFG) & PLL_CONTROL_CFG_LOCK_M;
+}
+
+#define FS1_PLL_MAX_WAIT_TIME	5
+
+void hl_pll_cfg(struct hl_pll_cfg *cfg)
+{
+	u32 reg = 0;
+
+	/* put the PLL into reset by writing 0 to CTRL_CFG.EN and CTRL_CFG.FOUTPOSTDIVEN */
+	hl_pll_reset(cfg, true);
+
+	/* FBDIV - bit 0
+	 * POSTDIV1 - bit 16
+	 * POSTDIV2 - bit 20
+	 * REFDIV - bit 24
+	 */
+	reg = cfg->nr << PLL_CONTROL_CFG_DIV_REFDIV_S | cfg->nb << PLL_CONTROL_CFG_DIV_POSTDIV2_S |
+	      cfg->od << PLL_CONTROL_CFG_DIV_POSTDIV1_S | cfg->nf;
+
+	regmap_writel(reg, cfg->base + mmPLL_CONTROL_CFG_DIV);
+	regmap_writel(PLL_CONTROL_VCO_CTRL_VCOSEL_M, cfg->base + mmPLL_CONTROL_VCO_CTRL);
+
+	/* after updating the coefficients we need to wait >=1us */
+	udelay(FS1_PLL_MAX_WAIT_TIME);
+
+	/* enable the PLL by writing a value of 1 to CTRL_CFG.EN and CTRL_CFG.FOUTPOSTDIVEN */
+	hl_pll_reset(cfg, false);
+}
+
+void hl_pll_clk_relax_reset(struct hl_pll_cfg *cfg) {}
+
+void hl_pll_clear_lock_status(struct hl_pll_cfg *cfg) {}
+
+void hl_pll_ack_int(struct hl_pll_cfg *cfg) {}
+
+void hl_pll_set_div_sel_delay(struct hl_pll_cfg *cfg)
+{
+	regmap_writel(1, cfg->base + mmPLL_CONTROL_DIV_SEL_DELAY);
+}
+
+void hl_pll_vco_en(struct hl_pll_cfg *cfg)
+{
+	u32 val;
+
+	val = regmap_readl(cfg->base + mmPLL_CONTROL_VCO_CTRL);
+	val |= PLL_CONTROL_VCO_CTRL_FOUTVCOEN_M;
+	regmap_writel(val, cfg->base + mmPLL_CONTROL_VCO_CTRL);
+}
+
+u32 hl_get_pll_div_fix(void) { return 1; }
