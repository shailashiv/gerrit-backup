[SW-56706] gaudi3: add new dma api for dma above 64kb

In Gaudi3, there is possibility of ppboot size might
increase above 64KB. Current DMA api implemented in ROM
only supports maximum DMA size of 64KB. The SPI DMA
hardware has a limitation on max size of 0xFFFC bytes and
also has alignment required of address and size.

This patch add new DMA API which support size upto 128KB - 8 bytes
and does check all alignment requirement and also performs all
adjustments to overcome SPI DMA max size limitation. This API
waits internally for DMA to finish and reports success/failure
accordingly. This API will also be exported for ppboot to reuse
while performing copy of preboot image.

+// SPDX-License-Identifier:     GPL-2.0+
+/*
+ * Copyright (C) 2022 HabanaLabs Ltd.
+ * All Rights Reserved.
+ */
+
+#include <arcboot.h>
+#include <dma.h>
+#include <dev_dma.h>
+
+enum dma_error dma_copy_image_v2(uint32_t image_size, uint32_t flash_offset,
+				 uint64_t dest_addr, enum dma_dest_type dest_type)
+{
+	uint32_t payload_size, copy_size;
+
+	if (image_size > MAX_IMAGE_SIZE)
+		return DMA_INVALID_IMG_SIZE;
+
+	if (dest_type >= DMA_DEST_MAX)
+		return DMA_INVALID_DEST_TYPE;
+
+	if (dest_addr % ADDR_SIZE_ALIGNMENT)
+		return DMA_UNALIGNED_DEST_ADDR;
+
+	if (image_size <=  (MAX_DMA_SIZE + CRC32_SIZE)) {
+		if ((image_size - CRC32_SIZE) % ADDR_SIZE_ALIGNMENT)
+			return DMA_UNALIGNED_IMG_SIZE;
+
+		dma_copy_image(image_size, flash_offset, dest_addr, dest_type);
+		if (dma_wait_finish() == DMA_CRC_ERROR)
+			return DMA_INVALID_CRC_ERROR;
+		else
+			return DMA_SUCCESS;
+	}
+
+	payload_size = image_size - (2 * CRC32_SIZE);
+	if (payload_size % ADDR_SIZE_ALIGNMENT)
+		return DMA_UNALIGNED_IMG_SIZE;
+
+	/* copy 1st chunk flash_off to dest_off */
+	copy_size = MAX_DMA_SIZE + CRC32_SIZE;
+	dma_copy_image(copy_size, flash_offset, dest_addr, dest_type);
+	if (dma_wait_finish() == DMA_CRC_ERROR)
+		return DMA_INVALID_CRC_ERROR;
+
+	/* copy 2nd chunk from flash_off + 0x10000 to dest_off + MAX_DMA_SIZE */
+	copy_size = payload_size - MAX_DMA_SIZE + CRC32_SIZE;
+	dma_copy_image(copy_size, (flash_offset + OFFSET_64KB),
+			(dest_addr + MAX_DMA_SIZE), dest_type);
+	if (dma_wait_finish() == DMA_CRC_ERROR)
+		return DMA_INVALID_CRC_ERROR;
+
+	return DMA_SUCCESS;
+}
