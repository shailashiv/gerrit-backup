+static int arcboot_rom_bringup_img_sub(int img, struct boot_desc *desc, enum fw_post_reset_cmd upd_status,
+		uint8_t cur_img)
+{
+	uint32_t act_img_size, image_len, loadaddr_arc;
+	unsigned char *image, *signature, *pub_key;
+	enum arcboot_rom_progress img_sts_base = PROGRESS_BRINGUP_PCI_PRI;
+	int ret;
+
+	arcboot_rom_global.curr_img = cur_img;
+
+	if (cur_img == ARCBOOT_SECONDARY)
+		img_sts_base = PROGRESS_BRINGUP_PCI_SEC;
+
+	arcboot_rom_status_update_bringup(img_sts_base, PROGRESS_BRINGUP_START);
+
+	arcboot_get_boot_desc(img, desc, upd_status);
+
+	security_report_img_fw(arcboot_rom_global.curr_img, desc->sec_ver);
+
+	act_img_size = desc->img_size - desc->pub_key_size - desc->sig_size - CRC32_SIZE;
+
+	arcboot_rom_status_update_bringup(img_sts_base, PROGRESS_BRINGUP_DMA);
+	/* TODO: Add DMA LOGIC */
+
+	if (!security_efuse_image_validation_en())
+		return ARCBOOT_BRINGUP_OK;
+
+	/* Derive key and signature parameters */
+	loadaddr_arc = (uint32_t)CONV_64B_TO_ARC(desc->loadaddr);
+	image = (unsigned char *)(loadaddr_arc);
+	image_len = act_img_size + desc->pub_key_size; /* img + public key*/
+	pub_key = (unsigned char *)(loadaddr_arc + act_img_size);
+	signature = (unsigned char *)(loadaddr_arc + image_len);
+
+	ret = secureboot_verify_image_v2(image, image_len,
+			pub_key, desc->pub_key_size, signature, desc->sig_size, desc->sec_ver);
+	if (ret != SECUREBOOT_VERIFY_SUCCESS) {
+		if (ret == SECUREBOOT_VERIFY_VER_FAILED)
+			security_report_img_status(IMG_STATUS_FW_VERSION_LOW, cur_img);
+		else if (ret == SECUREBOOT_VERIFY_KEY_REVOKED)
+			arcboot_rom_status_update_bringup(img_sts_base, PROGRESS_BRINGUP_KEY_REVOKED);
+		else if (ret == SECUREBOOT_VERIFY_KEY_FAILED)
+			arcboot_rom_status_update_bringup(img_sts_base, PROGRESS_BRINGUP_KEY_AUTH_FAIL);
+		else
+			security_report_img_status(IMG_STATUS_SIG_AUTH_FAIL, cur_img);
+		return ARCBOOT_BRINGUP_ERROR;
+	}
+
+	arcboot_rom_status_update_bringup(img_sts_base, PROGRESS_BRINGUP_SECURITY);
+
+	return ARCBOOT_BRINGUP_OK;
+}

+static int arcboot_rom_bringup_img(int img, uint64_t *load_addr)
+{
+	enum fw_post_reset_cmd fw_upd_status = regmap_readl(mmD0_PSOC_GLOBAL_CONF_COLD_RST_FLOPS_3);
+	int ret;
+	struct boot_desc desc;
+
+	ret = arcboot_rom_bringup_img_sub(img, &desc, fw_upd_status, ARCBOOT_PRIMARY);
+	if (ret == ARCBOOT_BRINGUP_OK)
+		goto image_bringup_success;
+
+	if (arcboot_rom_global.config_fuse &
+		EFUSE_SYSCFG_DISABLE_SECONDARY_IMG)
+		return ARCBOOT_BRINGUP_ERROR;
+
+	/* Try the secondary image by toggling upd_status variable */
+	fw_upd_status = (fw_upd_status == FLOPS_UPD_PENDING) ? FLOPS_UPD_NOOP :
+			 FLOPS_UPD_PENDING;
+
+	ret = arcboot_rom_bringup_img_sub(img, &desc, fw_upd_status, ARCBOOT_SECONDARY);
+	if (ret == ARCBOOT_BRINGUP_OK)
+		goto image_bringup_success;
+
+	return ARCBOOT_BRINGUP_ERROR;
+
+image_bringup_success:
+	*load_addr = desc.loadaddr;
+	return ARCBOOT_BRINGUP_OK;
+

+void arcboot_rom_bringup_pcie(void)
+{
+	void (*ppboot)();
+	uint64_t *loadaddr =
+			&arcboot_rom_global.boot_ptr_ctx[HL_FW_BOOT_PTR_SPI].load_addr;
+	int ret;
+
+	if (!boot_desc_init(HL_FW_BOOT_PTR_SPI)) {
+		arcboot_rom_status_update(PROGRESS_BRINGUP_PCI_BOOT_DESC_FAIL);
+		goto skip_bringup;
+	}
+
+	if (arcboot_rom_global.validation_pass)
+		goto skip_rom_validation;
+
+	/* Check for prim/sec failure */
+	ret = arcboot_rom_bringup_img(HL_FW_IMG_PPBOOT, loadaddr);
+	if (ret == ARCBOOT_BRINGUP_OK) {
+		arcboot_rom_validation_pass();
+	} else {
+		arcboot_rom_validation_fail();
+		goto skip_bringup;
+	}
+
+skip_rom_validation:
+	arcboot_rom_handle_tpm();
+
+	arcboot_rom_status_update(PROGRESS_PPBOOT_START);
+	if (loadaddr) {
+		ppboot = (void *)((uint32_t)CONV_64B_TO_ARC(*loadaddr));
+		ppboot();
+	}
+
+skip_bringup:
+	execute_pcie_recovery();
+	return;
+}
