Commit Message

[SW-24179] ppboot: goya2: initialize scratchpad sram

Currently ECC errors are observed for scratchpad memory while copying
ARC ICCM and DCCM images in two step process (SRAM -> scratchpad SRAM
-> ICCM/DCCM). These fake ECC errors are observed due to byte copy to
scratchpad memory which turns to be read-modify-write transaction.
In order to avoid these ECC errors, this patch adds complete
initialization of scratchpad SRAM using memcopy block.

+static void init_sp_sram(void)
+{
+	u32 mem_cfg, cpy_sts;
+	u8 retry = HL_INIT_SP_SRAM_RETRIES;
+
+	/* Use memblock to initialize scratchpad memory */
+	regmap_writel((u32)HL_SP_SRAM_BASE,
+			mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_L);
+	regmap_writel((u32)(HL_SP_SRAM_BASE >> 32),
+			mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_H);
+
+	mem_cfg = regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+	mem_cfg &= ~(PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_FLD_SIZE_MASK);
+	/* set size of 64k in granuality of 16 bytes */
+	mem_cfg |= (0x1000 & PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_FLD_SIZE_MASK);
+	mem_cfg |= (0x1 << PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_USE_CONST_SHIFT);
+
+	regmap_writel(mem_cfg, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+
+	/* Enable copy and wait for completion */
+	regmap_writel(0x1, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL);
+	while (1) {
+		cpy_sts = regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_STATUS);
+		if (cpy_sts)
+			break;
+		/*
+		 * Do not wait indefinitely on copy status
+		 * Wait for 5 seconds (50 retries with 100 milli-
+		 * seconds sleep), else bailout.
+		 */
+		retry--;
+		if (!retry) {
+			preboot_puts("Timed-out waiting for SP SRAM copy\n");
+			break;
+		}
+		udelay(100000);
+	}
+
