+#!/usr/bin/python3
+
+import json
+import sys
+from littlefs import LittleFS
+
+JSON_FILE_DEFAULT = 'gaudi3_serdes_config_flash_memory.json'
+OUTPUT_FILE_DEFAULT = 'FlashMemory.bin'
+
+def json_field_as_int(json, field_name):
+    val = json[field_name]
+
+    # if the value is a string, convert it to an integer in base 16
+    if isinstance(val, str):
+        val = int(val, 16)
+
+    return val
+
+def parse_config_json(file_path, output_file, debug=False):
+    with open(file_path, 'r') as file:
+        data = json.load(file)
+
+    fs = LittleFS(block_size=4096, block_count=16)
+    for serdes_entry in data['serdes_table']:
+        type_value = serdes_entry['type']
+        link_id_value = int(serdes_entry['link_id'], 2)
+        tx_lane_swap = [int(lane, 16) for lane in serdes_entry['tx_lane_swap']]
+
+        for config in serdes_entry['configurations']:
+            index = config['index']
+            tx_pol = json_field_as_int(config, 'tx_pol')
+            rx_pol = json_field_as_int(config, 'rx_pol')
+            enable_mask = json_field_as_int(config, 'enable_mask')
+            external_mask = json_field_as_int(config, 'external_mask')
+            autoneg_mask = json_field_as_int(config, 'autoneg_mask')
+
+            if debug:
+                print(f"Index: {index}")
+                print(f"Type: {type_value}")
+                print(f"Link ID: {link_id_value}")
+                print(f"TX Pol: {hex(tx_pol)}")
+                print(f"RX Pol: {hex(rx_pol)}")
+                print(f"Enable Mask: {hex(enable_mask)}")
+                print(f"External Mask: {hex(external_mask)}")
+                print(f"Autoneg Mask: {hex(autoneg_mask)}")
+                print(f"TX Lane Swap: {tx_lane_swap}")
+                print()
+
+            output_dir = '.'
+            file_name = f"{output_dir}/serdes-{link_id_value}-{index}"
+            with fs.open(file_name, 'wb') as fh:
+                fh.write(type_value.to_bytes(2, byteorder='little'))
+                fh.write(tx_pol.to_bytes(8, byteorder='little'))
+                fh.write(rx_pol.to_bytes(8, byteorder='little'))
+                fh.write(enable_mask.to_bytes(4, byteorder='little'))
+                fh.write(external_mask.to_bytes(4, byteorder='little'))
+                fh.write(autoneg_mask.to_bytes(4, byteorder='little'))
+                for lane in tx_lane_swap:
+                    fh.write(lane.to_bytes(2, byteorder='little'))
+
+    with open(output_file, 'wb') as fh:
+        fh.write(fs.context.buffer)
+
+def print_help():
+    print("Usage: parse_json_no_enums_names.py [<path_to_json_file>] [<output_file>] [--debug]")
+    print()
+    print("Arguments:")
+    print(f"  <path_to_json_file>  Optional path to the input JSON file containing Serdes configuration. Default is {JSON_FILE_DEFAULT}.")
+    print(f"  <output_file>        Optional path to the output file where the LittleFS image will be saved. Default is {OUTPUT_FILE_DEFAULT}.")
+    print("  --debug              Optional flag to enable debug mode for detailed output.")
+    print("  --help               Prints this help message.")
+
+if __name__ == "__main__":
+    if '--help' in sys.argv:
+        print_help()
+        sys.exit(0)
+
+    json_file_path = sys.argv[1] if len(sys.argv) > 1 and not sys.argv[1].startswith('--') else JSON_FILE_DEFAULT
+    output_file_path = sys.argv[2] if len(sys.argv) > 2 and not sys.argv[2].startswith('--') else OUTPUT_FILE_DEFAULT
+    debug_mode = '--debug' in sys.argv
+
+    print(f"Reading JSON file: {json_file_path}")
+    print(f"Output file: {output_file_path}")
+    if debug_mode:
+        print("Debug mode enabled")
+
+    parse_config_json(json_file_path, output_file_path, debug_mode)
