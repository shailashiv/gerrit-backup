+static void boot_rom_d2d_phy_wd_hw_errata_fix()
+{
+	uint32_t phy_off, inter_phy_offset, ctrl0, ctrl1, ctrl3;
+	uint8_t max_phy_count, phy_num, io_half1_off;
+
+	if(boot_rom_global.die_type == COMPUTE_DIE) {
+		inter_phy_offset = COMPUTE_INTER_PHY_DIST;
+		max_phy_count = COMPUTE_D2D_PHY_NUM;
+
+		ctrl0 = COMP_0_CFG_COMP_SOC_VD2D_PHY_0_COMP_DPHY_CTRL_BASE(mmCOMP_DPHY_CTRL_INIT_PLL_CTRL0);
+		ctrl1 = COMP_0_CFG_COMP_SOC_VD2D_PHY_0_COMP_DPHY_CTRL_BASE(mmCOMP_DPHY_CTRL_INIT_PLL_CTRL1);
+		ctrl3 = COMP_0_CFG_COMP_SOC_VD2D_PHY_0_COMP_DPHY_CTRL_BASE(mmCOMP_DPHY_CTRL_INIT_PLL_CTRL3);
+	} else {
+		inter_phy_offset = IO_INTER_PHY_DIST;
+		max_phy_count = IO_D2D_PHY_NUM;
+
+		ctrl0 = IO0_H0_SOC_0_ID2D_PHY_0_CTL0_BASE(mmD2D_CTL_INIT_PLL_CTRL0);
+		ctrl1 = IO0_H0_SOC_0_ID2D_PHY_0_CTL0_BASE(mmD2D_CTL_INIT_PLL_CTRL1);
+		ctrl3 = IO0_H0_SOC_0_ID2D_PHY_0_CTL0_BASE(mmD2D_CTL_INIT_PLL_CTRL3);
+	}
+
+	for (phy_num = 0; phy_num < max_phy_count; phy_num++) {
+		phy_off = phy_num * inter_phy_offset;
+
+		regmap_writel(PLL_LOCK_TIMER_VAL, ctrl0 + phy_off);
+		regmap_writel(OFFSET_CALLOCK_TIMER, ctrl1 + phy_off);
+		regmap_writel(TIMER_1US, ctrl3 + phy_off);
+
+		if(boot_rom_global.die_type != COMPUTE_DIE) {
+			io_half1_off = boot_rom_global.die_id + 1;
+
+			base_writel(PLL_LOCK_TIMER_VAL, IO_HALF1((ctrl0 + phy_off), io_half1_off));
+			base_writel(OFFSET_CALLOCK_TIMER, IO_HALF1((ctrl1 + phy_off), io_half1_off));
+			base_writel(TIMER_1US, IO_HALF1((ctrl3 + phy_off), io_half1_off));
+		}
+	}
+}

+static void boot_rom_d2d_phy_update_train_window()
+{
+	uint32_t val, inter_phy_offset, phy_off, reg_addr, h1_reg_addr;
+	uint8_t max_phy_count, phy_num;
+
+	if (!(boot_rom_global.config_fuse & EFUSE_SYSCFG_D2D_UPDATE_TRAIN_WINDOW))
+		return; /* skip d2d train window update */
+
+	if(boot_rom_global.die_type == COMPUTE_DIE) {
+		inter_phy_offset = COMPUTE_INTER_PHY_DIST;
+		max_phy_count = COMPUTE_D2D_PHY_NUM;
+		reg_addr = COMP_0_CFG_COMP_SOC_VD2D_PHY_0_IPT_D2D_N3_TOP_CH_BCAST_PHYCTRLREG_BASE(mmIPT_D2D_N3_PHYCTRLREG_RXBITSYNCTMR);
+	} else {
+		inter_phy_offset = IO_INTER_PHY_DIST;
+		max_phy_count = IO_D2D_PHY_NUM;
+		reg_addr = IO0_H0_ID2D_PHY_0_D2D_PHY0_CH_BCAST_PHYCTRLREG_BASE(mmIPT_D2D_PHYCTRLREG_RXBITSYNCTMR);
+	}
+
+	for (phy_num = 0; phy_num < max_phy_count; phy_num++) {
+		phy_off = phy_num * inter_phy_offset;
+
+		val = regmap_readl(reg_addr + phy_off);
+		val |= (BITSYNC_TRKWIN_VAL << IPT_D2D_PHYCTRLREG_RXBITSYNCTMR_BITSYNC_TRKWIN_S);
+		regmap_writel(val, reg_addr + phy_off);
+
+		if(boot_rom_global.die_type != COMPUTE_DIE) {
+			h1_reg_addr = IO_HALF1((reg_addr + phy_off), (boot_rom_global.die_id + 1));
+
+			val = base_readl(h1_reg_addr);
+			val |= (BITSYNC_TRKWIN_VAL << IPT_D2D_PHYCTRLREG_RXBITSYNCTMR_BITSYNC_TRKWIN_S);
+			base_writel(val, h1_reg_addr);
+		}
+	}
+}

+void boot_rom_d2d_link_init(void)
+{
+	uint64_t start;
+	uint32_t val;
+
+	if (!(boot_rom_global.config2_fuse & EFUSE_SYSCFG2_D2D_ENABLE))
+		return; /* skip d2d init */
+
+	boot_rom_status_update(PROGRESS_BOOT_ROM_D2D_LINK_START);
+
+	/* TODO: IO HALF0 FSM_STS and HALF1 FSM_STS are in sync */
+	/* Check BUSY bit for 0 value, expected to be 0 */
+	val = regmap_readl(IO0_H0_PSOC_GLOBAL_CONF(mmGLOBAL_CONF_DPHY_FSM_STS));
+	if (val & GLOBAL_CONF_DPHY_FSM_STS_BUSY_M) {
+		boot_rom_status_update(PROGRESS_BOOT_ROM_D2D_LINK_STS_BUSY);
+		return;
+	}
+
+	/*
+	 * TODO: Decided after FS1 IO Die POR.
+	 * SW WA for HW Errata GAUDI3_0314
+	 */
+	boot_rom_d2d_phy_wd_hw_errata_fix();
+
+	boot_rom_d2d_phy_update_train_window();
+
+	/* Master Die and Slave Die's sync for d2d rec init */
+	if (boot_rom_global.die_type == IO_DIE_MASTER) {
+		/* Indicate slave die's that d2d init has started */
+		boot_rom_notify_slave();
+	} else {
+		boot_rom_wait_for_recovery_in();
+	}
+
+	/* Inter die indication for d2d link init handshaking */
+	boot_rom_status_update(PROGRESS_BOOT_ROM_D2D_LINK_INTER_DIE_SYNC);
+
+	/* TODO: IO HALF0 FSM_CTL and HALF1 FSM_CTL are in sync */
+	/* Set START bit for D2D link initialization */
+	regmap_writel(GLOBAL_CONF_DPHY_FSM_CTRL_START_M, IO0_H0_PSOC_GLOBAL_CONF(mmGLOBAL_CONF_DPHY_FSM_CTRL));
+
+	/*
+	 * Poll 10ms for BUSY bit to be 0 indicating link initialization
+	 * is completed successfully.
+	 */
+	start = boot_rom_get_ts_us();
+	while (1) {
+		val = regmap_readl(IO0_H0_PSOC_GLOBAL_CONF(mmGLOBAL_CONF_DPHY_FSM_STS));
+		if (!(val & GLOBAL_CONF_DPHY_FSM_STS_BUSY_M))
+			break;
+
+		if (boot_rom_timeout_expired(start, D2D_LINK_STATUS_UDELAY)) {
+			boot_rom_status_update(PROGRESS_BOOT_ROM_D2D_LINK_INIT_TIMEDOUT);
+			return;
+		}
+	}
+
+	boot_rom_status_update(PROGRESS_BOOT_ROM_D2D_LINK_INIT_SUCCESS);
+
+	/*
+	 * TODO: D2D link init success acknowledgement to initiator
+	 * If no ack back, initiator has to blindly proceed that all Die's init success.
+	*/
+}
