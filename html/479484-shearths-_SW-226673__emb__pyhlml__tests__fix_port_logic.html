+def get_port_index_map(mask):
+    """
+    Given a 64-bit integer mask, returns a dictionary mapping each set bit (port number)
+    to its corresponding sequential index.
+
+    Each bit in the mask represents a port: if the bit is set (1), the port is considered active.
+    The ports are numbered from 0 (least significant bit) to 63 (most significant bit).
+    The index reflects the order in which set bits are encountered from LSB to MSB.
+
+    Parameters:
+        mask (int): A 64-bit integer bitmask representing active ports.
+
+    Returns:
+        dict: A dictionary where keys are port numbers (bit positions with value 1),
+              and values are their corresponding sequential indices.
+
+    Example:
+        >>> get_port_index_map(0b10000010010)
+        {1: 0, 4: 1, 10: 2}
+    """
+    port_index_map = {}
+    index = 0
+    bit_pos = 0
+
+    while mask:
+        if mask & 1:
+            port_index_map[bit_pos] = index
+            index += 1
+        mask >>= 1
+        bit_pos += 1
+
+    return port_index_map
