+
+void spi_flash_set_byte_swap(uint8_t enable)
+{
+	uint32_t reg_spi_direct_wr_rd_ctrl = 0;
+
+	/* clear BYTE SWAP in SPI DMA so we dont switch endianness */
+	reg_spi_direct_wr_rd_ctrl =
+		regmap_readl(IO0_HALF0_PSOC_GLOBAL_CONF(mmGLOBAL_CONF_SPI_DIRECT_WR_RD_CTRL));
+	if (enable)
+		reg_spi_direct_wr_rd_ctrl |=
+			GLOBAL_CONF_SPI_DIRECT_WR_RD_CTRL_BYTE_SWAP_M;
+	else
+		reg_spi_direct_wr_rd_ctrl &=
+			~GLOBAL_CONF_SPI_DIRECT_WR_RD_CTRL_BYTE_SWAP_M;
+
+	regmap_writel(reg_spi_direct_wr_rd_ctrl,
+		      IO0_HALF0_PSOC_GLOBAL_CONF(mmGLOBAL_CONF_SPI_DIRECT_WR_RD_CTRL));
+}
+
+void spi_flash_set_baud_rate(uint8_t baud_rate)
+{
+	regmap_writel(baud_rate, IO0_HALF0_PSOC_QSPI_BASE(mmSPI_BAUDR));
+}
+
+void spi_flash_set_write_limit(enum flash_write_limit write_limit)
+{
+	regmap_writel(write_limit, IO0_HALF0_PSOC_GLOBAL_CONF(mmGLOBAL_CONF_SPI_WRITE_LIMIT));
+}

+#include <spi_flash.h>
+
+void spi_flash_init(void)
+{
+	/* make sure spi controller doesn't swap bytes */
+	spi_flash_set_byte_swap(DISABLE_BYTE_SWAP);
+
+	spi_flash_set_baud_rate(SPI_FLASH_BAUDR_DIVISOR);
+
+	spi_flash_set_write_limit(write_en_2MB);
+}
