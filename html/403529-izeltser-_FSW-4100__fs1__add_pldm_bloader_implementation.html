Commit Message

[FSW-4100] fs1: add pldm bloader implementation

PLDM Bloader is a small bootloader for PLDM that runs instead of
arcboot in the partial boot flow, and performs following tasks:
1. init PLLs for prints and performance
2. init memory with fresh copy of preboot
3. init memory with bootinfo snapshot

This patch adds the implementation of PLDM Bloader for FS1.

+#define MEM_CPY_SIZE_SHIFT		4
+#define MEM_CPY_SIZE_SHIFT_MASK		(BIT(MEM_CPY_SIZE_SHIFT) - 1)
+#define HL_MEM_CPY_TIMEOUT_US		10000

+int fs1_mem_copy(uint64_t src, uint64_t dst, uint32_t size)
+{
+	u64 start = timestamp_get_us();
+	u32 cpy_sts, copy_sz;
+	u32 cp_err, wr_err, rd_err;
+
+	if (size > MEM_CPY_SIZE_LIMIT) {
+		preboot_puts("size exceeds MEM_CPY engine limit\n");
+		return -1;
+	}
+
+	regmap_writel((uint32_t)src >> MEM_CPY_SIZE_SHIFT,
+		      mmIO_0_CFG_HALF_0_PSOC_PSOC_CFG_MEM_COPY_BASE + mmMEM_COPY_SRC_ADDR_LSB);
+	regmap_writel((uint32_t)(src >> 32), mmIO_0_CFG_HALF_0_PSOC_PSOC_CFG_MEM_COPY_BASE +
+					     mmMEM_COPY_SRC_ADDR_MSB);
+
+	regmap_writel((uint32_t)dst >> MEM_CPY_SIZE_SHIFT,
+		      mmIO_0_CFG_HALF_0_PSOC_PSOC_CFG_MEM_COPY_BASE + mmMEM_COPY_DEST_ADDR_LSB);
+	regmap_writel((uint32_t)(dst >> 32), mmIO_0_CFG_HALF_0_PSOC_PSOC_CFG_MEM_COPY_BASE +
+					     mmMEM_COPY_DEST_ADDR_MSB);
+
+	/* set size of in granuality of 16 bytes */
+	copy_sz = size >> MEM_CPY_SIZE_SHIFT;
+	if (size & MEM_CPY_SIZE_SHIFT_MASK) {
+		preboot_puts("MEM_CPY size not aligned, rounding up\n");
+		copy_sz++;
+	}
+
+	regmap_writel(copy_sz, mmIO_0_CFG_HALF_0_PSOC_PSOC_CFG_MEM_COPY_BASE +
+			       mmMEM_COPY_COPY_SIZE);
+
+	/* Enable copy and wait for completion */
+	regmap_writel(MEM_COPY_COPY_EN_VAL_M, mmIO_0_CFG_HALF_0_PSOC_PSOC_CFG_MEM_COPY_BASE +
+					      mmMEM_COPY_COPY_EN);
+
+	/* let 'enable' command propagate before polling on status */
+	HL_MNG_W_BARRIER();
+
+	while (1) {
+		cpy_sts = regmap_readl(mmIO_0_CFG_HALF_0_PSOC_PSOC_CFG_MEM_COPY_BASE +
+				       mmMEM_COPY_COPY_EN);
+		/* COPY_EN is set by FW, and cleared by HW when copy ends */
+		if (!(cpy_sts & MEM_COPY_COPY_EN_VAL_M))
+			break;
+
+		if (timeout_expired(start, HL_MEM_CPY_TIMEOUT_US)) {
+			preboot_puts("error: MEM_CPY engine timeout\n");
+			return -1;
+		}
+	}
+
+	cp_err = regmap_readl(mmIO_0_CFG_HALF_0_PSOC_PSOC_CFG_MEM_COPY_BASE +
+			      mmMEM_COPY_COPY_ERR_STS);
+	wr_err = regmap_readl(mmIO_0_CFG_HALF_0_PSOC_PSOC_CFG_MEM_COPY_BASE +
+			      mmMEM_COPY_WR_ERR_STS);
+	rd_err = regmap_readl(mmIO_0_CFG_HALF_0_PSOC_PSOC_CFG_MEM_COPY_BASE +
+			      mmMEM_COPY_RD_ERR_STS);
+
+	if (cp_err | wr_err | rd_err) {
+		preboot_puts("MEM_CPY finished with errors\n");
+		return -1;
+	}
+
+	return 0;
+}

+static void btldr_copy_preboot_app(void)
+{
+	int err;
+	u32 val;
+
+	/*
+	 * Clean copy of the application is stored in MNG ICCM. In case the application binary
+	 * exceeds MNG ICCM size (256kb), the rest is stored in MISC ICCM.
+	 * An efuse bit is controlling whether the application binary is larger than MNG ICCM size.
+	 */
+	fs1_mem_copy(ARC_MNG_ICCM_FULL_BASE, PREBOOT_TEXT_BASE_FULL, ARC_MNG_ICCM_SIZE);
+
+	err = preboot_fuse_read(IO_EFUSE_PLDM_CFG_0_OFFSET, &val);
+	if (err)
+		val = BLOADER_COPY_LARGE_APP;
+
+	val &= BLOADER_COPY_LARGE_APP;
+
+	if (!val)
+		return;
+
+	fs1_mem_copy(ARC_MISC_ICCM_FULL_BASE, PREBOOT_TEXT_BASE_FULL + ARC_MNG_ICCM_SIZE,
+		     ARC_MISC_ICCM_SIZE);
+}
+
+static void btldr_copy_bootinfo(void)
+{
+	int err;
+	u32 val;
+
+	err = preboot_fuse_read(IO_EFUSE_PLDM_CFG_0_OFFSET, &val);
+	if (err)
+		val = BLOADER_COPY_BOOTINFO;
+
+	val &= BLOADER_COPY_BOOTINFO;
+
+	if (!val)
+		return;
+
+	/*
+	 * Normally bootinfo is first written to spsram by the arcboot, so preboot rtos expects some
+	 * bootinfo data to be available when it runs. On PLDM, when running standalone preboot,
+	 * pre-recorded boot info data is loaded to the SPSRAM.
+	 * Since SPSRAM content is erazed on cold reboot, fresh copy stored in MISC DCCM and copied
+	 * back to the SPSRAM as part of the preparation to run preboot rtos.
+	 */
+	fs1_mem_copy(ARC_MISC_DCCM_FULL_BASE, HL_SP_SRAM_BASE, BOOT_INFO_SIZE);
+}
