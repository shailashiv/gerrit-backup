Commit Message

[SW-36011] gaudi: configure nic phy regs

Configure NIC phy registers according to nic enable
bitmask. This configuration is done on receiving
fit on secured transport (ELBI). The patch disables
or enables the port according to the bitmask.
A later patch shall protect this registers access.
In this way, NIC ports are enabled from FW only,
if secured transport (ELBI) writes the NIC
enable fit.

+#define NIC_PHY_RX_CFG_OFFSET	0x10
+#define NIC_PORT_EN		0
+#define NIC_PORT_DIS		2
+
+void enable_nic_regs(u32 nic_en)
+{
+	u32 port, val, idx, block_of;
+
+	for (port = 0; port < GAUDI_NIC_ENGS_NUM; port++) {
+		val = (nic_en & (1 << port)) ? NIC_PORT_EN : NIC_PORT_DIS;
+		idx = port >> 1;
+		block_of = idx * GAUDI_NICS_DIST;
+
+		if (port & 1) {
+			regmap_writel(val, mmNIC0_PHY_PHY_RX_CFG_2 + block_of);
+			regmap_writel(val, mmNIC0_PHY_PHY_RX_CFG_3 + block_of);
+		} else {
+			regmap_writel(val, mmNIC0_PHY_PHY_RX_CFG_0 + block_of);
+			regmap_writel(val, mmNIC0_PHY_PHY_RX_CFG_1 + block_of);
+		}
+	}
+}

 void preboot_nic_en(u32 nic_en_)
 {
 	nic_en = nic_en_;
+	enable_nic_regs(nic_en);
