Commit Message

[SW-66407] gaudi3: pcie_fw: copy phy fw to pcie sram

Pcie phy fw is copied from flash to FW sram along with the PPBOOT
app by the bootrom. But in order to perform the Pcie Phy FW upgrade
it should be placed in the PCIE SRAM. On cold boot PPBOOT app
is copying the phy fw from FW SRAM to PCIE SRAM using the PSOC
MEM_CPY engine.

+#define MEM_CPY_SIZE_LIMIT		(SZ_1M - SZ_16)
+#define MEM_CPY_SIZE_SHIFT		4
+#define MEM_CPY_SIZE_SHIFT_MASK		(BIT(MEM_CPY_SIZE_SHIFT) - 1)
+#define HL_MEM_CPY_TIMEOUT_US		1000
+
+static int gaudi3_hbw_memcpy(uint64_t src, uint64_t dst, uint32_t size)
+{
+	u64 start = timestamp_get_us();
+	u32 mem_cfg, cpy_sts, copy_sz;
+
+	if (size > MEM_CPY_SIZE_LIMIT) {
+		preboot_puts("size exceeds MEM_CPY engine limit\n");
+		return -1;
+	}
+
+	regmap_writel((uint32_t)src, mmD0_PSOC_GLOBAL_CONF_MEM_CPY_START_ADDR_L);
+	regmap_writel((uint32_t)(src >> 32), mmD0_PSOC_GLOBAL_CONF_MEM_CPY_START_ADDR_H);
+	regmap_writel((uint32_t)dst, mmD0_PSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_L);
+	regmap_writel((uint32_t)(dst >> 32), mmD0_PSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_H);
+
+	mem_cfg = regmap_readl(mmD0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+	mem_cfg &= ~D0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_MEM_SIZE_M;
+	mem_cfg &= ~D0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_USE_CONST_M;
+
+	/* set size of in granuality of 16 bytes */
+	copy_sz = size >> MEM_CPY_SIZE_SHIFT;
+	if (size & MEM_CPY_SIZE_SHIFT_MASK) {
+		preboot_puts("MEM_CPY size not aligned, rounding up\n");
+		copy_sz++;
+	}
+	mem_cfg |= (copy_sz & D0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_MEM_SIZE_M);
+
+	/*
+	 * WA for HW errata GAUDI3_0315
+	 */
+	mem_cfg |= (MEM_CPY_ERRATA_WR_OS_VAL << D0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_WR_OS_S);
+
+	/*
+	 * WA for HW errata GAUDI3_0306
+	 */
+	mem_cfg |= (MEM_CPY_ERRATA_RD_OS_VAL << D0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_RD_OS_S);
+
+	regmap_writel(mem_cfg, mmD0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+
+	/* Enable copy and wait for completion */
+	regmap_writel(D0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL_EN_M, mmD0_PSOC_GLOBAL_CONF_MEM_CPY_CTRL);
+	while (1) {
+		cpy_sts = regmap_readl(mmD0_PSOC_GLOBAL_CONF_MEM_CPY_STATUS);
+		if (cpy_sts & D0_PSOC_GLOBAL_CONF_MEM_CPY_STATUS_DONE_M)
+			break;
+
+		if (timeout_expired(start, HL_MEM_CPY_TIMEOUT_US)) {
+			preboot_puts("error: MEM_CPY engine timeout\n");
+			return -1;
+		}
+	}
+
+	return 0;
+}

+void hl_init_pcie_phy_fw(void)
+{
+	struct boot_desc desc;
+	u64 src;
+	u32 idx, actual_size, rst_src = hl_rst_src_get();
+	unsigned int num_crc;
+	int err;
+
+	/* PHY FW is copied only after cold boot, remains in the memory after other resets */
+	if (rst_src != HL_COLD_RST)
+		return;
+
+	if (!boot_desc_init()) {
+		preboot_puts("No boot desc, err loading Phy FW\n");
+		return;
+	}
+
+	idx = boot_desc_get_img_desc(FLOPS_UPD_NOOP, HL_FW_IMG_PPBOOT, &desc);
+	if (idx == HL_FW_IMG_NOT_FOUND) {
+		preboot_puts("ppboot descriptor, err loading Phy FW\n");
+		return;
+	}
+
+	num_crc = ppboot_get_num_crc(desc.img_size);
+	actual_size = desc.img_size - (num_crc * CRC32_SIZE);
+	src = desc.loadaddr + actual_size - PHY_FW_SIZE;
+
+	err = gaudi3_hbw_memcpy(src, HL_PCIE_FW_SRAM_BASE_FULL, PHY_FW_SIZE);
+	if (err)
+		preboot_puts_hex("failed to copy phy fw to pcie sram, err 0x", err, false);
