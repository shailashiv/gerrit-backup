[SW-20154] gaudi2 : erom arcboot changes

The extended rom APIs are used by
arcboot based on the efuse setting.
Each API is a part its own individual
binary which is indexed in the flash
memory using a separate 'image pointer'
binary.

The 3 erom binaries are:
1) Alternate pcie config (erom_pcie.bin)
2) Recovery image (erom_recovery.bin)
3) TPM (erom_tpm.bin)

From the above binaries, 1) and 2) are
loaded based on efuse bit 13 (logic low)
while 3)is loaded based on efuse bit 11
(logic high).

+static int check_boundary(uint64_t addr, uint32_t size, uint64_t endaddr)
+{
+	return ((addr + size) <= endaddr) ? 0 : 1;

+/*
+ * check_load_addr_valid updates 'addr' with the 32bit ARC
+ * view point address used to jump to the code and execute.
+ * If address is valid and within bounds 0 is returned,
+ * else returns 1.
+ */
+static int check_load_addr_valid(struct boot_desc *p_desc, uint32_t *addr)
+{
+	uint32_t addr_low = LOW_DWORD(p_desc->loadaddr);
+	int ret;
+
+	switch (p_desc->loadaddr) {
+	case HL_ICCM_BASE ... HL_ICCM_END:
+		ret = check_boundary(p_desc->loadaddr, p_desc->img_size,
+				HL_ICCM_END);
+		*addr = addr_low & ICCM_OFFS_MASK;
+		break;
+	case HL_SP_SRAM_BASE ... HL_SP_SRAM_END:
+		ret = check_boundary(p_desc->loadaddr, p_desc->img_size,
+				HL_SP_SRAM_END);
+		*addr = HL_SCRATCHPAD_SRAM_BASE +
+			(addr_low & SP_SRAM_OFFS_MASK);
+		/*
+		 * Note that the offset of SP SRAM to be provided to perform
+		 * DMA must be in granuality of 4B. ie.
+		 * (right shift offset 0x1000 by 2).
+		 */
+		p_desc->loadaddr = (HL_SP_SRAM_BASE +
+				((addr_low & SP_SRAM_OFFS_MASK) >> 2));
+		break;

+		ret = check_boundary(p_desc->loadaddr, p_desc->img_size,
+				HL_SRAM_PHY_END);
