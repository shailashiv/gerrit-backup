+#define VERSION_MAX_LEN		128
+
+#define PREBOOT_VER_STR	"Preboot version " PREBOOT_VER \
+	" (" PREBOOT_DATE " - " PREBOOT_TIME ")"

+/* Stable ID can be 0 or 1, if not defined return primary offset (0) */
+static u32 stable_image_offset(void)
+{
+	u32 stable_image_id;
+
+	spi_copy(&stable_image_id, HL_SPI_NEXT_IMAGE_OFF,
+		 sizeof(stable_image_id), "NEXT");
+	stable_image_id = (stable_image_id & 0xFF) - '1';
+
+	if (stable_image_id == 1)
+		return HL_SPI_SECONDARY_BASE;
+
+	return HL_SPI_PRIMARY_BASE;
+}
+
+static u32 modifiable_image_offset(bool *pending)
+{
+	u32 ret = HL_SPI_PRIMARY_BASE;
+	u32 modifiable_id;
+	u32 upd_pending_reg = get_upd_pending_sts_reg();
+
+	spi_copy(&modifiable_id, HL_SPI_MODIFI_IMAGE_OFF,
+		 sizeof(modifiable_id), "MODIFIABLE");
+	modifiable_id = (modifiable_id & 0xFF) - '1';
+
+	if (modifiable_id < 2) {
+		*pending = true;
+		if (!modifiable_id) {
+			regmap_writel(FLOPS_UPD_NXT_IMG1, upd_pending_reg);
+		} else {
+			regmap_writel(FLOPS_UPD_NXT_IMG2, upd_pending_reg);
+			ret = HL_SPI_SECONDARY_BASE;
+		}
+	} else { /* No valid modifiable image ID */
+		regmap_writel(FLOPS_UPD_NOOP, upd_pending_reg);
+	}
+
+	return ret;
+}
+
+static u32 next_image_offset(bool *pending)
+{
+	u32 upd_pending_reg = get_upd_pending_sts_reg();
+	u32 image_pending = regmap_readl(upd_pending_reg);
+
+	/* Check for normal boot */
+	if (image_pending != FLOPS_UPD_PENDING)
+		return stable_image_offset();
+
+	/* There is an update pending */
+	return modifiable_image_offset(pending);
+}
+
+static void preboot_copy_spl(void)
+{
+	char *name = "SPL";
+	bool upgrade_pending = false;
+	u32 img_offset = next_image_offset(&upgrade_pending);
+	u32 spl_offset = img_offset + HL_SPI_SPL_OFF;
+	u32 uboot_offset = img_offset + HL_SPI_UBOOT_OFF;
+	char *spl_image = upgrade_pending ? "candidate" : "stable";
+
+	preboot_puts_raw("Preparing to run SPL ");
+	preboot_puts_raw(spl_image);
+	preboot_puts_hex_raw(" (offset: ", img_offset, false);
+	preboot_puts(")\n");
+
+	spi_copy((void *)SPL_TEXT_BASE, spl_offset, SPL_SIZE, name);
+
+	regmap_writel(uboot_offset, mmPSOC_GLOBAL_CONF_SCRATCHPAD_30);
+}
+
+static void preboot_pcie_init(void)
+{
+	if (hl_skip_pcie_init()) {
+		preboot_puts("PCIe initialization is skipped!\n");
+		return;
+	}
+
+	//preboot_puts("Waiting for PCIe link...\n");
+	preboot_pcie_wait_for_link();
+	preboot_puts("PCIe link is UP!\n");
+}
+
+#ifndef TARGET_IS_TESTER
+static void preboot_version(void)
+{
+	const char *btl_ver_tag = "BTL version ";
+	const char *preboot_ver_str = PREBOOT_VER_STR;
+	char *dst_version_str = (char *)PREBOOT_VER_BASE;
+	unsigned long offset = (unsigned long)dst_version_str;
+	u32 btlver = regmap_readl(mmPSOC_GLOBAL_CONF_SCRATCHPAD_28);
+	int i, btl_ver_sz;
+
+	preboot_puts("\n");
+	preboot_puts_raw(btl_ver_tag);
+	preboot_puth(btlver, true);
+	preboot_puts_raw(preboot_ver_str);
+	preboot_puts("\n");
+
+	/* Copy the BTL version tag to destination (SRAM) address */
+	for (i = 0; i < VERSION_MAX_LEN; i++) {
+		if (btl_ver_tag[i] == '\0')
+			break;
+
+		dst_version_str[i] = btl_ver_tag[i];
+	}
+
+	/* convert and copy the btl version */
+	i += preboot_uint2hexstring(&dst_version_str[i], btlver, true);
+
+	/* separate the btl from preboot */
+	dst_version_str[i++] = '\n';
+	btl_ver_sz = i;
+
+	/* copy the preboot version */
+	for (int j = 0; j < VERSION_MAX_LEN - btl_ver_sz - 1; j++) {
+		dst_version_str[i++] = preboot_ver_str[j];
+		if (preboot_ver_str[j] == '\0')
+			break;
+	}
+
+	regmap_writel((u32)(offset & ~SDRAM_BASE),
+			mmPSOC_GLOBAL_CONF_SCRATCHPAD_28);
+}
+#else	/* TARGET_IS_TESTER */
+static inline void preboot_version(void) {}
+#endif	/* TARGET_IS_TESTER */
+
+static void hl_jtag_init(void)
+{
+	u32 reg, nPRESETDBG = BIT(12);
+
+	/* Enable JTAG attach */
+	reg = regmap_readl(mmCPU_CA53_CFG_ARM_RST_CONTROL);
+	reg |= nPRESETDBG;
+	regmap_writel(reg, mmCPU_CA53_CFG_ARM_RST_CONTROL);
+}
+
+void preboot_main(void)
+{
+	void (*spl_start)(void) = (void *)SPL_TEXT_BASE;
+
+	regmap_writel(CPU_BOOT_STATUS_IN_PREBOOT, get_cpu_boot_status_reg());
+
+	hl_jtag_init();
+	preboot_sync_btl();
+	hl_uart_init();
+	preboot_pcie_init();
+	preboot_version();
+	preboot_delay();
+	hl_pll_init();
+	hl_pll_wa();
+	/* reinit uart after PLLs configured */
+	hl_uart_init();
+	preboot_copy_spl();
+
+	spl_start();
+}

+#define COMMON_PREBOOT_H_
+
+#define BIT(nr)					(1 << (nr))
+#define ARRAY_SIZE(x)				(sizeof(x) / sizeof((x)[0]))
+
+#ifdef __arc__
+#define HL_SPI_BASE				0x1A000000
+#define HL_CFG_BASE				0x1C000000
+#define SDRAM_BASE				0x10000000
+#define SPL_TEXT_BASE				0x10080000
+#endif
+
+#ifdef __aarch64__
+#define HL_SPI_BASE				0x7FF8000000ull
+#define HL_CFG_BASE				0x7FFC000000ull
+#define SDRAM_BASE				0x7FF0000000ull
+#define SPL_TEXT_BASE				0x7FF0080000ull
+#endif
+
+#define PREBOOT_VER_BASE			(SPL_TEXT_BASE - 0x1000)
+
+#define SPL_SIZE		0x10000		/* 64KB */
+#define HL_UART_BAUDRATE	115200
+#define HL_SPI_FLASH_MAX_FREQ	25000000
+#define HL_SPI_NEXT_IMAGE_OFF	0x60000
+#define HL_SPI_MODIFI_IMAGE_OFF	0x60004
+#define HL_SPI_SPL_OFF		0xA0000
+#define HL_SPI_UBOOT_OFF	0xB0000
+#define HL_SPI_PRIMARY_BASE	0
+#define HL_SPI_SECONDARY_BASE	0x800000
+
+typedef unsigned char u8;
+typedef unsigned int u32;
+typedef unsigned long long u64;
+
+static inline u8 regmap_readb(u32 offset)
+{
+	u32 reg = *((volatile u32 *)(HL_CFG_BASE + offset));
+
+	return (u8)(reg & 0xFF);
