+static void ext_rom_load_erom_image(enum hl_fw_img_type_erom img_type)
+{
+	enum arcboot_rom_progress img_sts_base = PROGRESS_ARCBOOT_EROM_PCIE;
+	enum dma_error dma_err = DMA_SUCCESS;
+	uint32_t idx, num_crc, img_len_nocrc;
+	enum dma_dest_type dst_type;
+	uint32_t loadaddr_arc;
+	struct boot_desc desc;
+	unsigned char *image;
+	void (*erom_func)();
+	int err;
+
+	if (img_type == HL_FW_EROM_RECOVERY)
+		img_sts_base = PROGRESS_ARCBOOT_EROM_RECOVERY;
+
+	arcboot_rom_status_update_bringup(img_sts_base, PROGRESS_EROM_BRINGUP_START);
+
+	idx = boot_desc_get_img_desc(HL_FW_BOOT_PTR_EROM, img_type, &desc, FLOPS_UPD_NOOP);
+	if (idx == HL_FW_IMG_NOT_FOUND)
+		return;
+
+	dst_type = arcboot_rom_get_dma_dest_type(desc.loadaddr);
+
+	/* TODO: check image descriptor values */
+	arcboot_rom_status_update_bringup(img_sts_base, PROGRESS_EROM_DMA_START);
+
+	dma_err = dma_copy_image_v2(desc.img_size, desc.spi_offset,
+				desc.loadaddr, dst_type, false);
+	if (dma_err != DMA_SUCCESS) {
+		if (dma_err == DMA_INVALID_CRC_ERROR) {
+			arcboot_rom_status_update_bringup(img_sts_base,
+					PROGRESS_EROM_DMA_CRC_FAILED);
+		} else if (dma_err == DMA_UNALIGNED_DEST_ADDR) {
+			arcboot_rom_status_update_bringup(img_sts_base,
+					PROGRESS_EROM_DMA_UNALIGNED_DEST_ADDR);
+		} else if (dma_err == DMA_UNALIGNED_IMG_SIZE) {
+			arcboot_rom_status_update_bringup(img_sts_base,
+					PROGRESS_EROM_DMA_UNALIGNED_IMG_SIZE);
+		} else {
+			arcboot_rom_status_update_bringup(img_sts_base,
+					PROGRESS_EROM_DMA_ERROR);
+		}
+
+		return;
+	}
+
+	arcboot_rom_status_update_bringup(img_sts_base, PROGRESS_EROM_DMA_FINISH);
+
+	if (!security_efuse_image_validation_en())
+		goto run_image;
+
+	num_crc = arcboot_rom_get_num_crc(desc.img_size);
+	loadaddr_arc = (uint32_t)CONV_64B_TO_ARC(desc.loadaddr);
+	image = (unsigned char *)(loadaddr_arc);
+
+	img_len_nocrc = desc.img_size - (num_crc * CRC32_SIZE);
+
+	err = secureboot_verify_image_hash384(desc.chksum, image, img_len_nocrc);
+	if (err) {
+		arcboot_rom_status_update_bringup(img_sts_base,
+					PROGRESS_EROM_IMG_VERIFICATION_FAIL);
+		return;
+	}
+
+run_image:
+	arcboot_rom_status_update_bringup(img_sts_base, PROGRESS_EROM_IMG_START);
+	erom_func = (void *)((uint32_t)CONV_64B_TO_ARC(desc.start_addr));
+	erom_func();
+	arcboot_rom_status_update_bringup(img_sts_base, PROGRESS_EROM_IMG_FINISH);
+}
