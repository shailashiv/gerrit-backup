+
+#include <arcboot.h>
+#include <spi_flash.h>
+#include <efuse.h>
+
+void spi_flash_init(void)
+{
+	if (arcboot_rom_global.die_id)
+		return;
+
+	/* make sure spi controller doesn't swap bytes */
+	spi_flash_set_byte_swap(DISABLE_BYTE_SWAP);
+
+	regmap_writel(SPI_FLASH_BAUDR_DIVISOR, mmQSPI_BAUDR);
+
+	regmap_writel(write_en_2MB, mmD0_PSOC_GLOBAL_CONF_SPI_WRITE_LIMIT);
+}
