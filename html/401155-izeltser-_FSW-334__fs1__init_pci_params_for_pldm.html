Commit Message

[FSW-334] fs1: init pci params for pldm

Add specific PCI configuration for PLDM.
On PLDM emulator there is no PHY and the configuration required for
PCI are very minimal, and mostly depend on vlsi image support.
To provide some flexibility, spare efuse bits are used to control over
some features, like nubmer of VFs, Cxl.cache capability and ATS
capability support. Other custom configs might be added in the future.

+static bool is_cxl_mode(void)
+{
+	u32 val;
+
+	val = regmap_readl(IO0_H0_PSOC_BOOT(mmPSOC_BOOT_CONF_BOOT_STRAP_PINS_H));
+	val &= PSOC_BOOT_CONF_BOOT_STRAP_PINS_H_CXL_MODE_M;
+
+	return !!val;
+}

+#define SRIOV_MAX_NUM_VFS	4
+#define SRIOV_DEFAULT_NUM_VFS	1
+
+static void pldm_pcie_dynamic_init(void)
+{
+	bool cxl_mode = is_cxl_mode();
+	u32 pci_efuse_cfg;
+	u8 num_vf = 0;
+	u32 val;
+
+	hl_fuse_read(mmIO_0_CFG_HALF_0_PSOC_PSOC_CFG_EFUSE_BASE, IO_EFUSE_PLDM_PCI_CFG_OFFSET,
+		     &pci_efuse_cfg);
+	preboot_puts_hex("efuse pci cfg = ", pci_efuse_cfg, false);
+
+	if (cxl_mode) {
+		/* CXL capability is enabled as reset default */
+
+		/* disable CXL.Cache capability if required */
+		if (!(pci_efuse_cfg & CXL_CACHE_CAP_ENABLED_M)) {
+			val = regmap_readl(IO0_H0_PCIE_DBI(
+						mmPCIE_DBI_CXL_RCRB_DVSEC_HDR_2_FLEXBUS_CAP_OFF));
+			val &= ~PCIE_DBI_CXL_RCRB_DVSEC_HDR_2_FLEXBUS_CAP_OFF_CACHE_CAP_M;
+			regmap_writel(val, IO0_H0_PCIE_DBI(
+						mmPCIE_DBI_CXL_RCRB_DVSEC_HDR_2_FLEXBUS_CAP_OFF));
+			val = regmap_readl(IO0_H0_PCIE_DBI(
+						mmPCIE_DBI_CXL_RCIEP_DVSEC_HDR_2_FLEXBUS_CAP_OFF));
+			val &= ~PCIE_DBI_CXL_RCIEP_DVSEC_HDR_2_FLEXBUS_CAP_OFF_CACHE_CAP_M;
+			regmap_writel(val, IO0_H0_PCIE_DBI(
+						mmPCIE_DBI_CXL_RCIEP_DVSEC_HDR_2_FLEXBUS_CAP_OFF));
+		}
+	} else { /* pci mode */
+		/* switch to PCI mode by disabling Alternative Protocol Support */
+		val = regmap_readl(IO0_H0_PCIE_DBI(mmPCIE_DBI_PL32G_CAPABILITY_REG));
+		val &= ~PCIE_DBI_PL32G_CAPABILITY_REG_MOD_TS_ALT_PROTOCOL_SUPPORT_M;
+		regmap_writel(val, IO0_H0_PCIE_DBI(mmPCIE_DBI_PL32G_CAPABILITY_REG));
+
+		/* set number of VFs */
+		num_vf = pci_efuse_cfg & SRIOV_NUM_OF_VFS_M;
+		if (num_vf > SRIOV_MAX_NUM_VFS) {
+			num_vf = SRIOV_DEFAULT_NUM_VFS;
+			preboot_puts("VF exceeds 4, using default of 1 VF\n");
+		}
+
+		val = num_vf << PCIE_DBI_TOTAL_VFS_INITIAL_VFS_REG_SRIOV_INITIAL_VFS_S |
+		      num_vf << PCIE_DBI_TOTAL_VFS_INITIAL_VFS_REG_SRIOV_TOTAL_VFS_S;
+		regmap_writel(val, IO0_H0_PCIE_DBI(mmPCIE_DBI_TOTAL_VFS_INITIAL_VFS_REG));
+	}
+
+	/* skip ATS capability */
+	if (!(pci_efuse_cfg & ATS_CAPABILITY_ENABLED_M)) {
+		val = regmap_readl(IO0_H0_PCIE_DBI(mmPCIE_DBI_SRIOV_BASE_REG));
+		val &= ~PCIE_DBI_SRIOV_BASE_REG_SRIOV_NEXT_OFFSET_M;
+		val |= mmPCIE_DBI_PRS_EXT_CAP_HDR_REG <<
+			PCIE_DBI_SRIOV_BASE_REG_SRIOV_NEXT_OFFSET_S;
+		regmap_writel(val, IO0_H0_PCIE_DBI(mmPCIE_DBI_SRIOV_BASE_REG));
+	}
+}
+
+#define BAR_TYPE_MEM	0xC
+#define BAR_SIZE_1GB	(((SZ_1G - 1) & PCIE_DBI_BAR0_REG_BAR0_START_M) | BAR_TYPE_MEM)
+
+static void pldm_pcie_params_init(void)
+{
+	u32 val;
+
+	/* enable writing to DBI RO regs */
+	val = regmap_readl(IO0_H0_PCIE_DBI(mmPCIE_DBI_MISC_CONTROL_1_OFF));
+	val |= PCIE_DBI_MISC_CONTROL_1_OFF_DBI_RO_WR_EN_M;
+	regmap_writel(val, IO0_H0_PCIE_DBI(mmPCIE_DBI_MISC_CONTROL_1_OFF));
+
+	/* set PF CFG BAR to 1GB */
+	regmap_writel(BAR_SIZE_1GB, IO0_H0_PCIE_DBI(mmPCIE_DBI_BAR0_REG));
+	regmap_writel(0, IO0_H0_PCIE_DBI(mmPCIE_DBI_BAR1_REG));
+
+	/* set SRIOV CFG BAR to 1GB */
+	regmap_writel(BAR_SIZE_1GB, IO0_H0_PCIE_DBI(mmPCIE_DBI_SRIOV_BAR0_REG));
+	regmap_writel(0, IO0_H0_PCIE_DBI(mmPCIE_DBI_SRIOV_BAR1_REG));
+
+	pldm_pcie_dynamic_init();
+
+	/* disable writing to DBI RO regs */
+	val = regmap_readl(IO0_H0_PCIE_DBI(mmPCIE_DBI_MISC_CONTROL_1_OFF));
+	val &= ~PCIE_DBI_MISC_CONTROL_1_OFF_DBI_RO_WR_EN_M;
+	regmap_writel(val, IO0_H0_PCIE_DBI(mmPCIE_DBI_MISC_CONTROL_1_OFF));
+}

+				      mmPCIE_AUX_LTSSM_EN);
