+	enum dma_error err = DMA_SUCCESS;

+	boot_rom_global.curr_img = cur_img;

+	boot_desc_get_img_desc(HL_FW_BOOT_PTR_SPI, img, desc, upd_status);
+	if (!boot_rom_is_valid_address_range(desc->loadaddr, desc->img_size)) {
+		boot_rom_status_update_bringup(img_sts_base, PROGRESS_BRINGUP_LOADADDR_INVALID);
+		return BOOT_ROM_BRINGUP_ERROR;
+	}
+
+	if (!boot_rom_is_valid_start_addr(desc)) {

+		return BOOT_ROM_BRINGUP_ERROR;

+	//boot_rom_security_report_img_fw(boot_rom_global.curr_img, desc->sec_ver); /* TODO: security_impl */

+static int boot_rom_bringup_img(int img, uint64_t *start_addr)
+{
+	enum fw_post_reset_cmd fw_upd_status = boot_rom_read_upd_status();
+	int ret;
+	struct boot_desc desc = {0};
+
+	ret = boot_rom_bringup_img_sub(img, &desc, fw_upd_status, BOOT_ROM_PRIMARY);
+	if (ret == BOOT_ROM_SUCCESS)
+		goto image_bringup_success;
+
+	if (boot_rom_secondary_img_disabled())
+		return BOOT_ROM_FAIL;
+
+	/* Try the secondary image by toggling upd_status variable */
+	fw_upd_status = (fw_upd_status == FLOPS_UPD_PENDING) ? FLOPS_UPD_NOOP :
+			 FLOPS_UPD_PENDING;
+
+	ret = boot_rom_bringup_img_sub(img, &desc, fw_upd_status, BOOT_ROM_SECONDARY);
+	if (ret == BOOT_ROM_SUCCESS)
+		goto image_bringup_success;
+
+	return BOOT_ROM_FAIL;
+
+image_bringup_success:
+	*start_addr = desc.start_addr;
+
+	return BOOT_ROM_SUCCESS;
