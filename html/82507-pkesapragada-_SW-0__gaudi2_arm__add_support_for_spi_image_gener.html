+#!/usr/bin/env bash
+
+function handle_btl {
+	local INF=$1
+	local OUTF=$2
+
+	for s in s?.bin; do
+		# Align to 4B
+		dd status=none conv=sync bs=4 if=${s} of=${s}.le
+	done
+
+	# Align to 64KB
+	dd status=none conv=sync bs=64K if=${INF} of=${INF}.tmp
+	swap_file ${INF}.tmp ${OUTF}
+}
+
+function create_spi_image {
+	local BTL_BIN=$1
+	local PREBOOT_BIN=$2
+	local IMG_PTR_BIN=$3
+	local OUTF=$4
+
+	# BTL
+	dd status=none conv=sync bs=64K		if=${BTL_BIN}		 > ${OUTF}
+	# Image pointers ARC and Security ARC
+	dd status=none conv=sync bs=128K count=1 if=128K-0xff.bin	>> ${OUTF}
+	# Image pointers ARM
+	dd status=none conv=sync bs=64K		if=${IMG_PTR_BIN}	>> ${OUTF}
+	# Security and U-Boot ENV
+	dd status=none conv=sync bs=128K count=1 if=128K-0xff.bin	>> ${OUTF}
+	# Preboot1
+	dd status=none conv=sync bs=64K	if=${PREBOOT_BINFILE}		>> ${OUTF}
+	# Preboot2
+	dd status=none conv=sync bs=64K	if=${PREBOOT_BINFILE}		>> ${OUTF}
+}
+
+PREBOOT_OFFSET=0xb0000
+PREBOOT2_OFFSET=0xc0000
+PREBOOT_TEXT_BASE=0x7FFD004000
