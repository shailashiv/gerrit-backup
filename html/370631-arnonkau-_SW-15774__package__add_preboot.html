+PREBOOT_WEST_DEST_DIR :=

+# Arguments:
+# - module name
+# - component dir
+# - component name

+define PREBOOT_DEFINE_MODULE_ENVVARS
+	$(eval MODULE_NAME=$(strip $1))
+	$(eval MODULE_NAME_UPPER_CASE=$(shell echo $(MODULE_NAME) | tr a-z A-Z))
+	$(eval MODULE_NAME_UPPER_CASE=$(strip $(MODULE_NAME_UPPER_CASE)))

+# Arguments:

+define PREBOOT_MODULE_VERSION_HOOK
+	$(eval MODULE_NAME_UPPER_CASE=$(shell echo $1 | tr a-z A-Z))
+	$(eval MODULE_NAME_UPPER_CASE=$(strip $(MODULE_NAME_UPPER_CASE)))
+	$(call HL_VERSION_GEN, ${${MODULE_NAME_UPPER_CASE}_SRCDIR}, ${${MODULE_NAME_UPPER_CASE}_DIR}, ${${MODULE_NAME_UPPER_CASE}_NAME})
+endef
+
+define PREBOOT_APP_VERSION_GEN
+	@$(call HL_VERSION_GEN, ${PREBOOT_SRCDIR},.,${PREBOOT_APP_PROD_NAME})
+	@$(call HL_VERSION_GEN, ${PREBOOT_SRCDIR},build,${PREBOOT_APP_PROD_NAME})
+	@$(foreach module,$(PREBOOT_APP_MODULES),$(call PREBOOT_MODULE_VERSION_HOOK,$(module)))
+endef

+	@echo "#define HL_VERSION_PATCH $$(echo $(PREBOOT_LOCAL_VERSION) | cut -d. -f3)" >> $(PREBOOT_BUILD_DIR)/include/hl_version.h
+	@echo "#endif" >>  $(PREBOOT_BUILD_DIR)/include/hl_version.h
+endef
