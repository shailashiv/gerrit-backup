+bool security_efuse_image_validation_en(void)
+{
+	/* check efuse for security validation enable */
+	uint32_t sig_val = boot_rom_global.sec_data.efuse_sys_cfg & 
+		EFUSE_SYS_CFG_SIGNATURE_VALIDATION_EN;
+
+	return (sig_val > 0);
+}
+
+int security_check_valid_fw_version(uint32_t fw_version)
+{
+	if ((fw_version < boot_rom_global.sec_data.sec_ver) ||
+		(fw_version > SECURITY_MAX_FW_SEC_VERSION)) {
+		return SECURITY_FW_VERSION_NOT_SECURE;
+	}
+
+	return SECURITY_FW_VERSION_SECURE;
+}
+
+int security_check_key_revoke_status(uint32_t key_index)
+{
+	uint8_t key_revoke;
+
+	if (key_index >= NUM_KEYS_IN_ROM)
+		return -1;
+
+	key_revoke = boot_rom_global.sec_data.key_revoke;
+
+	/* if the key is not revoked, return 1 */
+	if ((key_revoke & BIT(key_index)) == 0)
+		return 1;
+
+	/* no valid key found */
+	return -1;
+}
+
+int security_rom_get_key_hash_by_idx(uint32_t *hash, uint8_t key_index)
+{
+	uint32_t *ptr = (uint32_t *)rom_key_hashes[key_index];
+
+	memcpy(hash, ptr, SHA384_SUM_LEN);
+
+	return 0;
+}
+
+int secureboot_verify_key_hash(int key_index, uint8_t *key, uint32_t key_len)
+{
+	uint8_t expected_hash[SHA384_SUM_LEN];
+	uint8_t actual_hash[SHA384_SUM_LEN];
+
+	if (security_rom_get_key_hash_by_idx((uint32_t *)expected_hash, key_index))
+		return SECUREBOOT_VERIFY_KEY_FAILED;
+
+	sha384_csum(key, key_len, actual_hash, 0);
+
+	return memcmp(actual_hash, expected_hash, SHA384_SUM_LEN);
+}
+
+int secureboot_verify_image_signature(const uint8_t *image, uint32_t image_len, uint8_t *key,
+				uint32_t key_len, uint8_t *sig, uint32_t sig_len,
+				uint8_t sec_ver, uint8_t key_index)
+{
+	int ret;
+
+	if (!security_efuse_image_validation_en())
+		return SECUREBOOT_VERIFY_SUCCESS;
+
+	if ((key_len > TOTAL_PUB_KEY_PROPS_SZ) || (sig_len > SIG_SIZE))
+		return SECUREBOOT_VERIFY_FAILED;
+
+	if (security_check_valid_fw_version(sec_ver) != SECURITY_FW_VERSION_SECURE) {
+		return SECUREBOOT_VERIFY_VER_FAILED;
+	}
+
+	ret = security_check_key_revoke_status(key_index);
+	if (ret != 1)
+		return SECUREBOOT_VERIFY_KEY_REVOKED;
+
+	if (secureboot_verify_key_hash(key_index, key, key_len))
+		return SECUREBOOT_VERIFY_KEY_FAILED;
+
+	ret = rsa_verify(image, image_len, key, key_len, sig, sig_len);
+
+	/* 0 - SUCCESS, but arcboot main code, expects 1 for SUCCESS */
+	if (!ret)
+		return SECUREBOOT_VERIFY_SUCCESS;
+
+	return SECUREBOOT_VERIFY_FAILED;
+}
+
+int secureboot_verify_image_hash384(uint8_t *exp_hash, unsigned char *image, uint32_t image_len)
+{
+	uint8_t actual_hash[SHA384_SUM_LEN];
+
+	sha384_csum(image, image_len, actual_hash, 0);
+
+	return memcmp(actual_hash, exp_hash, SHA384_SUM_LEN);
+}

+static void
+sha512_transform(uint64_t *state, const uint8_t *input)

+	for (i = 0; i < 80; i += 8) {

+		t1 = g + e1(d) + Ch(d,e,f) + sha512_K[i+1] + W[(i & 15) + 1];
+		t2 = e0(h) + Maj(h,a,b);    c+=t1;    g=t1+t2;
+		t1 = f + e1(c) + Ch(c,d,e) + sha512_K[i+2] + W[(i & 15) + 2];
+		t2 = e0(g) + Maj(g,h,a);    b+=t1;    f=t1+t2;
+		t1 = e + e1(b) + Ch(b,c,d) + sha512_K[i+3] + W[(i & 15) + 3];
+		t2 = e0(f) + Maj(f,g,h);    a+=t1;    e=t1+t2;
+		t1 = d + e1(a) + Ch(a,b,c) + sha512_K[i+4] + W[(i & 15) + 4];

+static void sha512_base_do_update(sha512_context *sctx,
+					const uint8_t *data,
+					unsigned int len)
+{
+	unsigned int partial = sctx->count[0] % SHA512_BLOCK_SIZE;
+
+	sctx->count[0] += len;
+	if (sctx->count[0] < len)
+		sctx->count[1]++;
+
+	if (unlikely((partial + len) >= SHA512_BLOCK_SIZE)) {
+		int blocks;
+
+		if (partial) {
+			int p = SHA512_BLOCK_SIZE - partial;
+
+			memcpy(sctx->buf + partial, data, p);
+			data += p;
+			len -= p;
+
+			sha512_block_fn(sctx, sctx->buf, 1);
+		}
+
+		blocks = len / SHA512_BLOCK_SIZE;
+		len %= SHA512_BLOCK_SIZE;
+
+		if (blocks) {
+			sha512_block_fn(sctx, data, blocks);
+			data += blocks * SHA512_BLOCK_SIZE;
+		}
+		partial = 0;
+	}
+	if (len)

+static void sha512_base_do_finalize(sha512_context *sctx)
+{
+	const int bit_offset = SHA512_BLOCK_SIZE - sizeof(uint64_t[2]);
+	uint64_t *bits = (uint64_t *)(sctx->buf + bit_offset);
+	unsigned int partial = sctx->count[0] % SHA512_BLOCK_SIZE;

+	sctx->buf[partial++] = 0x80;

+		memset(sctx->buf + partial, 0x0, SHA512_BLOCK_SIZE - partial);
+		partial = 0;
