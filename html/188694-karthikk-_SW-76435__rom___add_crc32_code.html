+/*
+ * Compute CRC32 using 256 deep lookup table.
+ * Taken from Linux implementation of Sarwate algorithm with additional bit
+ * flip for pre and post CRC.
+ * This implementation is only good for little endian processor.
+ */
+
+#define CRC32_POLY	0xedb88320
+#define CRC_BITS	8
+#define TABLE_SIZE	256
+
+static uint32_t crc32table[256];
+static bool is_crc32_table_initialized;
+
+void crc32_init_lookup(void)
+{
+	unsigned int i, j;
+	uint32_t crc = 1;
+
+	crc32table[0] = 0;
+
+	for (i = TABLE_SIZE >> 1; i; i >>= 1) {
+		crc = (crc >> 1) ^ ((crc & 1) ? CRC32_POLY : 0);
+		for (j = 0; j < TABLE_SIZE; j += 2 * i)
+			crc32table[i + j] = crc ^ crc32table[j];
+	}
+	for (i = 0; i < TABLE_SIZE; i++) {
+		crc = crc32table[i];
+		crc = crc32table[crc & 0xff] ^ (crc >> 8);
+	}
+}
+
+uint32_t crc32(uint32_t crc, unsigned char const *p, size_t len)
+{
+	crc = ~crc;
+
+	if (!is_crc32_table_initialized) {
+		crc32_init_lookup();
+		is_crc32_table_initialized = true;
+	}
+
+	while (len--) {
+		crc ^= *p++;
+		crc = (crc >> 8) ^ crc32table[crc & 255];
+	}
+
+	return ~crc;
+}
