+static void preboot_secure_dma_if_regs(u32 base)
+{
+	u32 mask, prot_base = base;
+
+	/* 0 - 0x300 */
+	set_prot_bits_range_all(prot_base, SNGL_REG_PROT_SIZE * 6);
+
+	/*
+	 * 0x300 - 0x400 allowed
+	 * 0x400 - 0x500 not allowed access
+	 */
+	prot_base += SNGL_REG_PROT_SIZE * 8;
+	set_prot_bits_range_all(prot_base, SNGL_REG_PROT_SIZE * 2);
+
+	/* 0x500 - 0x600 allowed */
+	/* 0x600 - 0x700 not allowed */
+	prot_base += SNGL_REG_PROT_SIZE * 4;
+	set_prot_bits_range_all(prot_base, SNGL_REG_PROT_SIZE * 2);
+
+	/* 0x700 - 0x780 as following: */
+	prot_base += SNGL_REG_PROT_SIZE * 2;
+	mask = ~(PROT_BIT(mmDMA_IF_W_S_DMA0_HIT_RPROT) &
+		 PROT_BIT(mmDMA_IF_W_S_DMA0_HIT_WPROT) &
+		 PROT_BIT(mmDMA_IF_W_S_DMA1_HIT_RPROT) &
+		 PROT_BIT(mmDMA_IF_W_S_DMA1_HIT_WPROT));
+	regmap_writel(mask, PRIV_PROT_REG(prot_base));
+
+	/* 0x780 - 0xF80 as following: */
+	prot_base += SNGL_REG_PROT_SIZE;
+	set_prot_bits_range_all(prot_base, PROT_SIZE_MAX - (prot_base - base));
+}

+#define DMA_DIST		0x20000

+static void preboot_secure_dma(void)
+{
+	u32 i, base, prot_base;
+
+	for (i = 0; i < DMA_ENGINES; i++) {
+		base = i * DMA_DIST;
+		prot_base = base + (mmDMA_IF_W_S_BASE & ~HL_CFG_BASE);
+		preboot_secure_dma_if_regs(prot_base);
+		prot_base = base + (mmDMA_IF_W_S_DOWN_BASE & ~HL_CFG_BASE);
+		set_prot_bits_range_all(prot_base, PROT_SIZE_MAX);
+		prot_base = base + (mmDMA_IF_W_S_DOWN_CH0_BASE & ~HL_CFG_BASE);
+		preboot_secure_dma_if_down_regs(prot_base);
+		prot_base = base + (mmDMA_IF_W_S_DOWN_CH1_BASE & ~HL_CFG_BASE);
+		preboot_secure_dma_if_down_regs(prot_base);
+	}
