Commit Message

[SW-33054] greco : move rr lbw logic to common

As per errata GOYA2_0100, any access to PCIE_MSI region would
result in a freeze. This logic for rr was available in the
legacy flow. Hence minimal logic required to block accesses to
PCIE_MSI region is carried over to common code so that it
is also available in secureboot flow.

+static void preboot_secure_update_rr_reg(int rr_idx, u32 inst_offset, u32 min_reg, u32 max_reg,
+		u32 min, u32 max)
+{
+	u32 rr_base, rr_dest;
+
+	rr_base = rr_idx * sizeof(u32) + inst_offset;
+	rr_dest = rr_base + min_reg;
+	regmap_writel(min, rr_dest);
+	rr_dest = rr_base + max_reg;
+	regmap_writel(max, rr_dest);
+}
+
+static void preboot_secure_update_hit_reg(int rr_idx, struct preboot_sec_rrs_cfg *cfg, int inst_num,
+		bool write_only)
+{
+	u32 rr_base, rr_dest, hit;
+	/* Enable (hit) RR */
+	rr_base = cfg->instances_dist * inst_num;
+	rr_dest = rr_base + cfg->hit_wr_breg;
+	hit = regmap_readl(rr_dest);
+	regmap_writel(hit | BIT(rr_idx), rr_dest);
+
+	if (write_only)
+		return;
+
+	rr_dest = rr_base + cfg->hit_rd_breg;
+	hit = regmap_readl(rr_dest);
+	regmap_writel(hit | BIT(rr_idx), rr_dest);
+}

+static void preboot_secure_lbw_rr_gen(int rr_idx, u32 min, u32 max,
