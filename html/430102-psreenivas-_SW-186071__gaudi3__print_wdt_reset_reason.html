+void wd_cause_dump(void)
+{
+	struct gaudi3_cold_rst_data cold_rst_data;
+
+	cold_rst_data.data = regmap_readl(mmCOLD_RST_DATA);
+
+	if (cold_rst_data.wd_rst_cause_arcevent)
+		preboot_puts("wd cause: ARCEVENT heartbeat\n");
+
+	if (cold_rst_data.wd_rst_cause_arcpid)
+		preboot_puts("wd cause: ARCPID heartbeat\n");
+
+	if ((!cold_rst_data.wd_rst_cause_arcevent) && (!cold_rst_data.wd_rst_cause_arcpid))
+		preboot_puts("wd cause: ARCMGMT heartbeat\n");
+
+	cold_rst_data.wd_rst_cause_arcevent = 0;
+	cold_rst_data.wd_rst_cause_arcpid = 0;
+
+	regmap_writel(cold_rst_data.data, mmCOLD_RST_DATA);
