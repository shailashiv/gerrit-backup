[SW-44575] gaudi2: preboot: pcie dynamic security

In order to add security in firmware, the access of sram memory to
host are limited to mandatory regions (eg.fit images) and the others
are blocked. This is done by blocking the pcie access using pcie
range registers

+static void hl_secure_dyn_rr_cfg(int rr_idx, u64 min64, u64 max64)
+{
+	struct hl_rr_eng_select rr_sel;
+
+	memset(&rr_sel, 0, sizeof(struct hl_rr_eng_select));
+	/* Block pcie RR for dyn rr */

+	/*
+	 * PCIE shares noc rtrs.

+	 * rtrs.
+	 */
+	rr_sel.noc = true;
+
+	hl_secure_hbw_rr(rr_idx, min64, max64, rr_sel);
+}

+#define IMAGE_OFFSET_MASK		0xF00000

+static u64 hl_calc_mem_region_full_addr(uintptr_t img_base)
+{
+	u32 img_offs;
+
+	img_offs = img_base & IMAGE_OFFSET_MASK;
+
+	return SRAM_BASE_ADDR + img_offs;
+}

+void preboot_security_dynamic_init(void)
+{
+	struct preboot_sec_dyn_pcie_rr_cfg *cfg;
+	struct mem_desc mdesc;
+	u64 img_max, image_base_full;
+	int i;
+
+	if (!preboot_security_is_enabled())
+		return;
+
+	/* Initialize image related configuration with default settings */
+	preboot_get_mem_region(MEM_LKD_PCI, &mdesc);
+	image_base_full = hl_calc_mem_region_full_addr(mdesc.base);
+	img_max = image_base_full + mdesc.size;
+	preboot_sec_dyn_cfg[SRAM_BGN_TO_IMG].max = image_base_full;
+	preboot_sec_dyn_cfg[IMAGE].min = image_base_full;
+	preboot_sec_dyn_cfg[IMAGE].max = img_max;
+	preboot_sec_dyn_cfg[IMG_END_TO_SRAM_END].min = img_max;
+
+	/* Apply all already known configurations */
+	for (i = SRAM_BGN_TO_IMG; i <= IMG_END_TO_SRAM_END; i++) {
+		cfg = &preboot_sec_dyn_cfg[i];
+
+		if (cfg->rr >= 0 && cfg->min && cfg->max)
+			hl_secure_dyn_rr_cfg(cfg->rr, cfg->min, cfg->max);
+	}
+}
+
+void preboot_security_dynamic_img_sram_open(enum mem_region img_region,
+				enum preboot_mem_access_mode mode)
+{
+	struct preboot_sec_dyn_pcie_rr_cfg *cfg;
+	struct mem_desc mdesc;
+	u64 img_max, image_base_full;
+
+	if (!preboot_security_is_enabled())
+		return;
+
+	switch (img_region) {
+	case MEM_LKD_PCI:
+	case MEM_FWUPD_PCI:
+		if (preboot_get_mem_region(img_region, &mdesc))
+			preboot_puts_hex("SEC: failed requesting mem region ",
+					 img_region, false);
+		break;
+	default:
+		preboot_puts_hex("SEC: forbidden opening mem region ",
+				 img_region, false);
+		return;
+	}
+
+	/* Move the image window to the requested image memory region */
+	image_base_full = hl_calc_mem_region_full_addr(mdesc.base);
+	img_max = image_base_full + mdesc.size;
+	preboot_sec_dyn_cfg[SRAM_BGN_TO_IMG].max = image_base_full;
+	preboot_sec_dyn_cfg[IMAGE].min = image_base_full;
+	preboot_sec_dyn_cfg[IMAGE].max = img_max;
+	preboot_sec_dyn_cfg[IMG_END_TO_SRAM_END].min = img_max;
+
+	/* Apply the new window limits */
+	cfg = &preboot_sec_dyn_cfg[SRAM_BGN_TO_IMG];
+	hl_secure_dyn_rr_cfg(cfg->rr, cfg->min, cfg->max);
+	cfg = &preboot_sec_dyn_cfg[IMG_END_TO_SRAM_END];
+	hl_secure_dyn_rr_cfg(cfg->rr, cfg->min, cfg->max);
+
+	/* TODO: Check mode based opening possibility for gaudi2 */
+	/* Finally open the image memory space */
+	hl_secure_dyn_rr_cfg(preboot_sec_dyn_cfg[IMAGE].rr, 0, 0);
+}
