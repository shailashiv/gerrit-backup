Commit Message

[SW-31074] hl_pvt: implement shutdown seq

PVT controller is configured in run continuous mode for reporting
on-die temperature values through I2C slave interface. At preboot stage
TPC and MME plls are running at REF clock

As part of Zephyr boot, PLLs are configured at nominal values and PVT
registers are re-configured with new clock freq.

Mooretec team recommended to shutdown TS IP while pll clock changes are
done.
Implemented shutdown sequence to exit continuous mode and disable
data acquisitions

+void hl_ts_pvt_shutdown(int ts)
+{
+	u32 addr = hl_ts_data[ts].base;
+
+	/* Exit the continuous mode of operation */
+	if (hl_tmp_is_sdif_free(addr))
+		sdif_write(addr, HL_TS_IP_CTRL_RUN_ONCE, HL_TS_IP_CTRL, 1, 1);
+	else
+		preboot_puts_hex("Failed to exit continuous mode for ",
+					ts, false);
+
+	/* Ensure embedded IP is shut down ip_ctrl = 0x000001 */
+	if (hl_tmp_is_sdif_free(addr))
+		sdif_write(addr, HL_TS_IP_CTRL_SHUTDOWN, HL_TS_IP_CTRL, 1, 1);
+	else
+		preboot_puts_hex("Failed to shutdown TS IP ",
+					ts, false);
+
+	/* Disable TS data acquisitions and ensure clock is shutdown fully */
+	if (hl_tmp_is_sdif_free(addr))
+		regmap_writel(0, hl_ts_data[ts].base + HL_TS_CMN_CLK_SYNTH);
+	else
+		preboot_puts_hex("Failed to disable TS data acquisitions for ",
+					ts, false);
+}
+
+void hl_ts_reset(void)
+{
+	preboot_puts("Performing RST of TS UNIT\n");
+	/* Perform unit address */
+	regmap_writel(0, mmPSOC_GLOBAL_CONF_UNIT_RST_N);
+	regmap_writel(UNIT_RST_N_L_MSK, mmPSOC_GLOBAL_CONF_UNIT_RST_N_L);
+	regmap_writel(0, mmPSOC_GLOBAL_CONF_UNIT_RST_N_M);
+	regmap_writel(0, mmPSOC_GLOBAL_CONF_UNIT_RST_N_H);
+	regmap_writel(1, mmPSOC_GLOBAL_CONF_UNIT_RST_N);
+	udelay(1000);
+	regmap_writel(0, mmPSOC_GLOBAL_CONF_UNIT_RST_N);
+}
