Commit Message

[SW-70527] gaudi2: ras: sram repair support

This patch added support for sram repair flow on sram(48MB).
The flow contains 3 different operations namely
BIRA(Built-in Repair Analysis), BISR(Built-in Soft Repair) and
BIST(Built-in Self Test). Those operations are done sequentially
on 48 SRAM rings.

As BIST operations leaves the memory in dirty state w.r.t ECC,
so all the memory needs to be initialized again. This is done
by scrubbing the memory after BIST.

+static int gaudi2_sms_check_pattern(uint32_t value, uint32_t ones_mask, uint32_t zeros_mask)
+{
+	if ((value & ones_mask) != ones_mask) {
+		preboot_puts("ones check failed\n");
+		return 1;
+	}
+
+	if ((~value & zeros_mask) != zeros_mask) {
+		preboot_puts("zeros check failed\n");
+		return 1;
+	}
+
+	return 0;

+{

+	enum gaudi2_sram_ring_type ring_type;

+	gaudi2_sram_beginning_pattern();

+		for (ring = 0; ring < ring_count_table[dfx]; ring++) {

+				gaudi2_sram_id_sel();
+			}
+		}

+#define SRAM_NUM_MEM_CPY_XFERS			48 /* (48*1024*1024)/(65535*16) */
+static void gaudi2_mask_scrub_sram(void)
+{
+	u32 retry = 1000;
+	u32 mem_cfg, cpy_sts;
+	u32 size = 0xFFFF0; /* (65535*16) bytes */
+	u32 block;
+	u64 addr;
+
+	for (block = 0; block <= SRAM_NUM_MEM_CPY_XFERS; block++) {
+		addr = SDRAM_BASE_FULL + size * block;
+		if (block == SRAM_NUM_MEM_CPY_XFERS) /* last block */
+			size = 768; /* (48*1024*1024) - (48*65535*16) bytes */
+
+		/* Use memblock to initialize dccm memory */
+		regmap_writel((u32)addr, mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_L);
+		regmap_writel((u32)(addr >> 32), mmPSOC_GLOBAL_CONF_MEM_CPY_DEST_ADDR_H);
+
+		mem_cfg = regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+		mem_cfg &= ~(PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_MEM_SIZE_MASK);
+
+		/* set size of in granuality of 16 bytes */
+		mem_cfg |= ((size >> 4) & PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_MEM_SIZE_MASK);
+		mem_cfg |= BIT(PSOC_GLOBAL_CONF_MEM_CPY_CTRL2_USE_CONST_SHIFT);
+
+		regmap_writel(mem_cfg, mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL2);
+		/* Enable copy and wait for completion */
+		regmap_writel(BIT(0), mmPSOC_GLOBAL_CONF_MEM_CPY_CTRL);
+		while (1) {
+			cpy_sts = regmap_readl(mmPSOC_GLOBAL_CONF_MEM_CPY_STATUS);
+			if (cpy_sts)
+				break;
+
+			retry--;
+			if (!retry) {
+				preboot_puts("sram mem_cpy timedout");
+				break;
+			}
+			udelay(DELAY_100K_USEC);
+		}
+	}
+}
