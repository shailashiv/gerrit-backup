+# Arguments:
+# - module name
+# - component dir
+# - component name

+define AGENT_SERDES_DEFINE_MODULE_ENVVARS
+	$(eval MODULE_NAME=$(strip $1))
+	$(eval MODULE_NAME_UPPER_CASE=$(shell echo $(MODULE_NAME) | tr a-z A-Z))
+	$(eval MODULE_NAME_UPPER_CASE=$(strip $(MODULE_NAME_UPPER_CASE)))
+	$(eval $(MODULE_NAME_UPPER_CASE)_SRCDIR=$(EMBEDDED_DIR)/zephyr-modules/$(MODULE_NAME))

+$(foreach module,$(AGENT_SERDES_APP_MODULES),$(call AGENT_SERDES_DEFINE_MODULE_ENVVARS,$(module),$(AGENT_SERDES_APP_PROD_NAME)))

+# Arguments:
+# - module (mbedtls, eeprom, etc.)
+define AGENT_SERDES_MODULE_VERSION_HOOK
+	$(eval MODULE_NAME_UPPER_CASE=$(shell echo $1 | tr a-z A-Z))
+	$(eval MODULE_NAME_UPPER_CASE=$(strip $(MODULE_NAME_UPPER_CASE)))
+	$(call HL_VERSION_GEN, ${${MODULE_NAME_UPPER_CASE}_SRCDIR}, ${${MODULE_NAME_UPPER_CASE}_DIR}, ${${MODULE_NAME_UPPER_CASE}_NAME})
+endef
+
+define AGENT_SERDES_APP_VERSION_GEN
+	@$(call HL_VERSION_GEN, ${AGENT_SERDES_SRCDIR},.,${AGENT_SERDES_APP_PROD_NAME})
+	@$(call HL_VERSION_GEN, ${AGENT_SERDES_SRCDIR},build,${AGENT_SERDES_APP_PROD_NAME})
+	@$(foreach module,$(AGENT_SERDES_APP_MODULES),$(call AGENT_SERDES_MODULE_VERSION_HOOK,$(module)))
+endef
+
