+void gaudi_hbm_row_repair(int dev, int dma_pc, int sid, u32 bank, u32 page)
+{
+	u32 write_wdr_addr, base, hbm_ch, row;
+	int ch;
+
+	if (dma_pc < 0 || dma_pc >= (GAUDI_HBM_CHANNELS * PC_PER_CH)) {
+		preboot_puts("ERROR : hbm_row_repair: Illegal pc value\n");
+		return;
+	}
+
+	if (dev < 0 || dev >= GAUDI_HBM_DEVICES) {
+		preboot_puts("ERROR : hbm_row_repair: Illegal HBM device\n");
+		return;
+	}
+
+	row = bank << 14 | page;
+
+	if (dma_pc % 2)
+		row = row | BIT(HBM_ROW_SOFT_REP_DQ_BIT);
+	else
+		row = row & ~BIT(HBM_ROW_SOFT_REP_DQ_BIT);
+
+	if (sid)
+		row = row | BIT(HBM_ROW_SOFT_REP_SID_BIT);
+	else
+		row = row & ~BIT(HBM_ROW_SOFT_REP_SID_BIT);
+
+	row = row | BIT(HBM_ROW_SOFT_REP_EN_BIT);
+	base = GAUDI_HBM_CFG_BASE + dev * GAUDI_HBM_CFG_OFFSET;
+	write_wdr_addr = base + (GAUDI_HBM_CENTER_CHANNEL * 0x1000) + 0x100;
+	ch = hbm_dma_ch_to_map_ch(dev, (dma_pc / 2));
+	hbm_ch = hbm_ch_mapping[ch];
+
+	/* enter self refresh */
+	RMWREG32(base + 0x4, 1, 1 << 16);
+	WREG32(write_wdr_addr, row);
+	ieee1500_inst(dev, hbm_ch, ieee1500_soft_repair, EXTEST_WRITE_INST);
+
+	/*
+	 * WRCK is ~44ns. So the minimal delay is 44*300 = 13.5usec.
+	 * See MCBIST samsung doc page 21. increase to 1msec for safety.
+	 */
+	udelay(HBM_ROW_REP_WAIT_TIME);
+
+	/* Exit self refresh */
+	RMWREG32(base + 0x4, 1, 1 << 17);
+
+	/* WRSTN assert */
+	RMWREG32(base + GAUDI_HBM_CENTER_CHANNEL * 0x1000 + 0x14, 0x0, 0x1);
+}
