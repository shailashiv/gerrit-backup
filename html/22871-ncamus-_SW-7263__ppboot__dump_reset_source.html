+static void rst_src_dump(void)
+{
+	unsigned int rst_src = regmap_readl(mmPSOC_GLOBAL_CONF_RST_SRC);
+	char *cause;
+
+	switch (rst_src) {
+	case HL_COLD_RST:
+		cause = "cold";
+		break;
+	case HL_MANUAL_RST:
+		cause = "manual";
+		break;
+	case HL_PRSTN_RST:
+		cause = "prstn";
+		break;
+	case HL_WD_RST:
+		cause = "watch dog";
+		break;
+	case HL_SW_ALL_RST:
+		cause = "software all";
+		break;
+	default:
+		cause = "unknown";
+		break;
+	}
+	preboot_puts_raw("Last reset cause is: ");
+	preboot_puts_raw(cause);
+	preboot_puts(" reset\n");
+}
