[SW-117389] ppboot: gaudi2: store ppboot hash into tpm

If cpld resets tpm and pcie link has to be brought up
the tpm init in erom tpm is taking approx 50ms and
firmware wont be able to get pcie link up within
anticipated time. hence adding tpm init in ppboot post
invoking arc0 to start pcie initialization and before
enabling pcie and i2c memory access. The i2c slave
configuration is done in preboot.

+static void gaudi2_get_ppboot_data(uint8_t **offset)
+{
+	uint32_t ppboot_pcie_size;
+
+	ppboot_pcie_size = *(uint32_t *)(HL_FWSRAM_BASE);
+
+	ppboot_pcie_size -= SIG_SIZE;
+	ppboot_pcie_size -= CRC32_SIZE;
+	ppboot_pcie_size -= TOTAL_PUB_KEY_PROPS_SZ;
+	ppboot_pcie_size -= PPBOOT_SHA384_FOOTER;
+	ppboot_pcie_size -= PPBOOT_HASH_MAGIC_SIZE;
+
+	*offset = (uint8_t *)(HL_FWSRAM_BASE + ppboot_pcie_size);
+}
+
+static bool is_ppboot_tpm_extend_done(void)
+{
+	uint32_t reg = regmap_readl(mmPSOC_SECURITY_SW_STATUS_1);
+	uint32_t mask = (PROGRESS_ARCBOOT_EROM_TPM_FINISH & 0x1F);
+
+	return !!(reg & BIT(mask));
+}
+
+void ppboot_tpm_extend(void)
+{
+	uint8_t *data;
+	uint32_t magic;
+	uint8_t hash[SHA384_SUM_LEN] = { 0 };
+
+	/* return if erom has already done ppboot tpm extend */
+	if (is_ppboot_tpm_extend_done())
+		return;
+
+	if (tpm_init(true))
+		return;
+
+	if (tpm_startup())
+		return;
+
+	gaudi2_get_ppboot_data(&data);
+
+	memcpy(&magic, data, PPBOOT_HASH_MAGIC_SIZE);
+
+	if (magic != PPBOOT_SHA_MAGIC) {
+		preboot_puts("no ppboot sha magic found\n");
+		return;
+	}
+
+	memcpy(hash, data + PPBOOT_HASH_MAGIC_SIZE, SHA384_SUM_LEN);
+
+	tpm_pcr_extend(TPM2_PPBOOT_PCR_IDX, hash);
+}
