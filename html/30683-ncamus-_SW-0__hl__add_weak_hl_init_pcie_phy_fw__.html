+/*
+ * FW_SRAM layout:
+ * ---------------
+ * CRC32 [4B]
+ * Signature [384B]
+ * Public Key [392B]
+ * PCIe PHY FW
+ * PPboot
+ * PPboot size [4B]
+ * FW Version [4B]
+ * Layout Size [4B]
+ */
+#define FW_SRAM_META_SIZE	(4 + 4 + 4 + 392 + 384 + 4)
+void hl_init_pcie_phy_fw(void)
+{
+	int i;
+	u32 *fw_sram_base = (u32 *)FW_SRAM_BASE;
+	u32 image_size = *fw_sram_base;
+	u32 ppboot_size = *(fw_sram_base + 2);
+	u32 size = image_size - ppboot_size - FW_SRAM_META_SIZE;
+	u32 *src = (u32 *)((char *)(fw_sram_base + 3) + ppboot_size);
+	u32 *dst = (u32 *)PCIE_FW_SRAM_BASE;
+	unsigned int rst_src = regmap_readl(mmPSOC_GLOBAL_CONF_RST_SRC);
+
+	if (rst_src != HL_COLD_RST)
+		return;
+
+	for (i = 0; i < size; i++)
+		*(dst + i) = *(src + i);
+}
