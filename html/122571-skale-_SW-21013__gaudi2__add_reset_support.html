Commit Message

[SW-21013] gaudi2: add reset support

Compared to previous platform, multiple registers needs to be
configure to configure mask for given reset. This patch implements
hl_dev_reset routine for gaudi2 which configures mask and issues
sw_all reset as part of reset routine.

+static void gaudi2_hw_fini(void)
+{
+	u32 i, count;
+
+	/* Restart BTL/BLR upon hard-reset */
+	regmap_writel(0x770, mmPSOC_GLOBAL_CONF_BOOT_SEQ_RE_START);
+
+	udelay(1000);
+
+	/*
+	 * Note: Currently FW updates mask to all registers of
+	 * software reset. Once gaudi2 FW/SW is stabilized and all
+	 * resets are done by FW, optimize this loop to update
+	 * mask regsiters for which default value needs to be
+	 * change.
+	 */
+	count = sizeof(sw_rst) / sizeof(struct gaudi2_rst_mask);
+	for (i = 0; i < count; i++)
+		regmap_writel(sw_rst[i].val, sw_rst[i].offset);
+
+	regmap_writel(1, mmPSOC_RESET_CONF_SW_ALL_RST);
+}
+
+void hl_dev_reset(void)
+{
+	hl_pcie_disable_access();
+
+	gaudi2_hw_fini();
+	wfe();
+}
